cmake_minimum_required(VERSION 3.16)
project(ForgeCorrectionHistoryProbe LANGUAGES CXX)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(forge_core
    source/src/bitboard.cpp source/src/board.cpp source/src/book.cpp
    source/src/eval.cpp source/src/eval_params.cpp source/src/nnue.cpp
    source/src/search.cpp)
target_include_directories(forge_core PUBLIC source/include)
target_compile_options(forge_core PUBLIC -ffp-contract=off -march=x86-64-v3)
target_compile_options(forge_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(PROBE_SANITIZERS)
    target_compile_options(forge_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(forge_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(correction_probe probe.cpp)
target_link_libraries(correction_probe PRIVATE forge_core Threads::Threads)
