// Original Forge AL1 graph using documented MIT Bullet public APIs.
// The numerical inference contract is Forge 294dbed3, not another engine.
use std::{env, path::Path, process};
use bullet_lib::{
    game::inputs::Chess768,
    nn::optimiser::AdamW,
    trainer::{save::SavedFormat,
        schedule::{TrainingSchedule, TrainingSteps, lr, wdl},
        settings::LocalSettings},
    value::{ValueTrainerBuilder, loader::DirectSequentialDataLoader},
};

fn usage() -> ! {
    eprintln!("usage: forge-lock-bullet-smoke DATA OUTPUT_NEW_DIR BATCHES --labels-already-residual");
    eprintln!("DATA must contain signed STM (teacher_cp - HCE_cp) scores. This smoke uses score-only residual targets; do not pass original full-eval labels.");
    process::exit(2)
}

fn main() {
    let args: Vec<String> = env::args().collect();
    if args.len() != 5 || args[4] != "--labels-already-residual" { usage(); }
    let data = &args[1];
    let destination = &args[2];
    let batches: usize = args[3].parse().unwrap_or_else(|_| usage());
    if batches == 0 || batches > 100_000 { usage(); }
    if !Path::new(data).is_file() || Path::new(destination).exists() {
        eprintln!("input must exist and output must be a new directory"); process::exit(2);
    }
    #[cfg(not(any(feature="rocm", feature="cuda", feature="metal")))]
    {
        eprintln!("No GPU backend enabled. This build only type-checks the graph; it cannot train.");
        process::exit(2);
    }
    #[allow(unreachable_code)]
    let mut trainer = ValueTrainerBuilder::default()
        .seed(20261030)
        .dual_perspective()
        .inputs(Chess768)
        .optimiser(AdamW)
        .save_format(&[
            SavedFormat::id("ftw").round().quantise::<i16>(4096),
            SavedFormat::id("ftb").round().quantise::<i16>(4096),
            SavedFormat::id("outw").round().quantise::<i16>(4096),
            SavedFormat::id("outb").round().quantise::<i16>(4096),
        ])
        .loss_fn(|residual, target| residual.sigmoid().squared_error(target))
        .build(|builder, stm, opponent| {
            let transform = builder.new_affine("ft", 768, 256);
            let output = builder.new_affine("out", 512, 1);
            let ours = transform.forward(stm).screlu();
            let theirs = transform.forward(opponent).screlu();
            output.forward(ours.concat(theirs))
        });
    let schedule = TrainingSchedule {
        net_id: "forge_lock_residual_smoke".to_owned(),
        eval_scale: 400.0 / 2.25,
        steps: TrainingSteps {batch_size: 64, batches_per_superbatch: batches,
            start_superbatch: 1, end_superbatch: 1},
        wdl_scheduler: wdl::ConstantWDL { value: 0.0 },
        lr_scheduler: lr::StepLR { start: 0.0001, gamma: 1.0, step: 1 },
        save_rate: 1,
    };
    std::fs::create_dir(destination).expect("new output directory creation failed");
    let settings = LocalSettings { threads: 1, test_set: None,
        output_directory: destination, batch_queue_size: 2 };
    let loader = DirectSequentialDataLoader::new(&[data]);
    trainer.run(&schedule, &settings, &loader);
}
