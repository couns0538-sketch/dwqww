"""Fable 2026-11-23: recount of Astra's referee table for A19 (message 081746Z; Neon's recount 084238Z) from the raw
rows of the message. Stockfish 17.1 was asked, with the move forced, how it values the move the 2,000-node search
chose and the move the 8,000-node search chose, on every position where the two differ.
    a19_referee_recount.py <the csv cut out of the message> <new output file>
Per position: delta = ((2 W8 + D8) - (2 W2 + D2)) / 2000; a position on which both chose the same move counts 0.
Expected BEFORE the run (their figures): at 500,000 nodes 2,902 positions, 2,198 equal, 704 disputed, the referee
prefers the 8k move in 238, ties in 320, prefers the 2k move in 146; mean over ALL +0.41949 points [0.20644,
0.64774], over the disputed +1.72919 [0.85565, 2.65731]. At 2,000,000 nodes (k = 3): 382 positions, 284 equal, 98
disputed, 34 / 49 / 15; ALL +0.60576 [0.11697, 1.17016], disputed +2.36122 [0.46561, 4.53873]. The same 382 at
500,000 nodes: ALL +0.4429 [-0.0700, 0.9886]. Intervals: games drawn with replacement, 20,000 times, sums of
numerators and of denominators; theirs with NumPy default_rng(202611230747). My draws use the same generator and
seed, but the order of my calls need not be theirs: the intervals may differ in the third digit.
"""
import csv
import hashlib
import sys

import numpy as np

src, out_path = sys.argv[1:3]
out = open(out_path, "x", encoding="utf-8")


def say(t):
    print(t, flush=True)
    out.write(t + "\n")


say(f"csv sha256 {hashlib.sha256(open(src, 'rb').read()).hexdigest()} (the message names 2ab446db...31dce)")
rows = list(csv.DictReader(open(src, encoding="utf-8", newline="")))
pos = {}
problems = 0
for r in rows:
    key = (r["ref_nodes"], int(r["game"]), int(r["k"]))
    p = pos.setdefault(key, {})
    if r["problem"]:
        problems += 1
    if r["arm"] == "equal":
        assert not p, key
        p["equal"] = True
    else:
        assert r["arm"] in ("2", "8") and r["arm"] not in p and "equal" not in p, key
        w, d, l = int(r["W"]), int(r["D"]), int(r["L"])
        assert w + d + l == 1000, key
        assert r["returned"] == (r["move2k"] if r["arm"] == "2" else r["move8k"]), key
        p[r["arm"]] = 2 * w + d
say(f"{len(rows)} rows, {len(pos)} position-and-budget records, rows with a problem flag: {problems}")
games = sorted({k[1] for k in pos})
say(f"{len(games)} game ids that carry a record (the bootstrap draws from all 384 ids: an id without a record adds nothing)")


def scope(budget, only_k3=False, base_budget=None):
    """per game: (number of positions, sum of delta, number disputed, sum over disputed); and the three counts."""
    per = {g: [0, 0.0, 0, 0.0] for g in range(384)}
    pref = [0, 0, 0]
    n = eq = 0
    for (b, g, k), p in pos.items():
        if b != budget or (only_k3 and k != 3):
            continue
        n += 1
        if "equal" in p:
            eq += 1
            delta = None
        else:
            assert "2" in p and "8" in p, (b, g, k)
            delta = (p["8"] - p["2"]) / 2000.0
            pref[0 if delta > 0 else 1 if delta == 0 else 2] += 1
        per[g][0] += 1
        if delta is not None:
            per[g][1] += delta
            per[g][2] += 1
            per[g][3] += delta
    a = np.array([per[g] for g in range(384)])
    rng = np.random.default_rng(202611230747)
    idx = rng.integers(0, 384, size=(20000, 384))
    s = a[idx].sum(axis=1)
    all_pp = 100 * a[:, 1].sum() / a[:, 0].sum()
    dis_pp = 100 * a[:, 3].sum() / a[:, 2].sum()
    lo_a, hi_a = np.percentile(100 * s[:, 1] / s[:, 0], [2.5, 97.5])
    ok = s[:, 2] > 0
    lo_d, hi_d = np.percentile(100 * s[ok, 3] / s[ok, 2], [2.5, 97.5])
    return n, eq, n - eq, pref, all_pp, (lo_a, hi_a), dis_pp, (lo_d, hi_d)


for name, args, theirs in (
        ("500,000 nodes, all positions", ("500000",), "2,902 / 2,198 / 704; 238 / 320 / 146; ALL +0.41949 [0.20644, 0.64774]; disputed +1.72919 [0.85565, 2.65731]"),
        ("2,000,000 nodes, k = 3", ("2000000",), "382 / 284 / 98; 34 / 49 / 15; ALL +0.60576 [0.11697, 1.17016]; disputed +2.36122 [0.46561, 4.53873]"),
        ("500,000 nodes, k = 3 only", ("500000", True), "ALL +0.4429 [-0.0700, 0.9886]")):
    n, eq, dis, pref, all_pp, ci_a, dis_pp, ci_d = scope(*args)
    say(f"{name}: {n:,} positions, {eq:,} equal, {dis:,} disputed; the referee prefers the 8k move in {pref[0]}, ties in "
        f"{pref[1]}, prefers the 2k move in {pref[2]}; mean over ALL {all_pp:+.5f} points [{ci_a[0]:.5f}, {ci_a[1]:.5f}], "
        f"over the disputed {dis_pp:+.5f} [{ci_d[0]:.5f}, {ci_d[1]:.5f}]")
    say(f"    theirs: {theirs}")
out.close()
