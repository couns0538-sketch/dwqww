def say(t):

def whole(epoch, add=0):
    r = random.Random(202611230639 + add + epoch)
    ids = list(range(N))
    r.shuffle(ids)
    return [ids[b * K:(b + 1) * K] for b in range(B)]


def stratified(epoch, add=0):
    r = random.Random(202611230639 + add + epoch)
    strata = []
    for s in range(8):
        x = [i for i in range(N) if (i // 64) % 8 == s]
        r.shuffle(x)
        strata.append(x)
    return [[i for s in range(8) for i in strata[s][16 * b:16 * b + 16]] for b in range(B)]


def labels(batches):
    lab = np.empty(N, dtype=np.int64)
    seen = 0
    for b, rows in enumerate(batches):
        lab[rows] = b
        seen += len(rows)
    assert seen == N and len({i for rows in batches for i in rows}) == N
    return lab


def a22(passes):
    labs = [labels(p) for p in passes]
    best, where, ties = -1, None, 0
    for i, j in itertools.combinations(range(len(passes)), 2):
        t = np.zeros((B, B), dtype=np.int64)
        np.add.at(t, (labs[i], labs[j]), 1)
        num = int((t * (t - 1)).sum())
        if num > best:
            best, where, ties = num, (i, j), 1
        elif num == best:
            ties += 1
    return best, where, ties


def a20(passes):
    m = np.zeros((N, N), dtype=bool)
    for p in passes:
        for rows in p:
            idx = np.array(rows)
            m[np.ix_(idx, idx)] = True
    return float((m.sum(axis=1) - 1).mean())


def scramble(passes):
    res = []
    for e, p in enumerate(passes):
        r = random.Random(202611230822 + e)
        q = [list(rows) for rows in p]
        for rows in q:
            r.shuffle(rows)
        r.shuffle(q)
        res.append(q)
    return res
