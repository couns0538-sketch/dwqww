def strata_ids():
    return [[64 * g + row for g in range(r, 128, 8) for row in range(64)] for r in range(8)]


def fresh_stratified(seed):
    """One pass: every stratum shuffled in order 0..7 by ONE Random(seed), 16 rows of each stratum a batch.
    Returns the ordered row stream as 64 batches of 128 row IDs."""
    rng = random.Random(seed)
    st = strata_ids()
    for v in st:
        rng.shuffle(v)
    return [sum((v[16 * i:16 * i + 16] for v in st), []) for i in range(NB)]


def reorder(batches, seed):
    """The plants' disguise: batch order first, then the rows inside each batch, by Random(seed)."""
    rng = random.Random(seed)
    order = list(range(NB))
    rng.shuffle(order)
    out = []
    for i in order:
        rows = list(batches[i])
        rng.shuffle(rows)
        out.append(rows)
    return out


def judge(streams):
    """What a test of a real reader sees: for every pass the ordered row IDs in batches. Returns coverage,
    mean distinct co-batch partners a row over all passes, and the mean share of a row's partners kept at lag 1
    and at lag 2."""
    cover = all(sorted(x for b in s for x in b) == list(range(N)) and all(len(b) == B for b in s) for s in streams)
    union = np.zeros((N, N), dtype=bool)
    mats = []
    for s in streams:
        lab = np.empty(N, dtype=np.int16)
        for i, b in enumerate(s):
            lab[np.asarray(b)] = i
        m = lab[:, None] == lab[None, :]
        union |= m
        mats.append(lab)
    distinct = (union.sum() - N) / N

    def lag(k):
        tot = 0
        for e in range(len(mats) - k):
            a = mats[e][:, None] == mats[e][None, :]
            b = mats[e + k][:, None] == mats[e + k][None, :]
            tot += (a & b).sum() - N
        return tot / ((len(mats) - k) * N * (B - 1))

    return cover, float(distinct), float(lag(1)), float(lag(2))


def games_and_collision(stream):
    games = sum(len({x // 64 for x in b}) for b in stream) / NB
    tot = 0
    for b in stream:
        cnt = {}
        for x in b:
            cnt[x // 64] = cnt.get(x // 64, 0) + 1
        tot += sum(n * (n - 1) for n in cnt.values())
    return games, tot / (NB * B * (B - 1))

def slope(s):

def elo(s):

def constrained_logl(counts, mu):
    """As in n13_recount.py (the repaired, full-support form)."""
    n = sum(counts)
    used = [(c / n, x) for c, x in zip(counts, X) if c > 0]
    has0 = any(x == 0.0 for _f, x in used)
    has1 = any(x == 1.0 for _f, x in used)
    t_lo, t_hi = -1.0 / (1.0 - mu), 1.0 / mu

    def g(t):
        return sum(fi * (xi - mu) / (1.0 + t * (xi - mu)) for fi, xi in used)

    if not has0 and g(t_hi) >= 0:
        t = t_hi
    elif not has1 and g(t_lo) <= 0:
        t = t_lo
    else:
        a, b = t_lo + 1e-12, t_hi - 1e-12
        for _ in range(200):
            m = 0.5 * (a + b)
            if g(m) > 0:
                a = m
            else:
                b = m
        t = 0.5 * (a + b)
    p = [fi / (1.0 + t * (xi - mu)) for fi, xi in used]
    return sum(c * math.log(pi) for c, pi in zip([c for c in counts if c > 0], p))


def llr(counts, e0, e1):
    return constrained_logl(counts, score(e1)) - constrained_logl(counts, score(e0))

def wald_games(e0, v):
    ds = score(0.0) - score(e0)
    per_pair = ds * ds / (2.0 * v)
    return 2.0 * 0.9 * math.log(19.0) / per_pair
