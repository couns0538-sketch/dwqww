+using Job2Buckets = std::array<std::uint64_t, 32>;
+using Job2Counters = std::array<Job2Buckets, job2_event_names.size()>;
+
+thread_local Job2Counters* job2_counters = nullptr;
+
+void job2_count(CountEvent event, int depth, std::uint64_t amount = 1) {
+    if (job2_counters != nullptr) {
+        (*job2_counters)[static_cast<std::size_t>(event)][std::clamp(depth, 0, 31)] += amount;
+    }
+}
+
+void job2_uncount(CountEvent event, int depth) {
+    if (job2_counters != nullptr) {
+        --(*job2_counters)[static_cast<std::size_t>(event)][std::clamp(depth, 0, 31)];
+    }
+}
+
+void print_job2_counters(const Job2Counters& counters, std::uint64_t actual_nodes) {
+    std::uint64_t counted_nodes = 0;
+    for (const CountEvent event : {CountEvent::main_nodes, CountEvent::q_nodes}) {
+        for (const auto nodes : counters[static_cast<std::size_t>(event)]) counted_nodes += nodes;
+    }
+    const std::lock_guard<std::mutex> lock(uci_output_mutex());
+    if (counted_nodes != actual_nodes) {
+        std::cout << "info string ERROR CNT node accounting " << counted_nodes
+                  << " expected " << actual_nodes << std::endl;
+    }
+    std::cout << "info string CNT {";
+    for (std::size_t event = 0; event < counters.size(); ++event) {
+        if (event != 0) std::cout << ',';
+        std::cout << '"' << job2_event_names[event] << "\":[";
+        for (std::size_t bucket = 0; bucket < counters[event].size(); ++bucket) {
+            if (bucket != 0) std::cout << ',';
+            std::cout << counters[event][bucket];
+        }
+        std::cout << ']';
+    }
+    std::cout << '}' << std::endl;
+}
