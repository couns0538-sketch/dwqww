+[[nodiscard]] inline LesserThreats lesser_threats(const Board& board) {
+    const Color them = opposite(board.side_to_move());
+    const Bitboard occupancy = board.occupied();
+    const Bitboard by_pawn = pawn_attacks_bb(them, board.pieces(them, Pawn));
+    Bitboard by_minor = by_pawn;
+    for (Bitboard bb = board.pieces(them, Knight); bb != 0;) by_minor |= attacks::knight(pop_lsb(bb));
+    for (Bitboard bb = board.pieces(them, Bishop); bb != 0;) by_minor |= attacks::bishop(pop_lsb(bb), occupancy);
+    Bitboard by_rook = by_minor;
+    for (Bitboard bb = board.pieces(them, Rook); bb != 0;) by_rook |= attacks::rook(pop_lsb(bb), occupancy);
+    LesserThreats threats;
+    threats.by_type[static_cast<std::size_t>(Pawn)] = by_pawn;
+    threats.by_type[static_cast<std::size_t>(Knight)] = by_pawn;
+    threats.by_type[static_cast<std::size_t>(Bishop)] = by_pawn;
+    threats.by_type[static_cast<std::size_t>(Rook)] = by_minor;
+    threats.by_type[static_cast<std::size_t>(Queen)] = by_rook;
+    return threats;
+}

+// 0 or 1: is `square` (0..63) in `set`.
+[[nodiscard]] constexpr std::size_t square_bit(Bitboard set, int square) {
+    return static_cast<std::size_t>((set >> square) & 1U);
+}
+
+// Butterfly term of a quiet move: quarters/4 from the threat-keyed twin, the rest from the plain table.
+// |plain|, |threat| <= HistoryLimit and 0 <= quarters <= 4 give |numerator| <= 65536 and |result| <= HistoryLimit
+// (C++ division truncates toward zero).
+[[nodiscard]] constexpr int blend(int plain, int threat, int quarters) {
+    return (plain * (4 - quarters) + threat * quarters) / 4;
+}
+
+// Self-damping update of a joint-correction entry: entry + step - entry * |step| / limit. For |entry| <= limit and
+// |step| <= limit the result stays within +-limit; the clamp is a guard. |entry * step| <= limit^2 = 2^26 fits int32.
+[[nodiscard]] constexpr std::int32_t gravity(std::int32_t entry, std::int32_t step, std::int32_t limit) {
+    const std::int32_t magnitude = step < 0 ? -step : step;
+    const std::int32_t next = entry + step - entry * magnitude / limit;
+    return next < -limit ? -limit : (next > limit ? limit : next);
+}
+
+// Non-pawn material signature: 4 bits per (colour, knight/bishop/rook/queen) count, saturated at 15.
+[[nodiscard]] inline std::uint32_t non_pawn_material(const Board& board) {
+    std::uint32_t signature = 0;
+    for (const Color color : {White, Black}) {
+        for (int type = Knight; type <= Queen; ++type) {
+            const int count = popcount(board.pieces(color, static_cast<PieceType>(type)));
+            signature = (signature << 4U) | static_cast<std::uint32_t>(count < 15 ? count : 15);
+        }
+    }
+    return signature;
+}
+
+// SplitMix64 finaliser (a public-domain mixing function): every signature bit reaches the low index bits.
+[[nodiscard]] constexpr std::uint64_t mix64(std::uint64_t z) {
+    z = (z ^ (z >> 30U)) * 0xBF58476D1CE4E5B9ULL;
+    z = (z ^ (z >> 27U)) * 0x94D049BB133111EBULL;
+    return z ^ (z >> 31U);
+}
+
+// Key of the pawn-structure x non-pawn-material correction: the Zobrist pawn key mixed with the signature.
+[[nodiscard]] inline std::uint64_t joint_key(const Board& board) {
+    return board.pawn_key() ^ mix64(static_cast<std::uint64_t>(non_pawn_material(board)) + 1U);
+}

+// Ported from opus-r13-p1-nearmiss-ports:opus_r13/p1/b1/b1_port.patch (src/board.cpp), which was written from
+// M. van Kervinck's description (2013). One change: the table is built when the program starts instead of at
+// compile time, and upcoming_repetition() answers false unless the build came out complete with 3,668 entries.
+//
+// A position P that occurred i plies ago (i odd, inside the reversible window) can be reached again by one move of
+// the side to move exactly when (1) the opponent's men stand where they stood in P: the XOR of the opponent's key
+// deltas since P (side key removed) is zero, and (2) the key difference key ^ key(P) is the key of a single
+// reversible move of the side to move, which the table recognises in at most two probes, and the move is possible
+// now (mover on one end, the other end empty, no man in between). Such a move is legal: it recreates P, in which the
+// side to move was not in check.
+namespace {
+
+constexpr std::size_t CuckooSlots = 8192;
+
+struct CuckooMove {
+    std::uint8_t piece = 0;  // Piece (never a pawn); 0 = empty slot
+    std::uint8_t a = 0;      // lower square
+    std::uint8_t b = 0;      // higher square
+};
+
+struct CuckooTable {
+    std::array<std::uint64_t, CuckooSlots> keys{};  // 0 = empty slot
+    std::array<CuckooMove, CuckooSlots> moves{};
+    std::size_t entries = 0;
+    bool complete = true;  // every insertion found a slot
+};
+
+// Two independent 13-bit slot indices from different halves of the key.
+std::size_t cuckoo_slot_low(std::uint64_t key) { return static_cast<std::size_t>(key) & (CuckooSlots - 1); }
+std::size_t cuckoo_slot_high(std::uint64_t key) { return static_cast<std::size_t>(key >> 32U) & (CuckooSlots - 1); }
+
+// Whether a piece of this type connects squares a and b on an empty board.
+bool reversible_step(int type, int a, int b) {
+    const int df = file_of(a) - file_of(b);
+    const int dr = rank_of(a) - rank_of(b);
+    const int files = df < 0 ? -df : df;
+    const int ranks = dr < 0 ? -dr : dr;
+    const bool diagonal = files == ranks && files != 0;
+    const bool straight = (files == 0) != (ranks == 0);
+    switch (type) {
+    case Knight: return (files == 1 && ranks == 2) || (files == 2 && ranks == 1);
+    case Bishop: return diagonal;
+    case Rook: return straight;
+    case Queen: return diagonal || straight;
+    case King: return files <= 1 && ranks <= 1 && (files | ranks) != 0;
+    default: return false;
+    }
+}
+
+std::uint64_t reversible_move_key(int piece, int a, int b) {
+    return piece_key(a, static_cast<Piece>(piece)) ^ piece_key(b, static_cast<Piece>(piece)) ^ Z.side;
+}
+
+// Cuckoo hashing with two candidate slots per key: an insertion takes its first slot and evicts the occupant,
+// which moves to its other slot, and so on.
+CuckooTable build_cuckoo_table() {
+    CuckooTable table{};
+    for (int piece = WN; piece <= BK; ++piece) {
+        const int type = piece_type(static_cast<Piece>(piece));
+        if (type == Pawn) continue;
+        for (int a = 0; a < 64; ++a) {
+            for (int b = a + 1; b < 64; ++b) {
+                if (!reversible_step(type, a, b)) continue;
+                std::uint64_t key = reversible_move_key(piece, a, b);
+                CuckooMove move{static_cast<std::uint8_t>(piece), static_cast<std::uint8_t>(a),
+                                static_cast<std::uint8_t>(b)};
+                if (key == 0) table.complete = false;
+                std::size_t slot = cuckoo_slot_low(key);
+                for (int evictions = 0; key != 0; ++evictions) {
+                    if (evictions > 4096) {
+                        table.complete = false;
+                        break;
+                    }
+                    std::swap(table.keys[slot], key);
+                    std::swap(table.moves[slot], move);
+                    slot = slot == cuckoo_slot_low(key) ? cuckoo_slot_high(key) : cuckoo_slot_low(key);
+                }
+                ++table.entries;
+            }
+        }
+    }
+    return table;
+}
+
+// Built once at program start (Z is a constant, so it is ready before this runs).
+const CuckooTable CuckooMoves = build_cuckoo_table();
+// 2 colours x (N 168 + B 280 + R 448 + Q 728 + K 210) = 3,668 reversible moves.
+const bool CuckooOk = CuckooMoves.complete && CuckooMoves.entries == 3668;
+
+const CuckooMove* cuckoo_find(std::uint64_t key) {
+    if (key == 0) return nullptr;
+    std::size_t slot = cuckoo_slot_low(key);
+    if (CuckooMoves.keys[slot] == key) return &CuckooMoves.moves[slot];
+    slot = cuckoo_slot_high(key);
+    if (CuckooMoves.keys[slot] == key) return &CuckooMoves.moves[slot];
+    return nullptr;
+}

+bool Board::upcoming_repetition(std::size_t root_index) const {
+    if (!CuckooOk) return false;
+    // After one more reversible move, is_repetition(root_index) of the new position scans its earlier positions at
+    // even distances 4 .. span, where span is the smaller of its halfmove clock, its plies since a null move (both
+    // one more than here, with the same saturation as make_move) and its number of earlier positions. Seen from
+    // here those are the odd distances 3 .. span-1.
+    const std::size_t size = key_history_.size();
+    const std::size_t clock = std::min<std::size_t>(static_cast<std::size_t>(halfmove_) + 1, 65535);
+    const std::size_t since_null = std::min<std::size_t>(static_cast<std::size_t>(plies_from_null_) + 1, 1024);
+    const std::size_t span = std::min(std::min(clock, since_null), size);
+    if (span < 4) return false;
+    const std::uint64_t* const keys = key_history_.data();
+    const std::size_t now = size - 1;
+    // Running XOR of the opponent's move deltas (one ply each, side key removed) between the candidate earlier
+    // position and now.
+    std::uint64_t opponent = keys[now] ^ keys[now - 1] ^ Z.side;
+    for (std::size_t distance = 3; distance < span; distance += 2) {
+        opponent ^= keys[now - distance + 1] ^ keys[now - distance] ^ Z.side;
+        if (opponent != 0) continue;
+        const std::uint64_t earlier = keys[now - distance];
+        const CuckooMove* const move = cuckoo_find(key_ ^ earlier);
+        if (move == nullptr) continue;
+        const Piece piece = static_cast<Piece>(move->piece);
+        if (color_of(piece) != side_) continue;
+        int from = move->a;
+        int to = move->b;
+        if (squares_[static_cast<std::size_t>(from)] != piece) std::swap(from, to);
+        if (squares_[static_cast<std::size_t>(from)] != piece || squares_[static_cast<std::size_t>(to)] != Empty ||
+            (attacks::between(from, to) & occupied()) != 0) {
+            continue;
+        }
+        // As in is_repetition(root_index): one occurrence at or after the root is a draw; before the root a second
+        // occurrence inside the window is needed.
+        if (now - distance >= root_index) return true;
+        for (std::size_t further = distance + 2; further < span; further += 2) {
+            if (keys[now - further] == earlier) return true;
+        }
+    }
+    return false;
+}
