+// Where the knights and bishops of both sides stand. Each side's minors are mixed
+// with the splitmix64 finalizer, then the two sides are combined the same way, so a
+// move of one minor changes about half the bits of the result.
+[[nodiscard]] std::uint64_t minor_key(const Board& board) {
+    auto mix = [](std::uint64_t x) {
+        x += 0x9E3779B97F4A7C15ULL;
+        x = (x ^ (x >> 30)) * 0xBF58476D1CE4E5B9ULL;
+        x = (x ^ (x >> 27)) * 0x94D049BB133111EBULL;
+        return x ^ (x >> 31);
+    };
+    // Each piece type gets its OWN multiplier before mixing. Sharing one between
+    // knights and bishops would make a knight and a bishop on the same square the
+    // same key (they would XOR to zero), which the 100,000-position test found.
+    const std::uint64_t wk = mix(static_cast<std::uint64_t>(board.pieces(White, Knight)) ^ 0x243F6A8885A308D3ULL);
+    const std::uint64_t wb = mix(static_cast<std::uint64_t>(board.pieces(White, Bishop)) ^ 0x13198A2E03707344ULL);
+    const std::uint64_t bk = mix(static_cast<std::uint64_t>(board.pieces(Black, Knight)) ^ 0xA4093822299F31D0ULL);
+    const std::uint64_t bb = mix(static_cast<std::uint64_t>(board.pieces(Black, Bishop)) ^ 0x082EFA98EC4E6C89ULL);
+    return mix(mix(wk ^ wb) ^ (mix(bk ^ bb) * 0x9E3779B97F4A7C15ULL));
