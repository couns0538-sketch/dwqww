node <<'NODE'
const fs=require('fs'),path=require('path'),crypto=require('crypto');const root='/workspace/work/all-run/branch-review-r3-complete',out=root+'/ready-to-copy';const hash=b=>crypto.createHash('sha256').update(b).digest('hex');const provenanceFile=out+'/branch_review_evidence/COPY_PROVENANCE.json.txt';const provenance=JSON.parse(fs.readFileSync(provenanceFile,'utf8'));
