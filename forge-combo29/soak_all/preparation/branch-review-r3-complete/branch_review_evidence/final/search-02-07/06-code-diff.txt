+// Kings are left out: this engine's SEE value for a king is 0, so a king is
+// neither a material attacker nor a material victim.
+bool cheaper_enemy_attacks_dearer(const Board& board) {
+    const Color us = board.side_to_move();
+    const Bitboard enemy = board.occupied(opposite(us));
+    Bitboard ours = board.pieces(us, Knight) | board.pieces(us, Bishop) | board.pieces(us, Rook) |
+                    board.pieces(us, Queen);
+    while (ours != 0) {
+        const int square = pop_lsb(ours);
+        const int victim = SeePieceValue[static_cast<std::size_t>(piece_type(board.piece_at(square)))];
+        Bitboard hits = board.attackers_to(square) & enemy;
+        while (hits != 0) {
+            const int from = pop_lsb(hits);
+            const int attacker = SeePieceValue[static_cast<std::size_t>(piece_type(board.piece_at(from)))];
+            if (attacker > 0 && attacker < victim) return true;
+        }
+    }
+    return false;
+}
