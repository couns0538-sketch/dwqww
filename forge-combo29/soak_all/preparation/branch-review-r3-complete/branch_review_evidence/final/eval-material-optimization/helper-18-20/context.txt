   454	    MovePicker(const Board& board, const Worker& searcher, const SearchStack* ss, Move tt_move, Mode mode)
   455	        : board_(board), searcher_(searcher), ss_(ss), tt_move_(tt_move) {

   458	        if (mode == Mode::Quiescence && !in_check && !tt_move_.is_none() && !tt_move_.is_capture() &&
   459	            !tt_move_.is_promotion()) {
   460	            tt_move_ = Move::none();
   461	        }
   462	        if (in_check) stage_ = tt_move_.is_none() ? EvasionGenerate : EvasionTT;
   463	        else if (mode == Mode::Quiescence) stage_ = tt_move_.is_none() ? QsGenerate : QsTT;
   464	        else stage_ = tt_move_.is_none() ? GenerateCaptures : MainTT;

   483	            case GenerateCaptures:

   486	                [[fallthrough]];
   487	            case GoodCaptures:

   492	#if FORGE_SDK_BADCAP_PREFIX

   679	    // The selection keeps the lowest index on a tie (`>`), which is what the
   680	    // following move order depends on. Chunks of four compare inside the chunk
   681	    // first so the running best is not a loop-carried load on every move.
   682	    [[nodiscard]] Move pick_best() {
   683	        if (!sorted_) {
   684	            if (index_ - list_start_ < 4) {
   685	                if (seeded_ && index_ == list_start_) {
   686	                    if (seeded_at_ != index_) std::swap(moves_[index_], moves_[seeded_at_]);
   687	                    seeded_ = false;
   688	                    return moves_[index_++].move;
   689	                }
   690	                std::size_t best = index_;
   691	                int best_score = moves_[best].score;
   692	                std::size_t i = index_ + 1;
   693	                const std::size_t n = count_;
   694	#if defined(__AVX2__)
   695	                static_assert(sizeof(Scored) == 8, "score lane shuffle assumes an 8-byte record");
   696	                while (i + 4 <= n) {
   697	                    const __m256i packed = _mm256_loadu_si256(reinterpret_cast<const __m256i*>(&moves_[i]));
   698	                    const __m256i lanes = _mm256_setr_epi32(1, 3, 5, 7, 0, 2, 4, 6);
   699	                    const __m128i scores = _mm256_castsi256_si128(_mm256_permutevar8x32_epi32(packed, lanes));
   700	                    const __m128i swap_halves = _mm_shuffle_epi32(scores, _MM_SHUFFLE(1, 0, 3, 2));
   701	                    __m128i folded = _mm_max_epi32(scores, swap_halves);
   702	                    folded = _mm_max_epi32(folded, _mm_shuffle_epi32(folded, _MM_SHUFFLE(2, 3, 0, 1)));
   703	                    const int chunk_max = _mm_cvtsi128_si32(folded);
   704	                    if (chunk_max > best_score) {
   705	                        const __m128i equal = _mm_cmpeq_epi32(scores, _mm_set1_epi32(chunk_max));
   706	                        const int mask = _mm_movemask_ps(_mm_castsi128_ps(equal));
   707	                        best = i + static_cast<std::size_t>(__builtin_ctz(mask));
   708	                        best_score = chunk_max;
   709	                    }
   710	                    i += 4;
   711	                }
   712	#endif
   713	                for (; i < n; ++i) {
   714	                    const int score = moves_[i].score;
   715	                    if (score > best_score) {
   716	                        best_score = score;
   717	                        best = i;
   718	                    }
   719	                }
   720	                if (best != index_) std::swap(moves_[index_], moves_[best]);
   721	            } else {
   722	                std::sort(moves_.begin() + static_cast<std::ptrdiff_t>(index_),
   723	                          moves_.begin() + static_cast<std::ptrdiff_t>(count_),
   724	                          [](const Scored& a, const Scored& b) { return a.score > b.score; });
   725	                sorted_ = true;
   726	            }
   727	        }
   728	        return moves_[index_++].move;
   729	    }

    71	    const Bitboard enemy = board.pieces(Them, Pawn);

    74	    Bitboard pawns = own;
    75	    while (pawns != 0) {
   220	        const std::size_t bishop_index = static_cast<std::size_t>(_pext_u64(occupied, bishop_slider.mask));

   222	        const Bitboard queen_bishop = bishop_slider.table[bishop_index];

   241	template <bool Trace, Color Us>
   242	void evaluate_king(const Board& board, Accumulator<Trace>& acc, const EvalInfo& info) {
   243	    constexpr Color Them = opposite(Us);
   244	    const int king = info.king_square[Us];
   245	    const int king_file = file_of(king);
   246	    const int king_rank = relative_rank(Us, king);
   247	    const Bitboard own_pawns = board.pieces(Us, Pawn);
   248	    const Bitboard enemy_pawns = board.pieces(Them, Pawn);
   249	
   250	    int shelter = 0;
   251	    auto shelter_term = [&](int index) { shelter += acc.template add_mg<Us>(index); };
   252	    for (int file = std::max(0, king_file - 1); file <= std::min(7, king_file + 1); ++file) {
   253	        const Bitboard file_mask = file_bb(file) & attacks::forward_ranks(Us, king);
   254	        const Bitboard own = own_pawns & file_mask;
   255	        const Bitboard enemy = enemy_pawns & file_mask;
   256	        int own_distance = 0;
   257	        if (own != 0) {
   258	            const int nearest = Us == White ? lsb(own) : msb(own);
   259	            own_distance = std::min(3, relative_rank(Us, nearest) - king_rank);
   260	        }
   261	        shelter_term(ShelterDistance + own_distance);
   262	        if (enemy != 0) {
   263	            const int nearest = Us == White ? lsb(enemy) : msb(enemy);
   264	            const int distance = relative_rank(Us, nearest) - king_rank;
   265	            if (distance <= 3) shelter_term(StormDistance + std::max(1, distance));
   266	        }
   267	        if (((own_pawns | enemy_pawns) & file_bb(file)) == 0) shelter_term(KingFileOpen);
   268	    }
   269	
   270	    // King danger from pieces attacking the zone; requires two attackers
   271	    // unless a queen participates.
   272	    const int attackers = info.king_attackers[Us];
   273	    if (attackers >= 2 || (attackers >= 1 && board.pieces(Them, Queen) != 0)) {
   274	        int units = info.king_attack_units[Us];
   275	        const Bitboard zone = info.king_zone[Us];
   276	        const Bitboard weak = zone & info.attacked_by[Them][0] & ~info.attacked_twice[Us] &
   277	                              (~info.attacked_by[Us][0] | info.attacked_by[Us][King] | info.attacked_by[Us][Queen]);
   278	        units += king_safety.weak_square * popcount(weak);
   279	        const Bitboard safe = ~board.occupied(Them) & (~info.attacked_by[Us][0] | (weak & info.attacked_twice[Them]));
   280	        const Bitboard occ = board.occupied();
   281	        const attacks::PextSlider& rook_slider = attacks::RookPext[static_cast<std::size_t>(king)];
   282	        const attacks::PextSlider& bishop_slider = attacks::BishopPext[static_cast<std::size_t>(king)];
   283	        const std::size_t rook_index = static_cast<std::size_t>(_pext_u64(occ, rook_slider.mask));
   284	        const std::size_t bishop_index = static_cast<std::size_t>(_pext_u64(occ, bishop_slider.mask));
   285	        const Bitboard rook_rays = rook_slider.table[rook_index];
   286	        const Bitboard bishop_rays = bishop_slider.table[bishop_index];
   287	        const Bitboard rook_checks = rook_rays & safe & info.attacked_by[Them][Rook];
   288	        const Bitboard bishop_checks = bishop_rays & safe & info.attacked_by[Them][Bishop];
   289	        const Bitboard knight_checks = attacks::knight(king) & safe & info.attacked_by[Them][Knight];
   290	        const Bitboard queen_checks = (rook_rays | bishop_rays) & safe & info.attacked_by[Them][Queen];
   291	        units += king_safety.rook_check * popcount(rook_checks) + king_safety.bishop_check * popcount(bishop_checks) +
   292	                 king_safety.knight_check * popcount(knight_checks) + king_safety.queen_check * popcount(queen_checks);
   293	        if (board.pieces(Them, Queen) == 0) units /= 2;
   294	        units -= shelter / king_safety.shelter_divisor;
   295	        units = std::clamp(units, 0, 99);
   296	        acc.template add<Us>(KingDanger + units, -1);
   297	    }
   298	
   299	    // Space: safe squares on the central files in our half, counted once more
   300	    // when they lie behind one of our pawns, scaled by the number of pieces.
   301	    const int piece_count = popcount(board.occupied(Us) & ~own_pawns) - 1;
   302	    if (piece_count >= 2) {
   303	        constexpr Bitboard SpaceMask = Us == White ? (FileCBB | FileDBB | FileEBB | FileFBB) & (Rank2BB | Rank3BB | Rank4BB)
   304	                                                   : (FileCBB | FileDBB | FileEBB | FileFBB) & (Rank7BB | Rank6BB | Rank5BB);
   305	        const Bitboard safe = SpaceMask & ~own_pawns & ~info.attacked_by[Them][Pawn];
   306	        Bitboard behind = own_pawns;
   307	        behind |= Us == White ? behind >> 8 : behind << 8;
   308	        behind |= Us == White ? behind >> 16 : behind << 16;
   309	        const int count = popcount(safe) + popcount(behind & safe & ~info.attacked_by[Them][0]);
   310	        acc.template add<Us>(Space, count * piece_count / 4);
   311	    }
   312	
   313	    // King distance to the nearest own pawn (endgame term).
   314	    if (own_pawns != 0) {
   315	        acc.template add<Us>(KingPawnDistance, nearest_square_distance(king, own_pawns));
   316	    }
   317	}

   319	template <bool Trace, Color Us>
   320	void evaluate_threats(const Board& board, Accumulator<Trace>& acc, const EvalInfo& info) {
   321	    constexpr Color Them = opposite(Us);
   322	    const Bitboard enemy_pieces = board.occupied(Them) & ~board.pieces(Them, King);
   323	    const Bitboard enemy_non_pawns = enemy_pieces & ~board.pieces(Them, Pawn);
   324	
   325	    // Enemy pieces that are poorly defended relative to our attacks.
   326	    const Bitboard strongly_protected = info.attacked_by[Them][Pawn] | (info.attacked_twice[Them] & ~info.attacked_twice[Us]);
   327	    const Bitboard defended = enemy_non_pawns & strongly_protected;
   328	    const Bitboard weak = enemy_pieces & ~strongly_protected & info.attacked_by[Us][0];
   329	
   330	    if ((defended | weak) != 0) {
   331	        auto add_targets = [&](int base, Bitboard targets) {
   332	            if (targets == 0) return;
   333	            for (int piece = Pawn; piece <= Queen; ++piece) {
   334	                const int count = popcount(targets & board.pieces(Them, static_cast<PieceType>(piece)));
   335	                if (count != 0) acc.template add<Us>(base + piece, count);
   336	            }
   337	        };
   338	        add_targets(ThreatByMinor, (defended | weak) & (info.attacked_by[Us][Knight] | info.attacked_by[Us][Bishop]));
   339	        add_targets(ThreatByRook, weak & info.attacked_by[Us][Rook]);
   340	        if ((weak & info.attacked_by[Us][King]) != 0) acc.template add<Us>(ThreatByKing);
   341	        const Bitboard hanging = weak & (~info.attacked_by[Them][0] | (enemy_non_pawns & info.attacked_twice[Us]));
   342	        acc.template add<Us>(HangingPiece, popcount(hanging));
   343	    }
   344	
   345	    // Pieces attacked by safe pawns, and pieces our safe pawn pushes would attack.
   346	    const Bitboard safe_squares = ~info.attacked_by[Them][0] | info.attacked_by[Us][0];
   347	    const Bitboard safe_pawns = board.pieces(Us, Pawn) & safe_squares;
   348	    acc.template add<Us>(ThreatByPawn, popcount(pawn_attacks_bb(Us, safe_pawns) & enemy_non_pawns));
   349	
   350	    const Bitboard empty = ~board.occupied();
   351	    Bitboard pushes = pawn_push(Us, board.pieces(Us, Pawn)) & empty;
   352	    pushes |= pawn_push(Us, pushes & (Us == White ? Rank3BB : Rank6BB)) & empty;
   353	    pushes &= ~info.attacked_by[Them][Pawn] & safe_squares;
   354	    acc.template add<Us>(ThreatByPawnPush, popcount(pawn_attacks_bb(Us, pushes) & enemy_non_pawns));
   355	
   540	    }
   541	    nnue::Accumulator accumulator{};
   542	    nnue::refresh(board, accumulator);
   543	    parts.nnue = nnue::evaluate(board, accumulator);
   544	    if (!nnue::residual()) {
   545	        parts.total = parts.nnue;
   546	        return parts;
   547	    }
   548	    Evaluator fresh;
   549	    const RuntimePsqt psqt = refresh_runtime_psqt(board);
   550	    const int handcrafted = fresh.evaluate_runtime(board, psqt);
   551	    parts.total = std::clamp(handcrafted + parts.nnue, -MateInMaxPly + 1, MateInMaxPly - 1);
   552	    return parts;
   553	}

   555	Evaluator::RuntimePsqt Evaluator::refresh_runtime_psqt(const Board& board) {
   556	    RuntimePsqt state;
   557	    std::int64_t score = 0;
   558	    for (int type = Pawn; type <= King; ++type) {
   559	        Bitboard white = board.pieces(White, static_cast<PieceType>(type));
   560	        Bitboard black = board.pieces(Black, static_cast<PieceType>(type));
   561	        state.phase += PhaseValue[static_cast<std::size_t>(type)] * (popcount(white) + popcount(black));
   562	        while (white != 0) score += params::table[static_cast<std::size_t>(params::Psqt + (type - 1) * 64 + pop_lsb(white))];
   563	        while (black != 0) score -= params::table[static_cast<std::size_t>(params::Psqt + (type - 1) * 64 + flip_square(pop_lsb(black)))];
   564	    }
   565	    // Do not introduce an overflowing delta intermediate for unusual tool values.
   566	    state.valid = score >= std::numeric_limits<Score>::min() && score <= std::numeric_limits<Score>::max();
   567	    if (state.valid) state.score = static_cast<Score>(score);
   568	    return state;
   569	}

   571	void Evaluator::update_runtime_psqt(const RuntimePsqt& parent, RuntimePsqt& child, const nnue::DirtyPieces& dirty) {
   572	    child = parent;
   573	    if (!parent.valid) return;
   574	    std::int64_t score = parent.score;
   575	    for (int i = 0; i < dirty.count; ++i) {
   576	        const nnue::PieceChange& change = dirty.changes[static_cast<std::size_t>(i)];
   577	        const int type = piece_type(change.piece);
   578	        const int relative = color_of(change.piece) == White ? change.square : flip_square(change.square);
   579	        const std::int64_t term = params::table[static_cast<std::size_t>(params::Psqt + (type - 1) * 64 + relative)];
   580	        const std::int64_t signed_term = color_of(change.piece) == White ? term : -term;
   581	        score += change.add ? signed_term : -signed_term;
   582	        child.phase += change.add ? PhaseValue[static_cast<std::size_t>(type)] : -PhaseValue[static_cast<std::size_t>(type)];
   583	    }
   584	    child.valid = score >= std::numeric_limits<Score>::min() && score <= std::numeric_limits<Score>::max();
   585	    if (child.valid) child.score = static_cast<Score>(score);
   586	}

   589	    PawnEntry& entry = pawn_cache_[static_cast<std::size_t>(board.pawn_key()) & (PawnCacheSize - 1U)];
   590	    return evaluate_position<false>(board, &entry, nullptr, &psqt);
   591	}
