   852	int Worker::static_evaluate(const Board& board, SearchStack* ss) {
   853	    if (!nnue::enabled || !nnue::available()) return evaluator_.evaluate(board);
   854	    const bool residual = nnue::residual();
   855	    // Features used for this evaluation. A null move keeps the parent's
   856	    // accumulator: the piece set is unchanged, and a float copy would be the
   857	    // same bits without being needed.
   858	    const nnue::Accumulator* features = &ss->accumulator;
   859	    const Evaluator::RuntimePsqt* psqt = &ss->runtime_psqt;
   860	    int network = 0;
   861	    bool have_network = false;
   862	    if (!ss->accumulator.valid) {
   863	        // Walk back to the nearest ply whose features were already built.
   864	        int back = 0;
   865	        while (back < ss->ply && !(ss - back - 1)->accumulator.valid) ++back;
   866	        if (back >= ss->ply && !(ss - back - 1)->accumulator.valid) {
   867	            nnue::refresh(board, ss->accumulator);
   868	            if (residual) ss->runtime_psqt = Evaluator::refresh_runtime_psqt(board);
   869	        } else {
   870	            const nnue::Accumulator* source_features = &(ss - back - 1)->accumulator;
   871	            const Evaluator::RuntimePsqt* source_psqt = &(ss - back - 1)->runtime_psqt;
   872	            for (int i = back; i >= 0; --i) {
   873	                SearchStack* frame = ss - i;
   874	                if (frame->entered_null) continue;
   875	                ensure_piece_diff(frame);
   876	                if (residual) Evaluator::update_runtime_psqt(*source_psqt, frame->runtime_psqt, frame->dirty);
   877	                if (i == 0) {
   878	                    // The position being scored. Fold the output layer into the
   879	                    // feature update so those units are not reloaded.
   880	                    network = nnue::apply_and_evaluate(*source_features, frame->accumulator, frame->dirty, board);
   881	                    have_network = true;
   882	                } else {
   883	                    nnue::update(*source_features, frame->accumulator, frame->dirty);
   884	                }
   885	                source_features = &frame->accumulator;
   886	                source_psqt = &frame->runtime_psqt;
   887	            }
   888	            features = source_features;
   889	            psqt = source_psqt;
   890	        }
   891	    }
   892	    if (!have_network) network = nnue::evaluate(board, *features);
   893	    if (!residual) return network;
   894	    if (!psqt->valid) {
   895	        ss->runtime_psqt = Evaluator::refresh_runtime_psqt(board);
   896	        psqt = &ss->runtime_psqt;
   897	    }
   898	    // A residual network is a learned correction on top of the hand-crafted evaluation.
   899	    return std::clamp(evaluator_.evaluate_runtime(board, *psqt) + network, -MateInMaxPly + 1, MateInMaxPly - 1);

    55	## What changed, and where the idea came from
    56	
    57	Every change is local to this engine. Nothing was taken from another engine's source, constants, or tables. The profile above is the reason each spot was touched.
    58	
    59	Move picker. The first move handed out is the lowest-index move with the strictly highest score. That index is known as soon as the moves are scored, so the first pick no longer walks the list. Later picks still walk, four scores at a time, and a tie still keeps the earlier index. The tail is still `std::sort` with `score >`. A stable insertion sort was tried in its place. Bench 11 moved from 312866 nodes to 292627, so the insertion sort was removed and `std::sort` stayed.
    60	
    61	Move generation. Captures, quiets, and the full list are separate compilations of one generator, so the capture-or-quiet flag is fixed in each one. The order of moves is the order the old generator emitted. Slider attacks use the BMI2 indexer that was already in `bitboard.cpp` when the compiler target defines `__BMI2__`. The required build is `-march=x86-64-v3`, which has BMI2. That indexer returns the same attack sets as the multiply magics and does not search for magic numbers at startup. A CPU without BMI2 still gets the multiply indexer. Empty-board rook and bishop rays are constant per square, so they are stored once at startup and read back. They are the same bitboards `rook(square, 0)` and `bishop(square, 0)` already produced, and they are what king-line refresh and the exchange pin test were computing.
    62	
    63	Make and unmake. After a move, the checkers are the mover's attacks on the opponent king. The other side's pieces cannot be checkers, so they are not gathered. The bitboard is the one `attackers_to` was masked down to.

    65	Legality and discovered check. The usual question is a pin test or a check-square test on lines that were already built for the position. Those two tests are inlined at the call. Check, en passant, king moves, castling, and promotions still use the full test.
