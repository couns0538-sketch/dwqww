+std::uint64_t mix(std::uint64_t& state) {
+    state += 0xA5A5A5A5A5A5A5A5ULL;
+    state = (state << 7) | (state >> 57);
+    state ^= state >> 13;
+    state *= 0x6C62272E07BB0141ULL;
+    return state;
+}

+bool move_is_legal(const forge::Board& board, forge::Move move) {
+    if (move.is_none()) return board.legal_moves().empty();

+int search_walk(forge::Searcher& searcher, std::uint64_t seed, int games, int max_plies) {
+    forge::SearchLimits limits;
+    limits.depth = 10;
+    std::uint64_t state = seed;
+    int illegal = 0;
+    int in_check = 0;
+    int raised = 0;
+    for (int n = 0; n < games; ++n) {
+        forge::Board board;
+        if (!board.set_fen(forge::Board::StandardStartFen)) {
+            expect(false, "start position");
+            return games;
+        }
+        const int plies = static_cast<int>(mix(state) % static_cast<std::uint64_t>(max_plies));
+        for (int ply = 0; ply < plies; ++ply) {
+            const forge::MoveList moves = board.legal_moves();
+            if (moves.empty()) break;
+            forge::Undo undo;
+            const forge::Move chosen = moves[mix(state) % moves.size()];
+            if (!board.make_move(chosen, undo)) {
+                expect(false, "make_move rejected a legal move");
+                return games;
+            }
+        }
+        if ((n % 5) == 4) {

+            std::string error;

+                expect(false, "reclock " + error);
+                return games;
+            }

+        }
+        if (board.in_check()) ++in_check;
+        searcher.clear();
+        const forge::SearchInfo info = searcher.search(board, limits, false);
+        if (!move_is_legal(board, info.best_move)) {
+            ++illegal;

+                      << forge::Board::move_to_uci(info.best_move) << std::endl;

+bool dead_material_score(const Board& board, int& score) {
+    // Heavy pieces and pawns are ordinary positions; the network still scores them.
+    if ((board.pieces(Pawn) | board.pieces(Rook) | board.pieces(Queen)) != 0) return false;

+    const int black_knights = popcount(board.pieces(Black, Knight));

+    const int black_minors = black_knights + black_bishops;

+    }

+    }

+    const bool black_pair = black_knights == 1 && black_bishops == 1 && white_minors == 0;
+    if (!white_pair && !black_pair) return false;

+    // king there, and bring the stronger king closer, with the corner weighted

   200	forge::Board played_position(int index) {
   201	    static constexpr std::string_view roots[] = {
   202	        forge::Board::StandardStartFen,
   203	        "8/8/8/4k3/8/4K3/8/4R3 w - - 0 1",
   204	        "4k3/8/8/8/8/8/3Q4/4K3 w - - 0 1",
   205	        "8/2k5/8/2K5/8/8/2R5/8 w - - 0 1",
   206	    };
   207	    forge::Board board = board_from(roots[static_cast<std::size_t>(index) % 4]);
   208	    std::uint32_t mix = 0xC0FFEEU ^ static_cast<std::uint32_t>(index) * 0x45D9F3BU;
   209	    const int steps = 4 + (index % 9);
   210	    for (int step = 0; step < steps; ++step) {
   211	        const forge::MoveList legal = board.legal_moves();
   212	        if (legal.empty()) break;
   213	        mix ^= mix << 13;
   214	        mix ^= mix >> 17;
   215	        mix ^= mix << 5;
   216	        const forge::Move move = legal[mix % legal.size()];
   217	        forge::Undo undo;
   218	        if (!board.make_move(move, undo)) break;
   219	    }
   220	    if (board.legal_moves().empty()) board = board_from(forge::Board::StandardStartFen);
   221	    return board;
   222	}

# EvalRule50Damp

The fifty-move rule draws a game after 100 half-moves with no pawn move and no capture. This engine already treats that clock as a draw, and treats checkmate on the hundredth half-move as a win. A static score is a claim about a conversion that still has to happen. As the clock rises, fewer quiet moves remain, so the static score a node reads should shrink toward zero. Clock 0 keeps the whole score. Each half-move already played drops one percent, so a clock near 100 keeps only a sliver (clock 90 keeps 10, clock 99 keeps 1, clock 100 keeps 0). Mate scores are not scaled: a forced mate is not a positional edge the clock can expire.

That percent is the shape asked for here. It is not a scale table taken from another engine. `Board::compute_key` hashes pieces, castling, en passant, and side to move, and does not hash the half-move clock, so one hash entry is shared by the same placement at many clocks. The raw evaluation stored in that entry stays unscaled. Each node scales it for its own clock when it reads it. A search score was built from evals already scaled for the clock that stored it, so with the option on that score is used only when the stored half-move clock matches. With the option off the extra check accepts every score the old gate accepted.

## Option

`EvalRule50Damp`, check, default false. It is printed in the `uci` answer. `setoption name EvalRule50Damp value true` turns it on and clears the hash. `false` turns it off and clears the hash. Any other value is rejected and the flag is left as it was.

The scale is `rule50_damped_static` in `forge-combo29/include/forge/search.hpp`. `Worker::qsearch` and `Worker::search` call it where the corrected static evaluation is copied into the node, and on the ply-limit return that would otherwise hand back a raw static evaluation. `ValueNone` and mate scores are returned unchanged. The hash store still writes `ss->raw_eval`, the unscaled evaluation, into the eval slot.

# DeadMaterial

## Where the idea comes from

This is the ordinary endgame fact, written for this engine and not taken from another engine's code, constants, or tables.

King against king, king and one knight or one bishop against king, and king and two knights against king have no forced mate, so the score is 0. King, bishop and knight against king is won by driving the lone king into a corner of the bishop's colour. In this program `DarkSquares` has bit 0 set, so a1 is dark. A dark-squared bishop uses a1 and h8. A light-squared bishop uses h1 and a8.

The attacker score, in centipawns, is

`48 + 28 * (7 - corner distance) + 12 * (7 - distance between the kings)`.

`corner distance` is the Chebyshev distance from the lone king to the nearer of those two corners. The corner term is larger than the king term, so a step toward that corner still raises the score when the kings move farther apart. The number is from the attacker's point of view, then negated when the lone king is the side to move.

## What the option does

`DeadMaterial` is a UCI check option, default false, and it is listed in the `uci` answer.

With it off, evaluation and the draw shortcut are the previous code. Bench and perft stay on the lock.

With it on, `Worker::static_evaluate` returns the score above before it refreshes an accumulator or calls the network. `corrected_eval` does not add correction history on top of that score. The `eval` command uses the same rule and leaves the hand-crafted and network fields at 0.

Bare kings and king plus one minor are no longer cut off as insufficient material, so the 0 comes from this evaluation and those king moves are searched. Repetition, the fifty-move rule, stalemate, and other insufficient material (such as bishops all on one colour) still draw the old way. Two knights against a king were not an insufficient-material cutoff; they now score 0 when evaluated. Bishop and knight against a king was not a cutoff either; it now gets the corner score instead of the network.

Nothing here is a strength claim. No games were played.

# DrawJitter

Repetition leaves and fifty-move leaves can return a draw score a few centipawns off zero. The offset follows the node count. Mate scores are not adjusted.

`DrawJitter` is a UCI spin option, listed in the `uci` answer. Default 0, minimum 0, maximum 8. Zero keeps today's scores: those leaves return exactly 0.

## Where the idea comes from

A search that scores every draw as 0 cannot tell two drawing lines apart. Engines have avoided that with contempt, a constant added to draw scores. Stockfish exposed the constant as Contempt and later removed it. This is not that option. No contempt value, margin, or table was copied.

Only two kinds of leaf move. A repetition, and a position whose half-move clock has reached 100 and is not checkmate, score

    (nodes % (2 * DrawJitter + 1)) - DrawJitter

That stays inside [-DrawJitter, +DrawJitter] and changes as nodes are counted. The step was written for this search. Plain stalemate and insufficient material stay at 0. A stalemate whose clock is already 100 is a fifty-move leaf, so it uses the same step. Checkmate, including checkmate on the hundredth half-move, still returns a mate score.

If the root itself is already one of those draws, no nodes have been counted, so the score is -DrawJitter.
