   77 struct TTEntry {
   78     std::uint32_t key32 = 0;
   79     Move move{};
   80     std::int16_t score = 0;
   81     std::int16_t eval = 0;
   82     std::uint8_t depth = 0;
   83     // Bits 0..1 bound, bit 2 was-PV, bits 3..7 generation.
   84     std::uint8_t gen_bound = 0;
   85     std::uint8_t rule50_clock = 0; // Capped at100; belongs to admitted score payload.
   86     std::uint8_t padding = 0;
   87 
   88     [[nodiscard]] Bound bound() const { return static_cast<Bound>(gen_bound & 3U); }
   89     [[nodiscard]] bool pv() const { return (gen_bound & 4U) != 0; }
   90     [[nodiscard]] std::uint8_t generation() const { return static_cast<std::uint8_t>(gen_bound >> 3U); }
   91     [[nodiscard]] bool occupied() const { return bound() != BoundNone; }
   92     // Existing below90 policy, plus exact near50 context; not a repetition key.
   93     [[nodiscard]] bool score_usable(std::uint16_t current_clock) const {
   94         const int capped = current_clock < 100 ? current_clock : 100;
   95         return (rule50_clock < 90 && capped < 90) || rule50_clock == capped;
   96     }
   97 };

  120 // Search reads and writes are coherent under striped locks. clear/resize/
  121 // new_search/destruction require all accessors quiescent (including diagnostic
  122 // readers and prefetch callers) and invalidate old probes. Handles must not
  123 // outlive the table; lifecycle operations are not safe concurrently with search.
  124 class TranspositionTable {
  125 public:
  126     explicit TranspositionTable(std::size_t mb = 32) { resize(mb); }
  127     void resize(std::size_t mb);
  128     void clear();
  129     void new_search() { generation_ = static_cast<std::uint8_t>((generation_ + 1U) & 31U); ++epoch_; }
  130     [[nodiscard]] TTProbe probe(std::uint64_t key) const;
  131     void store(const TTProbe& probe, std::uint64_t key, int score, int eval, Move move, int depth, Bound bound, bool pv, std::uint16_t halfmove_clock);
  132     [[nodiscard]] int hashfull() const;
  133     [[nodiscard]] std::uint8_t generation() const { return generation_; }
  134     [[nodiscard]] std::size_t cluster_count() const { return cluster_count_; }
  135     void prefetch(std::uint64_t key) const {
  136 #if defined(__GNUC__) || defined(__clang__)
  137         __builtin_prefetch(&cluster_for(key));
  138 #else
  139         (void)key;
  140 #endif
  141     }
  142 
  143 private:
  144     friend class Searcher;
  145     static constexpr std::size_t StripeCount = 1024;
  146     struct alignas(64) Stripe { std::mutex mutex; };
  147     // Opt-in controller ownership: request exclusive access, drain each stripe,
  148     // then publish TLS. Lifecycle mutations still require all users quiescent.
  149     class ExclusiveLease {
  150     public:
  151         explicit ExclusiveLease(TranspositionTable& table);
  152         ~ExclusiveLease();
  153         ExclusiveLease(const ExclusiveLease&) = delete;
  154         ExclusiveLease& operator=(const ExclusiveLease&) = delete;
  155         ExclusiveLease(ExclusiveLease&&) = delete;
  156         ExclusiveLease& operator=(ExclusiveLease&&) = delete;
  157     private:
  158         friend class TranspositionTable;
  159         TranspositionTable& table_;
  160         const ExclusiveLease* previous_;
  161         std::unique_lock<std::mutex> gate_;
  162     };
  163     static thread_local const ExclusiveLease* active_lease_;
  164     [[nodiscard]] bool has_exclusive_access() const;
  165     [[nodiscard]] std::unique_lock<std::mutex> lock_stripe(std::size_t index) const;
  166 #if FORGE_SPD_TT_NOTLS
  167     // One body per access with a compile-time "take the stripe lock" parameter.
  168     // probe / store keep lock_stripe; probe_owned / store_owned run the same body
  169     // without it and are only for the Worker on the lease's thread (tt_owned_).
  170     friend class Worker;
  171     template <bool TakeLock>
  172     [[nodiscard]] TTProbe probe_body(std::uint64_t key) const;
  173     template <bool TakeLock>
  174     void store_body(const TTProbe& probe, std::uint64_t key, int score, int eval, Move move, int depth, Bound bound,
  175                     bool pv, std::uint16_t halfmove_clock);
  176     [[nodiscard]] TTProbe probe_owned(std::uint64_t key) const;
  177     void store_owned(const TTProbe& probe, std::uint64_t key, int score, int eval, Move move, int depth, Bound bound,
  178                      bool pv, std::uint16_t halfmove_clock);
  179 #endif
  180     mutable std::array<Stripe, StripeCount> stripes_{};
  181     mutable std::mutex exclusive_gate_;
  182     std::atomic<bool> exclusive_requested_{false};
  183     std::uint64_t epoch_ = 0;
  184     std::unique_ptr<TTCluster[]> clusters_;
  185     std::size_t cluster_count_ = 0;
  186     std::uint8_t generation_ = 0;
  187 
  188     [[nodiscard]] static constexpr std::uint32_t verification_key(std::uint64_t key) {
  189 #if defined(__SIZEOF_INT128__)
  190         // Multiply-high indexing consumes high key bits. Keep the independent
  191         // low bits as the tag, rather than reusing the bucket's index bits.
  192         return static_cast<std::uint32_t>(key);
  193 #else
  194         // Modulo indexing consumes low bits for power-of-two table sizes.
  195         return static_cast<std::uint32_t>(key >> 32U);
  196 #endif
  197     }
  198 
  199     [[nodiscard]] TTCluster& cluster_for(std::uint64_t key) const {
  200 #if defined(__SIZEOF_INT128__)
  201         __extension__ using u128 = unsigned __int128;
  202         return clusters_[static_cast<std::size_t>((static_cast<u128>(key) * cluster_count_) >> 64U)];
  203 #else
  204         return clusters_[static_cast<std::size_t>(key % cluster_count_)];
  205 #endif
  206     }
  207 };

  423 // Public search facade: a shared transposition table plus one or more workers.
  424 class Searcher {
  425 public:
  426     // Exclusive is for a controller with no concurrent diagnostics: at one
  427     // worker it drains the stripes and holds an exclusive gate. Foreign TT calls
  428     // block for the whole search and must not hold locks needed by search.
  429     // Multiple workers and the default library entry retain shared locking.
  430     enum class TTAccess { Shared, Exclusive };
  431     explicit Searcher(std::size_t hash_mb = 32, int threads = 1);
  432     ~Searcher();
  433     Searcher(const Searcher&) = delete;
  434     Searcher& operator=(const Searcher&) = delete;
  435 
  436     SearchInfo search(Board board, const SearchLimits& limits, bool emit_info = false,
  437                       TTAccess table_access = TTAccess::Shared);
  438     void stop() { shared_.stop.store(true, std::memory_order_relaxed); }
  439     void reset_stop() {
  440         shared_.stop.store(false, std::memory_order_relaxed);
  441         shared_.ponder_hit_ns.store(0, std::memory_order_relaxed);
  442     }
  443     void ponderhit() {
  444         const auto now_ns = std::chrono::duration_cast<std::chrono::nanoseconds>(
  445             std::chrono::steady_clock::now().time_since_epoch()).count();
  446         std::int64_t expected = 0;
  447         (void)shared_.ponder_hit_ns.compare_exchange_strong(expected, now_ns);
  448     }
  449     void clear();
  450     void clear_hash();
  451     void resize_hash(std::size_t hash_mb);
  452     void set_threads(int threads);
  453     [[nodiscard]] int threads() const { return static_cast<int>(workers_.size()); }
  454     [[nodiscard]] int hashfull_permille() const { return shared_.tt.hashfull(); }
  455     [[nodiscard]] static int evaluate(const Board& board) { return Evaluator::evaluate_uncached(board); }
  456     [[nodiscard]] Worker& main_worker() { return *workers_.front(); }
  457     [[nodiscard]] TranspositionTable& transposition_table() { return shared_.tt; }
  458 
  459 private:
  460     SharedSearchState shared_;
  461     std::vector<std::unique_ptr<Worker>> workers_;
  462 };
