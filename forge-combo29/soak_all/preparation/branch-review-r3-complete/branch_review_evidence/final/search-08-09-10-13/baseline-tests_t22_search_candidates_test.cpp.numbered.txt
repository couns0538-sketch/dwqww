   55 std::uint64_t run_bench(int depth) {
   56     std::uint64_t total_nodes = 0;
   57     forge::Searcher searcher(16, 1);
   58     for (const std::string_view fen : BenchFens) {
   59         forge::Board board;
   60         if (!board.set_fen(fen)) return 0;
   61         forge::SearchLimits limits;
   62         limits.depth = depth;
   63         searcher.clear();
   64         total_nodes += searcher.search(board, limits, false, forge::Searcher::TTAccess::Exclusive).nodes;
   65     }
   66     return total_nodes;
   67 }

   69 std::vector<forge::Board> build_200_positions() {
   70     std::vector<forge::Board> positions;
   71     positions.reserve(200);
   72     std::uint64_t rng = 0x9E3779B97F4A7C15ULL;
   73     auto next_u64 = [&]() {
   74         rng ^= rng >> 12U;
   75         rng ^= rng << 25U;
   76         rng ^= rng >> 27U;
   77         return rng * 0x2545F4914F6CDD1DULL;
   78     };
   79 
   80     for (std::size_t s = 0; s < SeedFens.size() && positions.size() < 200; ++s) {
   81         forge::Board board;
   82         if (!board.set_fen(SeedFens[s])) continue;
   83         for (int step = 0; step < 8 && positions.size() < 200; ++step) {
   84             const forge::MoveList legal = board.legal_moves();
   85             if (legal.empty()) {
   86                 (void)board.set_fen(SeedFens[(s + 1) % SeedFens.size()]);
   87             }
   88             if (!board.legal_moves().empty()) {
   89                 positions.push_back(board);
   90             }
   91             const forge::MoveList step_moves = board.legal_moves();
   92             if (step_moves.empty()) break;
   93             const forge::Move chosen = step_moves[static_cast<std::size_t>(next_u64() % step_moves.size())];
   94             forge::Undo undo;
   95             (void)board.make_move(chosen, undo);
   96         }
   97     }
   98     return positions;
   99 }

  101 void reset_candidates() {

  107 struct Candidate {
  108     const char* name;
  109     bool* flag;
  110 };
  111 
  112 int test_candidate(const Candidate& cand, const std::vector<forge::Board>& positions) {
  113     int failures = 0;
  114     reset_candidates();
  115     *cand.flag = true;
  116 
  117     const std::uint64_t bench13_on = run_bench(13);
  118     std::cout << "candidate " << cand.name << " bench13_on " << bench13_on
  119               << " bench13_off " << ExpectedBench13 << '\n';
  120     if (bench13_on == 0 || bench13_on == ExpectedBench13) {
  121         std::cerr << "FAIL " << cand.name << ": bench 13 did not change when enabled\n";
  122         ++failures;
  123     }
  124 
  125     forge::Searcher searcher(16, 1);
  126     std::uint64_t total_nodes = 0;
  127     for (std::size_t i = 0; i < positions.size(); ++i) {
  128         // Clear every 25 searches so cross-move history and correction tables
  129         // accumulate within mini-games as well as starting clean across games.
  130         if (i % 25 == 0) searcher.clear();
  131         const forge::Board& pos = positions[i];
  132         forge::SearchLimits limits;
  133         limits.depth = 10;
  134         const forge::SearchInfo info =
  135             searcher.search(pos, limits, false, forge::Searcher::TTAccess::Exclusive);
  136         total_nodes += info.nodes;
  137 
  138         const forge::MoveList legal = pos.legal_moves();
  139         bool found = false;
  140         for (const forge::Move m : legal) {
  141             if (m == info.best_move) {
  142                 found = true;
  143                 break;
  144             }
  145         }
  146         if (info.best_move.is_none() || !pos.is_legal(info.best_move) || !found) {
  147             std::cerr << "FAIL " << cand.name << ": illegal best_move at position " << i
  148                       << " (" << forge::Board::move_to_uci(info.best_move) << ")\n";
  149             ++failures;
  150             break;
  151         }
  152     }
  153     std::cout << "candidate " << cand.name << " searches " << positions.size()
  154               << " depth 10 legal_ok " << (failures == 0 ? 1 : 0)
  155               << " total_nodes " << total_nodes << '\n';
  156 
  157     reset_candidates();
  158     return failures;
  159 }

  163 int main() {
  164     int failures = 0;
  165     reset_candidates();
  166 
  167     const std::uint64_t b11 = run_bench(11);
  168     const std::uint64_t b13 = run_bench(13);
  169     forge::Board start;
  170     (void)start.set_fen(forge::Board::StandardStartFen);
  171     const std::uint64_t p5 = start.perft(5);
  172 
  173     std::cout << "defaults bench11 " << b11 << " bench13 " << b13 << " perft5 " << p5 << '\n';
  174     if (b11 != ExpectedBench11 || b13 != ExpectedBench13 || p5 != ExpectedPerft5) {
  175         std::cerr << "FAIL baseline parity mismatch with options off\n";
  176         ++failures;
  177     }
  178 
  179     const std::vector<forge::Board> positions = build_200_positions();
  180     if (positions.size() != 200) {
  181         std::cerr << "FAIL expected 200 test positions, built " << positions.size() << '\n';
  182         return 1;
  183     }
  184 
  185     const std::array<Candidate, 3> candidates{{
  186         {"ContHist4PlyRead", &forge::tune::ContHist4PlyRead},
  187         {"NonPawnCorrection", &forge::tune::NonPawnCorrection},
  188         {"LmrDeeperShallower", &forge::tune::LmrDeeperShallower},
  189     }};
  190 
  191     for (const Candidate& cand : candidates) {
  192         failures += test_candidate(cand, positions);
  193     }
  194 
  195     const std::uint64_t b13_after = run_bench(13);
  196     std::cout << "defaults_after bench13 " << b13_after << '\n';
  197     if (b13_after != ExpectedBench13) {
  198         std::cerr << "FAIL bench 13 changed after resetting options to false\n";
  199         ++failures;
  200     }
  201 
  202     if (failures != 0) {
  203         std::cerr << "FAIL total failures: " << failures << '\n';
  204         return 1;
  205     }
  206     std::cout << "PASS t22 search_candidates_test\n";
  207     return 0;
  208 }
