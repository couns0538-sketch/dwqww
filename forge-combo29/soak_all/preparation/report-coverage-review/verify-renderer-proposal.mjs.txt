import fs from 'node:fs';
import assert from 'node:assert/strict';
import {renderReport as original} from './render-report.original.mjs';
import {renderReport as proposed} from './render-report.proposed.mjs';
const dir = new URL('.', import.meta.url);
const state = JSON.parse(fs.readFileSync(new URL('state-snapshot.json.txt', dir), 'utf8'));
const before = JSON.stringify(state);
const testDir = '/workspace/work/all-run/test-results';
const tests = fs.readdirSync(testDir).filter(x => x.endsWith('.json')).flatMap(x => JSON.parse(fs.readFileSync(testDir + '/' + x, 'utf8')));
const audit = JSON.parse(fs.readFileSync(state.latestAudit, 'utf8'));
fs.writeFileSync(new URL('test-results-snapshot.json.txt', dir), JSON.stringify(tests, null, 2) + '\n');
fs.writeFileSync(new URL('audit-snapshot.json.txt', dir), JSON.stringify(audit, null, 2) + '\n');
const options = {generatedAt:'2026-11-14T15:10:45Z',testResults:tests,auditLink:'preparation/latest-audit.txt'};
const oldText = original(state, audit, options);
const newText = proposed(state, audit, options);
fs.writeFileSync(new URL('SOAK_ALL_REPORT.original-render.md', dir), oldText);
fs.writeFileSync(new URL('SOAK_ALL_REPORT.proposed.md', dir), newText);
assert.equal(JSON.stringify(state), before);
assert(newText.includes('Initial Part 1 KRK coverage: release T1 100, release T2 400, ASAN T1 100, ASAN T2 40.'));
assert(newText.includes('Later completed ASAN split retries (2KRKsplit): 2 invocations, T1 200 reference checks and T2 400 trials; T1 misses 0, T2 misses 0, reported sanitizer diagnostics 0.'));
assert(newText.includes('T1 300 reference checks, T2 440 trials.'));
assert(newText.includes('Timed-out single 40×10 invocations remain incomplete and contribute no completed coverage.'));
const between = (s,a,b) => s.slice(s.indexOf(a), s.indexOf(b));
assert.equal(between(newText,'## Mate observations','## Existing tests and limitations'), between(oldText,'## Mate observations','## Existing tests and limitations'));
assert.equal(newText.slice(newText.indexOf('## Incomplete attempts and retries')), oldText.slice(oldText.indexOf('## Incomplete attempts and retries')));
const noSplit = structuredClone(state);
delete noSplit.totals['2KRKsplit'];
assert(!proposed(noSplit,audit,options).includes('Combined completed ASAN KRK coverage'));
const missing = structuredClone(state);
delete missing.totals['2KRKsplit'].counts.main_runs;
assert(!proposed(missing,audit,options).includes('Combined completed ASAN KRK coverage'));
assert(proposed(missing,audit,options).includes('T2 — trials'));
console.log(JSON.stringify({stateUnmodified:true,testRecords:tests.length,initialAsanT1:100,initialAsanT2:40,splitAsanT1:200,splitAsanT2:400,combinedAsanT1:300,combinedAsanT2:440,mateCostMatchRuntimeSectionsUnchanged:true,incompleteAndEvidenceAppendicesUnchanged:true,missingCountsNotZero:true,noEngineWorkload:true}));
console.log(newText.slice(newText.indexOf('## Existing tests and limitations'),newText.indexOf('<details>',newText.indexOf('## Existing tests and limitations'))));
