function once(from, to, label) {
    if (source.split(from).length !== 2) throw new Error('Non-unique anchor: ' + label);
    source = source.replace(from, to);

}
function section(start, end, transform, label) {
    const a = source.indexOf(start), b = source.indexOf(end, a + start.length);
    if (a < 0 || b < 0) throw new Error('Missing section ' + label);
    const from = source.slice(a, b);
    let value = from;
    const edits = [];
    const replace = (old, next, name, count = 1) => {
        if (value.split(old).length !== count + 1) throw new Error('Bad section anchor ' + label + ':' + name);
        value = value.split(old).join(next);
        edits.push(name);
    };
    transform(replace);
    once(from, value, label + ' [' + edits.join(', ') + ']');
}
