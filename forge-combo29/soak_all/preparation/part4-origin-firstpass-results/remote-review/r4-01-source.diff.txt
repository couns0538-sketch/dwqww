+// One move of clock time (wtime/btime, increment, movestogo, Move Overhead).
+// movetime is a separate limit and is not planned here.
+struct ClockPlan {
+    std::int64_t soft_ms = 0;
+    std::int64_t hard_ms = 0;
+    // Nothing remains after Move Overhead. A hard limit of 0 would mean the
+    // search is not on a clock, so the caller must stop instead of searching.
+    bool reserve_empty = false;
+};
+
+// usable = max(0, time - overhead), with negatives treated as zero.
+// The plan never asks for more than usable. Soft never exceeds hard.
+// When usable is positive both limits stay in 1..usable, using the same
+// sudden-death and moves-to-go split as before. That split is left alone
+// when it already obeyed the bounds. It is pulled back when no time remains
+// after overhead, or when increment would place soft above hard.
+// soft_divisor must be positive; the UCI spin starts at 5.
+[[nodiscard]] inline ClockPlan plan_clock_budget(std::int64_t time_ms, std::int64_t increment_ms,
+                                                 std::int64_t moves_to_go, std::int64_t overhead_ms,
+                                                 int soft_divisor, int increment_percent, int hard_percent,
+                                                 int hard_soft_multiple) {
+    const std::int64_t overhead = overhead_ms > 0 ? overhead_ms : 0;
+    const std::int64_t time = time_ms > 0 ? time_ms : 0;
+    const std::int64_t increment = increment_ms > 0 ? increment_ms : 0;
+    if (time <= overhead) return ClockPlan{0, 0, true};
+
+    const std::int64_t available = time - overhead;
+    std::int64_t soft = 0;
+    std::int64_t hard = 0;
+    if (moves_to_go > 0) {
+        const std::int64_t mtg = moves_to_go < 50 ? moves_to_go : 50;
+        soft = available / (mtg + 1) + increment * 3 / 4;
+        hard = std::min(available * 4 / 5, soft * 4);
+    } else {
+        const int divisor = soft_divisor > 0 ? soft_divisor : 1;
+        soft = available / divisor + increment * increment_percent / 100;
+        hard = std::min(available * hard_percent / 100, soft * hard_soft_multiple);
+    }
+    soft = std::max<std::int64_t>(1, std::min(soft, available));
+    hard = std::max<std::int64_t>(1, std::min(hard, available));
+    if (soft > hard) soft = hard;
+    return ClockPlan{soft, hard, false};
+}

+#pragma once
+
+#include <cstdint>
+
+namespace forge {

+// Clock decision for one iterative-deepening step.
+//
+// prev_ms and last_ms are the durations, in milliseconds, of the two iterations
+// that just finished. last_ms is the later one. remaining_ms is how much of
+// this move's budget is left.
+//
+// Steady growth (last no more than twice prev) is extended one step:
+// last * last / prev. A larger jump is one deep iteration, not a new rate.
+// Compounding it predicted several times the real next iteration on this
+// engine's own depth curves (one measured step was about 84 ms after 16 ms,
+// and the following iteration was about 38 ms, not about 440). After that
+// kind of jump the estimate is the geometric mean of the two durations,
+// sqrt(last * prev). The iteration cannot finish when the estimate is
+// strictly longer than remaining_ms. The caller then keeps the move from the
+// iteration that already finished.
+//
+// A non-positive duration is not a measurement, so this returns false and the
+// search does what it did without the estimate. A budget that is already used
+// up cannot hold another iteration.
+[[nodiscard]] constexpr bool iteration_will_not_finish(std::int64_t prev_ms, std::int64_t last_ms,
+                                                        std::int64_t remaining_ms) noexcept {
+    if (remaining_ms <= 0) return true;
+    if (prev_ms <= 0 || last_ms <= 0) return false;
+    const auto prev = static_cast<unsigned __int128>(static_cast<std::uint64_t>(prev_ms));
+    const auto last = static_cast<unsigned __int128>(static_cast<std::uint64_t>(last_ms));
+    const auto left = static_cast<unsigned __int128>(static_cast<std::uint64_t>(remaining_ms));
+    // Twice prev still fits in 64 bits: prev_ms is a non-negative int64.
+    const bool spike = static_cast<std::uint64_t>(last_ms) > static_cast<std::uint64_t>(prev_ms) * 2ULL;
+    if (spike) return last * prev > left * left;
+    return last * last > left * prev;
+}

+// One finished clock iteration, with timings the caller already chose.
+// best_nodes is the current best root move. root_nodes is the sum of the
+// per-root-move counters (not the whole tree).
+struct NodeShareTiming {
+    std::int64_t soft_ms = 0;
+    std::int64_t hard_ms = 0;
+    std::int64_t elapsed_ms = 0;
+    std::uint64_t best_nodes = 0;
+    std::uint64_t root_nodes = 0;
+};
+
+// soft_ms is the soft target after the node-share scale.
+// reached means elapsed_ms has met that target. Search still multiplies the
+// target by its older stability factor before it stops.
+struct NodeShareTimingDecision {
+    std::int64_t soft_ms = 0;
+    bool reached = false;
+};
+
+// Lengthen the soft target when the best root move holds under half of the
+// root nodes, shorten it when the move holds more than half, and leave a tie
+// alone. The scale is linear in thousandths of the node share:
+// 0% -> 3/2 of soft, 50% -> soft, 100% -> 1/2 of soft.
+// A positive result is at least 1 ms. When hard_ms is positive the result is
+// also at most hard_ms, so a lengthened target cannot pass the hard limit.
+// A non-positive soft_ms is unchanged and is not treated as reached.
+[[nodiscard]] constexpr NodeShareTimingDecision decide_node_share_timing(NodeShareTiming in) noexcept {
+    NodeShareTimingDecision out;
+    if (in.soft_ms <= 0) {
+        out.soft_ms = in.soft_ms;
+        out.reached = false;
+        return out;
+    }
+    std::int64_t numer = 1000;
+    if (in.root_nodes > 0) {
+        constexpr std::uint64_t kShareScale = 1000;
+        const bool dominant = in.best_nodes >= in.root_nodes || in.best_nodes > (UINT64_MAX / kShareScale);
+        const std::uint64_t share = dominant ? kShareScale : (in.best_nodes * kShareScale) / in.root_nodes;
+        numer = static_cast<std::int64_t>(kShareScale) + (500 - static_cast<std::int64_t>(share));
+    }
+    const std::int64_t whole = in.soft_ms / 1000;
+    const std::int64_t part = in.soft_ms % 1000;
+    std::int64_t adjusted = whole * numer + (part * numer) / 1000;
+    if (adjusted < 1) adjusted = 1;
+    if (in.hard_ms > 0 && adjusted > in.hard_ms) adjusted = in.hard_ms;
+    out.soft_ms = adjusted;
+    out.reached = in.elapsed_ms >= adjusted;
+    return out;
+}
+
+// Older clock factor piece: clamp(base_percent/100 - effort, 0.6, 1.4),
+// where effort is the best move's share of nodes. TmNodeShare applies that
+// share itself, so this piece stays 1 while the option is on and the share
+// is counted once. Off, the older clamp is unchanged.
+[[nodiscard]] constexpr double effort_share_factor(double effort, int base_percent, bool node_share_on) noexcept {
+    if (node_share_on) return 1.0;
+    const double raw = static_cast<double>(base_percent) / 100.0 - effort;
+    if (raw < 0.6) return 0.6;
+    if (raw > 1.4) return 1.4;
+    return raw;
+}

+bool dead_material_score(const Board& board, int& score) {
+    // Heavy pieces and pawns are ordinary positions; the network still scores them.
+    if ((board.pieces(Pawn) | board.pieces(Rook) | board.pieces(Queen)) != 0) return false;
+    const int white_knights = popcount(board.pieces(White, Knight));
+    const int black_knights = popcount(board.pieces(Black, Knight));
+    const int white_bishops = popcount(board.pieces(White, Bishop));
+    const int black_bishops = popcount(board.pieces(Black, Bishop));
+    const int white_minors = white_knights + white_bishops;
+    const int black_minors = black_knights + black_bishops;
+    if (white_minors == 0 && black_minors == 0) {
+        score = 0;
+        return true;
+    }
+    // King and one knight, or king and one bishop, against a bare king.
+    if ((white_minors == 1 && black_minors == 0) || (black_minors == 1 && white_minors == 0)) {
+        score = 0;
+        return true;
+    }
+    // Two knights cannot force mate against a lone king.
+    if ((white_knights == 2 && white_bishops == 0 && black_minors == 0) ||
+        (black_knights == 2 && black_bishops == 0 && white_minors == 0)) {
+        score = 0;
+        return true;
+    }
+    const bool white_pair = white_knights == 1 && white_bishops == 1 && black_minors == 0;
+    const bool black_pair = black_knights == 1 && black_bishops == 1 && white_minors == 0;
+    if (!white_pair && !black_pair) return false;
+    // Mate is possible only in a corner of the bishop's colour. Pull the lone
+    // king there, and bring the stronger king closer, with the corner weighted
+    // more so a step toward that corner still helps when it leaves the kings farther apart.
+    const Color attacker = white_pair ? White : Black;
+    const int lone = board.king_square(opposite(attacker));
+    const int helper = board.king_square(attacker);
+    const bool dark = (board.pieces(attacker, Bishop) & DarkSquares) != 0;
+    const int corner_a = dark ? 0 : 7;    // a1 or h1
+    const int corner_b = dark ? 63 : 56;  // h8 or a8
+    const int to_corner = std::min(square_distance(lone, corner_a), square_distance(lone, corner_b));
+    const int king_gap = square_distance(helper, lone);
+    constexpr int CornerWeight = 28;
+    constexpr int KingWeight = 12;
+    // Same taper the hand-written eval uses for this material (phase 2 of 24).
+    constexpr int minor_phase = PhaseValue[Knight] + PhaseValue[Bishop];
+    constexpr int minor_mg = PieceValueMg[Knight] + PieceValueMg[Bishop];
+    constexpr int minor_eg = PieceValueEg[Knight] + PieceValueEg[Bishop];
+    constexpr int material = (minor_mg * minor_phase + minor_eg * (MaxPhase - minor_phase)) / MaxPhase;
+    const int drive = (7 - to_corner) * CornerWeight + (7 - king_gap) * KingWeight;
+    const int attacker_cp = material + drive;
+    score = board.side_to_move() == attacker ? attacker_cp : -attacker_cp;
+    return true;
+}

+// destination is. The destination test removes the mover from occupancy, so a
+// slider standing on the move's line is not hidden by the piece that is about
+// to leave. "Cheaper" follows SeePieceValue, except a king is never cheaper

+// mover, the piece that would stand there.
+int quiet_threat_index(const Board& board, Move move) {
+    const int type = piece_type(board.piece_at(move.from));
+    if (type <= Pawn) return 0;
+    const Color us = board.side_to_move();
+    const Color them = opposite(us);
+    const Bitboard from_occ = board.occupied();
+    const Bitboard to_occ = from_occ ^ square_bb(move.from);
+    const auto cheaper = [&](int square, Bitboard occ) {
+        if ((attacks::pawn(us, square) & board.pieces(them, Pawn)) != 0) return true;
+        if (type <= Bishop) return false;
+        if ((attacks::knight(square) & board.pieces(them, Knight)) != 0) return true;
+        if ((attacks::bishop(square, occ) & board.pieces(them, Bishop)) != 0) return true;
+        if (type <= Rook) return false;
+        if ((attacks::rook(square, occ) & board.pieces(them, Rook)) != 0) return true;
+        if (type <= Queen) return false;
+        const Bitboard queens = board.pieces(them, Queen);
+        return (attacks::bishop(square, occ) & queens) != 0 || (attacks::rook(square, occ) & queens) != 0;
+    };
+    return (cheaper(move.from, from_occ) ? 2 : 0) | (cheaper(move.to, to_occ) ? 1 : 0);
