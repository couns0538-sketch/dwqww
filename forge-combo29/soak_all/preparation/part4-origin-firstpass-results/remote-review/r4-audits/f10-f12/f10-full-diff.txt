+# R4-10 — `setoption` values with spaces or the word `value`
+
+Finding 10 from `audit/lock-source` at `302a08a2` says `setoption` drops spaces in a value and stops at the word `value`. Reproduced on base `e04556ddc3a430b2e751159b0d46111ef56c8b89`. The base already keeps those values. No engine source change.
+
+## What the parser does
+
+`forge-combo29/src/uci.cpp` reads the first word after `value` with `operator>>`, then `Book File` and `NnueFile` append the rest of the line with `std::getline`. Joining those two pieces keeps internal spaces, tabs, and a later `value` token. Whitespace immediately after the `value` keyword is the separator, so a leading run is not part of the path. Trailing spaces and `\r` are stripped. That is the same path text as the audit's proposed `getline` for these two string options.
+
+The proposed replacement was not applied. It would not change `Book File` or `NnueFile`, and it would reject a spin such as `Hash value 16 extra` instead of reading `16`. No advertised option name contains the token `value`. The name scan still treats the first `value` token as the UCI separator.
+
+## Proof
+
+`forge-combo29/tests_r4_10/check.cpp` is the check. It exits non-zero on failure. The engine was built with no `-DFORGE_*` override and linked with AddressSanitizer and UndefinedBehaviorSanitizer:

+std::string capture_command(const std::vector<std::string>& args) {
+    if (args.empty()) fail("empty command");
+    int pipes[2];
+    if (pipe(pipes) != 0) fail("pipe");
+    const pid_t child = fork();
+    if (child < 0) fail("fork");
+    if (child == 0) {
+        dup2(pipes[1], STDOUT_FILENO);
+        dup2(pipes[1], STDERR_FILENO);
+        close(pipes[0]);
+        close(pipes[1]);
+        std::vector<char*> argv;
+        argv.reserve(args.size() + 1);
+        for (const std::string& arg : args) argv.push_back(const_cast<char*>(arg.c_str()));
+        argv.push_back(nullptr);
+        execvp(argv[0], argv.data());
+        _exit(127);
+    }
+    close(pipes[1]);
+    std::string output;
+    char buffer[4096];
+    while (true) {
+        const ssize_t count = read(pipes[0], buffer, sizeof buffer);
+        if (count > 0) output.append(buffer, static_cast<std::size_t>(count));
+        else if (count == 0) break;
+        else if (errno != EINTR) break;
+    }
+    close(pipes[0]);
+    int status = 0;
+    waitpid(child, &status, 0);
+    if (!WIFEXITED(status) || WEXITSTATUS(status) != 0) fail("command failed: " + args[0]);
+    return output;
+}

+class Engine {
+public:
+    explicit Engine(const std::string& path) {
+        int input_pipe[2];
+        int output_pipe[2];
+        int error_pipe[2];
+        if (pipe(input_pipe) != 0 || pipe(output_pipe) != 0 || pipe(error_pipe) != 0) fail("pipe");
+        pid_ = fork();
+        if (pid_ < 0) fail("fork");
+        if (pid_ == 0) {
+            dup2(input_pipe[0], STDIN_FILENO);
+            dup2(output_pipe[1], STDOUT_FILENO);
+            dup2(error_pipe[1], STDERR_FILENO);
+            close(input_pipe[0]);
+            close(input_pipe[1]);
+            close(output_pipe[0]);
+            close(output_pipe[1]);
+            close(error_pipe[0]);
+            close(error_pipe[1]);
+            execl(path.c_str(), path.c_str(), static_cast<char*>(nullptr));
+            _exit(127);
+        }
+        close(input_pipe[0]);
+        close(output_pipe[1]);
+        close(error_pipe[1]);
+        to_child_ = input_pipe[1];
+        from_child_ = output_pipe[0];
+        err_child_ = error_pipe[0];
+        set_nonblock(from_child_);
+        set_nonblock(err_child_);
+    }
+
+    Engine(const Engine&) = delete;
+    Engine& operator=(const Engine&) = delete;
+
+    ~Engine() {
+        if (pid_ > 0) {
+            kill(pid_, SIGKILL);
+            waitpid(pid_, nullptr, 0);
+            pid_ = -1;
+        }
+        close_fds();
+    }
+
+    void send(const std::string& line) {
+        if (pid_ <= 0) fail("engine is not running before: " + line);
+        const std::string data = line + "\n";
+        std::size_t written = 0;
+        while (written < data.size()) {
+            const ssize_t count = write(to_child_, data.data() + written, data.size() - written);
+            if (count > 0) written += static_cast<std::size_t>(count);
+            else if (count < 0 && errno == EINTR) continue;
+            else fail("engine exited before: " + line + "; stderr=" + stderr_text());
+        }
+    }
+
+    std::string readline(int timeout_ms) {
+        const auto deadline = std::chrono::steady_clock::now() + std::chrono::milliseconds(timeout_ms);
+        while (true) {
+            if (const auto newline = stdout_pending_.find('\n'); newline != std::string::npos) {
+                std::string line = stdout_pending_.substr(0, newline);
+                stdout_pending_.erase(0, newline + 1);
+                if (!line.empty() && line.back() == '\r') line.pop_back();
+                return line;
+            }
+            if (stdout_eof_) fail("engine closed stdout; stderr=" + stderr_text());
+            const int remain = static_cast<int>(std::chrono::duration_cast<std::chrono::milliseconds>(
+                                                     deadline - std::chrono::steady_clock::now())
+                                                     .count());
+            if (remain <= 0) fail("timed out after " + std::to_string(timeout_ms) + "ms; stderr=" + stderr_text());
+            pollfd fds[2] = {{from_child_, POLLIN, 0}, {err_child_, POLLIN, 0}};
+            const int ready = poll(fds, 2, remain);
+            if (ready < 0) {
+                if (errno == EINTR) continue;
+                fail("poll");
+            }
+            if (ready == 0) fail("timed out after " + std::to_string(timeout_ms) + "ms; stderr=" + stderr_text());
+            if ((fds[1].revents & (POLLIN | POLLHUP | POLLERR)) != 0) drain(err_child_, stderr_raw_, stderr_eof_);
+            if ((fds[0].revents & (POLLIN | POLLHUP | POLLERR)) != 0) drain(from_child_, stdout_pending_, stdout_eof_);
+        }
+    }
+
+    std::vector<std::string> exchange(const std::string& line, int timeout_ms = 60000) {
+        send(line);
+        send("isready");
+        std::vector<std::string> rows;
+        while (true) {
+            std::string row = readline(timeout_ms);
+            if (row == "readyok") return rows;
+            rows.push_back(std::move(row));
+        }
+    }
+
+    std::string stderr_text() const { return trim_copy(stderr_raw_); }
+
+    int finish() {
+        send("quit");
+        const auto deadline = std::chrono::steady_clock::now() + std::chrono::seconds(60);
+        while (std::chrono::steady_clock::now() < deadline) {
+            drain(from_child_, stdout_pending_, stdout_eof_);
+            drain(err_child_, stderr_raw_, stderr_eof_);
+            int status = 0;
+            const pid_t got = waitpid(pid_, &status, WNOHANG);
+            if (got == pid_) {
+                pid_ = -1;
+                close_fds();
+                if (WIFEXITED(status)) return WEXITSTATUS(status);
+                return 1;
+            }
+            if (got < 0 && errno != EINTR) fail("waitpid");
+            pollfd fds[2] = {{from_child_, POLLIN, 0}, {err_child_, POLLIN, 0}};
+            poll(fds, 2, 50);
+        }
+        fail("engine did not exit; stderr=" + stderr_text());
+    }
+
+private:
+    static void set_nonblock(int fd) {
+        const int flags = fcntl(fd, F_GETFL, 0);
+        if (flags < 0 || fcntl(fd, F_SETFL, flags | O_NONBLOCK) < 0) fail("fcntl");
+    }
+
+    static void drain(int fd, std::string& dest, bool& eof) {
+        if (fd < 0 || eof) return;
+        char buffer[4096];
+        while (true) {
+            const ssize_t count = read(fd, buffer, sizeof buffer);
+            if (count > 0) dest.append(buffer, static_cast<std::size_t>(count));
+            else if (count == 0) {
+                eof = true;
+                return;
+            } else if (errno == EINTR) {
+                continue;
+            } else if (errno == EAGAIN || errno == EWOULDBLOCK) {
+                return;
+            } else {
+                fail(std::string("read: ") + std::strerror(errno));
+            }
+        }
+    }
+
+    void close_fds() {
+        if (to_child_ >= 0) close(to_child_);
+        if (from_child_ >= 0) close(from_child_);
+        if (err_child_ >= 0) close(err_child_);
+        to_child_ = from_child_ = err_child_ = -1;
+    }
+
+    pid_t pid_ = -1;
+    int to_child_ = -1;
+    int from_child_ = -1;
+    int err_child_ = -1;
+    bool stdout_eof_ = false;
+    bool stderr_eof_ = false;
+    std::string stdout_pending_;
+    std::string stderr_raw_;
+};

+    const std::uint64_t bench11 = bench_nodes(engine, 11);
+    const std::uint64_t bench13 = bench_nodes(engine, 13);
+    const std::uint64_t perft5 = perft_nodes(engine, 5);
+    std::cout << "pins " << bench11 << " " << bench13 << " " << perft5 << std::endl;
+    if (bench11 != kPins11 || bench13 != kPins13 || perft5 != kPerft5) {
+        fail("pin mismatch got " + std::to_string(bench11) + " " + std::to_string(bench13) + " " + std::to_string(perft5));
+    }
+    std::cout << "PASS pins 312866 862329 4865609" << std::endl;
+}

+std::uint64_t next_u64(std::uint64_t& state) {
+    state ^= state >> 12U;
+    state ^= state << 25U;
+    state ^= state >> 27U;
+    state *= 0x2545F4914F6CDD1DULL;
+    return state;
+}
+
+std::string position_command(const std::string& fen, const std::vector<std::string>& moves) {
+    std::string command = "position fen " + fen;
+    if (!moves.empty()) {
+        command += " moves";
+        for (const std::string& move : moves) command += " " + move;
+    }
+    return command;
+}
+
+std::vector<std::string> legal_moves(Engine& engine, const std::string& fen, const std::vector<std::string>& moves) {
+    const std::vector<std::string> placed = engine.exchange(position_command(fen, moves), 30000);
+    for (const std::string& row : placed) {
+        if (row.rfind("info string ", 0) == 0) fail("position rejected " + fen + ": " + row);
+    }
+    const std::vector<std::string> rows = engine.exchange("go perft 1", 30000);
+    std::vector<std::string> found;
+    int total = -1;
+    for (const std::string& row : rows) {
+        if (row.rfind("nodes ", 0) == 0) total = std::stoi(split_ws(row).at(1));
+        else if (const std::size_t split = row.find(": "); split != std::string::npos) {
+            const std::string move = row.substr(0, split);
+            const std::string count = row.substr(split + 2);
+            if (count != "1") fail("perft 1 count " + row);
+            found.push_back(move);
+        }
+    }
+    if (total != static_cast<int>(found.size())) {
+        fail("perft 1 nodes " + std::to_string(total) + " listed " + std::to_string(found.size()));
+    }
+    return found;
+}
+
+std::vector<Position> build_positions(Engine& engine) {
+    std::vector<Position> positions;
+    std::uint64_t state = 0x9E3779B97F4A7C15ULL;
+    for (const std::string_view seed : kSeeds) {
+        const std::string fen(seed);
+        std::vector<std::string> moves;
+        for (int step = 0; step < 8; ++step) {
+            if (positions.size() >= 200) return positions;
+            std::vector<std::string> legal = legal_moves(engine, fen, moves);
+            if (legal.empty()) break;
+            positions.push_back(Position{fen, moves});
+            moves.push_back(legal[static_cast<std::size_t>(next_u64(state) % legal.size())]);
+        }
+    }
+    if (positions.size() < 200) fail("only gathered " + std::to_string(positions.size()) + " positions");
+    return positions;
+}
+
+void check_searches(Engine& engine) {
+    const std::vector<Position> positions = build_positions(engine);
+    if (positions.size() != 200) fail("expected 200 positions, got " + std::to_string(positions.size()));
+    const auto started = std::chrono::steady_clock::now();
+    for (std::size_t index = 0; index < positions.size(); ++index) {
+        const std::vector<std::string> legal = legal_moves(engine, positions[index].fen, positions[index].moves);
+        if (legal.empty()) fail("search " + std::to_string(index + 1) + " has no legal move");
+        engine.send("go depth 10");
+        std::string best;
+        while (true) {
+            const std::string row = engine.readline(90000);
+            if (row.rfind("bestmove ", 0) == 0) {
+                const std::vector<std::string> parts = split_ws(row);
+                if (parts.size() < 2) fail("search " + std::to_string(index + 1) + " bad bestmove line " + row);
+                best = parts[1];
+                break;
+            }
+        }
+        bool found = false;
+        for (const std::string& move : legal) {
+            if (move == best) found = true;
+        }
+        if (!found) fail("search " + std::to_string(index + 1) + " bestmove " + best + " is illegal");
+        const std::size_t number = index + 1;
+        if (number == 1 || number == 200 || number % 25 == 0) {
+            std::cout << "search " << number << " bestmove " << best << " legal " << legal.size() << std::endl;
+        }
+    }
+    const auto elapsed_ms = std::chrono::duration_cast<std::chrono::milliseconds>(std::chrono::steady_clock::now() - started).count();
+    std::cout << "PASS 200 depth-10 searches legal in " << (elapsed_ms / 1000) << "." << ((elapsed_ms % 1000) / 100) << "s" << std::endl;
+}

+int main(int argc, char** argv) {
+    signal(SIGPIPE, SIG_IGN);
+    if (argc != 2) {
+        std::cerr << "usage: check /path/to/forge" << std::endl;
+        return 2;
+    }
+    setenv("ASAN_OPTIONS", "halt_on_error=1:abort_on_error=1:detect_leaks=1", 1);
+    setenv("UBSAN_OPTIONS", "halt_on_error=1:abort_on_error=1:print_stacktrace=1", 1);
+    try {
+        require_sanitizers(argv[1]);
+        Engine engine(argv[1]);
+        try {
+            const std::vector<std::string> rows = engine.exchange("uci", 30000);
+            bool uciok = false;
+            for (const std::string& row : rows) {
+                if (row == "uciok") uciok = true;
+            }
+            if (!uciok) fail("uci did not answer uciok");
+            check_string_options(engine);
+            check_pins(engine);
+            check_searches(engine);
+            const int code = engine.finish();
+            if (code != 0) fail("engine exit " + std::to_string(code) + "; stderr=" + engine.stderr_text());
+            if (!engine.stderr_text().empty()) fail("sanitizer or engine stderr: " + engine.stderr_text());
+            std::cout << "PASS r4-10" << std::endl;
+            return 0;
+        } catch (const std::exception& error) {
+            std::cerr << "FAIL " << error.what() << std::endl;
+            std::cerr << engine.stderr_text() << std::endl;
+            return 1;
+        }
+    } catch (const std::exception& error) {
+        std::cerr << "FAIL " << error.what() << std::endl;
+        return 1;
+    }
