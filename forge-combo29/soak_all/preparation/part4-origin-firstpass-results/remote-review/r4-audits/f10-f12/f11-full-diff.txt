+# R4 finding 11 — `params::name` negative index

+Base: `e04556ddc3a430b2e751159b0d46111ef56c8b89`

+
+## Bug
+
+`forge::params::name(int)` in `src/eval_params.cpp` treated every `index < KnightMobility` as a piece-square name. A negative index was not rejected. For `index <= -64`, `index / 64` is negative, `static_cast<std::size_t>` wraps, and the 6-element `pieces` array is read before its start. For `-63 <= index <= -1` the read stays in the array and the function returns a name such as `PawnPsqt[-1]` instead of `Unknown`.
+
+## Reproduction on the base
+
+Built and run before the fix, from `e04556dd`, with the audit proof:
+

+## Fix

+One guard at the start of `name`. In-range names are unchanged. `index >= Count` already fell through to `Unknown`.

+std::vector<forge::Board> build_200_positions() {
+    std::vector<forge::Board> positions;
+    positions.reserve(200);
+    std::uint64_t rng = 0x9E3779B97F4A7C15ULL;
+    auto next_u64 = [&]() {
+        rng ^= rng >> 12U;
+        rng ^= rng << 25U;
+        rng ^= rng >> 27U;
+        return rng * 0x2545F4914F6CDD1DULL;
+    };
+
+    for (std::size_t s = 0; s < SeedFens.size() && positions.size() < 200; ++s) {
+        forge::Board board;
+        if (!board.set_fen(SeedFens[s])) continue;
+        for (int step = 0; step < 8 && positions.size() < 200; ++step) {
+            const forge::MoveList legal = board.legal_moves();
+            if (legal.empty()) {
+                (void)board.set_fen(SeedFens[(s + 1) % SeedFens.size()]);
+            }
+            if (!board.legal_moves().empty()) {
+                positions.push_back(board);
+            }
+            const forge::MoveList step_moves = board.legal_moves();
+            if (step_moves.empty()) break;
+            const forge::Move chosen = step_moves[static_cast<std::size_t>(next_u64() % step_moves.size())];
+            forge::Undo undo;
+            (void)board.make_move(chosen, undo);
+        }
+    }
+    return positions;
+}

+bool move_is_legal(const forge::Board& board, forge::Move move) {
+    if (move.is_none() || !board.is_legal(move)) return false;
+    const forge::MoveList legal = board.legal_moves();
+    for (const forge::Move candidate : legal) {
+        if (candidate == move) return true;
+    }
+    return false;
+}

+int check_searches(const std::vector<forge::Board>& positions) {
+    forge::Searcher searcher(16, 1);
+    int illegal = 0;
+    std::uint64_t total_nodes = 0;
+    for (std::size_t i = 0; i < positions.size(); ++i) {
+        if (i % 25 == 0) searcher.clear();
+        const forge::Board& pos = positions[i];
+        forge::SearchLimits limits;
+        limits.depth = 10;
+        const forge::SearchInfo info =
+            searcher.search(pos, limits, false, forge::Searcher::TTAccess::Exclusive);
+        total_nodes += info.nodes;
+
+        if (!move_is_legal(pos, info.best_move)) {
+            fail("illegal best_move at position " + std::to_string(i) + " (" +
+                 forge::Board::move_to_uci(info.best_move) + ")");
+            ++illegal;
+            break;
+        }
+
+        forge::Board cursor = pos;
+        for (std::size_t ply = 0; ply < info.pv.size(); ++ply) {
+            const forge::Move move = info.pv[ply];
+            if (!move_is_legal(cursor, move)) {
+                fail("illegal pv move at position " + std::to_string(i) + " ply " + std::to_string(ply) + " (" +
+                     forge::Board::move_to_uci(move) + ")");
+                ++illegal;
+                break;
+            }
+            forge::Undo undo;
+            if (!cursor.make_move(move, undo)) {
+                fail("pv move rejected at position " + std::to_string(i) + " ply " + std::to_string(ply));
+                ++illegal;
+                break;
+            }
+        }
+        if (illegal != 0) break;
+        if ((i + 1) % 25 == 0) {
+            std::cout << "searches " << (i + 1) << " legal_ok 1\n";
+        }
+    }
+    std::cout << "searches " << positions.size() << " depth 10 legal_ok " << (illegal == 0 ? 1 : 0)
+              << " total_nodes " << total_nodes << '\n';
+    return illegal;
+}
