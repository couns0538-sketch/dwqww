   328	    // Lifecycle operations themselves must not run concurrently with search.
   329	    if (probe.owner_ != this || probe.epoch_ != epoch_ || probe.key_ != key ||

   331	    const auto lock = TakeLock ? lock_stripe(probe.cluster_) : std::unique_lock<std::mutex>();
   332	    TTCluster& cluster = clusters_[probe.cluster_];
   333	    const std::uint32_t key32 = verification_key(key);
   334	    // Probe records a slot before the subtree runs. Another store can place this
   335	    // key in a different entry of the same cluster; writing the stale slot would
   336	    // duplicate it, and probe returns the first match, so the shallow copy hides
   337	    // the deeper one. Re-find the key under the stripe lock and update that entry.
   338	    TTEntry* entry = &cluster.entries[probe.slot_];
   339	    for (TTEntry& candidate : cluster.entries) {
   340	        if (candidate.key32 == key32 && candidate.occupied()) {
   341	            entry = &candidate;
   342	            break;
   343	        }
   344	    }
   345	    const bool same_position = entry->key32 == key32 && entry->occupied();
   346	    if (!move.is_none() || !same_position) entry->move = move;
   347	    if (bound == BoundExact || !same_position || depth + 4 > entry->depth || entry->generation() != generation_) {
   348	        entry->key32 = key32;
   349	        entry->score = static_cast<std::int16_t>(std::clamp(score, -32767, 32767));

   390	            }
   391	            return snapshot(entry, true);
   392	        }

   396	        return static_cast<int>((32 + generation_ - entry.generation()) & 31U);
