async function hash(file){
  if(hashCache.has(file))return hashCache.get(file);
  const size=(await fsp.stat(file)).size,sha=crypto.createHash('sha256'),blob=crypto.createHash('sha1').update('blob '+size+'\0');let bytes=0;
  for await(const data of fs.createReadStream(file)){sha.update(data);blob.update(data);bytes+=data.length;}
  const value={bytes,sha256:sha.digest('hex'),gitBlob:blob.digest('hex')};hashCache.set(file,value);return value;
}
