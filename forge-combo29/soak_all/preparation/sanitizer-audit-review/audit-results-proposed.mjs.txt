// std::ostream defaultfloat with precision 6 rounds to six significant digits.
// Each printed wall-time value describes an interval half a last place wide.
// Integer counters never use these measurement bounds.
export function sixSignificantHalfStep(value){return value===0?0:0.5*10**(Math.floor(Math.log10(Math.abs(value)))-5);}
export function wallMeasurementMatch(reported,rawSum,rawRoundingBound=0,terms=1){
 if(!Number.isFinite(reported)||!Number.isFinite(rawSum)||reported<0||rawSum<0||!Number.isFinite(rawRoundingBound)||rawRoundingBound<0)return false;
 const arithmeticBound=Number.EPSILON*(Math.abs(reported)+Math.abs(rawSum))*(terms+2);
 return Math.abs(reported-rawSum)<=sixSignificantHalfStep(reported)+rawRoundingBound+arithmeticBound;
}

function sameResult(a,b){return !!a&&!!b&&['bestmove','score_kind','score','nodes','bound'].every(k=>a[k]===b[k]);}

function runtimeDiagnosticLines(text,source){
 const found=[];
 const strings=(x,where)=>{
  if(!x||typeof x!=='object')return;
  for(const [k,v]of Object.entries(x)){
   if(['lock_t1_reference','merged_t1_reference','config','options','description','command','commandShell','environmentOverrides'].includes(k))continue;
   if(['stderr','stderr_output','error'].includes(k)&&typeof v==='string'){
    for(const [i,line]of v.split(/\r?\n/).entries())if(diagnostic.test(line))found.push({line:null,payload:where+'.'+k+':'+(i+1),text:line,trigger:null});
   }else if(v&&typeof v==='object')strings(v,where+'.'+k);
  }
 };
 if(/\.jsonl$/.test(source)){for(const [i,line]of text.split(/\r?\n/).entries())if(line.trim()){try{strings(JSON.parse(line),'record'+(i+1));}catch{}}return found;}
 if(/\.json$/.test(source)){try{strings(JSON.parse(text),'root');}catch{}return found;}
 let lastTrigger=null;
 for(const [i,line]of text.split(/\r?\n/).entries()){
  if(line.startsWith('TRIGGER ')){try{lastTrigger=JSON.parse(line.slice(8));}catch{}}
  if(/^(?:RUN_SPEC_|CWD |ENV_OVERRIDES |COMMAND |CONFIG |PENDING |CASE |ATTEMPT |STATUS |COMPLETED |RESULT |UNIT |SUMMARY |WORKERS_)/.test(line))continue;
  if(diagnostic.test(line))found.push({line:i+1,text:line,trigger:lastTrigger});
 }
 return found;
}
export function parseExistingKrkNative(raw,runsPerPosition){
 const lines=raw.replace(/\r/g,'').split('\n'),configs=[],t1=[],main=[],diagnostics=[];
 let output=false;const end=[],argv=[],environment=[],binaryHashes=[];
 for(const [n,line]of lines.entries()){
  let m;
  if(line==='OUTPUT'){output=true;continue;}
  if((m=line.match(/^END (\S+) EXIT (\d+|null) SIGNAL (\S+)$/))){end.push({utc:m[1],exit:m[2]==='null'?null:Number(m[2]),signal:m[3]==='null'?null:m[3]});output=false;continue;}
  if(!output){if(line.startsWith('ARGV '))argv.push(JSON.parse(line.slice(5)));if(line.startsWith('ENV '))environment.push(JSON.parse(line.slice(4)));if((m=line.match(/^BINARY_SHA256 ([0-9a-f]{64})$/)))binaryHashes.push(m[1]);continue;}
  if((m=line.match(/^config positions=(\d+) runs_per_pos=(\d+) threads=(\d+) hash_mb=(\d+) depth=(\d+) use_nnue=(true|false)$/)))configs.push({positions:+m[1],runs_per_pos:+m[2],threads:+m[3],hash_mb:+m[4],depth:+m[5],use_nnue:m[6]==='true'});
  if((m=line.match(/^threads_1_check: (\d+) \/ (\d+) misses$/)))t1.push({misses:+m[1],runs:+m[2]});
  if((m=line.match(/^SUMMARY: misses=(\d+) \/ (\d+) \(no_mate_score=(\d+), non_mating_move=(\d+)\)$/)))main.push({misses:+m[1],runs:+m[2],no_mate_score:+m[3],non_mating_move:+m[4]});
  if(diagnostic.test(line))diagnostics.push({line:n+1,text:line});
 }
 const c=configs[0],r=t1[0],m=main[0],safe=n=>Number.isSafeInteger(n)&&n>=0;
 const configValid=configs.length===1&&same(c,{positions:40,runs_per_pos:runsPerPosition,threads:2,hash_mb:16,depth:14,use_nnue:true});
 const t1Valid=t1.length===1&&safe(r.misses)&&r.runs===100&&r.misses<=r.runs;
 const mainValid=main.length===1&&Object.values(m).every(safe)&&m.runs===40*runsPerPosition&&m.no_mate_score<=m.misses&&m.non_mating_move<=m.misses&&m.misses<=Math.min(m.runs,m.no_mate_score+m.non_mating_move);
 return {configs,t1,main,diagnostics,end,argv,environment,binaryHashes,configValid,t1Valid,mainValid};
}
export function parseTsanWarnings(raw){
 const lines=raw.replace(/\r/g,'').split('\n'),warnings=[],footers=[];let current=null,lastTrigger=null;
 const normalize=line=>line.replace(/^! stderr /,'');
 const frameLocation=line=>{
  const engine=line.match(/(?:^|\s)((?:\S*\/)?forge-combo29\/(?:src|include)\/[^\s():]+):(\d+)(?::\d+)?/);
  if(engine)return {location:engine[1].slice(engine[1].indexOf('forge-combo29/'))+':'+engine[2],engine:true};
  const any=line.match(/(?:^|\s)((?:\/|[A-Za-z]:\/)[^\s():]+\.(?:c|cc|cpp|cxx|h|hh|hpp)):(\d+)(?::\d+)?/);
  return any?{location:any[1]+':'+any[2],engine:false}:null;
 };
 const finish=endLine=>{if(!current)return;current.end_line=endLine;
  const accesses=[];let access=null;
  for(const line of current.lines){const text=normalize(line),m=text.match(/^\s*(?:Previous\s+)?(?:Atomic\s+)?(Read|Write) of size /i);if(m){access={kind:m[1].toLowerCase(),frames:[]};accesses.push(access);continue;}if(access&&/^\s*#\d+\b/.test(text)){const loc=frameLocation(text);if(loc)access.frames.push(loc);}else if(access&&(!text.trim()||/^\s*(?:Thread .*created|Location is|SUMMARY: ThreadSanitizer|Mutex .*created)/.test(text)))access=null;}
  current.accesses=accesses.map(a=>({kind:a.kind,location:(a.frames.find(f=>f.engine)??a.frames[0])?.location??null}));
  current.source_pair=current.accesses.length>=2&&current.accesses[0].location&&current.accesses[1].location?[current.accesses[0].location,current.accesses[1].location].sort():null;
  // An unsymbolized block remains separate: do not invent a conflicting pair.
  current.group_key=current.source_pair?current.kind+'|'+current.source_pair.join('|'):'unresolved-warning-line-'+current.start_line;
  current.raw=current.lines.join('\n')+'\n';delete current.lines;warnings.push(current);current=null;
 };
 for(const [i,original]of lines.entries()){
  if(original.startsWith('TRIGGER ')){try{lastTrigger=JSON.parse(original.slice(8));}catch{}}
  const line=normalize(original),start=line.match(/^WARNING: ThreadSanitizer: (.+?)(?: \(pid=\d+\))?$/);
  if(start){finish(i);current={start_line:i+1,kind:start[1],trigger:lastTrigger,lines:[original]};continue;}
  const footer=line.match(/^ThreadSanitizer: reported (\d+) warnings?$/);if(footer)footers.push({line:i+1,count:Number(footer[1])});
  if(current){if(/^={8,}$/.test(line)){finish(i);continue;}current.lines.push(original);}
 }
 finish(lines.length);
 const groups=new Map();for(const w of warnings){const g=groups.get(w.group_key)??{group_key:w.group_key,kind:w.kind,source_pair:w.source_pair,reported_warning_blocks:0,occurrences:[]};g.reported_warning_blocks++;g.occurrences.push({start_line:w.start_line,end_line:w.end_line,trigger:w.trigger});groups.set(w.group_key,g);}
 return {reported_warning_blocks:warnings.length,groups:[...groups.values()],warnings,footers,footer_count_consistent:footers.length===1?footers[0].count===warnings.length:null,last_trigger:lastTrigger,counting_note:'One primary merged runtime log only; mirrored artifacts and WARNING/SUMMARY/footer marker lines are not extra warning events. Source-line pairs are evidence groups, not proven unique bugs. Runtime duplicate suppression may omit repeat manifestations.'};
}
export function classifySelfPlayProcess(exit,signal,metadata,summary,markerCount){

 else if(exit===tsanExit){classification=markerCount?'sanitizer_diagnostic_nonzero_exit':'nonzero_sanitizer_exit_code_without_runtime_diagnostic';separateFault=true;}

 }
