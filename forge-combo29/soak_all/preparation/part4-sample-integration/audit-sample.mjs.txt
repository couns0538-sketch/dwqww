export function expectedRoots(count,seed){
 const anchor='8/8/8/8/8/1R6/2k5/5K2 w - - 0 1',answer=[anchor];
 const coordinate=sq=>[sq%8,Math.floor(sq/8)];
 const text=(wk,wr,bk)=>{const ranks=[];for(let rank=7;rank>=0;rank--){let result='',empty=0;for(let file=0;file<8;file++){const sq=rank*8+file,piece=sq===wk?'K':sq===wr?'R':sq===bk?'k':null;if(piece){if(empty)result+=empty;empty=0;result+=piece;}else empty++;}if(empty)result+=empty;ranks.push(result);}return ranks.join('/')+' w - - 0 1';};
 for(let i=0;answer.length<count&&i<262144;i++){
  const state=(i*176123+seed)%262144,wk=Math.floor(state/4096),wr=Math.floor(state/64)%64,bk=state%64;
  if(new Set([wk,wr,bk]).size!==3)continue;
  const [wx,wy]=coordinate(wk),[rx,ry]=coordinate(wr),[bx,by]=coordinate(bk);
  if(Math.max(Math.abs(wx-bx),Math.abs(wy-by))<=1)continue;
  const between=(x,a,b)=>x>Math.min(a,b)&&x<Math.max(a,b);
  if((rx===bx&&!(wx===rx&&between(wy,ry,by)))||(ry===by&&!(wy===ry&&between(wx,rx,bx))))continue;
  const fen=text(wk,wr,bk);if(fen!==anchor)answer.push(fen);
 }
 if(answer.length!==count)throw Error('fixed permutation exhausted');return answer;
}

export function auditSequence(reports,{requireStartZero=true}={}){
 const errors=[],totals=zero(),resumed=[],pending=[];let cursor=reports[0]?.offset??0,completed=0;const first=reports[0];
 if(requireStartZero&&cursor!==0)errors.push('sequence does not start at root zero');
 for(let i=0;i<reports.length;i++){
  const r=reports[i];if(!r.accounting_valid)errors.push('batch '+i+' not independently audit-valid');
  for(const k of ['phase','seed','corpus_sha256','available_units'])if(r[k]!==first[k])errors.push('batch '+i+' differs in '+k);
  if(r.offset!==cursor||r.next_offset!==r.offset+r.completed_units)errors.push('batch '+i+' overlaps/skips or has wrong cursor');
  for(const p of pending)if(p.resolvedBy===undefined&&r.units.some(u=>u.completed&&u.root===p.root)){p.resolvedBy=i;resumed.push({...p});}
  if(r.partial_root!==null)pending.push({batch:i,root:r.partial_root});
  cursor=r.next_offset;completed+=r.completed_units;add(totals,r.counts);
 }
 return {schema:'part4sample_readonly_sequence_v1',valid:errors.length===0,errors,batches:reports.length,completed_units:completed,next_offset:cursor,counts:totals,partial_resumed:resumed,partial_reproduction_pending:pending.filter(p=>p.resolvedBy===undefined),state_changed:false};
}
