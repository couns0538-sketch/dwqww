import fs from 'node:fs';import path from 'node:path';import {fileURLToPath} from 'node:url';const d=path.dirname(fileURLToPath(import.meta.url)),label=process.argv[2];if(!['off','on'].includes(label))throw Error('Invalid variant');
const load=name=>({name,summary:JSON.parse(fs.readFileSync(path.join(d,'runs',name,'summary.json.txt'),'utf8')),stdout:fs.readFileSync(path.join(d,'runs',name,'stdout.txt'),'utf8'),stderr:fs.readFileSync(path.join(d,'runs',name,'stderr.txt'),'utf8')});const base=load('baseline-r1'),a=load(label+'-r1'),b=load(label+'-r2');
const norm=s=>s.replace(/ nps [0-9]+/g,'').replace(/ time [0-9]+/g,'');const withoutNewOption=s=>s.replace('option name SoundMatePruning type check default false\n','');const complete=r=>r.summary.code===0&&r.summary.completed&&!r.summary.timedOut&&!r.summary.spawnError;
const checks={complete:[base,a,b].every(complete),stderrEmpty:[base,a,b].every(r=>r.stderr===''),repeatBinary:a.summary.binarySha256===b.summary.binarySha256,repeatedNormalizedStdoutEqual:norm(a.stdout)===norm(b.stdout),optionCondition:a.summary.settings.SoundMatePruning===(label==='on')&&b.summary.settings.SoundMatePruning===(label==='on'),optionAdvertised:a.stdout.includes('option name SoundMatePruning type check default false')&&b.stdout.includes('option name SoundMatePruning type check default false')};
if(label==='off')checks.defaultOffMatchesBaselineExceptNewOptionAdvertisement=norm(base.stdout)===norm(withoutNewOption(a.stdout));
const final=r=>r.stdout.split('\n').findLast(l=>l.startsWith('info depth 13 '));console.log(JSON.stringify({label,checks,baselineFinal:final(base),variantFinal:final(a),variantBestmove:a.stdout.match(/^bestmove .*$/m)?.[0],normalization:'Only nps/time removed; off-baseline additionally removes exact new-option advertisement line.'},null,2));if(Object.values(checks).some(x=>!x))process.exitCode=1;
