import fs from 'node:fs';import path from 'node:path';import crypto from 'node:crypto';import {spawnSync} from 'node:child_process';import {fileURLToPath} from 'node:url';
const d=path.dirname(fileURLToPath(import.meta.url)),plan=JSON.parse(fs.readFileSync(path.join(d,'native-plan.json.txt'),'utf8')),i=Number(process.argv[2]);if(!Number.isInteger(i)||!plan.steps[i])throw Error('Invalid step');
for(const[file,sha]of Object.entries(plan.sourceFiles))if(crypto.createHash('sha256').update(fs.readFileSync(file)).digest('hex')!==sha)throw Error('Pinned branch source changed');
const logs=path.join(d,'native-logs');fs.mkdirSync(logs,{recursive:true});const file=j=>path.join(logs,String(j).padStart(2,'0')+'-'+plan.steps[j].name+'.json.txt');if(fs.existsSync(file(i)))throw Error('Refusing overwrite');if(i&&JSON.parse(fs.readFileSync(file(i-1),'utf8')).exit!==0)throw Error('Previous step failed');
const argv=plan.steps[i].argv,started=new Date().toISOString(),t=Date.now();const r=spawnSync(argv[0],argv.slice(1),{cwd:d,encoding:'utf8',timeout:445000,maxBuffer:64*1024*1024});const out={started,ended:new Date().toISOString(),wallMs:Date.now()-t,cwd:d,argv,exit:r.status,signal:r.signal,error:r.error?String(r.error):null,stdout:r.stdout,stderr:r.stderr};fs.writeFileSync(file(i),JSON.stringify(out,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify(out,null,2));if(r.status!==0)process.exitCode=1;
