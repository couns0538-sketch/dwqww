+// mirror positions emit exactly corresponding lists. Colour-relative square = sq ^ 56 for Black, sq for White.
+namespace {
+inline void canonicalise_moves(const Board& board, MoveList& list) {
+    const int flip = board.side_to_move() == Black ? 56 : 0;
+    std::sort(list.begin(), list.end(), [flip](Move a, Move b) {
+        const int af = (a.from ^ flip), at = (a.to ^ flip);
+        const int bf = (b.from ^ flip), bt = (b.to ^ flip);
+        if (af != bf) return af < bf;
+        if (at != bt) return at < bt;
+        return a.promotion < b.promotion;
+    });
