+inline int monarch(const Position& p, Color c) {
+    for (int s = 0; s < 64; ++s)
+        if (p[static_cast<std::size_t>(s)] != Empty && team(p[static_cast<std::size_t>(s)]) == c &&
+            kind(p[static_cast<std::size_t>(s)]) == King) return s;

+inline bool attack(const Position& p, int from, int to, int type, Color c) {
+    if (from < 0 || to < 0 || from == to) return false;
+    const int dx = (to & 7) - (from & 7), dy = (to >> 3) - (from >> 3);
+    const int ax = std::abs(dx), ay = std::abs(dy);
+    if (type == Pawn) return ax == 1 && dy == (c == White ? 1 : -1);
+    if (type == Knight) return (ax == 1 && ay == 2) || (ax == 2 && ay == 1);
+    if (type == King) return ax <= 1 && ay <= 1;
+    const bool diagonal = ax == ay && ax != 0;
+    const bool straight = (dx == 0) != (dy == 0);
+    if (!((type == Bishop && diagonal) || (type == Rook && straight) ||
+          (type == Queen && (diagonal || straight)))) return false;
+    const int sx = sign(dx), sy = sign(dy);
+    for (int x = (from & 7) + sx, y = (from >> 3) + sy; x != (to & 7) || y != (to >> 3); x += sx, y += sy)
+        if (p[static_cast<std::size_t>(8 * y + x)] != Empty) return false;
+    return true;
+}
+inline bool threatened(const Position& p, int target, Color by) {
+    if (target < 0) return false;
+    for (int s = 0; s < 64; ++s) {
+        const Piece piece = p[static_cast<std::size_t>(s)];
+        if (piece != Empty && team(piece) == by && attack(p, s, target, kind(piece), by)) return true;
+    }
+    return false;
+}

+inline bool gives_check(const Board& b, const Move& m) {
+    const Color us = b.side_to_move();
+    const Position p = after(copy(b), m, us);
+    return threatened(p, monarch(p, opposite(us)), us);
+}
+inline bool direct_check(const Board& b, const Move& m) {
+    // Deliberate control: only the piece arriving on m.to counts. In castling
+    // this is the king; rook checks are intentionally omitted with discoveries.
+    const Color us = b.side_to_move();
+    const Position p = after(copy(b), m, us);
+    return attack(p, m.to, monarch(p, opposite(us)), kind(p[m.to]), us);
+}
+inline bool legal(const Board& b, const Move& m) {
+    const Color us = b.side_to_move(), them = opposite(us);
+    const Position before = copy(b);
+    if (m.is_castle()) {
+        if (threatened(before, m.from, them)) return false;
+        Position transit = before;
+        const int through = static_cast<int>(m.from) + sign(static_cast<int>(m.to) - static_cast<int>(m.from));
+        transit[static_cast<std::size_t>(through)] = transit[m.from];
+        transit[m.from] = Empty;
+        if (threatened(transit, through, them)) return false;
+    }
+    const Position p = after(before, m, us);
+    return !threatened(p, monarch(p, us), them);
+}
+struct Lines {
+    std::array<Bitboard, 7> checks{};
+    Bitboard pins = 0;
+    Bitboard discoveries = 0;
+};
+inline Bitboard blockers(const Position& p, int target, Color slider_side, Color blocker_side) {
+    if (target < 0) return 0;
+    Bitboard result = 0;
+    for (int dx = -1; dx <= 1; ++dx) for (int dy = -1; dy <= 1; ++dy) {
+        if (dx == 0 && dy == 0) continue;
+        int first = -1;
+        for (int x = (target & 7) + dx, y = (target >> 3) + dy;
+             x >= 0 && x < 8 && y >= 0 && y < 8; x += dx, y += dy) {
+            const int square = y * 8 + x;
+            const Piece piece = p[static_cast<std::size_t>(square)];
+            if (piece == Empty) continue;
+            if (first < 0) { first = square; continue; }
+            const int type = kind(piece);
+            const bool slider = type == Queen || ((dx != 0 && dy != 0) ? type == Bishop : type == Rook);
+            if (slider && team(piece) == slider_side && team(p[static_cast<std::size_t>(first)]) == blocker_side)
+                result |= Bitboard{1} << first;
+            break;
+        }
+    }
+    return result;
+}
+inline Lines reconstruct(const Board& b) {
+    const Position p = copy(b);
+    const Color us = b.side_to_move(), them = opposite(us);
+    const int enemy = monarch(p, them), own = monarch(p, us);
+    Lines result;
+    if (enemy >= 0) {
+        const int ex = enemy & 7, ey = enemy >> 3;
+        const int pawn_rank = ey + (us == White ? -1 : 1);
+        for (int dx : {-1, 1}) if (ex + dx >= 0 && ex + dx < 8 && pawn_rank >= 0 && pawn_rank < 8)
+            result.checks[Pawn] |= Bitboard{1} << (pawn_rank * 8 + ex + dx);
+        for (int dx = -2; dx <= 2; ++dx) for (int dy = -2; dy <= 2; ++dy) {
+            if (std::abs(dx) * std::abs(dy) != 2) continue;
+            const int x = ex + dx, y = ey + dy;
+            if (x >= 0 && x < 8 && y >= 0 && y < 8) result.checks[Knight] |= Bitboard{1} << (8 * y + x);
+        }
+        for (int dx = -1; dx <= 1; ++dx) for (int dy = -1; dy <= 1; ++dy) {
+            if (dx == 0 && dy == 0) continue;
+            const int type = dx != 0 && dy != 0 ? Bishop : Rook;
+            for (int x = ex + dx, y = ey + dy; x >= 0 && x < 8 && y >= 0 && y < 8; x += dx, y += dy) {
+                const int square = y * 8 + x;
+                result.checks[static_cast<std::size_t>(type)] |= Bitboard{1} << square;
+                if (p[static_cast<std::size_t>(square)] != Empty) break;
+            }
+        }
+        result.checks[Queen] = result.checks[Bishop] | result.checks[Rook];
+    }
+    // NoPieceType and King entries intentionally remain zero, matching the
+    // represented geometry contract, not because kings lack real attacks.
+    result.pins = blockers(p, own, them, us);
+    result.discoveries = blockers(p, enemy, us, us);
+    return result;
+}
