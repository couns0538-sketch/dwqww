import pathlib,runpy,json,hashlib,re,contextlib,sys,subprocess
root=pathlib.Path('forge-dot-job-20261118/forge-combo29/dot/job3');m=runpy.run_path('forge-dot-job-20261118/forge-combo29/dot/job2/runner.py.txt')
exe,label,blend=sys.argv[1],sys.argv[2],int(sys.argv[3]);extra=[('FailHighBlend',blend)]

uci=subprocess.run([exe],input='uci\nquit\n',text=True,capture_output=True,timeout=10).stdout
names={re.match(r'option name (.*?) type ',s).group(1) for s in uci.splitlines() if s.startswith('option name ')}
for n,v in extra:assert n in names,n
meta={'exe':exe,'sha256':hashlib.sha256(pathlib.Path(exe).read_bytes()).hexdigest(),'id name':[s for s in uci.splitlines() if s.startswith('id name ')],'settings_sent_in_order':list(m['PC20'].items())+extra,'depth':12,'positions_sha256':hashlib.sha256(pathlib.Path('forge-dot-evidence/job3_O.epd').read_bytes()).hexdigest(),'protocol':'one process per setting; ucinewgame + isready/wait before each FEN'}

rows=[]

 for n,v in extra:e.send('setoption name '+n+' value '+str(v))

  with (root/(label+'.rows.txt')).open('a') as f:f.write(json.dumps(row)+'\n')
 e.close()
summary={'label':label,'sha256':meta['sha256'],'settings':extra,'rows':len(rows),'sum':sum(r['nodes'] for r in rows),'first_three':[r['nodes'] for r in rows[:3]],'bestmoves_first_three':[r['bestmove'] for r in rows[:3]]}
(root/(label+'.summary.txt')).write_text(json.dumps(summary,indent=2)+'\n');print(json.dumps(summary,indent=2))
