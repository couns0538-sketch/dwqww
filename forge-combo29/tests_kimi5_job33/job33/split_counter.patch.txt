+// HP33 (job 33 counting copy): at one ELIGIBLE history-prune test (reached with
+// lmr_depth <= HistoryPruneDepth) recount every part from the same entries the engine just read,
+// verify h == blend(M, W, 3) + C1 + C2, and count pivotal parts and staleness. Never touches the search.
+void Worker::hp33_count_test(const Board& board, const SearchStack* ss, Move move, const int h, const int t) {
+    ++hp33_eligible_;
+    const std::size_t us = static_cast<std::size_t>(board.side_to_move());
+    const Piece piece = board.moved_piece(move);
+    const int m = (*main_history_)[us][move.from][move.to];
+    const int w = rework_entry(board, ss, move);
+    const int c1 = (ss - 1)->continuation != nullptr ? (*(ss - 1)->continuation)[static_cast<std::size_t>(piece)][move.to] : 0;
+    const int c2 = (ss - 2)->continuation != nullptr ? (*(ss - 2)->continuation)[static_cast<std::size_t>(piece)][move.to] : 0;
+    const int quarters = b4::threat_quarters(tune::HistRework);  // 3 under PC20 (HistRework 2)
+    const int b = b4::blend(m, w, quarters);
+    if (h != (tune::HistRework != 0 ? b : m) + c1 + c2) ++hp33_mismatch_;
+    const bool stale_m = (*hp33_main_stamp_)[us][move.from][move.to] != hp33_search_no_;
+    const bool stale_w = hp33_rework_stamp(board, ss, move) != hp33_search_no_;
+    if (stale_w && w != 0) ++hp33_elig_stale_nzw_;
+    if (stale_m && m != 0) ++hp33_elig_stale_nzm_;
+    if (h >= t) return;
+    ++hp33_fired_;
+    // The three "without" tests in the same int arithmetic and against the same t.
+    const bool no_m = b4::blend(0, w, quarters) + c1 + c2 < t;
+    const bool no_w = b4::blend(m, 0, quarters) + c1 + c2 < t;
+    const bool no_c = b < t;
+    const bool piv_m = !no_m;
+    const bool piv_w = !no_w;
+    const bool piv_c = !no_c;
+    if (piv_m) ++hp33_piv_m_;
+    if (piv_w) ++hp33_piv_w_;
+    if (piv_c) ++hp33_piv_c_;
+    if (no_m && no_w && no_c) ++hp33_none_;
+    if (m < 0) ++hp33_m_neg_;
+    if (w < 0) ++hp33_w_neg_;
+    if (c1 + c2 < 0) ++hp33_c_neg_;
+    hp33_sum_m_ += m;
+    hp33_sum_w_ += w;
+    hp33_sum_c_ += c1 + c2;
+    if (stale_m) ++hp33_fired_stale_m_;
+    if (stale_w) ++hp33_fired_stale_w_;
