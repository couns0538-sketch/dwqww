# setoption while a search is running must stop that search first.
position startpos
setoption name OwnBook value false
setoption name Threads value 1
go infinite
setoption name Hash value 32
go infinite
setoption name Threads value -1
go infinite
setoption name Not During Search value 1
go infinite
setoption name Threads value 1
setoption name Clear Hash
go movetime 10000
setoption name Move Overhead value 10
quit
