+// Hash of the minor-piece placements of BOTH sides (knights and bishops,
+// kings excluded), for the minor-piece correction history. The four placement
+// bitboards are multiplied by distinct odd constants derived from the golden
+// ratio constant, XOR-folded, then run through a xorshift-multiply finalizer;
+// every step is its own mixing choice, no table is needed.
+[[nodiscard]] std::uint64_t minor_key(const Board& board) {
+    constexpr std::uint64_t golden = 0x9E3779B97F4A7C15ULL;
+    std::uint64_t key = board.pieces(White, Knight) * golden;
+    key ^= board.pieces(White, Bishop) * (golden * 3);
+    key ^= board.pieces(Black, Knight) * (golden * 5);
+    key ^= board.pieces(Black, Bishop) * (golden * 7);
+    key ^= key >> 29U;
+    key *= golden * 9;
+    key ^= key >> 32U;
+    return key;
+}

+// The pieces of one side (kings excluded) that are attacked by a lesser enemy
+// piece: anything by an enemy pawn; a rook or queen by an enemy knight or
+// bishop; a queen by an enemy rook. Equal-value attackers do not count, and a
+// king is never a victim. Same notion of "cheaper" as quiet_threat_index.
+[[nodiscard]] Bitboard hanging_pieces(const Board& board, Color side) {
+    const Color them = opposite(side);
+    const Bitboard occ = board.occupied();
+    Bitboard victims = 0;
+    for (int t = Pawn; t <= Queen; ++t) {
+        victims |= board.pieces(side, static_cast<PieceType>(t));
+    }
+    Bitboard out = 0;
+    while (victims != 0) {
+        const int sq = pop_lsb(victims);
+        const Bitboard sqbb = square_bb(sq);
+        if ((attacks::pawn(side, sq) & board.pieces(them, Pawn)) != 0) {
+            out |= sqbb;
+            continue;
+        }
+        const int t = piece_type(board.piece_at(sq));
+        if (t == static_cast<int>(Rook) || t == static_cast<int>(Queen)) {
+            if ((attacks::knight(sq) & board.pieces(them, Knight)) != 0) {
+                out |= sqbb;
+                continue;
+            }
+            if ((attacks::bishop(sq, occ) & board.pieces(them, Bishop)) != 0) {
+                out |= sqbb;
+                continue;
+            }
+        }
+        if (t == static_cast<int>(Queen) && (attacks::rook(sq, occ) & board.pieces(them, Rook)) != 0) {
+            out |= sqbb;
+        }
+    }
+    return out;
+}

+// Index of the two hanging-sets (the side to move's first), mixed with odd

+[[nodiscard]] std::uint64_t threat_key(const Board& board) {

+    key ^= key >> 29U;
+    key *= golden * 9;

+    return key;
