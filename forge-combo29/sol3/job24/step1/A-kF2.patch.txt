+Board::QuietReference Board::job24_quiet_reference(const Move& move) const {
+    auto placement = squares_;
+    const Color us = side_;
+    const Color them = opposite(us);
+    const Piece moving = placement[move.from];
+    placement[move.from] = Empty;
+    if ((move.flags & EnPassant) != 0U) {
+        placement[move.to - 8 * pawn_forward(us)] = Empty;
+    }
+    if (move.is_castle()) {
+        const bool king_side = (move.flags & KingCastle) != 0U;
+        const int home = us == White ? 0 : 56;
+        const int rook_from = home + (king_side ? 7 : 0);
+        const int rook_to = home + (king_side ? 5 : 3);
+        placement[rook_to] = placement[rook_from];
+        placement[rook_from] = Empty;
+    }
+    placement[move.to] = move.is_promotion() ? make_piece(us, move.promotion) : moving;
+
+    const int own_king = moving == make_piece(us, King) ? static_cast<int>(move.to) :
+        std::countr_zero(pieces(us, King));
+    const int other_king = std::countr_zero(pieces(them, King));
+    const Bitboard attacking_other_king = raw_attackers(placement, other_king, us, false);
+    return {
+        raw_attackers(placement, own_king, them, true) == 0,
+        attacking_other_king != 0,
+        (attacking_other_king & bit_at(move.to)) != 0,
+    };
