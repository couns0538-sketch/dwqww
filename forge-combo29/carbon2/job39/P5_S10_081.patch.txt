+int Evaluator::pawn_structure_cp(const Board& board) {
+    PawnEntry& entry = pawn_cache_[static_cast<std::size_t>(board.pawn_key()) & (PawnCacheSize - 1U)];
+    if (!entry.valid || entry.key != board.pawn_key()) {
+        Accumulator<false> pawn_acc;
+        Bitboard white_passed = 0;
+        Bitboard black_passed = 0;
+        evaluate_pawn_structure<false, White>(board, pawn_acc, white_passed);
+        evaluate_pawn_structure<false, Black>(board, pawn_acc, black_passed);
+        entry.key = board.pawn_key();
+        entry.score = pawn_acc.score;
+        entry.passed = {white_passed, black_passed};
+        entry.valid = true;
+    }
+    int phase = 0;
+    for (int type = Knight; type <= Queen; ++type) {
+        phase += PhaseValue[static_cast<std::size_t>(type)] *
+                 (popcount(board.pieces(White, static_cast<PieceType>(type))) +
+                  popcount(board.pieces(Black, static_cast<PieceType>(type))));
+    }
+    phase = std::min(phase, MaxPhase);
+    const int mg = mg_value(entry.score);
+    const int eg = eg_value(entry.score);
+    const int tapered = (mg * phase + eg * (MaxPhase - phase)) / MaxPhase;
+    return board.side_to_move() == White ? tapered : -tapered;
+}

+int Worker::restricted_quiet_score(Board& board, SearchStack* ss) {
+    const Move saved_move = ss->current_move;
+    const Piece saved_piece = ss->moved_piece;
+    int best = -Infinite;
+    bool any = false;
+    const MoveList moves = board.legal_moves();
+    for (const Move& move : moves) {
+        if (move.is_capture() || move.is_promotion()) continue;
+        if (piece_type(board.moved_piece(move)) == Pawn) continue;
+        ss->current_move = move;
+        ss->moved_piece = board.moved_piece(move);
+        (ss + 1)->ply = ss->ply + 1;
+        record_move(ss, board, move);
+        Undo undo;
+        (void)board.make_move(move, undo);
+        const int child = static_evaluate(board, ss + 1);
+        board.unmake_move(move, undo);
+        if (is_mate_score(child)) continue;
+        const int score = -child;
+        if (score > best) best = score;
+        any = true;
+    }
+    ss->current_move = saved_move;
+    ss->moved_piece = saved_piece;
+    (ss + 1)->accumulator.valid = false;
+    (ss + 1)->runtime_psqt.valid = false;
+#if FORGE_NNUE_V3
+    if (v3_state_) v3_state_->valid_flags[static_cast<std::size_t>(ss->ply + 5)] = 0;
+#endif
+    return any ? best : ValueNone;
+}

+bool Worker::defender_holds(Board& board, SearchStack* ss, Color defender) {
+    auto holds = [&](Board& b, SearchStack* frame) {
+        const Move previous = frame->ply >= 2 ? (frame - 2)->current_move : Move::none();
+        const MoveList moves = b.legal_moves();
+        for (const Move& move : moves) {
+            const bool shuffle = !move.is_capture() && !move.is_promotion() &&
+                                 piece_type(b.moved_piece(move)) != Pawn;
+            if (!shuffle) continue;
+            if (!previous.is_none() && move.from == previous.to && move.to == previous.from) return true;
+            Undo undo;
+            (void)b.make_move(move, undo);
+            const bool repeated = b.is_repetition(root_history_index_) || b.is_repetition();
+            b.unmake_move(move, undo);
+            if (repeated) return true;
+        }
+        return false;
+    };
+    if (defender == board.side_to_move()) return holds(board, ss);
+    NullUndo undo;
+    board.make_null_move(undo);
+    (ss + 1)->ply = ss->ply + 1;
+    const bool ok = holds(board, ss + 1);
+    board.unmake_null_move(undo);
+    return ok;
+}

+bool Worker::idea_s10_applies(Board& board, SearchStack* ss, int depth, int eval) {
+    if (depth < tune::IdeaS10_Depth) return false;
+    const int look = tune::IdeaS10_Look;
+    if (look < 1 || ss->ply < look) return false;
+    if (board.halfmove_clock() < look) return false;
+    const int eval_abs = eval < 0 ? -eval : eval;
+    if (eval_abs < tune::IdeaS10_Margin) return false;
+    for (int k = 1; k <= look; ++k) {
+        if (!is_shuffle_piece_move((ss - k)->current_move, (ss - k)->moved_piece)) return false;
+    }
+    const int pawn_cp = evaluator_.pawn_structure_cp(board);
+    const int pawn_abs = pawn_cp < 0 ? -pawn_cp : pawn_cp;
+    if (pawn_abs < tune::IdeaS10_Margin) return false;
+    if ((pawn_cp > 0) != (eval > 0)) return false;
+
+    const bool winner_to_move = pawn_cp > 0;
+    int probe = ValueNone;
+    if (winner_to_move) {
+        probe = restricted_quiet_score(board, ss);
+    } else {
+        NullUndo undo;
+        board.make_null_move(undo);
+        record_null_move(ss);
+        (ss + 1)->ply = ss->ply + 1;
+        const int theirs = restricted_quiet_score(board, ss + 1);
+        board.unmake_null_move(undo);
+        (ss + 1)->accumulator.valid = false;
+        (ss + 1)->runtime_psqt.valid = false;
+#if FORGE_NNUE_V3
+        if (v3_state_) v3_state_->valid_flags[static_cast<std::size_t>(ss->ply + 5)] = 0;
+#endif
+        if (theirs != ValueNone) probe = -theirs;
+    }
+    if (probe == ValueNone || is_mate_score(probe)) return false;
+    const int gap = probe > eval ? probe - eval : eval - probe;
+    if (gap > tune::IdeaS10_Flat) return false;
+    const Color defender = winner_to_move ? opposite(board.side_to_move()) : board.side_to_move();
+    return defender_holds(board, ss, defender);
+}
