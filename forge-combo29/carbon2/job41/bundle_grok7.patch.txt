+int Evaluator::pawn_structure_cp(const Board& board) {
+    PawnEntry& entry = pawn_cache_[static_cast<std::size_t>(board.pawn_key()) & (PawnCacheSize - 1U)];
+    if (!entry.valid || entry.key != board.pawn_key()) {
+        Accumulator<false> pawn_acc;
+        Bitboard white_passed = 0;
+        Bitboard black_passed = 0;
+        evaluate_pawn_structure<false, White>(board, pawn_acc, white_passed);
+        evaluate_pawn_structure<false, Black>(board, pawn_acc, black_passed);
+        entry.key = board.pawn_key();
+        entry.score = pawn_acc.score;
+        entry.passed = {white_passed, black_passed};
+        entry.valid = true;
+    }
+    int phase = 0;
+    for (int type = Knight; type <= Queen; ++type) {
+        phase += PhaseValue[static_cast<std::size_t>(type)] *
+                 (popcount(board.pieces(White, static_cast<PieceType>(type))) +
+                  popcount(board.pieces(Black, static_cast<PieceType>(type))));
+    }
+    phase = std::min(phase, MaxPhase);
+    const int mg = mg_value(entry.score);
+    const int eg = eg_value(entry.score);
+    const int tapered = (mg * phase + eg * (MaxPhase - phase)) / MaxPhase;
+    return board.side_to_move() == White ? tapered : -tapered;
+}

+// IdeaB1. Quiet-move ordering prior from the current placement, not a
+// piece-square table. Pressure is attacks on the enemy army (weight 2) plus
+// attacks into the enemy king zone (weight 1). The bonus is that change from
+// the move's origin to its destination, then shrunk as this search writes
+// main history for the same from-to: Scale * delta * 1024 / (1024 + Fade * |h|).
+// Callers add it only to quiet scores, after hash, capture, and killer slots.
+[[nodiscard]] int fading_board_prior(const Board& board, Move move, int main_history) {
+    const int type = piece_type(board.piece_at(move.from));
+    const Color us = board.side_to_move();
+    const Color them = opposite(us);
+    const Bitboard enemy = board.occupied(them);
+    const int their_king = board.king_square(them);
+    const Bitboard zone = their_king >= 0 ? attacks::king_zone(their_king) : Bitboard{0};
+    const Bitboard occ = board.occupied();
+    const Bitboard occ_to = (occ ^ square_bb(move.from)) | square_bb(move.to);
+    auto pressure = [&](int square, Bitboard occupancy) {
+        Bitboard att = 0;
+        switch (type) {
+        case Pawn:
+            att = attacks::pawn(us, square);
+            break;
+        case Knight:
+            att = attacks::knight(square);
+            break;
+        case Bishop:
+            att = attacks::bishop(square, occupancy);
+            break;
+        case Rook:
+            att = attacks::rook(square, occupancy);
+            break;
+        case Queen:
+            att = attacks::queen(square, occupancy);
+            break;
+        case King:
+            att = attacks::king(square);
+            break;
+        default:
+            att = 0;
+            break;
+        }
+        return 2 * popcount(att & enemy) + popcount(att & zone);
+    };
+    const int delta = pressure(move.to, occ_to) - pressure(move.from, occ);
+    const int mag = main_history >= 0 ? main_history : -main_history;
+    const int denom = 1024 + tune::IdeaB1_Fade * mag;
+    if (denom <= 0) return 0;
+    return delta * tune::IdeaB1_Scale * 1024 / denom;
+}

+// pieces that a cheaper enemy piece attacks. The attacker is the cheapest such
+// piece, lowest square on a tie. Kings are not attackers (SEE prices a king at 0)
+// and pawns are not victims (nothing material is cheaper than a pawn).
+struct RoleThreat {
+    bool present = false;
+    int victim_sq = 0;
+    int attacker_sq = 0;
+    int victim_type = 0;
+    int attacker_type = 0;
+    int victim_value = 0;

+[[nodiscard]] Bitboard attacks_after_quiet(const Board& board, Move move) {
+    const int mover = piece_type(board.piece_at(move.from));
+    const Bitboard occ = board.occupied() ^ square_bb(move.from);
+    const Color us = board.side_to_move();
+    if (mover == Pawn) return attacks::pawn(us, move.to);
+    if (mover == Knight) return attacks::knight(move.to);
+    if (mover == Bishop) return attacks::bishop(move.to, occ);
+    if (mover == Rook) return attacks::rook(move.to, occ);
+    if (mover == Queen) return attacks::queen(move.to, occ);
+    if (mover == King) return attacks::king(move.to);
+    return 0;
+}

+[[nodiscard]] RoleThreat find_role_threat(const Board& board) {
+    RoleThreat best;
+    const Color us = board.side_to_move();
+    const Color them = opposite(us);
+    const Bitboard enemy = board.occupied(them);
+    for (int type = Knight; type <= Queen; ++type) {
+        const int victim_value = SeePieceValue[static_cast<std::size_t>(type)];
+        Bitboard ours = board.pieces(us, static_cast<PieceType>(type));
+        while (ours != 0) {
+            const int square = pop_lsb(ours);
+            Bitboard hits = board.attackers_to(square) & enemy;
+            int cheap_sq = -1;
+            int cheap_value = 0;
+            int cheap_type = 0;
+            while (hits != 0) {
+                const int from = pop_lsb(hits);
+                const int attacker_type = piece_type(board.piece_at(from));
+                const int attacker_value = SeePieceValue[static_cast<std::size_t>(attacker_type)];
+                if (attacker_value <= 0 || attacker_value >= victim_value) continue;
+                if (cheap_sq < 0 || attacker_value < cheap_value ||
+                    (attacker_value == cheap_value && from < cheap_sq)) {
+                    cheap_sq = from;
+                    cheap_value = attacker_value;
+                    cheap_type = attacker_type;
+                }
+            }
+            if (cheap_sq < 0) continue;
+            if (!best.present || victim_value > best.victim_value ||
+                (victim_value == best.victim_value && square < best.victim_sq)) {
+                best.present = true;
+                best.victim_sq = square;
+                best.attacker_sq = cheap_sq;
+                best.victim_type = type;
+                best.attacker_type = cheap_type;
+                best.victim_value = victim_value;
+            }
+        }
+    }
+    return best;

+// One role per quiet. First match wins: step away, block a slider, attack the
+// attacker, attack a dearer enemy piece, otherwise none. This engine has no
+// piece-value bonus for stepping off an attacked square, so there is nothing
+// to subtract; the step-away role is only a learned bucket.
+[[nodiscard]] int quiet_threat_role(const Board& board, Move move, const RoleThreat& threat) {
+    if (!threat.present) return -1;
+    if (move.from == threat.victim_sq) return 0;
+    if (threat.attacker_type == Bishop || threat.attacker_type == Rook || threat.attacker_type == Queen) {
+        if ((attacks::between(threat.attacker_sq, threat.victim_sq) & square_bb(move.to)) != 0) return 1;
+    }
+    const Bitboard hits = attacks_after_quiet(board, move);
+    if ((hits & square_bb(threat.attacker_sq)) != 0) return 2;
+    const Color them = opposite(board.side_to_move());
+    for (int type = Queen; type >= Knight; --type) {
+        if (SeePieceValue[static_cast<std::size_t>(type)] <= threat.victim_value) break;
+        if ((hits & board.pieces(them, static_cast<PieceType>(type))) != 0) return 3;

+    return 4;

+int Worker::threat_role_b8b_of(const Board& board, Move move, const RoleThreatB& threat) {
+    if (!threat.present) return 0;

+    Bitboard attacks_bb = 0;

+    case Rook:
+        attacks_bb = attacks::rook(move.to, occupancy);
+        break;

+    case King:

+        break;

+        break;
+    }
+    }

+    const int mover_worth = SeePieceValue[static_cast<std::size_t>(mover_type)];
+    if (mover_worth <= 0) return 0;

+    Bitboard targets = attacks_bb & them;
+    while (targets != 0) {
+        const int square = pop_lsb(targets);
+        const int enemy_type = piece_type(board.piece_at(square));
+        const int enemy_worth = enemy_type == King ? 100000 : SeePieceValue[static_cast<std::size_t>(enemy_type)];
