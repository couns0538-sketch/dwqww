+// One cheaper-attacker threat in the position. The victim is the dearest of our
+// pieces that a cheaper enemy piece attacks. The attacker is the cheapest such
+// piece, lowest square on a tie. Kings are not attackers (SEE prices a king at 0)
+// and pawns are not victims (nothing material is cheaper than a pawn).
+struct RoleThreat {
+    bool present = false;
+    int victim_sq = 0;
+    int attacker_sq = 0;
+    int victim_type = 0;
+    int attacker_type = 0;
+    int victim_value = 0;
+};

+[[nodiscard]] Bitboard attacks_after_quiet(const Board& board, Move move) {
+    const int mover = piece_type(board.piece_at(move.from));
+    const Bitboard occ = board.occupied() ^ square_bb(move.from);
+    const Color us = board.side_to_move();
+    if (mover == Pawn) return attacks::pawn(us, move.to);
+    if (mover == Knight) return attacks::knight(move.to);
+    if (mover == Bishop) return attacks::bishop(move.to, occ);
+    if (mover == Rook) return attacks::rook(move.to, occ);
+    if (mover == Queen) return attacks::queen(move.to, occ);
+    if (mover == King) return attacks::king(move.to);
+    return 0;
+}

+[[nodiscard]] RoleThreat find_role_threat(const Board& board) {
+    RoleThreat best;
+    const Color us = board.side_to_move();
+    const Color them = opposite(us);
+    const Bitboard enemy = board.occupied(them);
+    for (int type = Knight; type <= Queen; ++type) {
+        const int victim_value = SeePieceValue[static_cast<std::size_t>(type)];
+        Bitboard ours = board.pieces(us, static_cast<PieceType>(type));
+        while (ours != 0) {
+            const int square = pop_lsb(ours);
+            Bitboard hits = board.attackers_to(square) & enemy;
+            int cheap_sq = -1;
+            int cheap_value = 0;
+            int cheap_type = 0;
+            while (hits != 0) {
+                const int from = pop_lsb(hits);
+                const int attacker_type = piece_type(board.piece_at(from));
+                const int attacker_value = SeePieceValue[static_cast<std::size_t>(attacker_type)];
+                if (attacker_value <= 0 || attacker_value >= victim_value) continue;
+                if (cheap_sq < 0 || attacker_value < cheap_value ||
+                    (attacker_value == cheap_value && from < cheap_sq)) {
+                    cheap_sq = from;
+                    cheap_value = attacker_value;
+                    cheap_type = attacker_type;
+                }
+            }
+            if (cheap_sq < 0) continue;
+            if (!best.present || victim_value > best.victim_value ||
+                (victim_value == best.victim_value && square < best.victim_sq)) {
+                best.present = true;
+                best.victim_sq = square;
+                best.attacker_sq = cheap_sq;
+                best.victim_type = type;
+                best.attacker_type = cheap_type;
+                best.victim_value = victim_value;
+            }
+        }
+    }
+    return best;
+}

+// One role per quiet. First match wins: step away, block a slider, attack the
+// attacker, attack a dearer enemy piece, otherwise none. This engine has no
+// piece-value bonus for stepping off an attacked square, so there is nothing
+// to subtract; the step-away role is only a learned bucket.
+[[nodiscard]] int quiet_threat_role(const Board& board, Move move, const RoleThreat& threat) {
+    if (!threat.present) return -1;
+    if (move.from == threat.victim_sq) return 0;
+    if (threat.attacker_type == Bishop || threat.attacker_type == Rook || threat.attacker_type == Queen) {
+        if ((attacks::between(threat.attacker_sq, threat.victim_sq) & square_bb(move.to)) != 0) return 1;
+    }
+    const Bitboard hits = attacks_after_quiet(board, move);
+    if ((hits & square_bb(threat.attacker_sq)) != 0) return 2;
+    const Color them = opposite(board.side_to_move());
+    for (int type = Queen; type >= Knight; --type) {
+        if (SeePieceValue[static_cast<std::size_t>(type)] <= threat.victim_value) break;
+        if ((hits & board.pieces(them, static_cast<PieceType>(type))) != 0) return 3;
+    }
+    return 4;
+}

+[[nodiscard]] RoleThreat cached_role_threat(bool present, int victim_sq, int attacker_sq, int victim_type,
+                                            int attacker_type, int victim_value) {
+    RoleThreat threat;

+    threat.victim_sq = victim_sq;
+    threat.attacker_sq = attacker_sq;
+    threat.victim_type = victim_type;
+    threat.attacker_type = attacker_type;
+    threat.victim_value = victim_value;
+    return threat;
+}

+int Worker::threat_role_bonus(const Board& board, Move move) const {
+    if (!tune::IdeaB8) return 0;
+    ensure_role_threat(board);
+    if (!threat_role_present_) return 0;
+    const RoleThreat threat = cached_role_threat(threat_role_present_, threat_role_victim_sq_, threat_role_attacker_sq_,
+                                                 threat_role_victim_type_, threat_role_attacker_type_,
+                                                 threat_role_victim_value_);
+    const int role = quiet_threat_role(board, move, threat);
+    if (role < 0 || role > 4) return 0;
+    if (threat.victim_type < 0 || threat.victim_type > 6 || threat.attacker_type < 0 || threat.attacker_type > 6) return 0;
+    const int raw = (*threat_role_history_)[static_cast<std::size_t>(board.side_to_move())]
+                                           [static_cast<std::size_t>(threat.victim_type)]
+                                           [static_cast<std::size_t>(threat.attacker_type)]
+                                           [static_cast<std::size_t>(role)];
+    return raw * tune::IdeaB8_Weight / 1024;
+}

+    if (threat.victim_type < 0 || threat.victim_type > 6 || threat.attacker_type < 0 || threat.attacker_type > 6) return;
+    apply_history((*threat_role_history_)[static_cast<std::size_t>(board.side_to_move())]
+                                         [static_cast<std::size_t>(threat.victim_type)]
+                                         [static_cast<std::size_t>(threat.attacker_type)]
+                                         [static_cast<std::size_t>(role)],
+                  bonus);

+void Worker::note_threat_swing(const Board& board, SearchStack* ss, Move move, int depth) {
+    if (!tune::IdeaB8) return;
+    if (!(ss + 1)->role_eval_ready) return;
+    if (ss->static_eval == ValueNone || (ss + 1)->static_eval == ValueNone) return;
+    // The child score is from the opponent's side. A drop there is a gain for us.
+    const int gained = -(ss + 1)->static_eval - ss->static_eval;
+    if (gained < tune::IdeaB8_Swing) return;
+    update_threat_role(board, move, history_bonus(std::max(1, depth)) / 2);
+}

+const Worker::RoleThreatB& Worker::role_b8b_threat(const Board& board) const {
+    const std::uint64_t key = board.key();
+    if (role_b8b_key_valid_ && role_b8b_key_ == key) return role_b8b_threat_;
+    role_b8b_key_ = key;
+    role_b8b_key_valid_ = true;
+    RoleThreatB best;
+    int best_victim_worth = 0;
+    int best_attacker_worth = 0;
+    constexpr int KingWorth = 100000;
+    const Color us = board.side_to_move();
+    const Bitboard enemy = board.occupied(opposite(us));
+    Bitboard ours = board.pieces(us, Knight) | board.pieces(us, Bishop) | board.pieces(us, Rook) |
+                    board.pieces(us, Queen) | board.pieces(us, King);
+    while (ours != 0) {
+        const int square = pop_lsb(ours);
+        const int victim_type = piece_type(board.piece_at(square));
+        const int victim_worth = victim_type == King ? KingWorth : SeePieceValue[static_cast<std::size_t>(victim_type)];
+        Bitboard hits = board.attackers_to(square) & enemy;
+        while (hits != 0) {
+            const int from = pop_lsb(hits);
+            const int attacker_type = piece_type(board.piece_at(from));
+            const int attacker_worth = SeePieceValue[static_cast<std::size_t>(attacker_type)];
+            if (attacker_worth <= 0 || attacker_worth >= victim_worth) continue;
+            const bool better = !best.present || victim_worth > best_victim_worth ||
+                                (victim_worth == best_victim_worth &&
+                                 (attacker_worth < best_attacker_worth ||
+                                  (attacker_worth == best_attacker_worth &&
+                                   (square < best.victim_sq || (square == best.victim_sq && from < best.attacker_sq)))));
+            if (!better) continue;
+            best.present = true;
+            best.victim_sq = square;
+            best.attacker_sq = from;
+            best.victim_type = victim_type;
+            best.attacker_type = attacker_type;
+            best.victim_worth = victim_worth;
+            best_victim_worth = victim_worth;
+            best_attacker_worth = attacker_worth;
+        }
+    }
+    role_b8b_threat_ = best;
+    return role_b8b_threat_;
+}

+int Worker::threat_role_b8b_of(const Board& board, Move move, const RoleThreatB& threat) {
+    if (!threat.present) return 0;
+    if (move.from == threat.victim_sq) return 1;
+    if (threat.attacker_type == Bishop || threat.attacker_type == Rook || threat.attacker_type == Queen) {
+        if ((attacks::between(threat.attacker_sq, threat.victim_sq) & square_bb(move.to)) != 0) return 2;
+    }
+    const int mover_type = piece_type(board.piece_at(move.from));
+    const Color us = board.side_to_move();
+    const Bitboard occupancy = (board.occupied() ^ square_bb(move.from)) | square_bb(move.to);
+    Bitboard attacks_bb = 0;
+    switch (mover_type) {
+    case NoPieceType:
+        break;
+    case Pawn:
+        attacks_bb = attacks::pawn(us, move.to);
+        break;
+    case Knight:
+        attacks_bb = attacks::knight(move.to);
+        break;
+    case Bishop:
+        attacks_bb = attacks::bishop(move.to, occupancy);
+        break;
+    case Rook:
+        attacks_bb = attacks::rook(move.to, occupancy);
+        break;
+    case Queen:
+        attacks_bb = attacks::queen(move.to, occupancy);
+        break;
+    case King:
+        attacks_bb = attacks::king(move.to);
+        break;
+    default: {
+        const int unexpected = mover_type;
+        (void)unexpected;
+        break;
+    }
+    }
+    if ((attacks_bb & square_bb(threat.attacker_sq)) != 0) return 3;
+    const int mover_worth = SeePieceValue[static_cast<std::size_t>(mover_type)];
+    if (mover_worth <= 0) return 0;
+    const Bitboard them = board.occupied(opposite(us));
+    Bitboard targets = attacks_bb & them;
+    while (targets != 0) {
+        const int square = pop_lsb(targets);
+        const int enemy_type = piece_type(board.piece_at(square));
+        const int enemy_worth = enemy_type == King ? 100000 : SeePieceValue[static_cast<std::size_t>(enemy_type)];
+        if (enemy_worth > threat.victim_worth && mover_worth < enemy_worth) return 4;
+    }
+    return 0;
+}

+int Worker::threat_role_b8b_bonus(const Board& board, Move move) const {

+void Worker::note_role_b8b_swing(const Board& board, SearchStack* ss, Move move, int depth) {
