+const Worker::B8FastEntry& Worker::ensure_b8_fast(const Board& board) const {
+    const std::uint64_t key = board.key();
+    B8FastEntry& slot = b8_fast_cache_[static_cast<std::size_t>(key & 255ULL)];
+    if (slot.key == key && slot.occ_w == board.occupied(White) && slot.occ_b == board.occupied(Black)) {
