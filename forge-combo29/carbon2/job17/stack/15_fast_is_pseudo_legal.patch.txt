+inline bool Board::is_pseudo_legal(const Move& move) const {
+    if ((move.from | move.to) >= 64U || move.from == move.to) return false;
+    const Color us = side_;
+    const Piece moving = squares_[move.from];
+    if (moving == Empty || color_of(moving) != us) return false;
+    constexpr std::uint8_t kSpecialFlags = DoublePush | KingCastle | QueenCastle | EnPassant | Promotion;
+    if ((move.flags & kSpecialFlags) != 0U) return is_pseudo_legal_special(move);
+    if (move.promotion != 0) return false;
+    const Piece target = squares_[move.to];
+    const bool is_cap = (move.flags & Capture) != 0U;
+    if (is_cap != (target != Empty)) return false;
+    if (is_cap && (color_of(target) == us || piece_type(target) == King)) return false;
+    const int type = piece_type(moving);
+    const Bitboard to_bb = square_bb(move.to);
+    if (type == Pawn) {
+        if (relative_rank(us, move.from) == 6) return false;
+        if (is_cap) return (attacks::pawn(us, move.from) & to_bb) != 0;
+        return static_cast<int>(move.to) == static_cast<int>(move.from) + pawn_push_delta(us);
+    }
+    if (type == Knight) return (attacks::knight(move.from) & to_bb) != 0;
+    if (type == King) return (attacks::king(move.from) & to_bb) != 0;
+    const bool diag_ok = (type != Rook) && (attacks::bishop_ray(move.from) & to_bb) != 0;
+    const bool orth_ok = (type != Bishop) && (attacks::rook_ray(move.from) & to_bb) != 0;
+    return (diag_ok || orth_ok) && (attacks::between(move.from, move.to) & occupied()) == 0;
