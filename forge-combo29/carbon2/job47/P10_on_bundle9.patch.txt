+int Worker::role_b8b_fast(const Board& b, Piece moved_piece, Move m, const B8bFastEntry& e) {
+    if (m.from == e.victim_sq) return 1;
+    if ((e.between_bb & square_bb(m.to)) != 0) return 2;
+    const int mover = piece_type(moved_piece);
+    const Bitboard occ = b.occupied() ^ square_bb(m.from);
+    Bitboard hits = 0;
+    switch (mover) {
+    case Pawn: hits = attacks::pawn(b.side_to_move(), m.to); break;
+    case Knight: hits = attacks::knight(m.to); break;
+    case Bishop: hits = attacks::bishop(m.to, occ); break;
+    case Rook: hits = attacks::rook(m.to, occ); break;
+    case Queen: hits = attacks::queen(m.to, occ); break;
+    case King: hits = attacks::king(m.to); break;
+    default: return 0;
+    }
+    if ((hits & e.attacker_bb) != 0) return 3;
+    if ((hits & e.role4_mask[static_cast<std::size_t>(mover)]) != 0) return 4;
+    return 0;
+}

+const Worker::B8bFastEntry& Worker::ensure_b8b_fast(const Board& board) const {
+    const std::uint64_t key = board.key();
+    B8bFastEntry& slot = b8b_fast_cache_[static_cast<std::size_t>(key & 255ULL)];
+    if (slot.key == key && slot.occ_w == board.occupied(White) && slot.occ_b == board.occupied(Black)) {
+        return slot;
+    }
+    slot = make_b8b_fast_entry(board);
+    return slot;
+}

+const Worker::B8bFastEntry* Worker::prepare_b8b_fast_scorer(const Board& board, std::array<int, 5>& scaled) const {
+    const B8bFastEntry& e = ensure_b8b_fast(board);
+    if (!e.present) return nullptr;
+    const auto& row = (*role_b8b_history_)[static_cast<std::size_t>(board.side_to_move())]
+                                          [static_cast<std::size_t>(e.victim_type)]
+                                          [static_cast<std::size_t>(e.attacker_type)];
+    const int w = tune::IdeaB8b_Weight;
+    for (std::size_t r = 0; r < 5; ++r) {
+        scaled[r] = row[r] * w / 1024;
+    }
+    return &e;
+}

+int Worker::threat_role_b8b_bonus_fast(const Board& board, Piece piece, Move move) const {
+    const B8bFastEntry& e = ensure_b8b_fast(board);
+    if (!e.present) return 0;

+                                        [static_cast<std::size_t>(e.victim_type)]
+                                        [static_cast<std::size_t>(e.attacker_type)]
+                                        [static_cast<std::size_t>(role)];
+    return raw * tune::IdeaB8b_Weight / 1024;
+}
