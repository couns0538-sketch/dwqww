+std::string Board::sg_fen() const {
+    char buf[128];
+    char* p = buf;
+    for (int rank = 7; rank >= 0; --rank) {
+        int empty = 0;
+        for (int file = 0; file < 8; ++file) {
+            const Piece piece = squares_[static_cast<std::size_t>(rank * 8 + file)];
+            if (piece == Empty) {
+                ++empty;
+            } else {
+                if (empty != 0) *p++ = static_cast<char>('0' + empty);
+                empty = 0;
+                *p++ = piece_to_char(piece);
+            }
+        }
+        if (empty != 0) *p++ = static_cast<char>('0' + empty);
+        if (rank != 0) *p++ = '/';
+    }
+    *p++ = ' ';
+    *p++ = (side_ == White) ? 'w' : 'b';
+    *p++ = ' ';
+    if (castling_ == 0) {
+        *p++ = '-';
+    } else {
+        if ((castling_ & WhiteKingSide) != 0U) *p++ = 'K';
+        if ((castling_ & WhiteQueenSide) != 0U) *p++ = 'Q';
+        if ((castling_ & BlackKingSide) != 0U) *p++ = 'k';
+        if ((castling_ & BlackQueenSide) != 0U) *p++ = 'q';
+    }
+    *p++ = ' ';
+    if (ep_square_ < 0 || !has_legal_en_passant_capture()) {
+        *p++ = '-';
+    } else {
+        *p++ = static_cast<char>('a' + file_of(ep_square_));
+        *p++ = static_cast<char>('1' + rank_of(ep_square_));
+    }
+    *p++ = ' ';
+    const auto put_u16 = [&](std::uint16_t val) {
+        char digits[8];
+        int nd = 0;
+        do {
+            digits[nd++] = static_cast<char>('0' + (val % 10U));
+            val = static_cast<std::uint16_t>(val / 10U);
+        } while (val != 0U);
+        while (nd > 0) *p++ = digits[--nd];
+    };
+    put_u16(halfmove_);
+    *p++ = ' ';
+    put_u16(fullmove_);
+    return std::string(buf, static_cast<std::size_t>(p - buf));
+}

+bool Board::sg_insufficient_material() const {
+    const auto side_without_mating = [&](Color c) -> bool {
+        if ((pieces(c, Pawn) | pieces(c, Rook) | pieces(c, Queen)) != 0) return false;
+        const Color them = opposite(c);
+        if (pieces(c, Knight) != 0) {
+            return popcount(occupied(c)) <= 2 &&
+                   (occupied(them) & ~(pieces(them, King) | pieces(them, Queen))) == 0;
+        }
+        if (pieces(c, Bishop) != 0) {
+            constexpr Bitboard DarkSquares = 0xAA55AA55AA55AA55ULL;
+            constexpr Bitboard LightSquares = ~DarkSquares;
+            const Bitboard bishops = pieces(Bishop);
+            const bool one_color = (bishops & DarkSquares) == 0 || (bishops & LightSquares) == 0;
+            return one_color && (pieces(Pawn) | pieces(Knight)) == 0;
+        }
+        return true;
+    };
+    return side_without_mating(White) && side_without_mating(Black);
+}

+[[nodiscard]] bool sg_is_ws(char c) noexcept {
+    return c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\v' || c == '\f';
+}

+[[nodiscard]] std::string_view sg_next_token(std::string_view& sv) noexcept {
+    std::size_t i = 0;
+    while (i < sv.size() && sg_is_ws(sv[i])) ++i;
+    sv.remove_prefix(i);
+    if (sv.empty()) return {};
+    std::size_t j = 0;
+    while (j < sv.size() && !sg_is_ws(sv[j])) ++j;
+    const std::string_view tok = sv.substr(0, j);
+    sv.remove_prefix(j);
+    return tok;
+}

+[[nodiscard]] bool sg_parse_int(std::string_view tok, int& out) noexcept {
+    if (tok.empty()) return false;
+    long long wide = 0;
+    const char* first = tok.data() + (tok[0] == '+' ? 1 : 0);
+    if (first >= tok.data() + tok.size()) return false;
+    const auto res = std::from_chars(first, tok.data() + tok.size(), wide);
+    if (res.ptr != tok.data() + tok.size()) return false;
+    if (res.ec == std::errc::result_out_of_range) {
+        wide = tok[0] == '-' ? std::numeric_limits<int>::min() : std::numeric_limits<int>::max();
+    } else if (res.ec != std::errc{}) {
+        return false;
+    }
+    out = static_cast<int>(std::clamp<long long>(wide, std::numeric_limits<int>::min(),
+                                                 std::numeric_limits<int>::max()));
+    return true;
+}
+
+[[nodiscard]] bool sg_parse_u64(std::string_view tok, std::uint64_t& out) noexcept {
+    if (tok.empty()) return false;
+    if (tok[0] == '-') {
+        int dummy = 0;
+        if (!sg_parse_int(tok, dummy)) return false;
+        out = 0;
+        return true;
+    }
+    const char* first = tok.data() + (tok[0] == '+' ? 1 : 0);
+    if (first >= tok.data() + tok.size()) return false;
+    std::uint64_t val = 0;
+    const auto res = std::from_chars(first, tok.data() + tok.size(), val);
+    if (res.ptr != tok.data() + tok.size()) return false;
+    if (res.ec == std::errc::result_out_of_range) {
+        val = std::numeric_limits<std::uint64_t>::max();
+    } else if (res.ec != std::errc{}) {
+        return false;
+    }
+    out = val;
+    return true;
+}

+[[nodiscard]] SgConfig parse_selfgame_config(std::string_view line) noexcept {
+    SgConfig cfg;
+    (void)sg_next_token(line);
+    for (;;) {
+        const std::string_view key = sg_next_token(line);
+        if (key.empty()) break;
+        const std::string_view val = sg_next_token(line);
+        if (val.empty()) break;
+        if (key == "softnodes") {
+            std::uint64_t parsed = 0;
+            if (sg_parse_u64(val, parsed)) cfg.soft_nodes = parsed;
+        } else if (key == "evallimit") {
+            int parsed = 0;
+            if (sg_parse_int(val, parsed)) cfg.eval_limit = parsed;
+        } else if (key == "maxply") {
+            int parsed = 0;
+            if (sg_parse_int(val, parsed)) cfg.max_ply = parsed;
+        } else if (key == "writeminply") {
+            int parsed = 0;
+            if (sg_parse_int(val, parsed)) cfg.write_min_ply = parsed;
+        } else if (key == "drawply") {
+            int parsed = 0;
+            if (sg_parse_int(val, parsed)) cfg.draw_ply = parsed;
+        } else if (key == "drawscore") {
+            int parsed = 0;

+        } else if (key == "drawcount") {
+            int parsed = 0;

+        }
+    }
+    return cfg;

+void run_selfgame(forge::Board& board, forge::Searcher& searcher, int move_overhead_ms, int multipv,
+                  const SgConfig& cfg) {
+    struct SgRecord {
+        int ply = 0;
+        int score = 0;
+        std::string move;
+        std::string fen;
+    };
+
+    int ply = static_cast<int>(board.history_size() > 0 ? board.history_size() - 1 : 0);
+    int run = 0;
+    std::uint64_t searches = 0;
+    std::uint64_t total_nodes = 0;
+    std::vector<SgRecord> records;
+    records.reserve(256);
+    std::vector<std::string> loop_moves;
+    loop_moves.reserve(256);
+
+    int end_result = 0;
+    const char* end_reason = "plylimit";
+
+    for (;;) {
+        const SgState st = evaluate_sg_state(board);

+            if (!st.check) {

+                end_reason = "stalemate";
+            } else {

+                end_reason = "mate";
+            }
+            break;
+        }
+
+        const bool rule_draw = st.material || st.clock >= 100 || st.rep3;
+        if (rule_draw || ply >= cfg.max_ply) {

+        }

+        limits.nodes_specified = true;

+        limits.multipv = multipv;

+
