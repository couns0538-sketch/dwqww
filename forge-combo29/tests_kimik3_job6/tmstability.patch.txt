+// TmStability: an extra soft-limit scale from how settled the root search is,
+// on top of the factors the engine already applies. Two readings, one number.
+//
+// (a) Best-move stability: stable_iterations counts finished iterations in a
+// row that kept the same best move. A move that just changed (0) gets
+// unstable_pct percent of the soft target; one stable for 8 or more iterations
+// gets stable_pct percent; between them the scale falls linearly.
+//
+// (b) Score trend: score_drop_cp is how much the score fell against the
+// previous iteration (positive when falling, 0 otherwise), capped at 200 cp.
+// Each 100 cp of drop adds fall_per100_pct percent on top.
+//
+// The product is clamped to [min_pct, max_pct] percent and returned in
+// thousandths (1000 = soft target unchanged). Constants are the author's
+// first guess, deliberately milder than the engine's always-on stability and
+// falling-eval terms (which stay in place either way).
+[[nodiscard]] constexpr std::int64_t tm_stability_thousandths(std::int64_t stable_iterations, std::int64_t score_drop_cp,
+                                                              std::int64_t unstable_pct, std::int64_t stable_pct,
+                                                              std::int64_t fall_per100_pct, std::int64_t min_pct,
+                                                              std::int64_t max_pct) noexcept {
+    std::int64_t stab = stable_pct;
+    if (stable_iterations <= 0) stab = unstable_pct;
+    else if (stable_iterations < 8) stab = unstable_pct + (stable_pct - unstable_pct) * stable_iterations / 8;
+    std::int64_t drop = score_drop_cp;
+    if (drop < 0) drop = 0;
+    if (drop > 200) drop = 200;
+    const std::int64_t total = stab * (1000 + fall_per100_pct * 10 * drop / 100) / 1000;
+    if (total < min_pct) return min_pct * 10;
+    if (total > max_pct) return max_pct * 10;
+    return total * 10;
+}
