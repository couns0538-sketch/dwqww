+void fill(Event& e, int a, bool quiet, bool tt_pv, bool cut_node, int depth, int move_count, bool not_improving,
+          bool tt_capture_quiet, bool gives_check, bool in_check_node, bool cutoff_fired, int hist_value,
+          std::uint64_t n0) {
+    e.a = a;
+    e.quiet = quiet;
+    e.kind = tt_pv ? 0 : (cut_node ? 1 : 2);
+    e.db = depth <= 3 ? 0 : depth <= 6 ? 1 : depth <= 9 ? 2 : 3;
+    e.mb = move_count <= 4 ? 0 : move_count <= 8 ? 1 : move_count <= 16 ? 2 : 3;
+    e.hist = quiet ? hist_value : HIST_NA;
+    e.fired[0] = tt_pv;
+    e.fired[1] = not_improving;
+    e.fired[2] = cut_node;
+    e.fired[3] = tt_capture_quiet;
+    e.fired[4] = gives_check;
+    e.fired[5] = in_check_node;
+    e.fired[6] = cutoff_fired;
+    e.n0 = n0;
+}

+void after_reduced(Event& e, bool fh, std::uint64_t now) {
+    e.fh = fh;
+    e.nodes_red = now - e.n0;

+void after_research(Event& e, bool conf, std::uint64_t nodes) {

+void record(const Event& e) {
+    const int ac = std::min(e.a, 5);
+    const int q = e.quiet ? 1 : 0;
+    const bool conf = e.researched ? e.conf : (e.a == 0 && e.fh);
+    ++passes;
+    research2023 += e.researched ? 1 : 0;
+    ++cell_n[ac][q][e.kind][e.db][e.mb];
+    cell_fh[ac][q][e.kind][e.db][e.mb] += e.fh ? 1 : 0;
+    cell_conf[ac][q][e.kind][e.db][e.mb] += conf ? 1 : 0;
+    cell_rn[ac][q][e.kind][e.db][e.mb] += e.nodes_red;
+    cell_re[ac][q][e.kind][e.db][e.mb] += e.nodes_re;
+    for (int t = 0; t < 7; ++t) {
+        const int f = e.fired[t] ? 1 : 0;
+        ++term_n[t][ac][f];
+        term_fh[t][ac][f] += e.fh ? 1 : 0;
+        term_conf[t][ac][f] += conf ? 1 : 0;
+    }
+    if (e.quiet) {
+        const int h = e.hist + 2;
+        ++hist_n[ac][h];
+        hist_fh[ac][h] += e.fh ? 1 : 0;
+        hist_conf[ac][h] += conf ? 1 : 0;
+    } else {
+        ++hist_noisy[ac];
+    }
+}
