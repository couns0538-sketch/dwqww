+// This diagnostic applies to one single-threaded fixed-depth search of position P.
+// Path identity comes from the actual moves on the stack, including null moves.
+thread_local bool trace_position = false;

+
+const char* mate_trace_path(const SearchStack* ss) {
+    if (!trace_position || ss->ply > 3) return nullptr;
+    if (ss->ply == 0) return "root";
+    const SearchStack* root = ss - ss->ply;
+    if (Board::move_to_uci(root[0].current_move) != "g3g6") return nullptr;
+    if (ss->ply == 1) return "g3g6";
+    const auto reply = Board::move_to_uci(root[1].current_move);
+    if (root[1].current_move.is_none()) {
+        if (ss->ply == 2) return "g3g6/null";
+        if (Board::move_to_uci(root[2].current_move) == "g6h7") return "g3g6/null/g6h7";
+    } else if (reply == "f7g6") {
+        if (ss->ply == 2) return "g3g6/f7g6";
+        if (Board::move_to_uci(root[2].current_move) == "e5g6") return "g3g6/f7g6/e5g6";
+    } else if (reply == "c6e5") {
+        if (ss->ply == 2) return "g3g6/c6e5";
+        if (Board::move_to_uci(root[2].current_move) == "g6h7") return "g3g6/c6e5/g6h7";
+    }
+    return nullptr;
+}

+struct MateTrace {
+    const char* path;
+    const int& live_alpha;
+    const int& live_beta;
+    int id = 0;
+    int searched_static = ValueNone;
+    const char* termination = "exhausted";
+
+    MateTrace(const Board& board, SearchStack* ss, const char* kind, int iteration, int depth, const int& alpha, const int& beta)
+        : path(mate_trace_path(ss)), live_alpha(alpha), live_beta(beta) {
+        if (!path) return;
+        id = ++trace_serial;
+        // A fresh evaluator gives an independent raw static score even when a TT hit
+        // returns before the engine evaluates. Keep this separate from its actual score.
+        const auto parts = Evaluator::static_eval_components(board);
+        std::cout << "AS3TRACE id=" << id << " event=enter path=" << path << " kind=" << kind
+                  << " iteration=" << iteration << " depth=" << depth << " alpha=" << alpha << " beta=" << beta
+                  << " raw_static=" << parts.total << " in_check=" << parts.in_check
+                  << " excluded=" << Board::move_to_uci(ss->excluded_move) << '\n';
+    }
+
+    void move_event(Move move, const char* event, int amount = 0, bool all = false) const {
+        if (!path) return;
+        const auto name = Board::move_to_uci(move);
+        const std::string_view prefix(path);
+        if (!all && !((prefix == "root" && name == "g3g6") ||
+                      (prefix == "g3g6" && (name == "f7g6" || name == "c6e5")) ||
+                      (prefix == "g3g6/f7g6" && name == "e5g6") ||
+                      ((prefix == "g3g6/c6e5" || prefix == "g3g6/null") && name == "g6h7"))) return;
+        std::cout << "AS3TRACE id=" << id << " event=" << event << " move=" << name << " value=" << amount << '\n';
+    }
+
+    void tt(bool hit, Bound bound, Move move, int score, int depth) const {
+        if (!path) return;
+        std::cout << "AS3TRACE id=" << id << " event=tt hit=" << hit << " move=" << Board::move_to_uci(move)
+                  << " bound=" << (bound == BoundLower ? "lower" : bound == BoundUpper ? "upper" :
+                                   bound == BoundExact ? "exact" : "none")
+                  << " tt_score=" << score << " tt_depth=" << depth << '\n';
+    }
+
+    int finish(const char* reason, int score) const {
+        if (path) {
+            std::cout << "AS3TRACE id=" << id << " event=return reason=" << reason << " score=" << score
+                      << " exit_alpha=" << live_alpha << " exit_beta=" << live_beta << " search_static=";
+            if (searched_static == ValueNone) std::cout << "NA";
+            else std::cout << searched_static;
+            std::cout << '\n';
+        }
+        return score;
+    }
+};
