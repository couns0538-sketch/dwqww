+// Cheapest enemy piece attacking the capture's landing square, as a history class.
+// 0: nobody, or only a king (this engine's SEE value for a king is 0).
+// 1..5: pawn, knight, bishop, rook, queen. Equal SEE values take the lower piece type.
+// This does not decide whether the capture passes the exchange gate.
+int landing_attacker_class(const Board& board, int to) {
+    const Color them = opposite(board.side_to_move());
+    Bitboard hits = board.attackers_to(to) & board.occupied(them);
+    int best_value = 0;
+    int best_type = 0;
+    while (hits != 0) {
+        const int from = pop_lsb(hits);
+        const int type = piece_type(board.piece_at(from));
+        const int value = SeePieceValue[static_cast<std::size_t>(type)];
+        if (value <= 0) continue;
+        if (best_type == 0 || value < best_value || (value == best_value && type < best_type)) {
+            best_value = value;
+            best_type = type;
+        }
+    }
+    return best_type;
+}
