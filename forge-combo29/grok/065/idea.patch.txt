+// Idea S15, read on the position before the move is made. A rook ending here is
+// one rook on each side and no queen, bishop, or knight. The rook move leaves
+// its file and lands on the file of a passed pawn, so it reaches that file in
+// one move. The pawn's next step costs one move when the stop square is empty
+// and two when something stands there. One is never more than that, which is
+// the tempo test. The caller then sets the reduction to 0. This does not
+// extend and it does not touch the evaluation. Only called when IdeaS15 is on.
+[[nodiscard]] bool idea_s15_rook_meets_passer(const Board& board, Move move, Piece moved) {
+    if (piece_type(moved) != Rook) return false;
+    if (file_of(move.from) == file_of(move.to)) return false;
+    if (popcount(board.occupied()) < tune::IdeaS15_MinPieces) return false;
+    if ((board.pieces(Queen) | board.pieces(Bishop) | board.pieces(Knight)) != 0) return false;
+    if (popcount(board.pieces(White, Rook)) != 1 || popcount(board.pieces(Black, Rook)) != 1) return false;
+    constexpr int kRookMovesToFile = 1;
+    Bitboard pawns = board.pieces(Pawn) & file_bb(file_of(move.to));
+    while (pawns != 0) {
+        const int square = pop_lsb(pawns);
+        const Color us = color_of(board.piece_at(square));
+        const Color them = opposite(us);
+        if ((attacks::passed_pawn_mask(us, square) & board.pieces(them, Pawn)) != 0) continue;
+        if ((board.pieces(us, Pawn) & attacks::forward_file(us, square)) != 0) continue;
+        const int stop = square + pawn_push_delta(us);
+        const bool stop_empty = stop >= 0 && stop < 64 && board.piece_at(stop) == Empty;
+        const int pawn_steps = stop_empty ? 1 : 2;
+        if (kRookMovesToFile <= pawn_steps) return true;
+    }
+    return false;
+}
