+// IdeaS15. A rook ending is kings, rooks and pawns only. A rook move meets a
+// passer when it ends on that pawn's file. Reaching the file costs 0 moves if
+// the rook is already there and 1 move otherwise. The pawn's next step costs
+// 1 move when the stop square is empty and 2 when a piece blocks it. The
+// caller clears a positive reduction and does not add an extension.
+[[nodiscard]] bool idea_s15_rook_meets_passer(const Board& board, Move move) {
+    if (piece_type(board.moved_piece(move)) != Rook) return false;
+    if ((board.pieces(Queen) | board.pieces(Bishop) | board.pieces(Knight)) != 0) return false;
+    if (popcount(board.occupied()) < tune::IdeaS15_MinPieces) return false;
+    const int to_file = file_of(move.to);
+    const int rook_moves = file_of(move.from) == to_file ? 0 : 1;
+    for (const Color color : {White, Black}) {
+        Bitboard pawns = board.pieces(color, Pawn);
+        while (pawns != 0) {
+            const int square = pop_lsb(pawns);
+            if (file_of(square) != to_file) continue;
+            const Bitboard enemy_pawns = board.pieces(opposite(color), Pawn);
+            const Bitboard own_ahead = board.pieces(color, Pawn) & attacks::forward_file(color, square);
+            if ((attacks::passed_pawn_mask(color, square) & enemy_pawns) != 0 || own_ahead != 0) continue;
+            const int stop = square + pawn_push_delta(color);
+            const int pawn_steps = (stop >= 0 && stop < 64 && board.piece_at(stop) == Empty) ? 1 : 2;
+            if (rook_moves <= pawn_steps) return true;
+        }
+    }
+    return false;
+}
