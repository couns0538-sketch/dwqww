+// Idea S15. A rook ending is kings, exactly one rook each, and pawns. A pawn is
+// passed when no enemy pawn stands on its file or the next files in front, and
+// no friendly pawn stands in front of it on the same file. The rook meets that
+// passer when this move's destination file is the pawn's file. Reaching the
+// file costs 0 moves when the rook is already there and 1 move otherwise. The
+// pawn's next step costs 1 move when the square in front is empty and 2 when
+// it is occupied. Full depth means the reduction is cleared. Nothing is added
+// to the extension and nothing is added to the evaluation.
+[[nodiscard]] bool rook_meets_passer(const Board& board, Move move) {
+    if (popcount(board.pieces(White, Rook)) != 1 || popcount(board.pieces(Black, Rook)) != 1) return false;
+    if (board.pieces(Queen) != 0 || board.pieces(Bishop) != 0 || board.pieces(Knight) != 0) return false;
+    if (popcount(board.occupied()) < tune::IdeaS15_MinPieces) return false;
+    const int to_file = file_of(move.to);
+    const int rook_moves = file_of(move.from) == to_file ? 0 : 1;
+    for (const Color color : {White, Black}) {
+        Bitboard pawns = board.pieces(color, Pawn);
+        const Color them = opposite(color);
+        const int step = color == White ? 8 : -8;
+        while (pawns != 0) {
+            const int sq = pop_lsb(pawns);
+            if (file_of(sq) != to_file) continue;
+            if ((board.pieces(color, Pawn) & attacks::forward_file(color, sq)) != 0) continue;
+            if ((board.pieces(them, Pawn) & attacks::passed_pawn_mask(color, sq)) != 0) continue;
+            const int front = sq + step;
+            const int pawn_steps = (front >= 0 && front < 64 && board.piece_at(front) == Empty) ? 1 : 2;
+            if (rook_moves <= pawn_steps) return true;
+        }
+    }
+    return false;
+}
