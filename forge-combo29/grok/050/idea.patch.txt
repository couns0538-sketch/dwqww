+// IdeaK11. Today's cutoff bonus, then multiplied by (clamped margin + Base) / Divisor.
+// margin is how far the score beat beta, capped. A mate cutoff uses Cap so a mate
+// distance cannot blow the bonus up. Divisor 160 and Base 64 keep a 96 cp margin at 1.0.
+[[nodiscard]] int k11_scaled_bonus(int depth, int best_score, int beta) {
+    const int base = history_bonus(depth);
+    int margin;
+    if (is_mate_score(best_score) || is_mate_score(beta)) {
+        margin = tune::IdeaK11_Cap;
+    } else {
+        margin = best_score - beta;
+        if (margin < 0) margin = 0;
+        if (margin > tune::IdeaK11_Cap) margin = tune::IdeaK11_Cap;
+    }
+    int divisor = tune::IdeaK11_Divisor;
+    if (divisor < 1) divisor = 1;
+    const long long scaled =
+        static_cast<long long>(base) * static_cast<long long>(margin + tune::IdeaK11_Base) / divisor;
+    if (scaled > tune::HistoryBonusMax) return tune::HistoryBonusMax;
+    if (scaled < -tune::HistoryBonusMax) return -tune::HistoryBonusMax;
+    return static_cast<int>(scaled);
