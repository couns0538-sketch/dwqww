+// Quiet-history part of LMR, in whole plies. The integer engine divides history
+// by LmrHistoryDivisor and clamps to +/-2, so under S14 (divisor 5643) every
+// history in [-5642, 5642] contributes 0. IdeaA1 scores that same quotient in
+// 1/IdeaA1_Scale of a ply and rounds half toward zero: strictly more than half
+// a ply changes the reduction, and an exact half does not. The cap is
+// IdeaA1_Clamp whole plies, the same +/-2 the integer term uses at its default.
+[[nodiscard]] int idea_a1_history_plies(int history) {
+    const int scale = tune::IdeaA1_Scale;
+    const int divisor = tune::LmrHistoryDivisor;
+    long long units = static_cast<long long>(history) * scale / divisor;
+    const long long cap = static_cast<long long>(tune::IdeaA1_Clamp) * scale;
+    if (units > cap) units = cap;
+    if (units < -cap) units = -cap;
+    const long long sign = units < 0 ? -1 : 1;
+    const long long absolute = units < 0 ? -units : units;
+    long long plies = absolute / scale;
+    const long long remainder = absolute % scale;
+    if (remainder * 2 > static_cast<long long>(scale)) ++plies;
+    return static_cast<int>(sign * plies);
+}
