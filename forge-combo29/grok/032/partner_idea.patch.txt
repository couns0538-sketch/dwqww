+// History term of a quiet late-move reduction, in fractions of a ply.
+// credit_in is the unused fraction from earlier quiets on this line (positive
+// means reduce less). This move adds history * Scale / LmrHistoryDivisor,
+// capped at ±Clamp plies. Whole plies are applied to `reduction` now. The
+// leftover, in (-Scale, Scale), is credit_out for the child. Called only
+// when IdeaA1 is on, so the integer division below stays the off path.
+void apply_fractional_history_lmr(int history, int credit_in, int& reduction, int& credit_out) {
+    const int scale = tune::IdeaA1_Scale;
+    const int clamp_plies = tune::IdeaA1_Clamp;
+    if (scale <= 0) {
+        credit_out = 0;
+        return;
+    }
+    const long long limit = static_cast<long long>(clamp_plies) * static_cast<long long>(scale);
+    const int divisor = tune::LmrHistoryDivisor;
+    long long raw = 0;
+    if (divisor != 0) raw = static_cast<long long>(history) * static_cast<long long>(scale) / divisor;
+    if (raw > limit) raw = limit;
+    if (raw < -limit) raw = -limit;
+    long long credit = static_cast<long long>(credit_in) + raw;
+    int whole = static_cast<int>(credit / scale);
+    if (whole > clamp_plies) whole = clamp_plies;
+    if (whole < -clamp_plies) whole = -clamp_plies;
+    credit -= static_cast<long long>(whole) * static_cast<long long>(scale);
+    reduction -= whole;
+    credit_out = static_cast<int>(credit);
