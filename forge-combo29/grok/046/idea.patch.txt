+[[nodiscard]] bool pair_moves_ready(const SearchStack* ss) {
+    if (ss == nullptr) return false;
+    const SearchStack* recent = ss - 1;
+    const SearchStack* older = ss - 2;
+    if (recent->current_move.is_none() || recent->moved_piece == Empty) return false;
+    if (older->current_move.is_none() || older->moved_piece == Empty) return false;
+    return static_cast<unsigned>(recent->moved_piece) < PairPieceCount &&
+           static_cast<unsigned>(older->moved_piece) < PairPieceCount &&
+           recent->current_move.to < PairSquareCount && older->current_move.to < PairSquareCount;
