// least beta. The flag is read first: when it is off this returns score.
[[nodiscard]] int idea_a2_score(int score, int beta, int depth, bool pv_node) {
    if (!tune::IdeaA2 || pv_node || depth < tune::IdeaA2_MinDepth || score < beta || is_mate_score(score))
        return score;
    return static_cast<int>((static_cast<long long>(score) * depth + beta) / (depth + 1));
}
