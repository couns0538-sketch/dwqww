+// IdeaS15. A rook ending is kings, rooks and pawns, with a rook on each side,
+// and at least IdeaS15_MinPieces pieces on the board. A pawn is passed when no
+// enemy pawn stands on its file or the next files in front, and no own pawn
+// stands in front on the same file. The rook "meets" that pawn when this move
+// stays on its file (0 moves still to go) or steps onto its file (1 move).
+// The pawn's next step costs 1 when the square in front is empty and 2 when
+// that square is occupied. The rook is in time when its remaining moves are
+// no greater than that cost, and no greater than IdeaS15_MaxRookMoves.
+// This does not extend the move. Call it only when IdeaS15 is on.
+[[nodiscard]] bool s15_rook_meets_passer(const Board& board, Move move, Piece moved) {
+    if (piece_type(moved) != Rook) return false;
+    if (popcount(board.occupied()) < tune::IdeaS15_MinPieces) return false;
+    if ((board.pieces(Knight) | board.pieces(Bishop) | board.pieces(Queen)) != 0) return false;
+    if (board.pieces(White, Rook) == 0 || board.pieces(Black, Rook) == 0) return false;
+    const int rook_moves = file_of(move.from) == file_of(move.to) ? 0 : 1;
+    if (rook_moves > tune::IdeaS15_MaxRookMoves) return false;
+    const int to_file = file_of(move.to);
+    for (const Color color : {White, Black}) {
+        Bitboard pawns = board.pieces(color, Pawn);
+        while (pawns != 0) {
+            const int square = pop_lsb(pawns);
+            if (file_of(square) != to_file) continue;
+            if ((attacks::passed_pawn_mask(color, square) & board.pieces(opposite(color), Pawn)) != 0) continue;
+            if ((board.pieces(color, Pawn) & attacks::forward_file(color, square)) != 0) continue;
+            const int stop = square + pawn_push_delta(color);
+            if (stop < 0 || stop > 63) continue;
+            const int pawn_steps = (board.occupied() & square_bb(stop)) == 0 ? 1 : 2;
+            if (rook_moves <= pawn_steps) return true;
+        }
+    }
+    return false;
+}
