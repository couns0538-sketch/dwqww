+// ---------------------------------------------------------------------------
+// IdeaS6. At a quiet root, the friendly non-king piece whose removal changes
+// the network evaluation the least is idle. Quiet non-captures of that piece
+// are not late-move reduced. The piece is chosen once; its square is tracked
+// as it moves. Nothing here runs unless the option is on.
+
+namespace {
+
+[[nodiscard]] int probe_network_eval(const Board& board) {
+    return Evaluator::static_eval_components(board).total;
+}
+
+// -1 when the root is not quiet, or when the side to move has no non-king piece.
+// Quiet: not in check, and no legal capture, promotion, or check gains more than
+// IdeaS6_Margin centipawns over the static evaluation (S5's shallow agreement).
+[[nodiscard]] int choose_idle_square(const Board& board) {
+    if (board.in_check()) return -1;
+    const int quiet_eval = probe_network_eval(board);
+    int best_tactical = quiet_eval;
+    const MoveList legal = board.legal_moves();
+    for (const Move move : legal) {
+        if (!move.is_capture() && !move.is_promotion() && !board.gives_check(move)) continue;
+        Board child = board;
+        Undo undo;
+        if (!child.make_move(move, undo)) continue;
+        const int score = -probe_network_eval(child);
+        if (score > best_tactical) best_tactical = score;
+    }
+    if (best_tactical - quiet_eval > tune::IdeaS6_Margin) return -1;
+    int best_square = -1;
+    int least = std::numeric_limits<int>::max();
+    const Color us = board.side_to_move();
+    for (int square = 0; square < 64; ++square) {
+        const Piece piece = board.piece_at(square);
+        if (piece == Empty || color_of(piece) != us || piece_type(piece) == King) continue;
+        Board dropped = board;
+        dropped.lift_piece_for_probe(square);
+        const int score = probe_network_eval(dropped);
+        const int change = score > quiet_eval ? score - quiet_eval : quiet_eval - score;
+        if (change < least) {
+            least = change;
+            best_square = square;
+        }
+    }
+    return best_square;
+}

+void relay_idle_square(SearchStack* child, const SearchStack* parent, Move move, bool null_move) {
+    if (!tune::IdeaS6) return;
+    if (null_move || parent->idle_square < 0) {
+        child->idle_square = parent->idle_square;
+        return;
+    }
+    if (static_cast<int>(move.from) == parent->idle_square) child->idle_square = static_cast<int>(move.to);
+    else if (move.is_capture() && static_cast<int>(move.to) == parent->idle_square) child->idle_square = -1;
+    else child->idle_square = parent->idle_square;
+}
