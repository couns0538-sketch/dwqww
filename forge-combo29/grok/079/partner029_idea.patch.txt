+std::uint64_t Worker::effort_key(const Board& board, Move excluded) const {
+    std::uint64_t key = board.key();

+int Worker::effort_repay_plies(const Board& board, Move excluded, Move move, int withheld) const {
+    if (withheld <= 0 || path_repaid_ >= tune::IdeaB10_Cap || effort_marks_ == nullptr) return 0;
+    const std::uint64_t key = effort_key(board, excluded);
+    const EffortMark& mark = effort_marks_[key & (EffortSlots - 1U)];
+    if (mark.key != key || mark.dominance < static_cast<std::uint16_t>(tune::IdeaB10_Ratio)) return 0;
+    static_assert(sizeof(Move) == 4);
+    if (std::bit_cast<Move>(mark.move_word) != move) return 0;
+    const int excess = static_cast<int>(mark.dominance) - 256;
+    if (excess <= 0) return 0;
+    int repay = (withheld * excess + 255) / 256;
+    if (repay > withheld) repay = withheld;
+    const int room = tune::IdeaB10_Cap - path_repaid_;
+    if (repay > room) repay = room;
+    return repay > 0 ? repay : 0;
+}

+void Worker::remember_effort(const Board& board, Move excluded, Move move, std::uint64_t best_rate,
+                             std::uint64_t second_rate) {
+    if (effort_marks_ == nullptr || second_rate == 0 || best_rate == 0 || move.is_none()) return;
+    const std::uint64_t ratio = static_cast<std::uint64_t>(tune::IdeaB10_Ratio);
+    if (best_rate / second_rate > 256U) {
+        // Far above any stored dominance. Keep the cap below.
+    } else if (best_rate * 256U < second_rate * ratio) {
+        return;
+    }
+    std::uint64_t dominance = best_rate / second_rate > 256U ? 65535U : (best_rate * 256U) / second_rate;
+    if (dominance > 65535U) dominance = 65535U;
+    if (dominance < ratio) return;
+    const std::uint64_t key = effort_key(board, excluded);
+    EffortMark& mark = effort_marks_[key & (EffortSlots - 1U)];
+    mark.key = key;
+    mark.move_word = std::bit_cast<std::uint32_t>(move);
+    mark.dominance = static_cast<std::uint16_t>(dominance);
+}
