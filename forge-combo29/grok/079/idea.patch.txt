+void Worker::remember_effort(std::uint64_t key, int node_depth, const EffortSample* samples, int count) {
+    if (!tune::IdeaB10 || effort_table_ == nullptr || samples == nullptr || count < 2 || node_depth <= 0) return;
+    int best = -1;
+    int second = -1;
+    auto rate_beats = [&](int a, int b) {
+        const std::uint64_t left = static_cast<std::uint64_t>(samples[a].nodes) * static_cast<std::uint64_t>(samples[b].depth);
+        const std::uint64_t right = static_cast<std::uint64_t>(samples[b].nodes) * static_cast<std::uint64_t>(samples[a].depth);
+        return left > right;
+    };
+    for (int i = 0; i < count; ++i) {
+        if (samples[i].depth <= 0 || samples[i].nodes == 0 || samples[i].move.is_none()) continue;
+        if (best < 0 || rate_beats(i, best)) {
+            second = best;
+            best = i;
+        } else if (second < 0 || rate_beats(i, second)) {
+            second = i;
+        }
+    }
+    if (best < 0 || second < 0) return;
+    if (samples[best].nodes < static_cast<std::uint32_t>(std::max(tune::IdeaB10_MinNodes, 1))) return;
+    const int ratio = std::max(tune::IdeaB10_Ratio, 1);
+    const std::uint64_t left = static_cast<std::uint64_t>(samples[best].nodes) *
+                               static_cast<std::uint64_t>(samples[second].depth) * 100ULL;
+    const std::uint64_t right = static_cast<std::uint64_t>(samples[second].nodes) *
+                                static_cast<std::uint64_t>(samples[best].depth) *
+                                static_cast<std::uint64_t>(ratio);
+    if (left < right) return;
+    const std::uint64_t denom = static_cast<std::uint64_t>(samples[second].nodes) *
+                                static_cast<std::uint64_t>(samples[best].depth);
+    if (denom == 0) return;
+    std::uint64_t dominance = left / denom;
+    if (dominance > 1000000ULL) dominance = 1000000ULL;
+    const int step = std::max(tune::IdeaB10_Step, 1);
+    int repay = 1;
+    if (dominance > static_cast<std::uint64_t>(ratio)) {
+        repay += static_cast<int>((dominance - static_cast<std::uint64_t>(ratio)) / static_cast<std::uint64_t>(step));
+    }
+    const int cap = std::max(tune::IdeaB10_Cap, 0);
+    if (repay > cap) repay = cap;
+    if (repay > 8) repay = 8;
+    if (repay < 1) return;
+    const int stored_depth = node_depth > 65535 ? 65535 : node_depth;
+    EffortMark& slot = effort_table_->slots[static_cast<std::size_t>(key & (EffortTable::Slots - 1U))];
+    // A shallower mark is the one a later visit is more likely to meet. Do not
+    // let a deep one-off node evict it.
+    if (slot.occupied && slot.key != key && static_cast<int>(slot.depth) < stored_depth) return;
+    slot.key = key;
+    slot.move = samples[best].move;
+    slot.depth = static_cast<std::uint16_t>(stored_depth);
+    slot.repay = static_cast<std::uint8_t>(repay);
+    slot.occupied = true;
+}
