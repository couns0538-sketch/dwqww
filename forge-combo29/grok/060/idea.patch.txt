+// IdeaS24. Overlap, in centipawns, back into the previous root window.
+// 0 when recent root scores are steady. Half the old width when their
+// population stddev reaches IdeaS24_Swing. -1 keeps today's update: a mate
+// score is not a centipawn bounce, and a window with an open bound has no width.
+[[nodiscard]] int s24_overlap(int old_alpha, int old_beta, int fail_score, const int (&samples)[16], int count) {
+    if (is_mate_score(fail_score)) return -1;
+    if (old_alpha <= -Infinite || old_beta >= Infinite) return -1;
+    const int width = old_beta - old_alpha;
+    if (width <= 1) return 0;
+    long long sum = 0;
+    long long sumsq = 0;
+    int n = 0;
+    const int seen = std::min(count, 16);
+    for (int i = 0; i < seen; ++i) {
+        if (is_mate_score(samples[i])) continue;
+        const long long sample = samples[i];
+        sum += sample;
+        sumsq += sample * sample;
+        ++n;
+    }
+    const long long failed = fail_score;
+    sum += failed;
+    sumsq += failed * failed;
+    ++n;
+    if (n < 2) return 0;
+    const long long var_num = n * sumsq - sum * sum;
+    const long long var = var_num <= 0 ? 0 : var_num / (static_cast<long long>(n) * n);
+    std::uint64_t x = static_cast<std::uint64_t>(var);
+    std::uint64_t root = x;
+    std::uint64_t next = (root + 1) / 2;
+    while (next < root) {
+        root = next;
+        next = (root + x / root) / 2;
+    }
+    const int stddev = static_cast<int>(root > 1000000ULL ? 1000000ULL : root);
+    const int swing = std::max(1, tune::IdeaS24_Swing);
+    const int ratio = stddev < swing ? stddev : swing;
+    const int half = width / 2;
+    return static_cast<int>((static_cast<long long>(half) * ratio) / swing);
+}
+
+void s24_note(int (&samples)[16], int& count, int value) {
+    if (!tune::IdeaS24 || is_mate_score(value)) return;
+    const int keep = std::clamp(tune::IdeaS24_Samples, 2, 16);
+    if (count < keep) {
+        samples[count++] = value;
+        return;
+    }
+    for (int i = 1; i < keep; ++i) samples[i - 1] = samples[i];
+    samples[keep - 1] = value;
+    count = keep;
+}
