+// IdeaS24. After a root fail, the failing bound is still the true score shifted
+// by the current delta (a bound sitting on the score fails again forever).
+// The other bound keeps an overlap back into the previous window: none when
+// the recent resolved root scores are steady, up to half the old width when
+// they are swinging. The same rule is used on fail-high and fail-low.
+// Variance is the integer mean of squared deviations from the integer mean.
+void s24_note(int* scores, int& count, int value) {
+    if (is_mate_score(value)) return;
+    if (count < 16) {
+        scores[count++] = value;
+        return;
+    }
+    for (int i = 1; i < 16; ++i) scores[i - 1] = scores[i];
+    scores[15] = value;
+}
+
+[[nodiscard]] int s24_overlap(int old_alpha, int old_beta, const int* scores, int count) {
+    const int width = old_beta - old_alpha;
+    if (width <= 0 || count < 2) return 0;
+    const int half = width / 2;
+    int keep = tune::IdeaS24_Samples;
+    if (keep < 2) keep = 2;
+    if (keep > 16) keep = 16;
+    const int used = count < keep ? count : keep;
+    if (used < 2) return 0;
+    const int* window = scores + (count - used);
+    long long sum = 0;
+    for (int i = 0; i < used; ++i) sum += window[i];
+    const long long mean = sum / used;
+    long long acc = 0;
+    for (int i = 0; i < used; ++i) {
+        const long long delta = window[i] - mean;
+        acc += delta * delta;
+    }
+    const long long var = acc / used;
+    const long long steady = tune::IdeaS24_Steady;
+    const long long swing = tune::IdeaS24_Swing;
+    if (var <= steady) return 0;
+    if (swing <= steady || var >= swing) return half;
+    return static_cast<int>((static_cast<long long>(half) * (var - steady)) / (swing - steady));
+}
