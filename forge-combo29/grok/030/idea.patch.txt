+// IdeaB1. Quiet-move ordering prior from the current placement, not a
+// piece-square table. Pressure is attacks on the enemy army (weight 2) plus
+// attacks into the enemy king zone (weight 1). The bonus is that change from
+// the move's origin to its destination, then shrunk as this search writes
+// main history for the same from-to: Scale * delta * 1024 / (1024 + Fade * |h|).
+// Callers add it only to quiet scores, after hash, capture, and killer slots.
+[[nodiscard]] int fading_board_prior(const Board& board, Move move, int main_history) {
+    const int type = piece_type(board.piece_at(move.from));
+    const Color us = board.side_to_move();
+    const Color them = opposite(us);
+    const Bitboard enemy = board.occupied(them);
+    const int their_king = board.king_square(them);
+    const Bitboard zone = their_king >= 0 ? attacks::king_zone(their_king) : Bitboard{0};
+    const Bitboard occ = board.occupied();
+    const Bitboard occ_to = (occ ^ square_bb(move.from)) | square_bb(move.to);
+    auto pressure = [&](int square, Bitboard occupancy) {
+        Bitboard att = 0;
+        switch (type) {
+        case Pawn:
+            att = attacks::pawn(us, square);
+            break;
+        case Knight:
+            att = attacks::knight(square);
+            break;
+        case Bishop:
+            att = attacks::bishop(square, occupancy);
+            break;
+        case Rook:
+            att = attacks::rook(square, occupancy);
+            break;
+        case Queen:
+            att = attacks::queen(square, occupancy);
+            break;
+        case King:
+            att = attacks::king(square);
+            break;
+        default:
+            att = 0;
+            break;
+        }
+        return 2 * popcount(att & enemy) + popcount(att & zone);
+    };
+    const int delta = pressure(move.to, occ_to) - pressure(move.from, occ);
+    const int mag = main_history >= 0 ? main_history : -main_history;
+    const int denom = 1024 + tune::IdeaB1_Fade * mag;
+    if (denom <= 0) return 0;
+    return delta * tune::IdeaB1_Scale * 1024 / denom;
+}
