+[[nodiscard]] int idea_a4_capture_shift(int history) {
+    const int divisor = tune::IdeaA4_Divisor < 1 ? 1 : tune::IdeaA4_Divisor;
+    int shift = history / divisor;
+    const int cap = tune::IdeaA4_Cap;
+    if (shift > cap) shift = cap;
+    if (shift < -cap) shift = -cap;
+    return shift;
