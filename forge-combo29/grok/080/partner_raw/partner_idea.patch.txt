+void QuietPrior::prepare(const Board& board, const Accumulator& accumulator, int lanes) {
+    ready_ = false;
+    w1_ = nullptr;
+    lanes_ = 0;
+    if (!enabled || !available() || !accumulator.valid) return;
+    int count = lanes;
+    if (count < 1) count = 1;
+    if (count > 64) count = 64;
+    if (count > HiddenSize) count = HiddenSize;
+    int stride = HiddenSize / count;
+    if (stride < 1) {
+        stride = 1;
+        count = HiddenSize;
+    }
+    const Weights& w = weights();
+    if (w.w1 == nullptr) return;
+    const Color us = board.side_to_move();
+    const float* ours = accumulator.values[static_cast<std::size_t>(us)].data();
+    const float* theirs = accumulator.values[static_cast<std::size_t>(opposite(us))].data();
+    const float* w2_ours = w.w2.data();
+    const float* w2_theirs = w.w2.data() + HiddenSize;
+    lanes_ = count;
+    w1_ = w.w1.get();
+    for (int i = 0; i < count; ++i) {
+        const int unit = i * stride;
+        units_[static_cast<std::size_t>(i)] = unit;
+        const float our_activation = ours[static_cast<std::size_t>(unit)];
+        const float their_activation = theirs[static_cast<std::size_t>(unit)];
+        // Derivative of clamp(x, 0, 1)^2 is 2x inside (0, 1) and 0 outside.
+        grad_us_[static_cast<std::size_t>(i)] =
+            (our_activation > 0.0F && our_activation < 1.0F) ? 2.0F * our_activation * w2_ours[static_cast<std::size_t>(unit)] : 0.0F;
+        grad_them_[static_cast<std::size_t>(i)] =
+            (their_activation > 0.0F && their_activation < 1.0F)
+                ? 2.0F * their_activation * w2_theirs[static_cast<std::size_t>(unit)]
+                : 0.0F;
+    }
+    ready_ = true;
+}

+int QuietPrior::centipawns(const Board& board, Move move) const {
+    if (!ready_ || w1_ == nullptr) return 0;
+    const DirtyPieces dirty = dirty_pieces(board, move);
+    const Color us = board.side_to_move();
+    float dot = 0.0F;
+    for (int change_index = 0; change_index < dirty.count; ++change_index) {
+        const PieceChange& change = dirty.changes[static_cast<std::size_t>(change_index)];
+        const float sign = change.add ? 1.0F : -1.0F;
+        for (int side = 0; side < 2; ++side) {
+            const Color perspective = side == 0 ? us : opposite(us);
+            const float* gradient = side == 0 ? grad_us_.data() : grad_them_.data();
+            const int index = feature_index(perspective, change.piece, change.square);
+            const float* column = w1_ + static_cast<std::size_t>(index) * static_cast<std::size_t>(HiddenSize);
+            for (int lane = 0; lane < lanes_; ++lane) {
+                const int unit = units_[static_cast<std::size_t>(lane)];
+                dot += sign * gradient[lane] * column[unit];
+            }
+        }
+    }
+    const float centipawns = dot * 400.0F / SigmoidK;
+    return static_cast<int>(std::clamp(centipawns, -30000.0F, 30000.0F));
+}

+// Build this position's network features into `out` without writing the stack.
+// A cold node can still order quiets when the hash already stored the eval.
+bool materialise_features(const Board& board, const SearchStack* ss, nnue::Accumulator& out) {
+    if (ss == nullptr || !nnue::enabled || !nnue::available()) return false;
+    if (ss->accumulator.valid) {
+        out = ss->accumulator;
+        return true;
+    }
+    int back = 0;
+    while (back < ss->ply && !(ss - back - 1)->accumulator.valid) ++back;
+    if (back >= ss->ply && !(ss - back - 1)->accumulator.valid) {
+        nnue::refresh(board, out);
+        return out.valid;
+    }
+    nnue::Accumulator rolling = (ss - back - 1)->accumulator;
+    for (int i = back; i >= 0; --i) {
+        const SearchStack* frame = ss - i;
+        if (frame->entered_null) continue;
+        nnue::DirtyPieces dirty;
+        if (frame->diff_known) {
+            dirty = frame->dirty;
+        } else if (!frame->entered_move.is_none() && frame->entered_piece != Empty) {
+            dirty = nnue::dirty_from_move(frame->entered_move, frame->entered_piece, frame->entered_capture,
+                                           static_cast<Color>(frame->entered_side));
+        } else {
+            nnue::refresh(board, out);
+            return out.valid;
+        }
+        nnue::Accumulator next;
+        nnue::update(rolling, next, dirty);
+        rolling = next;
+    }
+    out = rolling;
+    return out.valid;
+}

+// prior_cp is the network's first-order change. history is the quiet-history
+// sum already written for this move. The bonus is prior_cp * scale at history
+// 0 and shrinks as |history| grows. It never leaves [-cap, cap].
+int fading_board_prior(int prior_cp, int history) {
+    const int fade = tune::IdeaB1_Fade > 0 ? tune::IdeaB1_Fade : 1;
+    const int scale = tune::IdeaB1_Scale > 0 ? tune::IdeaB1_Scale : 1;
+    const int cap = tune::IdeaB1_Cap > 0 ? tune::IdeaB1_Cap : 0;
+    const int abs_history = history < 0 ? -history : history;
+    const long long scaled = static_cast<long long>(prior_cp) * static_cast<long long>(scale) *
+                             static_cast<long long>(fade);
+    int bonus = static_cast<int>(scaled / (static_cast<long long>(fade) + abs_history));
+    if (bonus > cap) bonus = cap;
+    if (bonus < -cap) bonus = -cap;
+    return bonus;
+}

+// Off: ready stays false and adjust is the identity, so quiet scores stay put.
+struct BoardPriorGate {
+    nnue::QuietPrior prior{};
+    bool ready = false;
+
+    BoardPriorGate(const Board& board, const SearchStack* ss) {
+        if (!tune::IdeaB1 || ss == nullptr) return;
+        nnue::Accumulator local;
+        if (!materialise_features(board, ss, local)) return;
+        prior.prepare(board, local, tune::IdeaB1_Lanes);
+        ready = prior.ready();
+    }
+
+    [[nodiscard]] int adjust(const Board& board, Move move, int history) const {
+        if (!ready) return history;
+        return history + fading_board_prior(prior.centipawns(board, move), history);
+    }
+};
