+// near the quiet evaluation. A real search here would write the transposition
+// table, so the gate is SEE: the node is quiet when it is not in check and no
+// capture or check wins more than `margin` centipawns. The idle piece is the
+// friendly non-king piece that has a quiet move and whose removal changes the
+// network evaluation least (lowest square wins a tie). -1 means leave the
+// search alone. Skipping late-move reduction on that piece never fired in this
+// engine: those quiet moves are not in the reduced tail. The caller therefore
+// searches them one ply deeper for this one iteration, which is the same
+// direction as withholding the reduction.
+[[nodiscard]] int idle_piece_square(const Board& board, int margin) {
+    if (board.in_check()) return -1;
+    if (!nnue::enabled || !nnue::available()) return -1;
+    const MoveList legal = board.legal_moves();
+    int best_see = -100000;
+    bool saw_tactical = false;
+    bool quiet_from[64] = {};
+    for (const Move move : legal) {
+        if (move.is_quiet()) quiet_from[move.from] = true;
+        if (!move.is_capture() && !board.gives_check(move)) continue;
+        saw_tactical = true;
+        const int see = board.static_exchange(move);
+        if (see > best_see) best_see = see;
+    }
+    if (saw_tactical && best_see > margin) return -1;
+
+    nnue::Accumulator base_acc;
+    nnue::refresh(board, base_acc);
+    const int base = nnue::evaluate(board, base_acc);
+    const Color us = board.side_to_move();
+    int best_sq = -1;
+    int best_delta = 0;
+    for (int type = Pawn; type <= Queen; ++type) {
+        Bitboard left = board.pieces(us, static_cast<PieceType>(type));
+        while (left != 0) {
+            const int square = pop_lsb(left);
+            if (!quiet_from[square]) continue;
+            const Piece piece = board.piece_at(square);
+            nnue::DirtyPieces dirty;
+            dirty.remove(piece, square);
+            nnue::Accumulator dropped_acc;
+            nnue::update(base_acc, dropped_acc, dirty);
+            const int dropped = nnue::evaluate(board, dropped_acc);
+            const int delta = dropped > base ? dropped - base : base - dropped;
+            if (best_sq < 0 || delta < best_delta || (delta == best_delta && square < best_sq)) {
+                best_sq = square;
+                best_delta = delta;
+            }
+        }
+    }
+    return best_sq;
