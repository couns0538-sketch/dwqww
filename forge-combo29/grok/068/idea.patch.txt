+// Idea S16. A friendly pawn that is not yet passed, with exactly one stopper
+// (an enemy pawn in its passed-pawn span), every stopper a lever (attacked by
+// a friendly pawn), and at least as many friendly pawn helpers as enemy pawn
+// attackers on the one-step push square. The move that clears that last
+// stopper is a quiet pawn push onto a square that attacks it, or a capture of
+// it whose static exchange is not losing. Nothing is added to the evaluation.
+[[nodiscard]] Bitboard last_lever_stoppers(const Board& board) {
+    const Color us = board.side_to_move();
+    const Color them = opposite(us);
+    const Bitboard our_pawns = board.pieces(us, Pawn);
+    const Bitboard their_pawns = board.pieces(them, Pawn);
+    const Bitboard our_pawn_attacks = pawn_attacks_bb(us, our_pawns);
+    Bitboard last = 0;
+    Bitboard pawns = our_pawns;
+    while (pawns != 0) {
+        const int sq = pop_lsb(pawns);
+        if (relative_rank(us, sq) >= 6) continue;
+        const Bitboard stoppers = their_pawns & attacks::passed_pawn_mask(us, sq);
+        if (stoppers == 0 || more_than_one(stoppers)) continue;
+        if ((stoppers & ~our_pawn_attacks) != 0) continue;
+        const int push = sq + pawn_push_delta(us);
+        const int helpers = popcount(attacks::pawn(them, push) & our_pawns);
+        const int attackers = popcount(attacks::pawn(us, push) & their_pawns);
+        if (helpers < attackers) continue;
+        last |= stoppers;
+    }
+    return last;
+}
+
+[[nodiscard]] bool clears_last_stopper(const Board& board, Move move, Bitboard last) {
+    if (last == 0 || move.is_none()) return false;
+    if (move.is_quiet()) {
+        if (piece_type(board.moved_piece(move)) != Pawn) return false;
+        return (attacks::pawn(board.side_to_move(), move.to) & last) != 0;
+    }
+    if (!move.is_capture()) return false;
+    int cap_sq = move.to;
+    if ((move.flags & EnPassant) != 0U) cap_sq = move.to - pawn_push_delta(board.side_to_move());
+    if (cap_sq < 0 || cap_sq > 63) return false;
+    if ((last & square_bb(cap_sq)) == 0) return false;
+    return board.see_ge(move, 0);
+}
+
+[[nodiscard]] int lever_order_bonus(const Board& board, Move move, Bitboard last) {
+    if (!clears_last_stopper(board, move, last)) return 0;
+    return tune::IdeaS16_Order;
+}
