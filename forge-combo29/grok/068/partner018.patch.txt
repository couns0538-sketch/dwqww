+// IdeaS16. A friendly pawn is a candidate when it is not passed, every enemy
+// pawn in its passed-pawn span (a stopper) is attacked by a friendly pawn (a
+// lever), and the square one step ahead has at least as many friendly
+// attackers as enemy attackers. The move that clears the last stopper is a
+// quiet pawn push that leaves that pawn's span empty, or a capture of a pawn
+// that is some candidate's only stopper and whose exchange is not losing.
+// Nothing is added to the static evaluation.
+[[nodiscard]] bool idea_s16_candidate(const Board& board, Color us, int pawn_sq) {
+    const Bitboard enemy_pawns = board.pieces(opposite(us), Pawn);
+    const Bitboard stoppers = attacks::passed_pawn_mask(us, pawn_sq) & enemy_pawns;
+    if (stoppers == 0) return false;
+    const Bitboard our_pawns = board.pieces(us, Pawn);
+    Bitboard left = stoppers;
+    while (left != 0) {
+        const int sq = pop_lsb(left);
+        if ((attacks::pawn(opposite(us), sq) & our_pawns) == 0) return false;
+    }
+    const int push_sq = pawn_sq + pawn_push_delta(us);
+    if (push_sq < 0 || push_sq > 63) return false;
+    const int helpers = popcount(board.attackers_to(push_sq) & board.occupied(us));
+    const int foes = popcount(board.attackers_to(push_sq) & board.occupied(opposite(us)));
+    return helpers >= foes;
+}
+
+[[nodiscard]] bool clears_last_stopper(const Board& board, Move move) {
+    const Color us = board.side_to_move();
+    const Piece mover = board.piece_at(move.from);
+    if (piece_type(mover) == Pawn && move.is_quiet()) {
+        if (!idea_s16_candidate(board, us, move.from)) return false;
+        const Bitboard after = attacks::passed_pawn_mask(us, move.to) & board.pieces(opposite(us), Pawn);
+        return after == 0;
+    }
+    if (!move.is_capture()) return false;
+    int victim_sq = move.to;
+    if ((move.flags & EnPassant) != 0U) victim_sq = move.to - pawn_push_delta(us);
+    else if (piece_type(board.piece_at(move.to)) != Pawn) return false;
+    if (!board.see_ge(move, 0)) return false;
+    const Bitboard victim_bb = square_bb(victim_sq);
+    Bitboard ours = board.pieces(us, Pawn);
+    while (ours != 0) {
+        const int pawn_sq = pop_lsb(ours);
+        if (!idea_s16_candidate(board, us, pawn_sq)) continue;
+        const Bitboard stoppers = attacks::passed_pawn_mask(us, pawn_sq) & board.pieces(opposite(us), Pawn);
+        if (popcount(stoppers) == 1 && (stoppers & victim_bb) != 0) return true;
+    }
+    return false;
+}

+[[nodiscard]] int idea_s16_order_bonus(const Board& board, Move move) {
+    if (!tune::IdeaS16 || !clears_last_stopper(board, move)) return 0;
+    return tune::IdeaS16_OrderBonus;
+}
