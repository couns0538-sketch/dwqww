// Prepared execution command only. This file was not invoked during preparation.
import fs from 'node:fs';import cp from 'node:child_process';import crypto from 'node:crypto';import assert from 'node:assert/strict';
if(process.argv[2]!=='--execute-authorized')throw Error('This executes ten native searches; explicit --execute-authorized required');
const root='/workspace/work/r4-krk-repeat',hash=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex'),provenance=JSON.parse(fs.readFileSync(root+'/preparation-provenance.txt'));
for(const [p,pin]of Object.entries(provenance.originalSourceAndObjectPins))assert.equal(hash(p),pin,'prepared input changed '+p);assert.equal(hash(provenance.preparedBinary),provenance.preparedBinarySha256);
const stdout=root+'/repeat-run.stdout.txt',stderr=root+'/repeat-run.stderr.txt',metadata=root+'/repeat-run.metadata.txt';for(const f of [stdout,stderr,metadata])assert.ok(!fs.existsSync(f),'fresh evidence required '+f);
const argv=['/usr/bin/timeout','-k','2','400',provenance.preparedBinary,'--run-ten'],started=new Date().toISOString();fs.writeFileSync(metadata,JSON.stringify({argv,started,status:'running',stdout,stderr,binarySha256:provenance.preparedBinarySha256},null,2)+'\n',{flag:'wx'});const fds=[fs.openSync(stdout,'wx'),fs.openSync(stderr,'wx')];const p=cp.spawnSync(argv[0],argv.slice(1),{stdio:['ignore',...fds],timeout:410000});fds.forEach(fd=>fs.closeSync(fd));
const text=fs.readFileSync(stdout,'utf8'),errors=fs.readFileSync(stderr,'utf8'),lines=text.split('\n');const rows=lines.filter(l=>l.startsWith('ROW ')).map(l=>JSON.parse(l.slice(4))),summaries=lines.filter(l=>l.startsWith('SUMMARY ')).map(l=>JSON.parse(l.slice(8))),configs=lines.filter(l=>l.startsWith('CONFIG ')).map(l=>JSON.parse(l.slice(7)));
const complete=[0,1].includes(p.status)&&p.signal===null&&rows.length===10&&summaries.length===1;const counts={completedSearches:rows.length,misses:rows.filter(r=>r.miss).length,noMateScore:rows.filter(r=>!r.has_mate_score).length,nonMatingMove:rows.filter(r=>!r.is_mating_move).length};const diagnostics=[...errors.matchAll(/AddressSanitizer|UndefinedBehaviorSanitizer|LeakSanitizer|ThreadSanitizer|runtime error:/g)].map(m=>m[0]);
const record={argv,started,ended:new Date().toISOString(),exit:p.status,signal:p.signal,error:p.error?.message,complete,requestedRepeats:10,counts,summary:summaries.length===1?summaries[0]:null,configs,sanitizerDiagnostics:diagnostics,stdoutSha256:hash(stdout),stderrSha256:hash(stderr),binarySha256:provenance.preparedBinarySha256,originalFaultRetained:true};fs.writeFileSync(metadata,JSON.stringify(record,null,2)+'\n');
if(complete){assert.equal(configs.length,1);assert.equal(configs[0].threads,2);assert.equal(configs[0].hash_mb,16);assert.equal(configs[0].depth,14);assert.equal(configs[0].clear_all,true);assert.equal(configs[0].use_nnue,true);assert.equal(configs[0].MateAlphaGuard,false);assert.equal(configs[0].HindsightDepth,false);assert.equal(configs[0].SoundMatePruning,false);rows.forEach((r,i)=>assert.equal(r.run,i));assert.equal(summaries[0].runs,10);assert.equal(summaries[0].misses,counts.misses);assert.equal(summaries[0].no_mate_score,counts.noMateScore);assert.equal(summaries[0].non_mating_move,counts.nonMatingMove);}
console.log(JSON.stringify(record));process.exitCode=complete&&!diagnostics.length?0:3;
