import fs from 'node:fs';import cp from 'node:child_process';import crypto from 'node:crypto';import assert from 'node:assert/strict';

const paths=['/workspace/work/r4-equality-attribution/ref-36729-1.metadata.txt','/workspace/work/r4-equality-attribution/guard-off-36729-1.metadata.txt'];const sourceRecords=paths.map(p=>JSON.parse(fs.readFileSync(p)));assert.equal(sourceRecords[0].fen,sourceRecords[1].fen);
