// Short sequential group; the final command can use its full declared timeout.
// No new command starts after15seconds, so15+450seconds stays below8minutes.
import fs from 'node:fs';import cp from 'node:child_process';
const [mode,label]=process.argv.slice(2),r='/workspace/work/r4-run';
if(!['enabled','asan'].includes(mode)||!/^suite-[a-z0-9-]+$/.test(label??''))throw Error('mode/label');
const state=JSON.parse(fs.readFileSync(r+'/state.json'));
if(!state.equalityTotals?.targetMet)throw Error('finish equality corpus before suite runtime');
const plan=JSON.parse(fs.readFileSync('/workspace/work/r4-krk-validation/test-plan/plan-'+mode+'.json.txt'));
const output=r+'/logs/'+label+'.txt';if(fs.existsSync(output))throw Error('fresh batch label required');
const started=Date.now(),record={mode,label,started:new Date(started).toISOString(),entries:[]};
fs.writeFileSync(output,JSON.stringify(record,null,2)+'\n',{flag:'wx'});
let pending=0;
for(const item of plan.runs){
 const meta=item.stdoutFile+'.metadata.txt';
 if(fs.existsSync(meta)){const previous=JSON.parse(fs.readFileSync(meta));if(!previous.ended)throw Error('unresolved existing attempt '+item.id);continue;}
 pending++;
 if(Date.now()-started>=15000)break;
 if(Date.now()+item.outerTimeoutMs+10000>Date.parse(state.soakBudgetEnd))break;
 const argv=['node',r+'/execute-suite-step.mjs',mode,'runs',item.id],entry={id:item.id,argv,started:new Date().toISOString()};
 record.entries.push(entry);fs.writeFileSync(output,JSON.stringify(record,null,2)+'\n');
 const result=cp.spawnSync(argv[0],argv.slice(1),{encoding:'utf8',timeout:item.outerTimeoutMs+3000,maxBuffer:2*1024*1024});
 Object.assign(entry,{ended:new Date().toISOString(),exit:result.status,signal:result.signal,error:result.error?.message,stdout:result.stdout,stderr:result.stderr});
 fs.writeFileSync(output,JSON.stringify(record,null,2)+'\n');
 if(result.signal||!fs.existsSync(meta)||!JSON.parse(fs.readFileSync(meta)).ended)throw Error('executor interrupted; retain and inspect '+item.id);
 const m=JSON.parse(fs.readFileSync(meta));console.log(JSON.stringify({mode,id:item.id,exit:m.exit,passed:m.passed,incomplete:m.incomplete,sanitizer:m.sanitizer,reviewMarkers:m.reviewMarkers}));
}
record.ended=new Date().toISOString();record.completedEntries=record.entries.length;
record.remaining=plan.runs.filter(x=>!fs.existsSync(x.stdoutFile+'.metadata.txt')).map(x=>x.id);
fs.writeFileSync(output,JSON.stringify(record,null,2)+'\n');
console.log(JSON.stringify({label,completedEntries:record.completedEntries,remaining:record.remaining.length,elapsedSeconds:(Date.now()-started)/1000}));
