import fs from 'node:fs';import crypto from 'node:crypto';
const here='/workspace/work/r4-krk-rate';fs.copyFileSync(here+'/hash-evidence.mjs',here+'/hash-evidence.mjs.txt');
for(const root of ['/workspace/work/r4-fixture-repeat',here]){const files=[];function visit(dir){for(const e of fs.readdirSync(dir,{withFileTypes:true})){const p=dir+'/'+e.name;if(e.isDirectory())visit(p);else if(e.isFile()&&p!==root+'/artifact-hashes.txt')files.push(p);}}visit(root);const lines=files.sort().map(p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex')+'  '+p.slice(root.length+1));fs.writeFileSync(root+'/artifact-hashes.txt',lines.join('\n')+'\n');console.log(JSON.stringify({root,hashedFiles:files.length}));}
