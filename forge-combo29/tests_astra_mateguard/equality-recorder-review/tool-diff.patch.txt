+      check(file.startsWith(path.resolve(destination) + '/'), 'package chunk escapes destination');
+      const bytes = fs.readFileSync(file);
+      check(chunk.offset === offset && chunk.bytes === bytes.length && chunk.sha256 === sha256(bytes), 'package chunk mismatch');
+      pieces.push(bytes); offset += bytes.length;
+    }
+    check(original.length === entry.bytes && sha256(original) === entry.sha256 && Buffer.concat(pieces).equals(original), 'package source/reconstruction mismatch');
+  }
+  return manifest;
+}
