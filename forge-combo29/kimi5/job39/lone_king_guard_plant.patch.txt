+bool lone_king_guard_score(const Board& board, int& score) {
+    // The lone side L has only its king; the other side S is scored.
+    const bool white_lone = (board.pieces(White, Pawn) | board.pieces(White, Knight) | board.pieces(White, Bishop) |
+                             board.pieces(White, Rook) | board.pieces(White, Queen)) == 0;
+    const bool black_lone = (board.pieces(Black, Pawn) | board.pieces(Black, Knight) | board.pieces(Black, Bishop) |
+                             board.pieces(Black, Rook) | board.pieces(Black, Queen)) == 0;
+    if (white_lone == black_lone) return false;
+    const Color lone = white_lone ? White : Black;
+    const Color strong = opposite(lone);
+    const int queens = popcount(board.pieces(strong, Queen));
+    const int rooks = popcount(board.pieces(strong, Rook));
+    const int bishops = popcount(board.pieces(strong, Bishop));
+    const int knights = popcount(board.pieces(strong, Knight));
+    const int pawns = popcount(board.pieces(strong, Pawn));
+    const Bitboard strong_bishops = board.pieces(strong, Bishop);
+    const bool heavy = queens > 0 || rooks > 0;
+    const bool both_colours = (strong_bishops & DarkSquares) != 0 && (strong_bishops & LightSquares) != 0;
+    // Guarded: a queen or a rook (anything else allowed), or, with no queen,
+    // rook or pawn, bishop+knight or bishops on both square colours.
+    if (!heavy && (pawns > 0 || (bishops == 0 || knights == 0) && !both_colours)) return false;
+    const int material = 900 * queens + 500 * rooks + 300 * (bishops + knights) + 100 * pawns;
+    const int lone_sq = board.king_square(lone);
+    const int strong_sq = board.king_square(strong);
+    const int lf = file_of(lone_sq);
+    const int lr = rank_of(lone_sq);
+    int edge = lf < lr ? lf : lr;
+    const int fe = 7 - lf < 7 - lr ? 7 - lf : 7 - lr;
+    if (fe < edge) edge = fe;
+    const int kd = square_distance(lone_sq, strong_sq);
+    // Corner case: no queen, no rook, a knight, and all bishops on one colour.
+    const bool corner_case = !heavy && knights > 0 && !both_colours;
+    int push;
+    if (corner_case) {
+        const bool dark = (strong_bishops & DarkSquares) != 0;
+        const int corner_a = dark ? 0 : 56;   // a1 or a8
+        const int corner_b = dark ? 63 : 7;   // h8 or h1
+        const int cd = std::min(square_distance(lone_sq, corner_a), square_distance(lone_sq, corner_b));
+        push = 50 * (7 - cd) + 20 * (7 - kd);
+    } else {
+        push = 100 * edge + 20 * (7 - kd);
+    }
+    const int value = 2000 + material + push;
+    if (board.side_to_move() == lone) {
+        // Stalemate: the lone side to move, not in check, with no legal move.
+        if (!board.in_check() && board.legal_moves().size() == 0) {
+            score = 0;
+            return true;
+        }
+        score = -value;
+    } else {
+        score = value;
+    }
+    return true;
+}
