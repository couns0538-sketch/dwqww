#!/usr/bin/env python3
"""Validate compressed ASCII evidence; regenerate original manifest/hash chunks."""
import argparse,base64,hashlib,json,lzma,pathlib,re,tempfile,shutil

def safe_name(s):
 if not isinstance(s,str) or not re.fullmatch(r'[A-Za-z0-9][A-Za-z0-9_.-]*',s) or '..' in s:raise ValueError('unsafe basename')
 return s

def safe_rel(s):
 if not isinstance(s,str) or not s or '\\' in s or ':' in s:raise ValueError('unsafe relative path')
 p=pathlib.PurePosixPath(s)
 if p.is_absolute() or '..' in p.parts or str(p)!=s:raise ValueError('unsafe relative path')
 return p

def digest(b):return hashlib.sha256(b).hexdigest()
def check(b,n,h):
 if not isinstance(n,int) or n<0 or len(b)!=n or digest(b)!=h:raise ValueError('size/hash mismatch')
def read_entry(root,e):
 q=root/safe_name(e['name'])
 if q.is_symlink():raise ValueError('symlink input refused')
 b=q.read_bytes();check(b,e['bytes'],e['sha256']);return b

def decode(root,output,manifest_root=None):
 if manifest_root is None:manifest_root=root
 if (root/'transport_index.json').exists():
  if (root/'transport_index.json').is_symlink():raise ValueError('symlink index refused')
  ib=(root/'transport_index.json').read_bytes()
 else:
  im=json.loads((root/'index.transport.json').read_text())
  if im['format']!='lossless-text-transport-parts-v1':raise ValueError('index transport format')
  ib=b''.join(read_entry(root,e) for e in im['parts']);check(ib,im['original_bytes'],im['original_sha256'])
 idx=json.loads(ib)
 if idx['format']!='xz-base64-original-text-stream-v1':raise ValueError('format')
 if output.exists() or output.is_symlink():raise FileExistsError('output must be a new directory')
 # Trusted index hashes are independently published; do not treat tampered index as authenticity.
 mt=read_entry(manifest_root,idx['manifest_transport']);meta=json.loads(mt)
 if meta['format']!='lossless-text-transport-parts-v1':raise ValueError('manifest transport format')
 mb=b''.join(read_entry(manifest_root,e) for e in meta['parts']);check(mb,idx['manifest_bytes'],idx['manifest_sha256']);check(mb,meta['original_bytes'],meta['original_sha256']);m=json.loads(mb)
 if m['format']!='utf8-concatenated-text-stream-v1' or m['stream_bytes']!=idx['stream_bytes']:raise ValueError('original manifest format/size')
 for e in m['archived_files']+m['already_remote']:safe_rel(e['path'])
 recon=read_entry(root,idx['original_reconstructor'])
 with tempfile.TemporaryFile() as compressed,tempfile.TemporaryFile() as stream:
  ch=hashlib.sha256();cn=0;seen=set()
  for i,e in enumerate(idx['parts']):
   if e['index']!=i or e['name'] in seen:raise ValueError('part order/duplicate')
   seen.add(e['name']);b=read_entry(root,e)
   if len(b)>100000 or len(b)%4:raise ValueError('base64 part sizing')
   raw=base64.b64decode(b,validate=True)
   if base64.b64encode(raw)!=b:raise ValueError('noncanonical base64')
   compressed.write(raw);ch.update(raw);cn+=len(raw)
  if cn!=idx['compressed_bytes'] or ch.hexdigest()!=idx['compressed_sha256']:raise ValueError('compressed aggregate mismatch')
  compressed.seek(0);sh=hashlib.sha256();sn=0
  with lzma.LZMAFile(compressed,'rb') as f:
   while b:=f.read(1024*1024):
    sn+=len(b)
    if sn>idx['stream_bytes']:raise ValueError('decompression exceeds declared size')
    sh.update(b);stream.write(b)
  if sn!=idx['stream_bytes'] or sh.hexdigest()!=idx['stream_sha256']:raise ValueError('stream mismatch')
  off=0
  for i,e in enumerate(m['chunks']):
   safe_rel(e['path']);name=pathlib.PurePosixPath(e['path']).name;safe_name(name)
   if name!=e['sha256']+'.txt' or e['index']!=i or e['offset']!=off:raise ValueError('chunk sequence/name')
   stream.seek(off);check(stream.read(e['bytes']),e['bytes'],e['sha256']);off+=e['bytes']
  if off!=sn:raise ValueError('chunk total')
  for e in m['archived_files']:
   off=e['stream_offset'];n=e['bytes']
   if not isinstance(off,int) or not isinstance(n,int) or off<0 or n<0 or off+n>sn:raise ValueError('archived file extent')
   stream.seek(off);b=stream.read(n);check(b,n,e['sha256']);b.decode('utf-8')
  # All content verified before creating any output. Fresh private temp child avoids symlink traversal.
  output.parent.mkdir(parents=True,exist_ok=True)
  with tempfile.TemporaryDirectory(prefix='forge-verified-',dir=output.parent) as td:
   dest=pathlib.Path(td)/'result';dest.mkdir();chunks=dest/'chunks';chunks.mkdir()
   for e in m['chunks']:
    stream.seek(e['offset']);q=chunks/pathlib.PurePosixPath(e['path']).name
    b=stream.read(e['bytes'])
    if q.exists():check(q.read_bytes(),e['bytes'],e['sha256'])
    else:q.write_bytes(b)
   (dest/'manifest.json').write_bytes(mb);(dest/'reconstruct.py.txt').write_bytes(recon)
   if output.exists() or output.is_symlink():raise FileExistsError('output appeared during validation')
   dest.rename(output)
 return len(m['archived_files']),len(m['chunks'])
if __name__=='__main__':
 p=argparse.ArgumentParser(description=__doc__);p.add_argument('transport',type=pathlib.Path);p.add_argument('--output',required=True,type=pathlib.Path);p.add_argument('--manifest-directory',type=pathlib.Path,help='reuse existing manifest.transport.json and9 parts from this directory');a=p.parse_args();n,c=decode(a.transport,a.output,a.manifest_directory);print('PASS',n,'archived files verified;',c,'original chunks regenerated.
