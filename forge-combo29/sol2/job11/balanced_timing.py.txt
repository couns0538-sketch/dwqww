#!/usr/bin/env python3
"""Measure groups of pinned Forge engines with alternating ABBA and BAAB blocks."""

import argparse
import hashlib
import json
import math
import os
from pathlib import Path
from statistics import median
import sys
from concurrent_measure import measure_group


NET_SHA256 = "c3d95131d7c1c89ccb5b000b52c7c37ed8665dbf8bb2a0a7a873809c85b9f7c7"
WORKLOAD = ("--games", "12", "--nodes", "20000", "--random-plies", "8", "--max-plies", "300")
SIGNATURE_FIELDS = ("games", "searches", "total_nodes", "trace_fnv1a64")
ORDER = ("ABBA", "BAAB")
SCHEMA = "forge-balanced-concurrent-v1"
METRICS = ("engine_cpu_s", "total_wall_ns")


def pinned_identity(path, expected):
    path = path.resolve(strict=True)
    with path.open("rb") as content:
        actual = hashlib.file_digest(content, "sha256").hexdigest()
    if actual != expected:
        raise ValueError(f"SHA-256 mismatch for {path}: expected {expected}, got {actual}")
    return {"path": str(path), "sha256": actual}


def write_json(path, content):
    with path.open("x", encoding="utf-8") as output:
        print(json.dumps(content, indent=2), file=output)


def summarize_runs(runs, requested_blocks, floor=None):
    blocks = []
    complete_runs = []
    by_parity = {parity: {key: [] for key in METRICS} for parity in ("odd", "even")}
    for offset in range(0, len(runs), 5):
        block_index = offset // 5 + 1
        order = ORDER[(block_index - 1) % len(ORDER)]
        group = runs[offset:offset + 5]
        if len(group) != 5 or not all(row["eligible"] for row in group):
            break
        warmup, *measured = group
        if not warmup["warmup"] or warmup["arm"] != order[0] or warmup["order_index"] != 0:
            raise ValueError("Invalid first-arm warm-up receipt")
        if "".join(row["arm"] for row in measured) != order or any(row["warmup"] for row in measured):
            raise ValueError("Invalid timed order or warm-up flag")
        block = {"block": block_index, "order": order}
        parity = "odd" if block_index % 2 else "even"
        for key in METRICS:
            total_a = sum(row[key] for row in measured if row["arm"] == "A")
            total_b = sum(row[key] for row in measured if row["arm"] == "B")
            block[key] = total_b / total_a
            average = (total_a + total_b) / len(measured)
            by_parity[parity][key].append([row[key] / average for row in measured])
        blocks.append(block)
        complete_runs.extend(measured)
    complete = len(blocks) == requested_blocks and len(runs) == requested_blocks * 5
    position_means = {
        parity: {
            key: [sum(column) / len(columns) for column in zip(*columns)]
            for key, columns in data.items()
        }
        for parity, data in by_parity.items()
    }
    metrics = {}
    for key in METRICS:
        if not blocks:
            break
        total_a = sum(row[key] for row in complete_runs if row["arm"] == "A")
        total_b = sum(row[key] for row in complete_runs if row["arm"] == "B")
        ratios = [block[key] for block in blocks]
        middle = median(ratios)
        slower = sum(value > 1 for value in ratios)
        faster = sum(value < 1 for value in ratios)
        result = {"pooled_b_over_a": total_b / total_a, "block_median": middle,
                  "block_min": min(ratios), "block_max": max(ratios), "below_one": faster,
                  "above_one": slower, "equal_one": len(ratios) - faster - slower}
        if floor is not None:
            low, high = min(floor[key]), max(floor[key])
            outside = middle < low or middle > high
            signs_agree = max(faster, slower) >= 10
            decision = "incomplete"
            if complete and requested_blocks == 12:
                decision = "MEASURED" if outside and signs_agree else "not measurable here"
            result.update({"floor_min": low, "floor_max": high, "median_outside_floor": outside,
                           "at_least_10_on_one_side": signs_agree, "decision": decision})
        metrics[key] = result
    return {"schema": SCHEMA, "order": ORDER, "comparison_direction": "B / A, less than 1 is faster",
            "requested_blocks": requested_blocks, "completed_blocks": len(blocks),
            "attempted_runs": len(runs), "included_runs": len(complete_runs),
            "included_warmups": len(blocks), "complete": complete, "blocks": blocks, "metrics": metrics,
            "order_position_means": position_means}


def load_floor(path, current):
    source = path.resolve(strict=True)
    data = source.read_bytes()
    metadata = {"path": str(source), "sha256": hashlib.sha256(data).hexdigest()}
    floor = json.loads(data)
    if not isinstance(floor, dict):
        raise ValueError("Expected one balanced summary object")
    if (floor.get("schema") != SCHEMA or floor.get("order") != list(ORDER)
            or not floor.get("identical_floor") or not floor.get("complete") or floor.get("error") is not None
            or floor.get("requested_blocks") != 12 or floor.get("completed_blocks") != 12
            or floor.get("attempted_runs") != 60 or floor.get("included_runs") != 48
            or floor.get("included_warmups") != 12
            or floor.get("expected_games_searches_nodes_trace") != list(current["expected_games_searches_nodes_trace"])
            or floor.get("worker_cpus") != current["worker_cpus"]
            or floor.get("worker_count") != current["worker_count"]
            or floor.get("worker_affinity") != current["worker_affinity"]):
        raise ValueError("Floor must be a complete, eligible twelve-block identical balanced campaign on the same workload/affinity")
    pins = floor["verified_inputs_after"]
    if (pins["A"]["sha256"] != pins["B"]["sha256"] or pins["A"]["path"] == pins["B"]["path"]
            or any(pins[name]["sha256"] != current["inputs"][name]["sha256"]
                   for name in ("net", "driver", "openings", "known_answer"))):
        raise ValueError("Floor must have byte-identical A/B executables at distinct paths and the same workload inputs")
    checked = {key: [] for key in METRICS}
    if len(floor["blocks"]) != 12:
        raise ValueError("Expected twelve raw floor block ratios")
    for expected_index, block in enumerate(floor["blocks"], start=1):
        if block["block"] != expected_index or block["order"] != ORDER[(expected_index - 1) % len(ORDER)]:
            raise ValueError("Unexpected floor block ordering")
        for key in METRICS:
            ratio = block[key]
            if type(ratio) not in (int, float) or not math.isfinite(ratio) or ratio <= 0:
                raise ValueError("Nonpositive or nonfinite floor ratio")
            checked[key].append(ratio)
    return checked, metadata


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("a", type=Path)
    parser.add_argument("b", type=Path)
    parser.add_argument("net", type=Path)
    parser.add_argument("output", type=Path, help="new output directory")
    parser.add_argument("--a-sha256", required=True)
    parser.add_argument("--b-sha256", required=True)
    parser.add_argument("--a-label", default="A")
    parser.add_argument("--b-label", default="B")
    parser.add_argument("--driver", type=Path, required=True)
    parser.add_argument("--driver-sha256", required=True)
    parser.add_argument("--openings", type=Path, required=True)
    parser.add_argument("--openings-sha256", required=True)
    parser.add_argument("--known-answer", type=Path, required=True)
    parser.add_argument("--known-answer-sha256", required=True)
    parser.add_argument("--worker-cpus", required=True)
    parser.add_argument("--blocks", type=int, choices=(12,), required=True)
    parser.add_argument("--identical-floor", action="store_true", help="mark this twelve-block A/copy experiment as the reference floor")
    parser.add_argument("--floor", type=Path, help="summary.txt from a completed --identical-floor campaign")
    args = parser.parse_args()
    cpus = [int(item) for item in args.worker_cpus.split(",")]
    if len(cpus) < 2 or len(cpus) != len(set(cpus)) or not set(cpus) <= os.sched_getaffinity(0):
        parser.error("Worker CPU list must contain distinct available logical CPUs")
    if args.identical_floor and args.floor:
        parser.error("--identical-floor and --floor are mutually exclusive")
    args.output.mkdir(parents=True, exist_ok=False)
    pins = {"A": (args.a, args.a_sha256), "B": (args.b, args.b_sha256),
            "net": (args.net, NET_SHA256), "driver": (args.driver, args.driver_sha256),
            "openings": (args.openings, args.openings_sha256),
            "known_answer": (args.known_answer, args.known_answer_sha256)}
    inputs = {key: pinned_identity(*spec) for key, spec in pins.items()}
    gate = json.loads(Path(inputs["known_answer"]["path"]).read_text(encoding="utf-8"))
    expected = tuple(gate[name] for name in SIGNATURE_FIELDS)
    if expected[0] != 12 or min(expected[1:3]) <= 0 or len(expected[3]) != 16:
        raise ValueError("Invalid fixed-node known-answer receipt")
    labels = {"A": args.a_label, "B": args.b_label}
    if args.identical_floor and (inputs["A"]["sha256"] != inputs["B"]["sha256"] or inputs["A"]["path"] == inputs["B"]["path"]):
        raise ValueError("The reference floor requires byte-identical executables at distinct paths")
    argv = {name: [inputs["driver"]["path"], "--engine", inputs[name]["path"],
                   "--net", inputs["net"]["path"], *WORKLOAD,
                   "--openings", inputs["openings"]["path"], "--json"] for name in "AB"}
    manifest = {"schema": SCHEMA, "inputs": inputs, "arm_labels": labels, "argv_by_arm": argv,
                "order": ORDER, "blocks_requested": args.blocks,
                "expected_games_searches_nodes_trace": expected, "identical_floor": args.identical_floor,
                "cpu_scope": "sum of the N separately pinned Forge children user plus system time",
                "wall_scope": "earliest worker start to latest worker end on the shared monotonic clock",
                "worker_affinity": cpus, "worker_cpus": cpus, "worker_count": len(cpus)}
    floor, floor_source = load_floor(args.floor, manifest) if args.floor else (None, None)
    manifest["floor"] = floor_source
    write_json(args.output / "inputs.txt", manifest)
    runs = []
    failure = None
    with (args.output / "runs.jsonl.txt").open("x", encoding="utf-8") as log:
        for block in range(1, args.blocks + 1):
            order = ORDER[(block - 1) % len(ORDER)]
            for index, arm in enumerate(order[0] + order):
                stem = f"b{block:02d}-r{index}-{arm}"
                row = {"block": block, "order_index": index, "warmup": index == 0, "arm": arm, "engine": labels[arm],
                       "engine_sha256": inputs[arm]["sha256"],
                       **measure_group(argv[arm], args.output, stem, expected, cpus)}
                runs.append(row)
                line = json.dumps(row)
                print(line, file=log, flush=True)
                print(line, flush=True)
                if not row["eligible"]:
                    failure = f"{stem}: {row['error']}"
                    break
            if failure:
                break
    try:
        after = {key: pinned_identity(*spec) for key, spec in pins.items()}
        if after != inputs:
            raise ValueError("An input path changed during the campaign")
        if floor_source and pinned_identity(Path(floor_source["path"]), floor_source["sha256"]) != floor_source:
            raise ValueError("The floor file changed during the campaign")
    except (OSError, ValueError) as exc:
        # Retain the original rows while withholding the measurements from the ratio.
        summary = summarize_runs([], args.blocks)
        summary["attempted_runs"] = len(runs)
        failure = f"Post-run identity failed: {exc}"
    else:
        summary = summarize_runs(runs, args.blocks, floor)
        summary["verified_inputs_after"] = after
    summary.update({"arm_labels": labels, "error": failure, "identical_floor": args.identical_floor,
                    "floor": floor_source, "expected_games_searches_nodes_trace": expected,
                    "worker_affinity": manifest["worker_affinity"],
                    "worker_cpus": cpus, "worker_count": len(cpus)})
    write_json(args.output / "summary.txt", summary)
    if failure:
        raise RuntimeError(failure)
    print(json.dumps(summary), flush=True)


if __name__ == "__main__":
    try:
        main()
    except (KeyError, OSError, RuntimeError, ValueError) as exc:
        print(exc, file=sys.stderr)
        sys.exit(1)
