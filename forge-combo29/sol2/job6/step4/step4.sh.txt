#!/usr/bin/env bash
set -euxo pipefail
job=/workspace/work/sol2-job6
base=/workspace/work/sol2-job2
before=/workspace/work/sol2-job5
previous=/workspace/work/sol2-job3
prof="$previous/profiler"
private="$job/profile-private"
out=/workspace/work/sol2-repo/forge-combo29/sol2/job6/step4
cmake=/workspace/work/sol2-tools/cmake/data/bin/cmake
exe="$job/build-k11-fp/forge"
control="$job/build-see-control/forge"
reference="$previous/build-r15d-fp/forge"
driver="$base/carbon2/selfplay_driver.py"
net="$base/net768.fnn3"
flags='-DCMAKE_CXX_FLAGS=-DFORGE_MATE_ALPHA_GUARD=1 -DFORGE_NNUE_V3=1 -g -fno-omit-frame-pointer'
args=(--net "$net" --games 60 --softnodes 2000 --random-plies 8 --max-plies 300 --seed 20261117 --json)
mkdir "$private"
sha256sum "$prof/cpu_sampler.so" "$driver" "$net" "$before/k11/CMakeLists.txt" "$reference" "$out/see-spin.patch.txt"
"$cmake" -S "$before/k11" -B "$job/build-k11-fp" -DCMAKE_BUILD_TYPE=Release "$flags"
"$cmake" --build "$job/build-k11-fp" --parallel 3 --verbose > "$out/k11-build.raw.txt" 2>&1
sha256sum "$exe"
printf 'uci\nquit\n' | "$exe" > "$out/k11-uci.raw.txt"
timeout 240 taskset -c 2 python -B "$driver" --engine "$exe" "${args[@]}" > "$out/frame-pointer-gate.stdout.txt" 2> "$out/frame-pointer-gate.stderr.txt"
python - "$out/frame-pointer-gate.stdout.txt" <<'PY'
from pathlib import Path
import sys,json
row=json.loads(Path(sys.argv[1]).read_text()); print(row)
assert (row['searches'],row['total_nodes'],row['trace_fnv1a64'])==(8612,29333343,'5c705841c3ea17b4')
PY
"$cmake" -S "$job/k11-control" -B "$job/build-see-control" -DCMAKE_BUILD_TYPE=Release "$flags"
"$cmake" --build "$job/build-see-control" --parallel 3 --verbose > "$out/control-build.raw.txt" 2>&1
sha256sum "$control"
run_profile() {
  local name=$1 binary=$2 prefix="$private/sampler-$1"
  date -u +'%FT%TZ'
  timeout 240 taskset -c 2 python -B "$job/profile_one.py" "$binary" "$prefix" "$out" "$name"
  cp "$prefix.engine.stderr.txt" "$out/$name.sampler.stderr.txt"
  python -B "$prof/symbolize.py" "$prefix" "$private/$name" --libc-debug "$base/profiler/libc.debuginfo"
  python -B "$prof/profile_groups.py" "$private/$name"
  for receipt in sampler_summary.txt top15.tsv groups.tsv; do
    cp "$private/$name/$receipt" "$out/$name.$receipt.txt"
  done
  date -u +'%FT%TZ'
}
run_profile control "$control"
run_profile k11-1 "$exe"
ln -s k11-1 "$private/clean-1"
python -B "$previous/aggregate.py" control "$private" "$private" | tee "$out/positive-control.raw.txt"
run_profile r15d-1 "$reference"
run_profile k11-2 "$exe"
run_profile r15d-2 "$reference"
run_profile r15d-3 "$reference"
run_profile k11-3 "$exe"
profile_args=()
for name in r15d k11; do
  for index in 1 2 3; do
    profile_args+=("--$name" "$private/$name-$index/symbolized_pcs.tsv" "$out/$name-$index.cpu.json.txt")
  done
done
python -B "$job/profile_compare.py" "${profile_args[@]}" --output "$out/compare"
