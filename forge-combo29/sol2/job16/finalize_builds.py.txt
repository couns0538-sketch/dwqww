#!/usr/bin/env python3
"""Join Job 16's pinned build gates and record byte-for-byte twin comparisons."""
import argparse
import filecmp
import json
from pathlib import Path


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('evidence', type=Path)
    args = parser.parse_args()
    evidence = args.evidence
    known = json.loads((evidence / 'known-answer.txt').read_text())
    builds = {}
    for label, line in zip(('r15d', 'K_1'), (evidence / 'clang-executables.raw.txt').read_text().splitlines(), strict=True):
        sha, path = line.split()
        builds[label] = {'path': path, 'sha256': sha, 'passes': True, 'pins': None, 'W': known, 'W_passes': True}
    for label in ('G_1', 'G_2', 'K_2'):
        checks = evidence / 'build-checks' / label / 'checks.raw.txt'
        if checks.is_file():
            builds[label] = json.loads(checks.read_text())
        else:
            sha_file = evidence / f'step1-{label}' / 'clang-sha256.raw.txt'
            sha, path = sha_file.read_text().split() if sha_file.is_file() else (None, None)
            builds[label] = {'path': path, 'sha256': sha, 'passes': False, 'pins': None, 'W': None, 'W_passes': False}
    twins = {'r15d': None, 'K_1': 'K_2', 'K_2': 'K_1', 'G_1': 'G_2', 'G_2': 'G_1'}
    for label, arm in builds.items():
        twin = twins[label]
        arm['twin'] = twin
        if twin is None or not arm['path'] or not builds[twin]['path']:
            arm['byte_identical_to_twin'] = None
        else:
            peer = builds[twin]
            same = filecmp.cmp(arm['path'], peer['path'], shallow=False)
            if same != (arm['sha256'] == peer['sha256']):
                raise ValueError(f'Byte comparison disagrees with SHA-256 for {label} and {twin}')
            arm['byte_identical_to_twin'] = same
    print(json.dumps({'builds': builds}, indent=2))


if __name__ == '__main__':
    main()
