def require(condition, message):
    if not condition:
        raise ValueError(message)


def load(path):
    return json.loads(path.read_text(encoding="utf-8"))


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def signature(row):
    return tuple(row.get(field) for field in TRACE_FIELDS)


def check(actual, expected, where):
    if isinstance(expected, dict):
        require(isinstance(actual, dict) and expected.keys() <= actual.keys(), f"missing fields at {where}")
        for key, value in expected.items():
            check(actual[key], value, f"{where}.{key}")
    elif isinstance(expected, (tuple, list)):
        require(isinstance(actual, (tuple, list)) and len(actual) == len(expected), f"list shape differs at {where}")
        for index, (one, two) in enumerate(zip(actual, expected)):
            check(one, two, f"{where}[{index}]")
    elif type(expected) is bool:
        require(actual is expected, f"wrong {where}: {actual!r}, expected {expected!r}")
    else:
        equal = math.isclose(actual, expected, rel_tol=1e-9, abs_tol=1e-12) if type(expected) is float else actual == expected
        require(equal, f"wrong {where}: {actual!r}, expected {expected!r}")


def read_rows(path, finished):
    if not path.is_file():
        return [], False
    lines = path.read_text(encoding="utf-8").splitlines(keepends=True)
    rows = []
    for index, line in enumerate(lines, 1):
        try:
            rows.append(json.loads(line))
        except json.JSONDecodeError:
            if not finished and index == len(lines) and not line.endswith("\n"):
                return rows, True
            raise
    return rows, False


def validate_run(directory, stem, row, expected):
    source = f"{directory}/{stem}"
    stdout, stderr = f"{stem}.driver.stdout.txt", f"{stem}.engine.stderr.txt"
    check(row, {"driver_stdout": stdout, "engine_stderr": stderr}, source)
    require((directory / stdout).is_file() and (directory / stderr).is_file(), f"missing driver/engine receipts for {source}")
    actual = signature(row)
    check(row["trace_matches"], actual == expected, f"{source} signature flag")
    start, end = row["wall_start_monotonic_ns"], row["wall_end_monotonic_ns"]
    user, system = row["engine_user_s"], row["engine_system_s"]
    check(row, {"total_wall_ns": end - start, "engine_cpu_s": user + system}, f"{source} durations")
    if row["eligible"] or any(value is not None for value in actual):
        driver = load(directory / stdout)
        check(row, driver, f"{source} driver stdout")
    if row["eligible"]:
        require(actual == expected and row["engine_exit"] == 0 and row["error"] is None
                and user >= 0 and system >= 0 and user + system > 0 and end > start,
                f"{source} marked eligible without correct identity, status and durations")
    else:
        require(bool(row["error"]), f"{source} excluded without a recorded error")


def validate_tape(path, receipt):
    rows, _ = read_rows(path, True)
    last_info_total = final_total = final_lines = zero_last = 0
    hash_value = 14695981039346656037
    current_book_line = 7000
    current_ply = 0
    seen = set()
    for row in rows:
        book_line = row["book_line"]
        require(7001 <= book_line <= 7020 and book_line >= current_book_line, f"unordered book line in {path}")
        current_ply = current_ply + 1 if book_line == current_book_line else 1
        current_book_line = book_line
        seen.add(book_line)
        check(row["ply"], current_ply, f"{path} game {book_line} ply")
        match = re.search(r"\bnodes (\d+)(?=\s|$)", row["last_info"] or "")
        check(row["nodes"], int(match[1]) if match else 0, f"{path} game {book_line} last-info nodes")
        move = row["bestmove_line"].split()
        require(move[:2] == ["bestmove", row["bestmove"]], f"bestmove differs from its UCI line in {path}")
        final = row["final_nodes"]
        if final is None:
            require(row["final_nodes_line"] is None and row["nodes"] == 0, f"missing final soft-node telemetry in {path}")
        else:
            require(final >= row["nodes"], f"final soft nodes below last-info nodes in {path}")
            check(row["final_nodes_line"], f"info string softnodes nodes {final}", f"{path} final soft-node line")
            final_lines += 1
            final_total += final
        last_info_total += row["nodes"]
        zero_last += row["nodes"] == 0
        wire = f"{row['bestmove']},{row['nodes']};"
        check(row["trace_input"], wire, f"{path} trace input")
        for byte in wire.encode("ascii"):
            hash_value = (hash_value ^ byte) * 1099511628211 & ((1 << 64) - 1)
        check(row["trace_fnv1a64"], f"{hash_value:016x}", f"{path} cumulative trace")
    require(seen == set(range(7001, 7021)), f"wrong W book lines in {path}")
    check(receipt, {"searches": len(rows), "total_nodes": last_info_total, "final_nodes": final_total,
                    "trace_fnv1a64": f"{hash_value:016x}", "final_lines": final_lines,
                    "missing_final_lines": len(rows) - final_lines, "zero_last_info_searches": zero_last},
          f"{path} tape summary")


def read_pins(path, labels="AB"):
    lines = path.read_text(encoding="utf-8").splitlines()
    require(len(lines) == len(labels), f"expected one executable receipt per engine in {path}")
    pins = {}
    for arm, line in zip(labels, lines, strict=True):
        sha256, executable = line.split(maxsplit=1)
        require(re.fullmatch(r"[0-9a-f]{64}", sha256), f"invalid executable digest in {path}")
        pins[arm] = {"path": executable, "sha256": sha256}
    return pins


def known_workload(root):
    path = root / "known-answer.txt"
    source = load(path)
    expected = signature(source)
    require(source.keys() == set(TRACE_FIELDS) and expected[0] == 20
            and all(type(value) is int and value > 0 for value in expected[:4])
            and expected[3] >= expected[2]
            and isinstance(expected[4], str) and re.fullmatch(r"[0-9a-f]{16}", expected[4]),
            f"invalid W signature in {path}")
    pins = read_pins(root / "clang-executables.raw.txt")
    for key, arm in zip("AB", ("r15d", "kF")):
        gate = root / "step2"
        data = load(gate / f"{arm}.result.raw.txt")
        validate_run(gate, arm, data, expected)
        check(data["engine_sha256"], pins[key]["sha256"], f"Step 2 {arm} executable")
        require(data["eligible"], f"Step 2 {arm} failed the W gate")
        validate_tape(gate / f"{arm}.searches.raw.txt", data)
    return Known(expected, {"net": NET_SHA256, "book": BOOK_SHA256, "known_answer": digest(path),
                            "driver": digest(root / "soft_book_driver.py.txt")}, pins)


def timing_math(groups, requested_blocks):
    blocks = []
    positions = {cohort: {key: [] for key, _ in METRICS} for cohort in ("odd", "even")}
    timed_runs = []
    for group in groups:
        measured = group[1:]
        timed_runs.extend(measured)
        block_index = group[0]["block"]
        cohort = "odd" if block_index % 2 else "even"
        block = {"block": block_index, "order": ORDERS[(block_index - 1) % len(ORDERS)]}
        for key, _ in METRICS:
            totals = {arm: math.fsum(row[key] for row in measured if row["arm"] == arm) for arm in "AB"}
            block[key] = totals["B"] / totals["A"]
            mean = fmean(row[key] for row in measured)
            positions[cohort][key].append([row[key] / mean for row in measured])
        blocks.append(block)
    means = {cohort: {key: [fmean(position) for position in zip(*series)] for key, series in data.items()}
             for cohort, data in positions.items()}
    metrics = {}
    count = len(blocks)
    for key, _ in METRICS:
        if count == 0:
            break
        values = [block[key] for block in blocks]
        ordered = sorted(values)
        block_mean = fmean(values)
        df = count - 1
        below, above = sum(value < 1 for value in values), sum(value > 1 for value in values)
        totals = {arm: math.fsum(row[key] for row in timed_runs if row["arm"] == arm) for arm in "AB"}
        data = {"block_mean": block_mean, "block_median": median(values), "block_min": ordered[0], "block_max": ordered[-1],
                "pooled_b_over_a": totals["B"] / totals["A"], "below_one": below, "above_one": above,
                "equal_one": count - below - above}
        if count == requested_blocks:
            halfwidth = T_CRITICAL[df] * stdev(values) / math.sqrt(count)
            data.update(block_mean_ci95=[block_mean - halfwidth, block_mean + halfwidth], ci_df=df)
        if count >= 3:
            data.update(block_middle_min=ordered[1], block_middle_max=ordered[-2])
        metrics[key] = data
    return blocks, metrics, means


def read_environment(record):
    wire = b"".join(f"{key}={value}".encode("ascii") + b"\0" for key, value in ENVIRONMENT.items())
    check(record, {"variables": ENVIRONMENT, "bytes": len(wire), "sha256": hashlib.sha256(wire).hexdigest()},
          "fixed engine environment")
    require(record["variables"] == ENVIRONMENT, "the engine environment must have just LC_ALL and TZ")
    return record


def engine_metadata(directory, stem, row, pin, environment):
    where = f"{directory}/{stem} launch"
    expected = {"engine_argv0": pin["path"], "engine_argv0_length": pin["length"],
                "engine_environment_bytes": environment["bytes"], "engine_environment_sha256": environment["sha256"]}
    if row["eligible"] or row["engine_pid"] is not None or not expected.keys().isdisjoint(row):
        check(row, expected, where)
    proc = row.get("engine_proc_receipt")
    require(proc in (None, "verified", "unavailable"), f"unexpected procfs receipt at {where}")
    if proc == "verified":
        check(row, {"engine_inode": pin["inode"], "engine_device": pin["device"]}, f"{where} procfs")


def interval_direction(interval):
    low, high = interval
    if high < 1:
        return "below"
    if low > 1:
        return "above"
    return "includes"

def verify_gate(root, known, frozen, environment, receipt_labels, *, render=False):
    pins = read_pins(root / "clang-executables.raw.txt", tuple(receipt_labels))
    for label, pin in pins.items():
        check(pin, {key: frozen[label][key] for key in ("path", "sha256")}, f"frozen launch pin {label} in {root}")
        check(len(pin["path"]), 72, f"Step 0 {label} path length")
    check(digest(root / "known-answer.txt"), known.hashes["known_answer"], f"{root} known-answer bytes")
    entries = [line.split(maxsplit=1)
               for line in (root / "workload-pins.raw.txt").read_text(encoding="utf-8").splitlines()]
    intake = {Path(path).name: sha256 for sha256, path in entries}
    require(len(intake) == len(entries), f"duplicate workload input pins in {root}")
    for name, key in (("soft_book_driver.py", "driver"), ("known-answer.txt", "known_answer"),
                      ("UHO_Lichess_4852_v1.epd", "book"), ("net768.fnn3", "net")):
        check(intake[name], known.hashes[key], f"{root} workload receipt {name}")
    compiler = root / "step0-command.raw.txt"
    compiler_ok = compiler.is_file() and re.search(r"\bclang version 19\.1\.7\b", compiler.read_text(encoding="utf-8"))
    if not compiler_ok:
        if render:
            print("Step 0 clang 19.1.7 receipt is missing or did not pass. [Compiler and Step 0](sol2/job23/step0-command.raw.txt)")
        return False
    gate = root / "step0"
    eligible = []
    pids = []
    for label, stored in receipt_labels.items():
        path = gate / f"{stored}.result.raw.txt"
        if not path.is_file():
            if render:
                print(f"Step 0 {label} receipt is missing.")
            continue
        row = load(path)
        validate_run(gate, stored, row, known.signature)
        check(row["engine_sha256"], pins[label]["sha256"], f"Step 0 {label} build")
        engine_metadata(gate, stored, row, frozen[label], environment)
        if not row["eligible"]:
            if render:
                print(f"Step 0 {label} failed with `{row['error']}`. [Receipt](sol2/job23/step0/{path.name})")
            continue
        pids.append(row["engine_pid"])
        eligible.append(f"[{label}](sol2/job23/step0/{path.name}#L2-L6)")
    if len(eligible) != len(receipt_labels):
        return False
    require(None not in pids and len(set(pids)) == len(receipt_labels), "Step 0 must use a fresh engine process per arm")
    if render:
        games, searches, nodes, final_nodes, trace = known.signature
        print(f"Both Step 0 builds matched W with {games} games, {searches:,} searches, {nodes:,} last-info nodes, "
              f"{final_nodes:,} final nodes and trace `{trace}`. W uses PC20, Threads 1, Hash 16 MiB, net768 and "
              f"`go softnodes 2000` on book lines 7001–7020. {' · '.join(eligible)} · [W signature](sol2/job23/known-answer.txt#L2-L6)")
    return True


def interval_text(metric):
    low, high = metric["block_mean_ci95"]
    return f"[{low:.4f}, {high:.4f}]"


def reading(metric):
    if metric is None:
        return None
    side = interval_direction(metric["block_mean_ci95"])
    return {"below": "PAYS", "above": "COSTS", "includes": "NOT SEEN"}[side]




def settings_reading(baseline_cpu, draw_cpu):
    if baseline_cpu is None or draw_cpu is None:
        return None
    baseline_lo, baseline_hi = baseline_cpu["block_mean_ci95"]
    draw_lo, draw_hi = draw_cpu["block_mean_ci95"]
    return "differs" if baseline_hi < draw_lo or draw_hi < baseline_lo else "same within the error"

def draw_options(root, workload):
    path = draw_path(root, workload)
    expected = dict(zip(SETTINGS, DRAW_VALUES[DRAW_LABELS[workload]], strict=True))
    pairs = json.loads(path.read_text(encoding="utf-8"), object_pairs_hook=list)
    require(isinstance(pairs, list) and all(isinstance(pair, tuple) and len(pair) == 2 for pair in pairs),
            f"{path} is not a JSON settings object")
    check(tuple(name for name, _ in pairs), SETTINGS, f"{path} ordered settings")
    require(all(type(value) is int for _, value in pairs), f"{path} values must be integers")
    observed = dict(pairs)
    check(observed, expected, f"{path} values")
    return observed


def read_uci(root, *, render=False):
    inventories = {}
    for label in REFERENCE_LABELS:
        path = root / "step0" / f"{label}.uci.stdout.txt"
        if not path.is_file():
            if render:
                print(f"The {label} UCI inventory is unavailable.")
            continue
        raw = path.read_text(encoding="utf-8").splitlines()
        require("uciok" in raw, f"{path} lacks uciok")
        options = {}
        for line in raw:
            match = re.fullmatch(r"option name (.+?) type spin default (-?\d+) min (-?\d+) max (-?\d+)", line.strip())
            if match and match[1] in SETTINGS:
                require(match[1] not in options, f"{path} repeats spin {match[1]}")
                options[match[1]] = tuple(map(int, match.groups()[1:]))
        missing = [name for name in SETTINGS if name not in options]
        if missing:
            if render:
                print(f"The {label} UCI inventory is missing required spin declarations {', '.join(missing)}. "
                      f"[Raw UCI inventory](sol2/job23/step0/{path.name})")
            continue
        for name, (default, lower, upper) in options.items():
            require(lower <= default <= upper, f"{path} spin {name} has an invalid default")
        inventories[label] = options
        if render:
            print(f"The {label} UCI inventory declares all {len(SETTINGS)} settings as spins. "
                  f"[Raw UCI inventory](sol2/job23/step0/{path.name})")
    return inventories


def check_workload_metadata(row, workload, root, where):
    check(row, {"book_start": 7001, "book_end": 7020, "softnodes": SOFT_LIMITS[workload]}, where)
    if workload not in DRAW_LABELS:
        require("option_receipts" not in row and "draw_sha256" not in row,
                f"{where} records unexpected draw settings")
        return
    verify_option_receipts(row, workload, root, where, finished=True)


def verify_option_receipts(row, workload, root, where, *, finished):
    expected = list(draw_options(root, workload).items())
    check(row["draw_sha256"], digest(draw_path(root, workload)), f"{where} draw digest")
    receipts = row["option_receipts"]
    require(isinstance(receipts, list) and len(receipts) <= len(expected), f"{where} invalid acknowledgment count")
    failure = row.get("draw_refusal")
    if finished:
        require(len(receipts) == len(expected) and failure is None, f"{where} incomplete acknowledgments")
    else:
        require(isinstance(failure, str) and bool(receipts), f"{where} missing failed acknowledgments")
    for index, (receipt, (name, value)) in enumerate(zip(receipts, expected)):
        check(receipt, {"name": name, "value": value}, f"{where} option {index + 1}")
        response = receipt["responses"]
        require(isinstance(response, list) and all(isinstance(line, str) for line in response),
                f"{where} invalid UCI response at {name}")
        refused = [line for line in response if REFUSAL.search(line)]
        ready = [index for index, line in enumerate(response) if line.strip() == "readyok"]
        last_failed = index == len(receipts) - 1 and not finished
        if last_failed:
            if refused:
                check(failure, f"Draw option refusal ({name}={value}): {refused[0]}",
                      f"{where} exact refused line")
            else:
                check(failure, f"Engine EOF before readyok for draw option {name}={value}",
                      f"{where} exact incomplete exchange")
                require(not ready, f"{where} says readyok was missing at {name}")
            require(not ready or ready == [len(response) - 1], f"{where} invalid readyok sequence at {name}")
        else:
            require(ready == [len(response) - 1] and not refused, f"{where} missing clean readyok at {name}")


def verify_refusal(directory, stem, row, workload, root):
    if workload not in DRAW_LABELS:
        return False
    path = directory / f"{stem}.driver.stdout.txt"
    error = row.get("error") or ""
    stopped = "Draw option refusal" in error
    if not path.stat().st_size:
        require(not stopped, f"{path} lacks the refusal receipt")
        return False
    observed = load(path)
    if "draw_refusal" not in observed:
        require(not stopped, f"{path} lacks the refusal line")
        return False
    failure = observed["draw_refusal"]
    require(not row["eligible"] and failure in error, f"{path} refusal disagrees with run status")
    verify_option_receipts(observed, workload, root, str(path), finished=False)
    check(stopped, failure.startswith("Draw option refusal"), f"{path} workload refusal status")
    return stopped


def read_draw_gates(root, frozen, environment):
    expected, gates = {"W": W_SIGNATURE}, {}
    for workload in ("WD1", "WD100", "W2200"):
        path = expected_path(root, workload)
        directory = root / "draw-gates" / workload
        values = None
        if path.is_file():
            known = load(path)
            values = signature(known)
            require(known.keys() == set(TRACE_FIELDS), f"unexpected known-answer fields in {path}")
            require(values[0] == 20 and all(type(value) is int and value > 0 for value in values[:4])
                    and values[3] >= values[2]
                    and isinstance(values[4], str) and re.fullmatch(r"[0-9a-f]{16}", values[4]),
                    f"invalid soft-node signature in {path}")
            expected[workload] = values
        first = {}
        for label, stem in (("r15d", "A.r15d"), ("K_1", "B.K_1")):
            receipt = directory / f"{stem}.result.raw.txt"
            if not receipt.is_file():
                continue
            require(workload != "W2200" or label == "r15d", f"unexpected K_1 control gate in {workload}")
            row = load(receipt)
            validate_run(directory, stem, row, values)
            check(row, {"engine_sha256": frozen[label]["sha256"], "workload": workload,
                        "expected_signature": None if label == "r15d" else values}, str(receipt))
            engine_metadata(directory, stem, row, frozen[label], environment)
            observed = signature(row)
            distinct = observed != W_SIGNATURE if row.get("trace_fnv1a64") else None
            check(row["differs_from_W"], distinct, f"{receipt} change from W")
            if row["eligible"] or row.get("games") is not None:
                check_workload_metadata(row, workload, root, str(receipt))
            if not row["eligible"]:
                verify_refusal(directory, stem, row, workload, root)
            first[label] = row
        if values is not None:
            require(first.get("r15d", {}).get("eligible"), f"{path} lacks a passed first r15d launch")
        pids = [row["engine_pid"] for row in first.values() if row.get("engine_pid") is not None]
        require(len(pids) == len(set(pids)), f"{workload} gates did not use fresh processes")
        gates[workload] = first
    return expected, gates


def signature_text(values):
    return " / ".join(f"{value:,}" if type(value) is int else f"`{value}`" if value is not None else "unavailable"
                      for value in values)

def verify_manifest(directory, manifest, expected, known, frozen, environment):
    step = directory.name
    workloads = dict(zip("AB", STEP_WORKLOADS[step], strict=True))
    signatures = {arm: expected[w] for arm, w in workloads.items()}
    labels = dict(zip("AB", labels_for(step), strict=True))
    check(manifest, {"schema": SCHEMA, "order": ORDERS, "blocks_requested": blocks_for(step),
                     "identical_floor": False, "floor": None, "expected_by_arm": signatures,
                     "workload_by_arm": workloads, "engine_environment": environment,
                     "worker_affinity": [2], "arm_labels": labels,
                     "expected_games_searches_nodes_trace": signatures["A"] if signatures["A"] == signatures["B"] else None},
          str(directory))
    inputs = manifest["inputs"]
    for key in ("net", "book"):
        check(inputs[key]["sha256"], known.hashes[key], f"{directory} {key} digest")
    driver = inputs["driver"]
    draw = workloads["A"] if workloads["A"] in DRAW_LABELS else None
    driver_pin = known.hashes["draw_driver"] if draw is not None else known.hashes["driver"]
    check(driver["sha256"], driver_pin, f"{directory} driver digest")
    for arm, label in labels.items():
        workload = workloads[arm]
        pin = frozen[label]
        check(inputs[arm], {key: pin[key] for key in ("path", "sha256")}, f"{directory} {arm} executable")
        for identity in ("device", "inode", "length"):
            if identity in inputs[arm]:
                check(inputs[arm][identity], pin[identity], f"{directory} {arm} {identity}")
        argv = [driver["path"], "--engine", pin["path"], "--net", inputs["net"]["path"],
                "--book", inputs["book"]["path"], *workload_argv(workload)]
        actual = manifest["argv_by_arm"][arm]
        if draw is not None:
            draw_input = inputs["draw"]
            expected_draw = draw_path(directory.parent, draw)
            check(draw_input["sha256"], digest(expected_draw), f"{directory} draw digest")
            check(Path(draw_input["path"]).name, expected_draw.name, f"{directory} draw filename")
            require(actual.count("--draw") == 1, f"{directory} {arm} draw argument count")
            draw_index = actual.index("--draw")
            check(actual[draw_index + 1], draw_input["path"], f"{directory} {arm} draw path")
            actual = actual[:draw_index] + actual[draw_index + 2:]
        check(actual, argv, f"{directory} {arm} argv")
        role = f"known_{arm}"
        check(inputs[role]["sha256"], digest(expected_path(directory.parent, workload)),
              f"{directory} {arm} known-answer digest")
    return labels, workloads, signatures


def verified_blocks(directory, rows, frozen, environment, labels, workloads, signatures):
    require(len(rows) <= LEGS * blocks_for(directory.name), f"too many raw runs in {directory}")
    end = 0
    pids = set()
    refusal = False
    for index, row in enumerate(rows):
        block, leg = divmod(index, LEGS)
        order = ORDERS[block % len(ORDERS)]
        arm = order[0] if leg == 0 else order[leg - 1]
        stem = f"b{block + 1:02d}-r{leg}-{arm}"
        label = labels[arm]
        check(row, {"block": block + 1, "order_index": leg, "arm": arm, "warmup": leg == 0,
                    "engine": label, "engine_sha256": frozen[label]["sha256"]}, stem)
        require(not refusal, f"{directory} continued timing after an option refusal")
        check(row["workload"], workloads[arm], f"{stem} workload")
        engine_metadata(directory, stem, row, frozen[label], environment)
        validate_run(directory, stem, row, signatures[arm])
        if row["eligible"] or row.get("games") is not None:
            check_workload_metadata(row, workloads[arm], directory.parent, f"{directory} {stem}")
        elif not row["eligible"]:
            refusal = verify_refusal(directory, stem, row, workloads[arm], directory.parent)
        pid = row.get("engine_pid")
        require(pid is None or pid not in pids, f"{directory} reused an engine process at {stem}")
        if pid is not None:
            pids.add(pid)
        require(row["wall_start_monotonic_ns"] >= end, f"overlapping run at {stem}")
        end = row["wall_end_monotonic_ns"]
    groups = [rows[offset:offset + LEGS] for offset in range(0, len(rows), LEGS)]
    return [group for group in groups if len(group) == LEGS and all(row["eligible"] for row in group)]


def audit(directory, expected, known, frozen, environment):
    manifest = load(directory / "inputs.txt")
    labels, workloads, signatures = verify_manifest(directory, manifest, expected, known, frozen, environment)
    summary_path = directory / "summary.txt"
    summary = load(summary_path) if summary_path.is_file() else None
    rows, truncated = read_rows(directory / "runs.jsonl.txt", summary is not None)
    groups = verified_blocks(directory, rows, frozen, environment, labels, workloads, signatures)
    if summary is not None and "verified_inputs_after" not in summary:
        require(str(summary["error"]).startswith("Post-run identity failed:"), f"missing post-run identity in {directory}")
        groups = []
    requested = blocks_for(directory.name)
    stopped = rows[-1].get("error") if rows and "Draw option refusal" in (rows[-1].get("error") or "") else None
    blocks, metrics, means = timing_math(groups, requested)
    complete = len(blocks) == requested and len(rows) == requested * LEGS
    if summary is not None:
        check(summary, {"schema": SCHEMA, "order": ORDERS, "requested_blocks": requested, "completed_blocks": len(blocks),
                        "comparison_direction": "B / A, less than 1 is faster",
                        "attempted_runs": len(rows), "included_runs": 4 * len(blocks), "included_warmups": len(blocks),
                        "complete": complete, "blocks": blocks, "order_position_means": means,
                        "stopped_on_refusal": stopped,
                        **{key: manifest[key] for key in ("arm_labels", "identical_floor", "floor", "worker_affinity",
                                                       "expected_by_arm", "workload_by_arm", "expected_games_searches_nodes_trace")}}, str(summary_path))
        require(summary["metrics"].keys() == metrics.keys(), f"unexpected metric coverage in {summary_path}")
        for key, values in metrics.items():
            actual = summary["metrics"][key]
            check(actual, values, f"{summary_path} {key}")
            require(complete or {"block_mean_ci95", "ci_df"}.isdisjoint(actual), f"incomplete {directory} has a full-degree CI")
        if "engine_environment" in summary:
            check(summary["engine_environment"], environment, f"{summary_path} environment")
        if "verified_inputs_after" in summary:
            check(summary["verified_inputs_after"], manifest["inputs"], f"{summary_path} post-run input pins")
        require(not complete or summary["error"] is None, f"completed campaign records an error in {summary_path}")
    return Campaign(directory, manifest, rows, blocks, metrics, means, summary, complete and summary is not None, truncated)


def timing_table(campaigns, steps, df):
    print(f"| Comparison | A | B | Workload A / B | Metric | Blocks | Median | Pooled B/A | Mean | 95% t interval, df {df} | Below 1 |")
    print("|---|---|---|---|---|---:|---:|---:|---:|---:|---:|")
    for step in steps:
        a, b = labels_for(step)
        workload = " / ".join(STEP_WORKLOADS[step])
        data = campaigns.get(step)
        needed = blocks_for(step)
        for key, label in METRICS:
            lead = f"| {step} | {a} | {b} | {workload} | {label} |"
            if data is None:
                print(f"{lead} Not run | — | — | — | — | — |")
            elif key not in data.metrics:
                print(f"{lead} 0/{needed} | — | — | — | — | — |")
            else:
                values = data.metrics[key]
                count = len(data.included_blocks)
                interval = interval_text(values) if data.complete else "—"
                print(f"{lead} {count}/{needed} | {values['block_median']:.4f} | {values['pooled_b_over_a']:.4f} | "
                      f"{values['block_mean']:.4f} | {interval} | {values['below_one']}/{count} |")


def show_campaign(step, data):
    print(f"\n### {step}\n")
    print(f"Logical CPU {data.manifest['worker_affinity'][0]}. {len(data.rows)} attempted runs, "
          f"{len(data.included_blocks)} included blocks, {len(data.included_blocks) * 4} included timed runs.")
    if data.included_blocks:
        print("\n| Block | Timed order | B/A CPU | B/A wall |")
        print("|---:|---|---:|---:|")
        for row in data.included_blocks:
            print(f"| {row['block']} | {row['order']} | {row['engine_cpu_s']:.4f} | {row['total_wall_ns']:.4f} |")
    for row in data.rows:
        if not row["eligible"]:
            block, leg, arm = row["block"], row["order_index"], row["arm"]
            expected = tuple(data.manifest["expected_by_arm"][arm])
            base = f"sol2/job23/{step}/b{block:02d}-r{leg}-{arm}"
            print(f"\nExcluded block {block}, leg {leg}, arm {arm}. Expected `{expected}`. Observed `{signature(row)}`. "
                  f"Recorded error `{row['error']}`. [Driver]({base}.driver.stdout.txt) · [Engine]({base}.engine.stderr.txt)")
    if not data.complete:
        print("\nThe campaign is incomplete.")
    if data.summary and data.summary["error"]:
        print(f"\nRecorded summary error `{data.summary['error']}`.")
    if data.truncated:
        print("\nThe last raw run record is partial. The preceding complete receipts were checked.")
    base = f"sol2/job23/{step}"
    links = [f"[Inputs]({base}/inputs.txt)"]
    if (data.directory / "runs.jsonl.txt").is_file():
        links.append(f"[Raw runs]({base}/runs.jsonl.txt)")
    if data.summary is not None:
        links.append(f"[Summary]({base}/summary.txt)")
    command = data.directory.parent / f"{step}-command.raw.txt"
    if command.is_file():
        links.append(f"[Command](sol2/job23/{command.name})")
    print("\n" + " · ".join(links))


def show_control(campaigns, signatures):
    print("\n## C23 control\n")
    timing_table(campaigns, ("C23",), 7)
    if "W2200" in signatures:
        print(f"\nChanging the configured soft-node limit from 2,000 to 2,200 gave "
              f"{signatures['W'][1]:,} and {signatures['W2200'][1]:,} searches. Each arm uses its own exact "
              "known answer. [W](sol2/job23/known-answer.txt#L2-L6) · "
              "[W2200](sol2/job23/known-answer-W2200.txt#L2-L6)")
    cpu = completed_metric(campaigns, "C23", "engine_cpu_s")
    if cpu is not None:

        print(f"\n**{status}**. The CPU interval must lie wholly above one for W2200 relative to W.")
    else:

    if "C23" in campaigns:
        show_campaign("C23", campaigns["C23"])


def show_readings(campaigns, steps, gates):
    print("\n## Declared readings and arithmetic\n")
    print("| Comparison | CPU reading | 95% t interval | Pooled B/A | Wall pooled B/A [95% t interval] |")
    print("|---|---|---:|---:|---:|")
    for step in steps:
        cpu = completed_metric(campaigns, step, "engine_cpu_s")
        wall = completed_metric(campaigns, step, "total_wall_ns")
        if cpu is None:
            label = "Incomplete" if step in campaigns else "Not run"
            print(f"| {step} | {label} | — | — | — |")
        else:
            print(f"| {step} | {reading(cpu)} | {interval_text(cpu)} | {cpu['pooled_b_over_a']:.4f} | "
                  f"{wall['pooled_b_over_a']:.4f} {interval_text(wall)} |")
    print("\nThe CPU interval wholly below one reads PAYS. Wholly above one reads COSTS. An interval containing one reads NOT SEEN.")
    print("\n| Comparison | CPU conclusion from 95% interval overlap | Draw minus PC20 pooled CPU ratio | "
          "Draw minus PC20 pooled wall ratio |")
    print("|---|---|---:|---:|")
    baseline_cpu = completed_metric(campaigns, "D0", "engine_cpu_s")
    baseline_wall = completed_metric(campaigns, "D0", "total_wall_ns")
    for step in (name for name in steps if name != "D0"):
        draw_cpu = completed_metric(campaigns, step, "engine_cpu_s")
        draw_wall = completed_metric(campaigns, step, "total_wall_ns")
        conclusion = settings_reading(baseline_cpu, draw_cpu)
        if conclusion is None:
            print(f"| {step} vs D0 | Unavailable | — | — |")
            continue
        cpu_delta = draw_cpu["pooled_b_over_a"] - baseline_cpu["pooled_b_over_a"]
        wall_delta = draw_wall["pooled_b_over_a"] - baseline_wall["pooled_b_over_a"]
        print(f"| {step} vs D0 | {conclusion} | {cpu_delta:+.6f} | {wall_delta:+.6f} |")
    print("\nThe pooled differences are descriptive. The CPU conclusion uses interval overlap as specified for this job.")
    for step, workload in (("D1", "WD1"), ("D2", "WD100")):
        first = gates.get(workload, {}).get("K_1")
        if first and not first["eligible"]:
            print(f"\n{step} stopped on the first K_1 identity check. Expected `{signature(gates[workload]['r15d'])}`. "
                  f"Observed `{signature(first)}`. [Raw first K_1](sol2/job23/draw-gates/{workload}/B.K_1.result.raw.txt)")


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("job23", type=Path, help="Job 23 raw receipts directory")
    parser.add_argument("job22", type=Path, help="Job 22 receipts directory, beside earlier jobs")
    parser.add_argument("--job13", type=Path, help="Job 13 W receipts, defaults beside Job 22")
    parser.add_argument("--job16", type=Path, help="Job 16 frozen launch arms, defaults beside Job 22")
    parser.add_argument("--through", choices=STEPS, default="D2", help="last Job 23 campaign to verify")
    args = parser.parse_args()
    root = args.job23
    anchor = args.job22.parent
    known = known_workload(args.job13 or anchor / "job13")
    check(known.signature, W_SIGNATURE, "declared Job 23 W signature")
    timed = any((root / step / "inputs.txt").is_file() for step in STEPS)
    if not (root / "clang-executables.raw.txt").is_file():
        require(not timed, "timing without Step 0 pins")
        print("Step 0 frozen executable receipts are unavailable.")
        return
    previous = load((args.job16 or anchor / "job16") / "arms.txt")
    environment = read_environment(previous["engine_environment"])
    frozen = {label: previous["arms"][label] for label in REFERENCE_LABELS}
    for label, arm in zip(REFERENCE_LABELS, "AB", strict=True):
        check(frozen[label]["sha256"], known.clang_inputs[arm]["sha256"], f"Job 13 original frozen digest {label}")
    reference = load(root / "reference-arms.txt")
    check(set(reference["arms"]), set(frozen), "Job 23 Step 0 launch inventory")
    check(reference["arms"], frozen, "Job 23 Step 0 launch receipt")
    check(read_environment(reference["engine_environment"]), environment, "Job 23 Step 0 environment")
    gate_ok = verify_gate(root, known, frozen, environment, {label: label for label in REFERENCE_LABELS}, render=True)
    inventories = read_uci(root, render=True)
    gate_ok = gate_ok and len(inventories) == len(REFERENCE_LABELS)
    require(gate_ok or not timed, "timing without passed Step 0 W and UCI gates")
    for workload in DRAW_LABELS:
        path = draw_path(root, workload)
        if path.is_file():
            pinned = draw_options(root, workload)
            for label, inventory in inventories.items():
                for name, value in pinned.items():
                    _, lower, upper = inventory[name]
                    require(lower <= value <= upper, f"{label} declared spin excludes {name}={value}")
    show_arms(frozen, environment, REFERENCE_LABELS)
    signatures, gates = read_draw_gates(root, frozen, environment)
    show_draw_gates(signatures, gates)
    driver = root / "draw_book_driver.py.txt"
    if driver.is_file():
        known.hashes["draw_driver"] = digest(driver)
        receipts = [line.split(maxsplit=1) for line in (root / "workload-pins.raw.txt").read_text().splitlines()]
        driver_pins = [sha for sha, path in receipts if Path(path).name == "draw_book_driver.py"]
        check(driver_pins, [known.hashes["draw_driver"]], "archived draw driver pin")
    campaigns = {}
    for step in STEPS:
        directory = root / step
        if (directory / "inputs.txt").is_file():
            if step != "C23":
                require(reading(completed_metric(campaigns, "C23", "engine_cpu_s")) == "COSTS",
                        f"{step} ran without a passed C23 control")
            if step in ("D1", "D2"):
                workload = STEP_WORKLOADS[step][0]
                require(gates.get(workload, {}).get("K_1", {}).get("eligible")
                        and signatures[workload] != signatures["W"],
                        f"{step} ran without matched, changed draw gates")
            campaigns[step] = audit(directory, signatures, known, frozen, environment)
        else:
            require(not (directory / "runs.jsonl.txt").is_file(), f"raw runs lack a manifest in {directory}")
        if step == args.through:
            break
    show_control(campaigns, signatures)
    print("\n## Timings D0–D2\n")
    steps = STEPS[1:]
    timing_table(campaigns, steps, 15)
    show_readings(campaigns, steps, gates)
    for step in steps:
        if step in campaigns:
            show_campaign(step, campaigns[step])


if __name__ == "__main__":
    main()
