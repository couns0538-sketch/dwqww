#!/usr/bin/env python3
"""Independently reconcile Job 15 pathname and stack results from raw W timing and identity receipts."""

import argparse
from dataclasses import dataclass
import hashlib
import json
import math
from pathlib import Path
import re
from statistics import fmean, median, stdev


SCHEMA = "forge-bal-arm-diagnostic-v1"
ORDERS = ("ABBA", "BAAB")
LEGS = 5
BLOCKS = 16
METRICS = (("engine_cpu_s", "Engine CPU"), ("total_wall_ns", "Wall"))
TRACE_FIELDS = ("games", "searches", "total_nodes", "final_nodes", "trace_fnv1a64")
NET_SHA256 = "c3d95131d7c1c89ccb5b000b52c7c37ed8665dbf8bb2a0a7a873809c85b9f7c7"
BOOK_SHA256 = "7a7f6470615a69c6cf23d565417701d38732876f480af90d67b42abade35644a"
WORKLOAD = ("--start", "7001", "--games", "20", "--softnodes", "2000", "--max-plies", "300", "--json")
T975 = 2.13144954556
ENVIRONMENT = {"LC_ALL": "C", "TZ": "UTC"}
W_SIGNATURE = (20, 2247, 7905938, 9189426, "7f78d3aec021ab01")
LENGTHS = (40, 56, 72, 88)
U_STEPS = tuple(f"U{length}" for length in LENGTHS)
ARMS = ("P0", "H1", *(f"{engine}{length}" for length in LENGTHS for engine in "RK"))
PAIRS = {
    "T1": ("P0", "H1"),
    "T2": ("H1", "P0"),
    **{f"U{length}": (f"R{length}", f"K{length}") for length in LENGTHS},
    "V40": ("K40", "R40"),
    "V88": ("K88", "R88"),
}
SE_DIVISOR = 2.131
Q_LIMITS = {2: 3.84, 3: 5.99, 4: 7.81}


@dataclass
class Known:
    signature: tuple
    hashes: dict
    clang_inputs: dict


@dataclass
class Campaign:
    directory: Path
    manifest: dict
    rows: list
    included_blocks: list
    metrics: dict
    position_means: dict
    summary: dict | None
    complete: bool
    truncated: bool


def require(condition, message):
    if not condition:
        raise ValueError(message)


def load(path):
    return json.loads(path.read_text(encoding="utf-8"))


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def signature(row):
    return tuple(row.get(field) for field in TRACE_FIELDS)


def check(actual, expected, where):
    if isinstance(expected, dict):
        require(isinstance(actual, dict) and expected.keys() <= actual.keys(), f"missing fields at {where}")
        for key, value in expected.items():
            check(actual[key], value, f"{where}.{key}")
    elif isinstance(expected, (tuple, list)):
        require(isinstance(actual, (tuple, list)) and len(actual) == len(expected), f"list shape differs at {where}")
        for index, (one, two) in enumerate(zip(actual, expected)):
            check(one, two, f"{where}[{index}]")
    elif type(expected) is bool:
        require(actual is expected, f"wrong {where}: {actual!r}, expected {expected!r}")
    else:
        equal = math.isclose(actual, expected, rel_tol=1e-9, abs_tol=1e-12) if type(expected) is float else actual == expected
        require(equal, f"wrong {where}: {actual!r}, expected {expected!r}")


def read_rows(path, finished):
    if not path.is_file():
        return [], False
    lines = path.read_text(encoding="utf-8").splitlines(keepends=True)
    rows = []
    for index, line in enumerate(lines, 1):
        try:
            rows.append(json.loads(line))
        except json.JSONDecodeError:
            if not finished and index == len(lines) and not line.endswith("\n"):
                return rows, True
            raise
    return rows, False


def validate_run(directory, stem, row, expected):
    source = f"{directory}/{stem}"
    stdout, stderr = f"{stem}.driver.stdout.txt", f"{stem}.engine.stderr.txt"
    check(row, {"driver_stdout": stdout, "engine_stderr": stderr}, source)
    require((directory / stdout).is_file() and (directory / stderr).is_file(), f"missing driver/engine receipts for {source}")
    actual = signature(row)
    check(row["trace_matches"], actual == expected, f"{source} signature flag")
    start, end = row["wall_start_monotonic_ns"], row["wall_end_monotonic_ns"]
    user, system = row["engine_user_s"], row["engine_system_s"]
    check(row, {"total_wall_ns": end - start, "engine_cpu_s": user + system}, f"{source} durations")
    if row["eligible"] or any(value is not None for value in actual):
        driver = load(directory / stdout)
        check(row, driver, f"{source} driver stdout")
    if row["eligible"]:
        require(actual == expected and row["engine_exit"] == 0 and row["error"] is None
                and user >= 0 and system >= 0 and user + system > 0 and end > start,
                f"{source} marked eligible without correct identity, status and durations")
    else:
        require(bool(row["error"]), f"{source} excluded without a recorded error")


def validate_tape(path, receipt):
    rows, _ = read_rows(path, True)
    last_info_total = final_total = final_lines = zero_last = 0
    hash_value = 14695981039346656037
    current_book_line = 7000
    current_ply = 0
    seen = set()
    for row in rows:
        book_line = row["book_line"]
        require(7001 <= book_line <= 7020 and book_line >= current_book_line, f"unordered book line in {path}")
        current_ply = current_ply + 1 if book_line == current_book_line else 1
        current_book_line = book_line
        seen.add(book_line)
        check(row["ply"], current_ply, f"{path} game {book_line} ply")
        match = re.search(r"\bnodes (\d+)(?=\s|$)", row["last_info"] or "")
        check(row["nodes"], int(match[1]) if match else 0, f"{path} game {book_line} last-info nodes")
        move = row["bestmove_line"].split()
        require(move[:2] == ["bestmove", row["bestmove"]], f"bestmove differs from its UCI line in {path}")
        final = row["final_nodes"]
        if final is None:
            require(row["final_nodes_line"] is None and row["nodes"] == 0, f"missing final soft-node telemetry in {path}")
        else:
            require(final >= row["nodes"], f"final soft nodes below last-info nodes in {path}")
            check(row["final_nodes_line"], f"info string softnodes nodes {final}", f"{path} final soft-node line")
            final_lines += 1
            final_total += final
        last_info_total += row["nodes"]
        zero_last += row["nodes"] == 0
        wire = f"{row['bestmove']},{row['nodes']};"
        check(row["trace_input"], wire, f"{path} trace input")
        for byte in wire.encode("ascii"):
            hash_value = (hash_value ^ byte) * 1099511628211 & ((1 << 64) - 1)
        check(row["trace_fnv1a64"], f"{hash_value:016x}", f"{path} cumulative trace")
    require(seen == set(range(7001, 7021)), f"wrong W book lines in {path}")
    check(receipt, {"searches": len(rows), "total_nodes": last_info_total, "final_nodes": final_total,
                    "trace_fnv1a64": f"{hash_value:016x}", "final_lines": final_lines,
                    "missing_final_lines": len(rows) - final_lines, "zero_last_info_searches": zero_last},
          f"{path} tape summary")


def read_pins(path):
    lines = path.read_text(encoding="utf-8").splitlines()
    require(len(lines) == 2, f"expected one executable receipt per engine in {path}")
    pins = {}
    for arm, line in zip("AB", lines):
        sha256, executable = line.split(maxsplit=1)
        require(re.fullmatch(r"[0-9a-f]{64}", sha256), f"invalid executable digest in {path}")
        pins[arm] = {"path": executable, "sha256": sha256}
    return pins


def known_workload(root):
    path = root / "known-answer.txt"
    source = load(path)
    expected = signature(source)
    require(source.keys() == set(TRACE_FIELDS) and expected[0] == 20
            and all(type(value) is int and value > 0 for value in expected[:4])
            and expected[3] >= expected[2]
            and isinstance(expected[4], str) and re.fullmatch(r"[0-9a-f]{16}", expected[4]),
            f"invalid W signature in {path}")
    pins = read_pins(root / "clang-executables.raw.txt")
    for key, arm in zip("AB", ("r15d", "kF")):
        gate = root / "step2"
        data = load(gate / f"{arm}.result.raw.txt")
        validate_run(gate, arm, data, expected)
        check(data["engine_sha256"], pins[key]["sha256"], f"Step 2 {arm} executable")
        require(data["eligible"], f"Step 2 {arm} failed the W gate")
        validate_tape(gate / f"{arm}.searches.raw.txt", data)
    return Known(expected, {"net": NET_SHA256, "book": BOOK_SHA256, "known_answer": digest(path),
                            "driver": digest(root / "soft_book_driver.py.txt")}, pins)


def timing_math(groups):
    blocks = []
    positions = {cohort: {key: [] for key, _ in METRICS} for cohort in ("odd", "even")}
    timed_runs = []
    for group in groups:
        measured = group[1:]
        timed_runs.extend(measured)
        block_index = group[0]["block"]
        cohort = "odd" if block_index % 2 else "even"
        block = {"block": block_index, "order": ORDERS[(block_index - 1) % len(ORDERS)]}
        for key, _ in METRICS:
            totals = {arm: math.fsum(row[key] for row in measured if row["arm"] == arm) for arm in "AB"}
            block[key] = totals["B"] / totals["A"]
            mean = fmean(row[key] for row in measured)
            positions[cohort][key].append([row[key] / mean for row in measured])
        blocks.append(block)
    means = {cohort: {key: [fmean(position) for position in zip(*series)] for key, series in data.items()}
             for cohort, data in positions.items()}
    metrics = {}
    count = len(blocks)
    for key, _ in METRICS:
        if count == 0:
            break
        values = [block[key] for block in blocks]
        ordered = sorted(values)
        block_mean = fmean(values)
        df = count - 1
        below, above = sum(value < 1 for value in values), sum(value > 1 for value in values)
        totals = {arm: math.fsum(row[key] for row in timed_runs if row["arm"] == arm) for arm in "AB"}
        data = {"block_mean": block_mean, "block_median": median(values), "block_min": ordered[0], "block_max": ordered[-1],
                "pooled_b_over_a": totals["B"] / totals["A"], "below_one": below, "above_one": above,
                "equal_one": count - below - above}
        if count == BLOCKS:
            halfwidth = T975 * stdev(values) / math.sqrt(count)
            data.update(block_mean_ci95=[block_mean - halfwidth, block_mean + halfwidth], ci_df=df)
        if count >= 3:
            data.update(block_middle_min=ordered[1], block_middle_max=ordered[-2])
        metrics[key] = data
    return blocks, metrics, means


def read_environment(record):
    wire = b"".join(f"{key}={value}".encode("ascii") + b"\0" for key, value in ENVIRONMENT.items())
    check(record, {"variables": ENVIRONMENT, "bytes": len(wire), "sha256": hashlib.sha256(wire).hexdigest()},
          "fixed engine environment")
    require(record["variables"] == ENVIRONMENT, "the engine environment must have just LC_ALL and TZ")
    return record


def read_arms(root, prior, known):
    path = root / "arms.txt"
    document = load(path)
    environment = read_environment(document["engine_environment"])
    previous = load(prior / "arms.txt")
    check(environment, previous["engine_environment"], "engine environment relative to Job 14")
    pins = document["arms"]
    require(set(pins) == set(ARMS), f"wrong arm names in {path}")
    for name, arm in pins.items():
        require(Path(arm["path"]).is_absolute() and type(arm["inode"]) is int and arm["inode"] > 0
                and type(arm["device"]) is int and arm["device"] >= 0, f"invalid path/inode/device for {name}")
        check(arm["length"], len(arm["path"]), f"{path} {name} path length")
        build = "B" if name.startswith("K") else "A"
        check(arm["sha256"], known.clang_inputs[build]["sha256"], f"{path} {name} build SHA-256")
        if name in ("P0", "H1"):
            old = previous["arms"][name]
            check(arm, {key: old[key] for key in ("path", "length", "sha256")}, f"{path} {name} Job 14 pin")
        else:
            check(arm["length"], int(name[1:]), f"{path} {name} exact length")
    check(pins["P0"], known.clang_inputs["A"], f"{path} P0 frozen Clang path")
    require(len({pin["path"] for pin in pins.values()}) == len(ARMS), "arm paths must be distinct")
    identities = {key: (pin["device"], pin["inode"]) for key, pin in pins.items()}
    check(identities["H1"], identities["P0"], "hard link identity")
    require(len(set(identities.values())) == len(ARMS) - 1, "only P0 and H1 may share recorded inode/device")
    return pins, environment


def verify_manifest(directory, manifest, known, pins, environment):
    check(manifest, {"schema": SCHEMA, "order": ORDERS, "blocks_requested": BLOCKS, "identical_floor": False,
                     "floor": None, "expected_games_searches_nodes_trace": known.signature,
                     "engine_environment": environment}, str(directory))
    read_environment(manifest["engine_environment"])
    affinity = manifest["worker_affinity"]
    require(len(affinity) == 1 and type(affinity[0]) is int and affinity[0] >= 0,
            f"{directory} is not pinned to exactly one logical CPU")
    identities = manifest["inputs"]
    for key, sha256 in known.hashes.items():
        check(identities[key]["sha256"], sha256, f"{directory} {key} SHA-256")
    for arm, label in zip("AB", PAIRS[directory.name]):
        check(manifest["arm_labels"][arm], label, f"{directory} arm {arm} label")
        pin = pins[label]
        check(identities[arm], {key: pin[key] for key in ("path", "sha256")}, f"{directory} arm {arm} pin")
        for key in ("length", "inode", "device"):
            if key in identities[arm]:
                check(identities[arm][key], pin[key], f"{directory} arm {arm} {key}")
        argv = [identities["driver"]["path"], "--engine", pin["path"], "--net", identities["net"]["path"],
                "--book", identities["book"]["path"], *WORKLOAD]
        check(manifest["argv_by_arm"][arm], argv, f"{directory} arm {arm} argv")


def engine_metadata(directory, stem, row, pin, environment):
    where = f"{directory}/{stem} launch"
    expected = {"engine_argv0": pin["path"], "engine_argv0_length": pin["length"],
                "engine_environment_bytes": environment["bytes"], "engine_environment_sha256": environment["sha256"]}
    if row["eligible"] or row["engine_pid"] is not None or not expected.keys().isdisjoint(row):
        check(row, expected, where)
    proc = row.get("engine_proc_receipt")
    require(proc in (None, "verified", "unavailable"), f"unexpected procfs receipt at {where}")
    if proc == "verified":
        check(row, {"engine_inode": pin["inode"], "engine_device": pin["device"]}, f"{where} procfs")


def verified_blocks(directory, manifest, rows, known, pins, environment):
    require(len(rows) <= LEGS * BLOCKS, f"too many raw runs in {directory}")
    end = 0
    for index, row in enumerate(rows):
        block, leg = divmod(index, LEGS)
        order = ORDERS[block % len(ORDERS)]
        arm = order[0] if leg == 0 else order[leg - 1]
        stem = f"b{block + 1:02d}-r{leg}-{arm}"
        label = manifest["arm_labels"][arm]
        check(row, {"block": block + 1, "order_index": leg, "arm": arm, "warmup": leg == 0,
                    "engine": label, "engine_sha256": pins[label]["sha256"]}, stem)
        engine_metadata(directory, stem, row, pins[label], environment)
        validate_run(directory, stem, row, known.signature)
        require(row["wall_start_monotonic_ns"] >= end, f"overlapping run at {stem}")
        end = row["wall_end_monotonic_ns"]
    groups = [rows[offset:offset + LEGS] for offset in range(0, len(rows), LEGS)]
    return [group for group in groups if len(group) == LEGS and all(row["eligible"] for row in group)]


def audit(directory, known, pins, environment):
    manifest = load(directory / "inputs.txt")
    verify_manifest(directory, manifest, known, pins, environment)
    summary_path = directory / "summary.txt"
    summary = load(summary_path) if summary_path.is_file() else None
    rows, truncated = read_rows(directory / "runs.jsonl.txt", summary is not None)
    groups = verified_blocks(directory, manifest, rows, known, pins, environment)
    if summary is not None and "verified_inputs_after" not in summary:
        require(str(summary["error"]).startswith("Post-run identity failed:"), f"missing post-run identity in {directory}")
        groups = []
    blocks, metrics, means = timing_math(groups)
    complete = len(blocks) == BLOCKS and len(rows) == BLOCKS * LEGS
    if summary is not None:
        check(summary, {"schema": SCHEMA, "order": ORDERS, "requested_blocks": BLOCKS, "completed_blocks": len(blocks),
                        "comparison_direction": "B / A, less than 1 is faster",
                        "attempted_runs": len(rows), "included_runs": 4 * len(blocks), "included_warmups": len(blocks),
                        "complete": complete, "blocks": blocks, "order_position_means": means,
                        "expected_games_searches_nodes_trace": known.signature,
                        **{key: manifest[key] for key in ("arm_labels", "identical_floor", "floor", "worker_affinity")}},
              str(summary_path))
        require(summary["metrics"].keys() == metrics.keys(), f"unexpected metric coverage in {summary_path}")
        for key, values in metrics.items():
            optional = ("block_middle_min", "block_middle_max")
            actual = summary["metrics"][key]
            check(actual, {item: value for item, value in values.items() if item not in optional or item in actual},
                  f"{summary_path} {key}")
            require(complete or {"block_mean_ci95", "ci_df"}.isdisjoint(actual), f"incomplete {directory} has a 15-df CI")
            require("decision" not in actual, f"unexpected per-campaign decision in {directory}")
        if "engine_environment" in summary:
            check(summary["engine_environment"], environment, f"{summary_path} environment")
        if "verified_inputs_after" in summary:
            check(summary["verified_inputs_after"], manifest["inputs"], f"{summary_path} post-run input pins")
        require(not complete or summary["error"] is None, f"completed campaign records an error in {summary_path}")
    return Campaign(directory, manifest, rows, blocks, metrics, means, summary, complete and summary is not None, truncated)


def show_gate(root, prior):
    known = known_workload(prior)
    check(known.signature, W_SIGNATURE, "declared Job 15 W signature")
    check(read_pins(root / "clang-executables.raw.txt"), known.clang_inputs, "Job15 frozen Clang build pins")
    step = root / "step0"
    eligible = []
    pids = []
    wire = b"LC_ALL=C\0TZ=UTC\0"
    for key, label in zip("AB", ("r15d", "kF")):
        path = step / f"{label}.result.raw.txt"
        if not path.is_file():
            print(f"Step 0 {label} receipt is missing.")
            continue
        row = load(path)
        validate_run(step, label, row, known.signature)
        check(row["engine_sha256"], known.clang_inputs[key]["sha256"], f"Step 0 {label} build")
        if not row["eligible"]:
            print(f"Step 0 {label} failed with `{row['error']}`. [Receipt](sol2/job15/step0/{path.name})")
            continue
        executable = known.clang_inputs[key]["path"]
        check(row, {"engine_argv0": executable, "engine_argv0_length": len(executable),
                    "engine_environment_bytes": len(wire), "engine_environment_sha256": hashlib.sha256(wire).hexdigest()},
              f"Step 0 {label} process launch")
        tape = step / f"{label}.searches.raw.txt"
        if tape.is_file():
            validate_tape(tape, row)
        pids.append(row["engine_pid"])
        eligible.append(f"[{label}](sol2/job15/step0/{path.name})")
    if len(eligible) != 2:
        return None
    require(None not in pids and len(set(pids)) == 2, "Step 0 must use a fresh engine process per arm")
    check(digest(root / "known-answer.txt"), known.hashes["known_answer"], "Job15 known-answer bytes")
    entries = [line.split(maxsplit=1)
               for line in (root / "workload-pins.raw.txt").read_text(encoding="utf-8").splitlines()]
    pins = {Path(path).name: sha256 for sha256, path in entries}
    require(len(pins) == len(entries) == 4, "Job15 must record the four workload input pins")
    for name, key in (("soft_book_driver.py", "driver"), ("known-answer.txt", "known_answer"),
                      ("UHO_Lichess_4852_v1.epd", "book"), ("net768.fnn3", "net")):
        check(pins[name], known.hashes[key], f"Job15 workload receipt {name}")
    games, searches, nodes, final_nodes, trace = known.signature
    print(f"Both Step 0 builds match W with {games} games, {searches:,} searches, {nodes:,} last-info nodes, "
          f"{final_nodes:,} final nodes and trace `{trace}`. W uses PC20, Threads 1, Hash 16 MiB, net768 and "
          f"`go softnodes 2000` on book lines 7001–7020. {' · '.join(eligible)} · [W signature](sol2/job15/known-answer.txt)")
    return known


def interval_direction(interval):
    low, high = interval
    if high < 1:
        return "below"
    if low > 1:
        return "above"
    return "includes"


def completed_metric(campaigns, step, key):
    data = campaigns.get(step)
    return data.metrics[key] if data is not None and data.complete else None


def swapped_ratio(first, second):
    if first is None or second is None:
        return None
    return math.sqrt(first["pooled_b_over_a"] / second["pooled_b_over_a"])


def heterogeneity(metrics):
    parts = {}
    for step in U_STEPS:
        record = metrics[step]
        if record is None:
            continue
        low, high = record["block_mean_ci95"]
        scale = (high - low) / (2 * SE_DIVISOR)
        parts[step] = {"mean": record["block_mean"], "se": scale, "weight": 1 / scale**2 if scale else None}
    limit = Q_LIMITS.get(len(parts))
    if not parts:
        return {"parts": parts, "weighted_mean": None, "q": None, "limit": limit, "status": "Incomplete"}
    if any(record["weight"] is None for record in parts.values()):
        return {"parts": parts, "weighted_mean": None, "q": None, "limit": limit,
                "status": "Not judged, zero interval half-width"}
    if len(parts) == 1:
        mean = next(iter(parts.values()))["mean"]
        return {"parts": parts, "weighted_mean": mean, "q": 0.0, "limit": None, "status": "Not judged, one length"}
    weight_total = math.fsum(record["weight"] for record in parts.values())
    mean = math.fsum(record["weight"] * record["mean"] for record in parts.values()) / weight_total
    q = math.fsum(((record["mean"] - mean) / record["se"])**2 for record in parts.values())
    return {"parts": parts, "weighted_mean": mean, "q": q, "limit": limit,
            "status": "THE SAME" if q <= limit else "DIFFERENT"}


def reading(campaigns, key):
    metrics = {step: completed_metric(campaigns, step, key) for step in PAIRS}
    t1, t2 = metrics["T1"], metrics["T2"]
    if t1 is None or t2 is None:
        path = "Incomplete"
    else:
        replicated = (interval_direction(t1["block_mean_ci95"]) == "below"
                      and interval_direction(t2["block_mean_ci95"]) == "above")
        path = "REPLICATED" if replicated else "Not replicated"
    tested = sum(metrics[step] is not None for step in U_STEPS)
    below = sum(metrics[step] is not None and interval_direction(metrics[step]["block_mean_ci95"]) == "below"
                for step in U_STEPS)
    if tested != len(U_STEPS):
        faster = f"Incomplete, {below}/{tested} completed intervals wholly below 1" if tested else "Incomplete"
    else:
        faster = "Faster at every length" if below == len(U_STEPS) else f"Criterion unmet, {below}/4 intervals wholly below 1"
    return {"path": path, "path_ratio": swapped_ratio(t1, t2), "faster": faster,
            "heterogeneity": heterogeneity(metrics),
            "extremes": {length: swapped_ratio(metrics[f"U{length}"], metrics[f"V{length}"]) for length in (40, 88)}}


def show_readings(campaigns):
    print("\n## Declared readings and arithmetic")
    print("\n| Metric | Path effect T1/T2 | H1/P0 swapped ratio | kF faster at every length | kF/r15d swapped at 40 | kF/r15d swapped at 88 |")
    print("|---|---|---:|---|---:|---:|")
    results = {name: reading(campaigns, key) for key, name in METRICS}

    def display(number, digits=9):
        return "—" if number is None else f"{number:.{digits}f}"

    for name, result in results.items():
        print(f"| {name} | {result['path']} | {display(result['path_ratio'], 6)} | {result['faster']} | "
              f"{display(result['extremes'][40], 6)} | {display(result['extremes'][88], 6)} |")
    print("\nEach swapped ratio uses the square root of the forward pooled ratio divided by the reverse pooled "
          "ratio. The extreme-length ratios are descriptive.")
    print("\n| Metric | Length | U mean m_L | CI half-width / 2.131 (s_L) | Weight w_L |")
    print("|---|---:|---:|---:|---:|")
    for name, result in results.items():
        parts = result["heterogeneity"]["parts"]
        for length in LENGTHS:
            record = parts.get(f"U{length}", {})
            print(f"| {name} | {length} | {display(record.get('mean'))} | {display(record.get('se'))} | "
                  f"{display(record.get('weight'), 6)} |")
    print("\n| Metric | Complete U lengths | Weighted mean m_w | Q | Q cutoff | Reading |")
    print("|---|---:|---:|---:|---:|---|")
    for name, result in results.items():
        combined = result["heterogeneity"]
        print(f"| {name} | {len(combined['parts'])} | {display(combined['weighted_mean'])} | "
              f"{display(combined['q'])} | {display(combined['limit'], 2)} | {combined['status']} |")
    print("\nWeights are 1/s_L². The weighted mean uses their sum as its denominator. Q sums "
          "((m_L − m_w)/s_L)² across completed lengths.")
    links = [f"[{step}](sol2/job15/{step}/summary.txt)" for step in PAIRS
             if step in campaigns and campaigns[step].complete]
    if links:
        print("\n" + " · ".join(links))


def show_arms(pins, environment):
    print("\n| Arm | Engine argv[0] | Characters | Device | Inode | SHA-256 |")
    print("|---|---|---:|---:|---:|---|")
    for key in ARMS:
        pin = pins[key]
        print(f"| {key} | `{pin['path']}` | {pin['length']} | {pin['device']} | {pin['inode']} | `{pin['sha256']}` |")
    print(f"\nEach engine receives `LC_ALL=C` and `TZ=UTC`, {environment['bytes']} bytes, "
          f"SHA-256 `{environment['sha256']}`. [Arms](sol2/job15/arms.txt)")


def show_overview(campaigns, through):
    print("\n| Comparison | A | B | Metric | Blocks | Median | Pooled B/A | Mean | 95% t interval, df 15 | Below 1 |")
    print("|---|---|---|---|---:|---:|---:|---:|---:|---:|")
    for step, (a, b) in PAIRS.items():
        data = campaigns.get(step)
        for key, label in METRICS:
            if data is None:
                print(f"| {step} | {a} | {b} | {label} | Not run | — | — | — | — | — |")
            elif key not in data.metrics:
                print(f"| {step} | {a} | {b} | {label} | 0/{BLOCKS} | — | — | — | — | — |")
            else:
                values = data.metrics[key]
                count = len(data.included_blocks)
                interval = "—"
                if data.complete:
                    lo, hi = values["block_mean_ci95"]
                    interval = f"[{lo:.4f}, {hi:.4f}]"
                print(f"| {step} | {a} | {b} | {label} | {count}/{BLOCKS} | {values['block_median']:.4f} | "
                      f"{values['pooled_b_over_a']:.4f} | {values['block_mean']:.4f} | {interval} | "
                      f"{values['below_one']}/{count} |")
        if step == through:
            break


def show_campaign(step, data):
    print(f"\n### {step}\n")
    print(f"Logical CPU {data.manifest['worker_affinity'][0]}. {len(data.rows)} attempted runs, "
          f"{len(data.included_blocks)} included blocks, {len(data.included_blocks) * 4} included timed runs.")
    if data.included_blocks:
        print("\n| Block | Timed order | B/A CPU | B/A wall |")
        print("|---:|---|---:|---:|")
        for row in data.included_blocks:
            print(f"| {row['block']} | {row['order']} | {row['engine_cpu_s']:.4f} | {row['total_wall_ns']:.4f} |")
    for row in data.rows:
        if not row["eligible"]:
            x = row["block"], row["order_index"], row["arm"]
            stem = f"b{x[0]:02d}-r{x[1]}-{x[2]}"
            base = f"sol2/job15/{step}/{stem}"
            print(f"\nExcluded block {x[0]}, leg {x[1]}, arm {x[2]}. Observed `{signature(row)}`. "
                  f"Recorded error `{row['error']}`. [Driver]({base}.driver.stdout.txt) · [Engine]({base}.engine.stderr.txt)")
    if not data.complete:
        print("\nThe campaign is incomplete.")
    if data.summary and data.summary["error"]:
        print(f"\nRecorded summary error `{data.summary['error']}`.")
    if data.truncated:
        print("\nThe last raw run record is partial. The preceding complete receipts were checked.")
    base = f"sol2/job15/{step}"
    links = [f"[Inputs]({base}/inputs.txt)"]
    if (data.directory / "runs.jsonl.txt").is_file():
        links.append(f"[Raw runs]({base}/runs.jsonl.txt)")
    if data.summary is not None:
        links.append(f"[Summary]({base}/summary.txt)")
    command = data.directory.parent / f"{step}-command.raw.txt"
    if command.is_file():
        links.append(f"[Command](sol2/job15/{command.name})")
    print("\n" + " · ".join(links))


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("job15", type=Path, help="Job 15 raw receipts directory")
    parser.add_argument("job14", type=Path, help="Job 14 arm receipts directory")
    parser.add_argument("--job13", type=Path, help="Job 13 W receipts, default beside Job 14")
    parser.add_argument("--through", choices=tuple(PAIRS), default="V88", help="last campaign to read")
    args = parser.parse_args()
    job13 = args.job13 or args.job14.parent / "job13"
    known = show_gate(args.job15, job13)
    if known is None or not (args.job15 / "arms.txt").is_file():
        require(not any((args.job15 / step / "inputs.txt").is_file() for step in PAIRS),
                "timing manifests lack valid Step 0 and Step 1 pins")
        return
    pins, environment = read_arms(args.job15, args.job14, known)
    show_arms(pins, environment)
    campaigns = {}
    affinity = None
    for step in PAIRS:
        directory = args.job15 / step
        if (directory / "inputs.txt").is_file():
            data = audit(directory, known, pins, environment)
            if affinity is None:
                affinity = data.manifest["worker_affinity"]
            check(data.manifest["worker_affinity"], affinity, f"{step} logical CPU across campaigns")
            campaigns[step] = data
        else:
            require(not (directory / "runs.jsonl.txt").is_file(), f"raw runs lack a manifest in {directory}")
        if step == args.through:
            break
    show_overview(campaigns, args.through)
    show_readings(campaigns)
    for step, data in campaigns.items():
        show_campaign(step, data)


if __name__ == "__main__":
    main()
