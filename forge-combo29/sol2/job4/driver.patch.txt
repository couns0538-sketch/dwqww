+def read_counted_search(stdout, location, expected_keys=None):
+    last_nodes = 0
+    last_score = ""
+    last_info = None
+    evalcount = None
+    counters = {}
+    while True:
+        line = stdout.readline()
+        if not line:
+            raise RuntimeError(f"{location}: engine EOF before bestmove")
+        if line.startswith("info depth "):
+            last_info = line.rstrip("\r\n")
+            parts = line.split()
+            for i, tok in enumerate(parts):
+                if tok == "nodes" and i + 1 < len(parts):
+                    last_nodes = int(parts[i + 1])
+                elif tok == "score" and i + 2 < len(parts):
+                    last_score = f"{parts[i + 1]}:{parts[i + 2]}"
+        elif line.startswith("info string evalcount") and line.split()[:3] == ["info", "string", "evalcount"]:
+            if evalcount is not None:
+                raise RuntimeError(f"{location}: duplicate evalcount line before bestmove")
+            evalcount = line.rstrip("\r\n")
+            tokens = line.split()[3:]
+            if not tokens:
+                raise RuntimeError(f"{location}: empty evalcount line")
+            for token in tokens:
+                key, separator, value = token.partition("=")
+                if (not separator or not re.fullmatch(r"[A-Za-z_][A-Za-z0-9_.-]*", key)
+                        or not re.fullmatch(r"[0-9]{1,20}", value) or key in counters
+                        or int(value) >= 2**64):
+                    raise RuntimeError(f"{location}: invalid or duplicate evalcount field {token!r}")
+                counters[key] = int(value)
+            if expected_keys is not None and tuple(counters) != expected_keys:
+                raise RuntimeError(f"{location}: evalcount keys {tuple(counters)} != {expected_keys}")
+        elif line.startswith("bestmove "):
+            if evalcount is None:
+                raise RuntimeError(f"{location}: missing evalcount line before bestmove")
+            return {
+                "bestmove": line.split()[1], "nodes": last_nodes, "score": last_score,
+                "uci_last_info": last_info, "uci_bestmove": line.rstrip("\r\n"),
+                "uci_evalcount": evalcount, "evalcounts": counters,
+            }
