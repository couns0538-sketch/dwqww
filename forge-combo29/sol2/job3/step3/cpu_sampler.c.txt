#define _GNU_SOURCE
#include <dlfcn.h>
#include <errno.h>
#include <stdbool.h>
#include <fcntl.h>
#include <pthread.h>
#include <signal.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <ucontext.h>
#include <unistd.h>

static int samples = -1;
static int profile_signal;
static const long period_us = 3989;
static _Thread_local unsigned thread_id __attribute__((tls_model("initial-exec")));
static _Thread_local uintptr_t stack_low __attribute__((tls_model("initial-exec")));
static _Thread_local uintptr_t stack_high __attribute__((tls_model("initial-exec")));

enum { CallerLimit = 6, TextMapLimit = 256 };
struct text_map { uintptr_t first, end; };
static struct text_map text_maps[TextMapLimit];
static size_t text_map_count;

static bool is_code(uintptr_t address) {
    for (size_t i = 0; i < text_map_count; ++i) {
        if (address >= text_maps[i].first && address < text_maps[i].end) return true;
    }
    return false;
}

static bool readable_stack(uintptr_t address, uintptr_t rsp, size_t bytes) {
    return (address & (sizeof(uintptr_t) - 1)) == 0 &&
           address >= rsp && address >= stack_low && stack_high >= bytes &&
           address <= stack_high - bytes;
}

static void fail(const char *message) {
    fprintf(stderr, "SOL2 sampler error: %s: %s\n", message, strerror(errno));
    _exit(125);
}

static char *hex(char *out, uint64_t value, int digits) {
    for (int i=digits-1; i>=0; --i) {
        out[i]="0123456789abcdef"[value & 15];
        value >>= 4;
    }
    return out + digits;
}

static void sample(int number, siginfo_t *info, void *data) {
    (void)number;
    if (info->si_code != SI_TIMER) return;
    int saved_errno = errno;
    const greg_t *regs = ((ucontext_t *)data)->uc_mcontext.gregs;
    const uintptr_t pc = (uintptr_t)regs[REG_RIP];
    const uintptr_t rsp = (uintptr_t)regs[REG_RSP];
    const uintptr_t initial_frame = (uintptr_t)regs[REG_RBP];
    uintptr_t stack_word = 0;
    uintptr_t callers[CallerLimit];
    size_t depth = 0;
    char status = 'L';

    if (rsp >= stack_low && rsp < stack_high) {
        if (readable_stack(rsp, rsp, sizeof(uintptr_t))) {
            const uintptr_t word = *(const volatile uintptr_t *)rsp;
            if (word && is_code(word - 1)) stack_word = word;
        }
        uintptr_t frame = initial_frame;
        for (; depth < CallerLimit; ++depth) {
            if (frame == 0) { status = 'Z'; break; }
            if (!readable_stack(frame, rsp, 2 * sizeof(uintptr_t))) {
                status = 'B';
                break;
            }
            const volatile uintptr_t *words = (const volatile uintptr_t *)frame;
            const uintptr_t next = words[0];
            const uintptr_t resume = words[1];
            if (!resume || !is_code(resume - 1)) { status = 'X'; break; }
            callers[depth] = resume;
            if (next && (next <= frame || next - frame < 2 * sizeof(uintptr_t))) {
                ++depth;
                status = 'M';
                break;
            }
            frame = next;
        }
    } else {
        status = 'S';
    }

    char row[84 + CallerLimit * 17];
    char *end = hex(row, thread_id, 8);
    *end++ = '\t';
    end = hex(end, pc, 16);
    *end++ = '\t';
    end = hex(end, (unsigned)info->si_overrun, 4);
    *end++ = '\t';
    end = hex(end, rsp, 16);
    *end++ = '\t';
    end = hex(end, initial_frame, 16);
    *end++ = '\t';
    *end++ = status;
    *end++ = '\t';
    end = hex(end, stack_word, 16);
    for (size_t i = 0; i < depth; ++i) {
        *end++ = '\t';
        end = hex(end, callers[i], 16);
    }
    *end++ = '\n';
    if (write(samples, row, (size_t)(end-row)) != end-row) _exit(126);
    errno = saved_errno;
}

static void start_thread_timer(timer_t *timer) {
    thread_id = (unsigned)gettid();
    pthread_attr_t attr;
    int rc = pthread_getattr_np(pthread_self(), &attr);
    if (rc != 0) { errno = rc; fail("pthread_getattr_np"); }
    void *base = NULL;
    size_t length = 0;
    rc = pthread_attr_getstack(&attr, &base, &length);
    if (rc != 0) { errno = rc; fail("pthread_attr_getstack"); }
    rc = pthread_attr_destroy(&attr);
    if (rc != 0) { errno = rc; fail("pthread_attr_destroy"); }
    stack_low = (uintptr_t)base;
    if (length == 0 || length > UINTPTR_MAX - stack_low) {
        errno = EINVAL;
        fail("thread stack bounds");
    }
    stack_high = stack_low + length;
    struct sigevent event = {.sigev_notify=SIGEV_THREAD_ID, .sigev_signo=profile_signal};
    event._sigev_un._tid = (int)thread_id;
    if (timer_create(CLOCK_THREAD_CPUTIME_ID, &event, timer)) fail("timer_create");
    struct itimerspec tick = {
        .it_interval = {.tv_nsec = period_us * 1000},
        .it_value = {.tv_nsec = period_us * 1000},
    };
    if (timer_settime(*timer, 0, &tick, NULL)) fail("timer_settime");
}

static void copy_maps(const char *prefix) {
    char *name = NULL;
    if (asprintf(&name, "%s.maps.txt", prefix) < 0) fail("maps path");
    int out = open(name, O_WRONLY | O_CREAT | O_EXCL | O_CLOEXEC, 0600);
    if (out < 0) fail("open output maps");
    free(name);
    FILE *in = fopen("/proc/self/maps", "re");
    if (!in) fail("open process maps");
    char *line = NULL;
    size_t capacity = 0;
    ssize_t n;
    while ((n = getline(&line, &capacity, in)) >= 0) {
        if (write(out, line, (size_t)n) != n) fail("write process maps");
        unsigned long first, end;
        char access[5];
        if (sscanf(line, "%lx-%lx %4s", &first, &end, access) == 3 && access[2] == 'x') {
            if (text_map_count == TextMapLimit) { errno = ENOSPC; fail("executable map limit"); }
            text_maps[text_map_count++] = (struct text_map){first, end};
        }
    }
    if (ferror(in)) fail("read process maps");
    free(line);
    fclose(in);
    close(out);
}

__attribute__((constructor)) static void initialize_sampler(void) {
    const char *prefix = getenv("SOL2_PROFILE_PREFIX");
    if (!prefix || !*prefix) return;
    char *name = NULL;
    if (asprintf(&name, "%s.samples.txt", prefix) < 0) fail("samples path");
    samples = open(name, O_WRONLY | O_CREAT | O_EXCL | O_APPEND | O_CLOEXEC, 0600);
    if (samples < 0) fail("open output samples");
    free(name);
    copy_maps(prefix);
    profile_signal = SIGRTMIN + 6;
    struct sigaction action = {.sa_sigaction=sample, .sa_flags=SA_SIGINFO | SA_RESTART};
    sigemptyset(&action.sa_mask);
    if (sigaction(profile_signal, &action, NULL)) fail("sigaction");
    timer_t timer;
    start_thread_timer(&timer);
    fprintf(stderr, "SOL2 sampler: per-thread CLOCK_THREAD_CPUTIME_ID, period %ld us, signal %d, x86-64 RBP chain up to six callers\n", period_us, profile_signal);
}

struct launch {
    void *(*entry)(void *);
    void *arg;
};

static void *start_sampled_thread(void *opaque) {
    struct launch request = *(struct launch *)opaque;
    free(opaque);
    timer_t timer;
    start_thread_timer(&timer);
    void *result = request.entry(request.arg);
    if (timer_delete(timer)) fail("timer_delete");
    return result;
}

int pthread_create(pthread_t *id, const pthread_attr_t *attrs, void *(*entry)(void *), void *arg) {
    int (*real_create)(pthread_t *, const pthread_attr_t *, void *(*)(void *), void *) = dlsym(RTLD_NEXT, "pthread_create");
    if (!real_create) {
        errno = ENOSYS;
        fail("pthread_create symbol");
    }
    if (samples < 0) return real_create(id, attrs, entry, arg);
    struct launch *request = malloc(sizeof(*request));
    if (!request) return ENOMEM;
    *request = (struct launch){entry,arg};
    int rc = real_create(id, attrs, start_sampled_thread, request);
    if (rc) free(request);
    return rc;
}
