#!/usr/bin/env bash
set -euxo pipefail
job=/workspace/work/sol2-job8
base=/workspace/work/sol2-job2
out=/workspace/work/sol2-repo/forge-combo29/sol2/job8
cmake=/workspace/work/sol2-tools/cmake/data/bin/cmake
tag=k13
source="$job/k13"
build="$job/build-pgo-$tag"
profiles="$job/pgo-data-$tag"
mkdir "$profiles"
flags='-DFORGE_MATE_ALPHA_GUARD=1 -DFORGE_NNUE_V3=1 -flto'
driver="$base/carbon2/selfplay_driver.py"
net="$base/net768.fnn3"
sha256sum "$source/CMakeLists.txt" "$driver" "$net"
c++ --version | head -1
"$cmake" -S "$source" -B "$build" -DCMAKE_BUILD_TYPE=Release "-DCMAKE_CXX_FLAGS=$flags -fprofile-generate=$profiles"
"$cmake" --build "$build" --parallel 3 --verbose > "$out/$tag-pgo-generate-build.raw.txt" 2>&1
sha256sum "$build/forge"
date -u +'%FT%TZ'
timeout 240 taskset -c 2 python -B "$job/capture_openings.py" "$driver" "$out/$tag-training-openings.txt" --engine "$build/forge" --net "$net" --games 20 --softnodes 2000 --random-plies 8 --max-plies 300 --seed 20261118 --json > "$out/$tag-training.raw.txt" 2> "$out/$tag-training.stderr.txt"
date -u +'%FT%TZ'
python - "$out/$tag-training.raw.txt" <<'PY'
import json,sys
from pathlib import Path
r=json.loads(Path(sys.argv[1]).read_text()); print(r)
assert (r['games'],r['searches'],r['total_nodes'],r['trace_fnv1a64'])==(20,2482,8391013,'c7d9520ffe04deda')
PY
shopt -s nullglob
data=("$profiles"/*.gcda)
test "${#data[@]}" -gt 0
sha256sum "${data[@]}" > "$out/$tag-gcda-hashes.raw.txt"
for file in "${data[@]}"; do gcov-dump -l "$file" | awk '/OBJECT_SUMMARY.*runs=1/ {print; seen=1} END {if (!seen) exit 1}'; done > "$out/$tag-gcda-inventory.raw.txt"
"$cmake" -S "$source" -B "$build" -DCMAKE_BUILD_TYPE=Release "-DCMAKE_CXX_FLAGS=$flags -fprofile-use=$profiles -Werror=missing-profile -Werror=coverage-mismatch"
"$cmake" --build "$build" --clean-first --parallel 3 --verbose > "$out/$tag-pgo-use-build.raw.txt" 2>&1
sha256sum "$build/forge"
sha256sum -c "$out/$tag-gcda-hashes.raw.txt"
date -u +'%FT%TZ'
timeout 240 taskset -c 2 python -B "$job/capture_openings.py" "$driver" "$out/$tag-timing-openings.txt" --engine "$build/forge" --net "$net" --games 60 --softnodes 2000 --random-plies 8 --max-plies 300 --seed 20261117 --json > "$out/$tag-identity.raw.txt" 2> "$out/$tag-identity.stderr.txt"
date -u +'%FT%TZ'
sha256sum -c "$out/$tag-gcda-hashes.raw.txt"
python - "$out" "$tag" <<'PY'
import json,sys
from pathlib import Path
root=Path(sys.argv[1]); tag=sys.argv[2]
row=json.loads((root/f'{tag}-identity.raw.txt').read_text()); print(row)
assert (row['games'],row['searches'],row['total_nodes'],row['trace_fnv1a64'])==(60,8612,29333343,'5c705841c3ea17b4')
a=json.loads((root/f'{tag}-training-openings.txt').read_text()); b=json.loads((root/f'{tag}-timing-openings.txt').read_text()); assert (len(a),len(b))==(20,60)
for key in ('moves','position_side_castling'):
    overlap={tuple(r[key]) for r in a}&{tuple(r[key]) for r in b}; print(key,'overlap',len(overlap)); assert not overlap
print('Known answer and disjoint opening groups PASS')
PY
