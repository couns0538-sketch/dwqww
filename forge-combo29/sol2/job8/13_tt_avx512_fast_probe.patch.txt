+[[gnu::noinline]] std::uint8_t TranspositionTable::replace_full_cluster_slot(const TTCluster& cluster,
+                                                                             std::uint8_t generation) noexcept {
+    std::uint8_t replace_slot = 0;
+    int replace_score = cluster.entries[0].depth -
+                        4 * static_cast<int>((32U + generation - cluster.entries[0].generation()) & 31U);
+    for (std::uint8_t s = 1; s < 4; ++s) {
+        const TTEntry& entry = cluster.entries[s];
+        const int age = static_cast<int>((32U + generation - entry.generation()) & 31U);
+        const int score = entry.depth - 4 * age;
+        if (score < replace_score) {
+            replace_slot = s;
+            replace_score = score;
+        }
+    }
+    return replace_slot;
+}
