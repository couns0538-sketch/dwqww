#!/usr/bin/env python3
"""Measure Forge engine CPU and whole driver wall time in six fixed ABBA blocks."""

import argparse
from contextlib import redirect_stdout
from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path
import resource
import runpy
import subprocess
import sys
import time
from unittest.mock import patch


JOB2 = Path(__file__).resolve().parent.parent / "sol2-job2"
DRIVER = JOB2 / "carbon2" / "selfplay_driver.py"
BASE_SHA256 = "bb37ed7c50e1716c5fc38f1b2fb92293c18e98f6131bf2be2b13d668603b4fc6"
NET_SHA256 = "c3d95131d7c1c89ccb5b000b52c7c37ed8665dbf8bb2a0a7a873809c85b9f7c7"
DRIVER_SHA256 = "c79b95bf45762fa9869d23140ed7f946cfaf0cb6bc4f74325a18d5f934bf1c45"
EXPECTED = (60, 8612, 29333343, "5c705841c3ea17b4")
WORKLOAD = ("--games", "60", "--softnodes", "2000", "--random-plies", "8", "--max-plies", "300",
            "--seed", "20261117", "--json")
ORDER = ("r15d", "k12", "k12", "r15d")
BLOCKS = 6
METRICS = ("engine_cpu_s", "total_wall_ns")


def now_utc():
    return datetime.now(timezone.utc).isoformat(timespec="microseconds").replace("+00:00", "Z")


def pinned_identity(path, expected):
    path = path.resolve(strict=True)
    with path.open("rb") as content:
        actual = hashlib.file_digest(content, "sha256").hexdigest()
    if actual != expected:
        raise ValueError(f"SHA-256 mismatch for {path}: expected {expected}, got {actual}")
    return {"path": str(path), "sha256": actual}


def write_json(path, content):
    with path.open("x", encoding="utf-8") as output:
        print(json.dumps(content, indent=2), file=output)


def matches(row):
    return (row.get("games"), row.get("searches"), row.get("total_nodes"), row.get("trace_fnv1a64")) == EXPECTED


def measure_driver(driver_main, argv, output, stem):
    stdout_path = output / f"{stem}.driver.stdout.txt"
    stderr_path = output / f"{stem}.engine.stderr.txt"
    real_popen = subprocess.Popen
    child = None
    error = None
    old_argv = sys.argv
    with (stdout_path.open("x", encoding="utf-8") as stdout,
          stderr_path.open("xb") as stderr):
        def launch(*args, **kwargs):
            nonlocal child
            if child is not None:
                raise RuntimeError("Pinned driver attempted a second child")
            kwargs["stderr"] = stderr
            child = real_popen(*args, **kwargs)
            return child

        sys.argv = argv
        try:
            with redirect_stdout(stdout), patch.object(subprocess, "Popen", launch):
                utc_start = now_utc()
                before = resource.getrusage(resource.RUSAGE_CHILDREN)
                start = time.monotonic_ns()
                try:
                    driver_main()
                except (Exception, SystemExit) as exc:
                    error = f"{type(exc).__name__}: {exc}"
                finally:
                    if child is not None and child.poll() is None:
                        child.kill()
                        child.wait()
                    finished = time.monotonic_ns()
                    after = resource.getrusage(resource.RUSAGE_CHILDREN)
                    utc_end = now_utc()
        finally:
            sys.argv = old_argv
    row = {}
    if error is None:
        try:
            row = json.loads(stdout_path.read_text(encoding="utf-8"))
            if not isinstance(row, dict):
                raise ValueError("driver output must be one JSON object")
        except (ValueError, UnicodeError) as exc:
            row = {}
            error = f"Unreadable driver JSON: {exc}"
    user = after.ru_utime - before.ru_utime
    system = after.ru_stime - before.ru_stime
    engine_exit = child.returncode if child is not None else None
    trace_valid = matches(row)
    if error is None:
        if engine_exit != 0:
            error = f"Expected one successful Forge child, got exit {engine_exit}"
        elif not trace_valid:
            error = f"Workload identity differs from {EXPECTED}"
        elif user < 0 or system < 0 or user + system <= 0 or finished <= start:
            error = "Nonpositive or inconsistent native time accounting"
    return {**row, "utc_start": utc_start, "utc_end": utc_end,
            "wall_start_monotonic_ns": start, "wall_end_monotonic_ns": finished,
            "total_wall_ns": finished - start, "engine_user_s": user, "engine_system_s": system,
            "engine_cpu_s": user + system, "engine_pid": child.pid if child is not None else None,
            "engine_exit": engine_exit, "trace_matches": trace_valid, "eligible": error is None,
            "error": error, "driver_stdout": stdout_path.name, "engine_stderr": stderr_path.name}


def summarize_runs(runs):
    blocks = []
    complete_runs = []
    for offset in range(0, len(runs), len(ORDER)):
        group = runs[offset:offset + len(ORDER)]
        if tuple(row["engine"] for row in group) != ORDER or not all(row["eligible"] for row in group):
            break
        a_first, b_first, b_last, a_last = group
        block = {"block": offset // len(ORDER) + 1}
        for key in METRICS:
            block[key] = {"k12_over_r15d": (b_first[key] + b_last[key]) / (a_first[key] + a_last[key]),
                          "r15d_first_over_last": a_first[key] / a_last[key]}
        blocks.append(block)
        complete_runs.extend(group)
    metrics = {}
    if blocks:
        for key in METRICS:
            total_a = sum(row[key] for row in complete_runs if row["engine"] == "r15d")
            total_b = sum(row[key] for row in complete_runs if row["engine"] == "k12")
            point = total_b / total_a
            ratios = [block[key]["k12_over_r15d"] for block in blocks]
            noise = [block[key]["r15d_first_over_last"] for block in blocks]
            inside = min(noise) <= point <= max(noise)
            metrics[key] = {"k12_over_r15d": point, "block_min": min(ratios), "block_max": max(ratios),
                            "aa_min": min(noise), "aa_max": max(noise), "inside_aa_spread": inside,
                            "interpretation": "not measurable here" if inside else "outside observed A/A spread"}
    return {"comparison_direction": "k12 / r15d, less than 1 is faster",
            "aa_direction": "first r15d / last r15d in each ABBA block",
            "requested_blocks": BLOCKS, "completed_blocks": len(blocks),
            "attempted_runs": len(runs), "included_runs": len(complete_runs),
            "complete": len(blocks) == BLOCKS, "blocks": blocks, "metrics": metrics}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("r15d", type=Path)
    parser.add_argument("k12", type=Path)
    parser.add_argument("net", type=Path)
    parser.add_argument("output", type=Path, help="new output directory")
    parser.add_argument("--k12-sha256", required=True, help="candidate identity from the Step 1 receipt")
    parser.add_argument("--driver", type=Path, default=DRIVER)
    args = parser.parse_args()
    args.output.mkdir(parents=True, exist_ok=False)
    pins = {"r15d": (args.r15d, BASE_SHA256), "k12": (args.k12, args.k12_sha256),
            "net": (args.net, NET_SHA256), "driver": (args.driver, DRIVER_SHA256)}
    inputs = {key: pinned_identity(*spec) for key, spec in pins.items()}
    argv = {name: [inputs["driver"]["path"], "--engine", inputs[name]["path"],
                   "--net", inputs["net"]["path"], *WORKLOAD] for name in dict.fromkeys(ORDER)}
    write_json(args.output / "inputs.txt", {"inputs": inputs, "argv_by_engine": argv, "order": ORDER,
              "blocks_requested": BLOCKS, "expected_games_searches_nodes_trace": EXPECTED,
              "cpu_scope": "RUSAGE_CHILDREN delta while pinned driver runs its one Forge child, setup through exit",
              "wall_scope": "monotonic wall for full driver call including opening generation, warmup and teardown",
              "worker_affinity": sorted(os.sched_getaffinity(0))})
    driver_main = runpy.run_path(inputs["driver"]["path"])["main"]
    runs = []
    failure = None
    with (args.output / "runs.jsonl.txt").open("x", encoding="utf-8") as log:
        for block in range(1, BLOCKS + 1):
            for index, name in enumerate(ORDER, start=1):
                stem = f"b{block:02d}-r{index}-{name}"
                row = {"block": block, "order_index": index, "engine": name,
                       "engine_sha256": inputs[name]["sha256"],
                       **measure_driver(driver_main, argv[name], args.output, stem)}
                runs.append(row)
                line = json.dumps(row)
                print(line, file=log, flush=True)
                print(line, flush=True)
                if not row["eligible"]:
                    failure = f"{stem}: {row['error']}"
                    break
            if failure:
                break
    try:
        after = {key: pinned_identity(*spec) for key, spec in pins.items()}
        if after != inputs:
            raise ValueError("An input path changed during the campaign")
    except (OSError, ValueError) as exc:
        # Retain the original rows while withholding the measurements from the ratio.
        summary = summarize_runs([])
        summary["attempted_runs"] = len(runs)
        failure = f"Post-run identity failed: {exc}"
    else:
        summary = summarize_runs(runs)
        summary["verified_inputs_after"] = after
    summary["error"] = failure
    write_json(args.output / "summary.txt", summary)
    if failure:
        raise RuntimeError(failure)
    print(json.dumps(summary), flush=True)


if __name__ == "__main__":
    try:
        main()
    except (KeyError, OSError, RuntimeError, ValueError) as exc:
        print(exc, file=sys.stderr)
        sys.exit(1)
