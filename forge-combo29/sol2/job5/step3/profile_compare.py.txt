import csv
from collections import Counter
from pathlib import Path
import sys

base, candidate, baseline_groups, candidate_groups, out = map(Path,sys.argv[1:])

def read(path):
    with path.open() as f:
        return list(csv.DictReader(f, delimiter='\t'))


def save(name, titles, records):
    with (out/name).open('x', newline='') as f:
        writer=csv.writer(f,delimiter='\t',lineterminator='\n')
        writer.writerow(titles)
        writer.writerows(records)

known={(r['image'], r['function']):float(r['percent_of_all_samples']) for r in read(base)}
rows=read(candidate/'symbolized_pcs.tsv')
total=sum(int(r['samples']) for r in rows)
new=Counter()
for r in rows:
    if r['outer_function']!='??': new[Path(r['image']).name,r['outer_function']]+=int(r['samples'])
changes=[(f,image,known.get((image,f),0.0),100*new[image,f]/total) for image,f in known.keys()|new.keys()]
changes.sort(key=lambda r:(r[3]-r[2],r[0]))
save('all-symbol-deltas.raw.tsv.txt',['function','image','r15d_pct','k12_pct','k12_minus_r15d_pp'],[(f,i,f'{a:.6f}',f'{b:.6f}',f'{b-a:.6f}') for f,i,a,b in changes])
save('top15-comparison.raw.tsv.txt',['function','image','k12_source','k12_samples','r15d_pct','k12_pct'],[(r['function'],r['image'],r['modal_sample_source_line'],r['samples'],f"{known.get((r['image'],r['function']),0):.6f}",r['percent_of_all_samples']) for r in read(candidate/'top15.tsv')])
a={r['group']:float(r['percent_of_all_samples']) for r in read(baseline_groups)}
b={r['group']:float(r['percent_of_all_samples']) for r in read(candidate_groups)}
save('subsystem-comparison.raw.tsv.txt',['group','r15d_pct','k12_pct'],[(g,f'{a.get(g,0):.6f}',f'{b.get(g,0):.6f}') for g in sorted(a.keys()|b.keys(),key=lambda g:-b.get(g,0))])
print('Clean k12 samples',total)
for f,i,a,b in changes[:3]: print('Largest drop',f,a,b,b-a)
