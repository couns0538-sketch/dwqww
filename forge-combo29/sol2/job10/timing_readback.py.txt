#!/usr/bin/env python3
"""Recount Job 10 fixed-node timings and compare the Job 9 soft-node campaign."""

import argparse
from dataclasses import dataclass
import hashlib
import json
import math
from pathlib import Path
import re
from statistics import fmean, median


SCHEMA = "forge-warm-balanced-v1"
ORDERS = ("ABBA", "BAAB")
LEGS = 5
BLOCKS = 12
TRACE_FIELDS = ("games", "searches", "total_nodes", "trace_fnv1a64")
SOFT_ARGS = ["--games", "60", "--softnodes", "2000", "--random-plies", "8", "--max-plies", "300",
             "--seed", "20261117", "--json"]
FIXED_ARGS = ["--games", "12", "--nodes", "20000", "--random-plies", "8", "--max-plies", "300"]
METRICS = (("engine_cpu_s", "Engine CPU"), ("total_wall_ns", "Wall"))
NET_SHA256 = "c3d95131d7c1c89ccb5b000b52c7c37ed8665dbf8bb2a0a7a873809c85b9f7c7"
DRIVER_SHA256 = "c79b95bf45762fa9869d23140ed7f946cfaf0cb6bc4f74325a18d5f934bf1c45"


@dataclass(frozen=True)
class Workload:
    expected: tuple
    pinned_files: dict
    fixed: bool = False

    def arguments(self, identities):
        if self.fixed:
            return [*FIXED_ARGS, "--openings", identities["openings"]["path"], "--json"]
        return SOFT_ARGS


SOFT = Workload((60, 8612, 29333343, "5c705841c3ea17b4"), {"net": NET_SHA256, "driver": DRIVER_SHA256})


@dataclass
class Audit:
    directory: Path
    manifest: dict
    summary: dict | None
    rows: list
    blocks: list
    metrics: dict
    position_means: dict
    complete: bool
    truncated_row: bool


def require(condition, message):
    if not condition:
        raise ValueError(message)


def load(path):
    return json.loads(path.read_text(encoding="utf-8"))


def same_number(actual, expected):
    return math.isclose(actual, expected, rel_tol=1e-10, abs_tol=1e-12)


def trace_of(record):
    return tuple(record.get(key) for key in TRACE_FIELDS)


def moves_of(data):
    entries = data["openings"] if isinstance(data, dict) else data
    return [row["moves"] if isinstance(row, dict) else row for row in entries]


def fixed_workload(job10, job9):
    answer_path = job10 / "known-answer.txt"
    answer = load(answer_path)
    expected = trace_of(answer)
    require(set(answer) == set(TRACE_FIELDS) and expected[0] == 12
            and all(type(n) is int and n > 0 for n in expected[:3])
            and isinstance(expected[3], str) and re.fullmatch(r"[0-9a-f]{16}", expected[3]),
            f"invalid fixed-node known answer in {answer_path}")
    for arm in ("r15d", "kF"):
        gate = job10 / f"gate-{arm}-driver.raw.txt"
        require(trace_of(load(gate)) == expected, f"frozen known answer differs from driver receipt {gate}")
    openings = job10 / "fixed-openings.txt"
    fixed_moves = moves_of(load(openings))
    prior_moves = moves_of(load(job9 / "kF-timing-openings.txt"))
    require(len(prior_moves) == 60 and len(fixed_moves) == 12 and prior_moves[:12] == fixed_moves,
            "fixed workload openings differ from the first twelve Job 9 kF timing openings")
    require(all(len(moves) == 8 and all(re.fullmatch(r"[a-h][1-8][a-h][1-8][nbrq]?", move) for move in moves)
                for moves in fixed_moves), "fixed workload has an invalid eight-ply opening sequence")
    files = {"known_answer": answer_path, "openings": openings, "driver": job10 / "fixed_nodes_driver.py.txt"}
    pins = {key: hashlib.sha256(path.read_bytes()).hexdigest() for key, path in files.items()}
    return Workload(expected, {"net": NET_SHA256, **pins}, fixed=True)


def verify_manifest(directory, manifest, requested, floor, workload):
    require(manifest["schema"] == SCHEMA and manifest["order"] == list(ORDERS), f"bad schema/order in {directory}")
    require(manifest["blocks_requested"] == requested
            and tuple(manifest["expected_games_searches_nodes_trace"]) == workload.expected,
            f"bad block count or workload in {directory}")
    identities = manifest["inputs"]
    for key, digest in workload.pinned_files.items():
        require(identities[key]["sha256"] == digest, f"bad {key} pin in {directory}")
    for arm in "AB":
        expected_argv = [identities["driver"]["path"], "--engine", identities[arm]["path"], "--net", identities["net"]["path"],
                         *workload.arguments(identities)]
        require(manifest["argv_by_arm"][arm] == expected_argv, f"workload argv differs for {arm} in {directory}")
    if floor is None:
        require(manifest["identical_floor"] and manifest["floor"] is None, f"missing identical-floor flag in {directory}")
        require(identities["A"]["path"] != identities["B"]["path"] and identities["A"]["sha256"] == identities["B"]["sha256"],
                f"floor did not name distinct paths with identical executable hashes in {directory}")
    else:
        source = floor.directory / "summary.txt"
        digest = hashlib.sha256(source.read_bytes()).hexdigest()
        require(floor.complete and not manifest["identical_floor"] and manifest["floor"]["sha256"] == digest,
                f"invalid floor binding in {directory}")
        require(manifest["worker_affinity"] == floor.manifest["worker_affinity"], f"floor CPU affinity differs in {directory}")


def inspect_rows(directory, manifest, rows, requested, expected):
    require(len(rows) <= requested * LEGS, f"unexpected extra runs in {directory}")
    first_failed = None
    previous_finished = 0
    for index, row in enumerate(rows):
        block, offset = divmod(index, LEGS)
        order = ORDERS[block % len(ORDERS)]
        is_warmup = offset == 0
        arm = order[0] if is_warmup else order[offset - 1]
        stem = f"b{block + 1:02d}-r{offset}-{arm}"
        require((row["block"], row["order_index"], row["arm"]) == (block + 1, offset, arm)
                and row["warmup"] is is_warmup,
                f"misordered leg {stem} in {directory}")
        require(row["engine"] == manifest["arm_labels"][arm] and row["engine_sha256"] == manifest["inputs"][arm]["sha256"],
                f"wrong run executable at {stem} in {directory}")
        start = row["wall_start_monotonic_ns"]
        end = row["wall_end_monotonic_ns"]
        require(end - start == row["total_wall_ns"] and start >= previous_finished, f"invalid/overlapping wall receipts at {stem}")
        previous_finished = end
        require(same_number(row["engine_user_s"] + row["engine_system_s"], row["engine_cpu_s"]),
                f"engine CPU does not equal user plus system at {stem}")
        require(row["driver_stdout"] == stem + ".driver.stdout.txt" and row["engine_stderr"] == stem + ".engine.stderr.txt",
                f"unexpected raw paths at {stem}")
        require((directory / row["engine_stderr"]).is_file() and (directory / row["driver_stdout"]).is_file(),
                f"missing raw output at {stem}")
        exact = trace_of(row) == expected
        require(row["trace_matches"] is exact, f"incorrect trace flag at {stem}")
        if row["eligible"]:
            require(first_failed is None and exact and row["error"] is None and row["engine_exit"] == 0,
                    f"invalid run eligible for analysis at {stem}")
            require(row["engine_cpu_s"] > 0 and row["total_wall_ns"] > 0 and min(row["engine_user_s"], row["engine_system_s"]) >= 0,
                    f"invalid raw durations at {stem}")
            driver = load(directory / row["driver_stdout"])
            require(all(row.get(key) == field for key, field in driver.items()), f"driver stdout does not match timing row at {stem}")
        else:
            require(first_failed is None and index == len(rows) - 1 and row["error"], f"invalid failure boundary at {stem}")
            first_failed = index
    return rows[:first_failed] if first_failed is not None else rows


def independent_summary(included, requested, floor):
    blocks = []
    position_ratios = {cohort: {metric: [] for metric, _ in METRICS} for cohort in ("odd", "even")}
    included_timed = [row for row in included if not row["warmup"]]
    for offset in range(0, len(included), LEGS):
        timed = included[offset + 1:offset + LEGS]
        block = 1 + offset // LEGS
        cohort = "odd" if block % 2 else "even"
        row = {"block": block, "order": ORDERS[(block - 1) % len(ORDERS)]}
        for metric, _ in METRICS:
            totals = {arm: math.fsum(leg[metric] for leg in timed if leg["arm"] == arm) for arm in "AB"}
            row[metric] = totals["B"] / totals["A"]
            block_mean = math.fsum(leg[metric] for leg in timed) / len(timed)
            position_ratios[cohort][metric].append([leg[metric] / block_mean for leg in timed])
        blocks.append(row)
    position_means = {cohort: {metric: [fmean(position) for position in zip(*values)]
                               for metric, values in metrics.items()}
                      for cohort, metrics in position_ratios.items()}
    stats = {}
    for metric, _ in METRICS:
        if not blocks:
            break
        values = [row[metric] for row in blocks]
        pooled = {arm: math.fsum(row[metric] for row in included_timed if row["arm"] == arm) for arm in "AB"}
        below, above = sum(n < 1 for n in values), sum(n > 1 for n in values)
        stats[metric] = {"pooled_b_over_a": pooled["B"] / pooled["A"], "block_median": median(values),
                         "block_min": min(values), "block_max": max(values), "below_one": below,
                         "above_one": above, "equal_one": len(values) - below - above}
        if floor is not None:
            low, high = floor.metrics[metric]["block_min"], floor.metrics[metric]["block_max"]
            outside = median(values) < low or median(values) > high
            consistent = max(below, above) >= 10
            decision = "incomplete"
            if len(blocks) == requested:
                decision = "MEASURED" if outside and consistent else "not measurable here"
            stats[metric].update(floor_min=low, floor_max=high, median_outside_floor=outside,
                                 at_least_10_on_one_side=consistent, decision=decision)
    return blocks, stats, position_means


def read_rows(path, finished):
    if not path.exists():
        return [], False
    rows = []
    lines = path.read_text(encoding="utf-8").splitlines(keepends=True)
    for number, line in enumerate(lines, 1):
        try:
            rows.append(json.loads(line))
        except json.JSONDecodeError:
            if not finished and number == len(lines) and not line.endswith("\n"):
                return rows, True
            raise
    return rows, False


def audit(directory, requested, floor=None, workload=SOFT):
    manifest = load(directory / "inputs.txt")
    verify_manifest(directory, manifest, requested, floor, workload)
    summary_path = directory / "summary.txt"
    recorded = load(summary_path) if summary_path.exists() else None
    rows, truncated = read_rows(directory / "runs.jsonl.txt", recorded is not None)
    eligible = inspect_rows(directory, manifest, rows, requested, workload.expected)
    included = eligible[:len(eligible) - len(eligible) % LEGS]
    if recorded and "verified_inputs_after" not in recorded:
        require(str(recorded["error"]).startswith("Post-run identity failed:"), f"missing verified input record in {directory}")
        included = []
    blocks, stats, position_means = independent_summary(included, requested, floor)
    complete = len(blocks) == requested
    if recorded is not None:
        require(recorded["schema"] == SCHEMA and recorded["order"] == list(ORDERS) and recorded["complete"] is complete,
                f"inconsistent summary header in {directory}")
        require(tuple(recorded["expected_games_searches_nodes_trace"]) == workload.expected
                and (not complete or recorded["error"] is None), f"inconsistent summary workload or exit in {directory}")
        require((recorded["requested_blocks"], recorded["completed_blocks"], recorded["attempted_runs"],
                 recorded["included_runs"], recorded["included_warmups"])
                == (requested, len(blocks), len(rows), len(blocks) * 4, len(blocks)),
                f"inconsistent summary counts in {directory}")
        require(all(recorded[key] == manifest[key] for key in ("arm_labels", "identical_floor", "floor", "worker_affinity")),
                f"inconsistent summary identities in {directory}")
        if "verified_inputs_after" in recorded:
            require(recorded["verified_inputs_after"] == manifest["inputs"], f"post-run input identity differs in {directory}")
        require(len(recorded["blocks"]) == len(blocks) and recorded["metrics"].keys() == stats.keys(),
                f"summary block/metric coverage differs in {directory}")
        for got, wanted in zip(recorded["blocks"], blocks):
            require((got["block"], got["order"]) == (wanted["block"], wanted["order"])
                    and all(same_number(got[key], wanted[key]) for key, _ in METRICS),
                    f"block ratio mismatch in {directory}: {wanted!r}")
        for metric, values in stats.items():
            for key, expected in values.items():
                got = recorded["metrics"][metric][key]
                same = same_number(got, expected) if type(expected) is float else got == expected
                require(same, f"summary mismatch in {directory}: {metric}.{key}, {got!r} versus {expected!r}")
        got_means = recorded["order_position_means"]
        require(got_means.keys() == position_means.keys(), f"order-position cohort mismatch in {directory}")
        for cohort, metrics in position_means.items():
            require(got_means[cohort].keys() == metrics.keys(), f"order-position metrics mismatch in {directory}: {cohort}")
            for metric, expected in metrics.items():
                got = got_means[cohort][metric]
                require(len(got) == len(expected) and all(same_number(a, b) for a, b in zip(got, expected)),
                        f"order-position mismatch in {directory}: {cohort}.{metric}")
    return Audit(directory, manifest, recorded, rows, blocks, stats, position_means,
                 complete and recorded is not None, truncated)


def report_campaign(heading, data):
    print(f"### {heading}\n")
    inputs = data.manifest["inputs"]
    labels = data.manifest["arm_labels"]
    print(f"B/A compares `{labels['B']}` (`{inputs['B']['sha256']}`) with `{labels['A']}` (`{inputs['A']['sha256']}`).\n")
    print("| Block | Leg | Engine | CPU (s) | Wall (s) | Trace | Exit | In ratios |")
    print("|---:|---|---|---:|---:|---|---:|---|")
    for index, row in enumerate(data.rows):
        included = "yes" if not row["warmup"] and index < len(data.blocks) * LEGS else "no"
        leg = "Warm-up" if row["warmup"] else row["order_index"]
        print(f"| {row['block']} | {leg} | {row['engine']} | {row['engine_cpu_s']:.3f} | "
              f"{row['total_wall_ns']/1e9:.3f} | `{row.get('trace_fnv1a64', 'pending')}` | {row['engine_exit']} | {included} |")
    if data.blocks:
        print("\n| Block | Timed order | B/A CPU | B/A wall |\n|---:|---|---:|---:|")
        for row in data.blocks:
            print(f"| {row['block']} | {row['order']} | {row['engine_cpu_s']:.4f} | {row['total_wall_ns']:.4f} |")
        print("\n| Metric | Median | Pooled | Block range | Below 1 | Above 1 | Reading |")
        print("|---|---:|---:|---:|---:|---:|---|")
        for metric, label in METRICS:
            row = data.metrics[metric]
            reading = row.get("decision", "Identical-binary floor") if data.complete else "incomplete"
            print(f"| {label} | {row['block_median']:.4f} | {row['pooled_b_over_a']:.4f} | "
                  f"{row['block_min']:.4f}–{row['block_max']:.4f} | {row['below_one']}/{len(data.blocks)} | "
                  f"{row['above_one']}/{len(data.blocks)} | {reading} |")
        print("\n| Block cohort | Metric | Timed leg 1 | Leg 2 | Leg 3 | Leg 4 |")
        print("|---|---|---:|---:|---:|---:|")
        for cohort, measurements in data.position_means.items():
            for metric, label in METRICS:
                means = measurements[metric]
                if means:
                    formatted = " | ".join(f"{value:.5f}" for value in means)
                    order = "ABBA" if cohort == "odd" else "BAAB"
                    print(f"| {cohort.capitalize()} ({order}) | {label} | {formatted} |")
        print("\nEach position is divided by the mean of the four timed legs in its block before averaging by cohort.")
    else:
        print("\nNo complete blocks were eligible.")
    if data.summary and data.summary["error"]:
        print(f"\nRecorded error `{data.summary['error']}`")
    if data.truncated_row:
        print("\nThe raw log ends with a partial run record. The complete preceding records appear above.")
    base = f"sol2/job10/{data.directory.name}"
    links = [f"[Inputs]({base}/inputs.txt)"]
    if (data.directory / "runs.jsonl.txt").is_file():
        links.append(f"[Raw runs]({base}/runs.jsonl.txt)")
    if data.summary is not None:
        links.append(f"[Summary]({base}/summary.txt)")
    command = f"{data.directory.name}-command.raw.txt"
    if (data.directory.parent / command).is_file():
        links.append(f"[Command](sol2/job10/{command})")
    print("\n" + " · ".join(links) + "\n")


def report_comparison(soft, fixed):
    for arm in "AB":
        soft_pin = soft.manifest["inputs"][arm]["sha256"]
        fixed_pin = fixed.manifest["inputs"][arm]["sha256"]
        require(soft_pin == fixed_pin, f"Job 9 and Job 10 name different executables for arm {arm}")
    print("### kF-pgo against r15d-pgo on both workloads\n")
    print("| Metric | Job 9 Step 3, softnodes 2000 | Job 10 Step 2, nodes 20000 |\n|---|---:|---:|")
    for index, label in enumerate(("Games per run", "Searches per run", "Last-iteration nodes per run", "Trace")):
        values = [row.manifest["expected_games_searches_nodes_trace"][index] for row in (soft, fixed)]
        values = [f"`{value}`" if index == 3 else f"{value:,}" for value in values]
        print(f"| {label} | {values[0]} | {values[1]} |")
    for metric, label in METRICS:
        options = (
            ("floor range", lambda values: f"{values['floor_min']:.4f}–{values['floor_max']:.4f}"),
            ("block median", lambda values: f"{values['block_median']:.4f}"),
            ("pooled B/A", lambda values: f"{values['pooled_b_over_a']:.4f}"),
            ("blocks below 1", lambda values: f"{values['below_one']}/{values['below_one'] + values['above_one'] + values['equal_one']}"),
            ("reading", lambda values: values["decision"]),
        )
        for title, format_value in options:
            values = [format_value(row.metrics[metric]) if row.complete else "Incomplete" for row in (soft, fixed)]
            print(f"| {label} {title} | {values[0]} | {values[1]} |")
    print("\nBoth sessions used the same two executable hashes and workload-specific floors. "
          "The fixed-node session uses the first 12 of the 60 openings from Job 9. "
          "[Job 9 raw comparison](sol2/job9/step3/runs.jsonl.txt) · "
          "[Job 10 raw comparison](sol2/job10/step2/runs.jsonl.txt)\n")


def available_campaign(root, name, floor=None, workload=SOFT):
    directory = root / name
    if not (directory / "inputs.txt").is_file():
        require(not (directory / "runs.jsonl.txt").is_file(), f"raw runs lack manifest in {directory}")
        return None
    return audit(directory, BLOCKS, floor, workload)


def main():
    parser = argparse.ArgumentParser(description=__doc__, epilog=(
        "Use --through 1 for the fixed-node floor only. After Step 2, use the default to read the candidate "
        "campaign and compare it with Job 9's soft-node result."
    ))
    parser.add_argument("job10", type=Path, help="directory containing Job 10's known answer and timing campaigns")
    parser.add_argument("job9", type=Path, help="directory containing Job 9's opening selection and timing campaigns")
    parser.add_argument("--through", type=int, choices=(1, 2), default=2, help="last campaign to read (default: 2)")
    args = parser.parse_args()
    if not (args.job10 / "step1" / "inputs.txt").is_file():
        require(not (args.job10 / "step1" / "runs.jsonl.txt").is_file(), "Job 10 floor raw runs lack a manifest")
        print("Step 1 has no timing receipts.\n")
        return
    workload = fixed_workload(args.job10, args.job9)
    floor = available_campaign(args.job10, "step1", workload=workload)
    games, searches, nodes, trace = workload.expected
    print(f"Job 10 uses PC20, Threads 1, Hash 16 MiB, net768 and `go nodes 20000`. Each eligible run uses "
          f"{games} openings and records {searches:,} searches, {nodes:,} last-iteration nodes and trace `{trace}`. "
          "[Known answer](sol2/job10/known-answer.txt) · [Openings](sol2/job10/fixed-openings.txt)\n")
    report_campaign("Step 1 — identical-binary fixed-node floor", floor)
    if args.through == 1 or not floor.complete:
        return
    candidate = available_campaign(args.job10, "step2", floor, workload)
    if candidate is None:
        print("Step 2 has no timing receipts.\n")
        return
    require(candidate.manifest["inputs"]["A"]["sha256"] == floor.manifest["inputs"]["A"]["sha256"],
            "Job 10 Step 2 baseline differs from its Step 1 floor executable")
    report_campaign("Step 2 — kF-pgo against r15d-pgo at 20,000 nodes", candidate)
    previous_floor = available_campaign(args.job9, "step1")
    require(previous_floor is not None and previous_floor.complete, "Job 9's completed floor is missing")
    previous = available_campaign(args.job9, "step3", previous_floor)
    require(previous is not None and previous.complete, "Job 9's completed whole-stack comparison is missing")
    report_comparison(previous, candidate)


if __name__ == "__main__":
    main()
