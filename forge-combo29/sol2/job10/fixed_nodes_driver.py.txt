#!/usr/bin/env python3
"""Forge PC20 fixed-node self-play over saved eight-ply move histories."""

import argparse
from contextlib import ExitStack
import json
import pathlib
import re
import subprocess
import sys
import time
from typing import TextIO

PC20_OPTIONS = [
    ("LmrQuietBase", "93"),
    ("LmrQuietDivisor", "205"),
    ("LmrNoisyBase", "13"),
    ("LmrNoisyDivisor", "333"),
    ("LmrHistoryDivisor", "5643"),
    ("NullMoveBase", "5"),
    ("LmpBase", "8"),
    ("HistoryPruneMultiplier", "2220"),
    ("HistoryBonusQuadratic", "24"),
    ("HistoryBonusLinear", "54"),
    ("HistoryBonusMax", "2212"),
    ("SingularDepth", "6"),
    ("IirDepth", "2"),
    ("NonPawnCorrection", "true"),
    ("CuckooRep", "true"),
    ("HistRework", "2"),
    ("TtPruneV2", "true"),
    ("CorrHistV2", "true"),
    ("MateAlphaGuard", "true"),
    ("OwnBook", "false"),
    ("Threads", "1"),
    ("Hash", "16"),
]


class MinimalBoardTracker:
    """Tracks board placement, castling, ep, halfmove clock, and 3-fold repetition for UCI moves."""

    def __init__(self):
        self.grid = [
            "R", "N", "B", "Q", "K", "B", "N", "R",
            "P", "P", "P", "P", "P", "P", "P", "P",
            ".", ".", ".", ".", ".", ".", ".", ".",
            ".", ".", ".", ".", ".", ".", ".", ".",
            ".", ".", ".", ".", ".", ".", ".", ".",
            ".", ".", ".", ".", ".", ".", ".", ".",
            "p", "p", "p", "p", "p", "p", "p", "p",
            "r", "n", "b", "q", "k", "b", "n", "r",
        ]
        self.stm = "w"
        self.castling = "KQkq"
        self.ep = -1
        self.halfmove = 0
        self.fullmove = 1
        self.counts = {}
        self._record()

    @staticmethod
    def sq(name: str) -> int:
        return (ord(name[1]) - ord("1")) * 8 + (ord(name[0]) - ord("a"))

    @staticmethod
    def sq_name(idx: int) -> str:
        return chr(ord("a") + (idx & 7)) + chr(ord("1") + (idx >> 3))

    def _key(self):
        return ("".join(self.grid), self.stm, self.castling, self.ep)

    def _record(self):
        k = self._key()
        self.counts[k] = self.counts.get(k, 0) + 1

    def fen(self) -> str:
        rows = []
        for r in range(7, -1, -1):
            row = ""
            empty = 0
            for f in range(8):
                p = self.grid[r * 8 + f]
                if p == ".":
                    empty += 1
                else:
                    if empty:
                        row += str(empty)
                        empty = 0
                    row += p
            if empty:
                row += str(empty)
            rows.append(row)
        c = self.castling if self.castling else "-"
        ep_str = self.sq_name(self.ep) if self.ep >= 0 else "-"
        return f"{'/'.join(rows)} {self.stm} {c} {ep_str} {self.halfmove} {self.fullmove}"

    def push_uci(self, uci: str):
        fr = self.sq(uci[0:2])
        to = self.sq(uci[2:4])
        promo = uci[4] if len(uci) == 5 else ""
        piece = self.grid[fr]
        target = self.grid[to]
        is_pawn = piece in ("P", "p")
        is_cap = target != "." or (is_pawn and to == self.ep)

        new_ep = -1
        if is_pawn and to == self.ep:
            cap_sq = to - 8 if self.stm == "w" else to + 8
            self.grid[cap_sq] = "."
        elif piece == "K" and fr == 4 and to == 6:
            self.grid[7] = "."
            self.grid[5] = "R"
        elif piece == "K" and fr == 4 and to == 2:
            self.grid[0] = "."
            self.grid[3] = "R"
        elif piece == "k" and fr == 60 and to == 62:
            self.grid[63] = "."
            self.grid[61] = "r"
        elif piece == "k" and fr == 60 and to == 58:
            self.grid[56] = "."
            self.grid[59] = "r"

        self.grid[fr] = "."
        if promo:
            self.grid[to] = promo.upper() if self.stm == "w" else promo.lower()
        else:
            self.grid[to] = piece

        if is_pawn and abs(to - fr) == 16:
            mid = (fr + to) // 2
            opp_pawn = "p" if self.stm == "w" else "P"
            f_file = to & 7
            has_ep = False
            if f_file > 0 and self.grid[to - 1] == opp_pawn:
                has_ep = True
            if f_file < 7 and self.grid[to + 1] == opp_pawn:
                has_ep = True
            if has_ep:
                new_ep = mid

        for ch, sq_idx in (("K", 4), ("Q", 4), ("K", 7), ("Q", 0), ("k", 60), ("q", 60), ("k", 63), ("q", 56)):
            if fr == sq_idx or to == sq_idx:
                self.castling = self.castling.replace(ch, "")

        self.ep = new_ep
        if is_pawn or is_cap:
            self.halfmove = 0
            self.counts.clear()
        else:
            self.halfmove += 1
        if self.stm == "b":
            self.fullmove += 1
        self.stm = "b" if self.stm == "w" else "w"
        self._record()

    def is_draw(self) -> bool:
        if self.halfmove >= 100:
            return True
        if self.counts.get(self._key(), 0) >= 3:
            return True
        return False


def fnv1a64_str(h: int, s: str) -> int:
    mask = (1 << 64) - 1
    for b in s.encode("ascii"):
        h = ((h ^ b) * 1099511628211) & mask
    return h


def send(proc: subprocess.Popen, text: str):
    assert proc.stdin is not None
    proc.stdin.write(text)
    proc.stdin.flush()


def wait_ready(proc: subprocess.Popen, check_errors: bool = True, receipt: TextIO | None = None):
    assert proc.stdout is not None
    while True:
        line = proc.stdout.readline()
        if not line:
            raise RuntimeError("Engine EOF before readyok")
        if receipt is not None:
            receipt.write("< " + line)
        s = line.strip()
        if check_errors and any(bad in s for bad in ("unknown option", "invalid value", "ERROR")):
            raise RuntimeError(f"Engine option/setup error: {s}")
        if s == "readyok":
            return


def load_openings(path, games, opening_plies):
    rows = json.loads(path.read_text(encoding="utf-8"))
    if len(rows) != games:
        raise ValueError("Expected the pinned number of opening histories")
    openings = []
    for row in rows:
        moves = row["moves"]
        if len(moves) != opening_plies or any(not re.fullmatch("[a-h][1-8][a-h][1-8][nbrq]?", move) for move in moves):
            raise ValueError("Invalid opening move vector")
        board = MinimalBoardTracker()
        for move in moves:
            board.push_uci(move)
        if board.fen().split()[:3] != row["position_side_castling"]:
            raise ValueError("Opening history and position receipt disagree")
        openings.append(moves)
    if len({tuple(moves) for moves in openings}) != games:
        raise ValueError("Duplicate pinned opening histories")
    return openings


def run_selfplay(
    engine_path: str,
    net_path: str,
    openings: list[list[str]],
    nodes: int,
    opening_plies: int,
    max_plies: int,
    trace_rows: TextIO | None,
    wire_log: TextIO | None,
):
    proc = subprocess.Popen(
        [engine_path],
        stdin=subprocess.PIPE,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
        bufsize=1,
    )
    def issue(command):
        send(proc, command)
        if wire_log is not None:
            for written in command.splitlines():
                print("> " + written, file=wire_log)

    go_command = f"go nodes {nodes}\n"
    num_games = len(openings)
    try:
        cfg_lines = ["uci\n"]
        for k, v in PC20_OPTIONS:
            cfg_lines.append(f"setoption name {k} value {v}\n")
        cfg_lines.append(f"setoption name NNUEArchFile value {net_path}\n")
        cfg_lines.append("setoption name NNUEArch value true\n")
        cfg_lines.append("isready\n")
        issue("".join(cfg_lines))
        wait_ready(proc, receipt=wire_log)

        # One untimed search faults in executable and allocator pages.
        issue("ucinewgame\nisready\n")
        wait_ready(proc, check_errors=True)
        issue(f"position startpos moves {' '.join(openings[0])}\n" + go_command)
        assert proc.stdout is not None
        while True:
            ln = proc.stdout.readline()
            if not ln:
                raise RuntimeError("Engine EOF during internal warm-up")
            if ln.startswith("info string softnodes"):
                raise RuntimeError("Engine reported soft-node mode during hard-node warm-up")
            if ln.startswith("bestmove "):
                break

        total_searches = 0
        total_nodes = 0
        trace_hash = 14695981039346656037

        t0 = time.perf_counter_ns()
        for g_idx, opening in enumerate(openings):
            issue("ucinewgame\nisready\n")
            wait_ready(proc, check_errors=False)

            moves = list(opening)
            tracker = MinimalBoardTracker()
            for mv in moves:
                tracker.push_uci(mv)

            for ply in range(opening_plies, max_plies):
                if tracker.is_draw():
                    break
                issue(f"position startpos moves {' '.join(moves)}\n" + go_command)

                last_nodes = 0
                last_score = ""
                bestmove = "0000"
                last_info = ""
                while True:
                    line = proc.stdout.readline()
                    if not line:
                        raise RuntimeError("Engine EOF during search")
                    if line.startswith("info string softnodes"):
                        raise RuntimeError("Engine reported soft-node mode during hard-node search")
                    if line.startswith("info depth "):
                        last_info = line
                        parts = line.split()
                        for i, tok in enumerate(parts):
                            if tok == "nodes" and i + 1 < len(parts):
                                last_nodes = int(parts[i + 1])
                            elif tok == "score" and i + 2 < len(parts):
                                last_score = f"{parts[i + 1]}:{parts[i + 2]}"
                    elif line.startswith("bestmove "):
                        bestmove = line.split()[1]
                        break

                if trace_rows is not None:
                    print(json.dumps({
                        "game": g_idx, "ply": ply, "bestmove": bestmove, "nodes": last_nodes,
                        "score": last_score, "last_info": last_info.rstrip("\n"), "bestmove_line": line.rstrip("\n"),
                    }), file=trace_rows)
                total_searches += 1
                total_nodes += last_nodes
                trace_hash = fnv1a64_str(trace_hash, f"{g_idx},{ply},{bestmove},{last_nodes},{last_score};")

                if bestmove == "0000" or last_nodes == 0:
                    break
                moves.append(bestmove)
                tracker.push_uci(bestmove)

        t1 = time.perf_counter_ns()
        issue("quit\n")
        if proc.wait(timeout=5) != 0:
            raise RuntimeError(f"Forge exited {proc.returncode}")
    finally:
        if proc.poll() is None:
            proc.kill()

    elapsed_s = (t1 - t0) / 1e9
    return {
        "games": num_games,
        "searches": total_searches,
        "total_nodes": total_nodes,
        "avg_nodes_per_search": round(total_nodes / max(1, total_searches), 2),
        "avg_searches_per_game": round(total_searches / max(1, num_games), 2),
        "elapsed_ms": round(elapsed_s * 1000.0, 3),
        "searches_per_sec": round(total_searches / elapsed_s, 2),
        "effective_nps": round(total_nodes / elapsed_s, 1),
        "trace_fnv1a64": f"{trace_hash:016x}",
    }


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--engine", required=True)
    parser.add_argument("--net", required=True)
    parser.add_argument("--games", required=True, type=int)
    parser.add_argument("--nodes", required=True, type=int)
    parser.add_argument("--random-plies", required=True, type=int)
    parser.add_argument("--max-plies", required=True, type=int)
    parser.add_argument("--openings", required=True, type=pathlib.Path)
    parser.add_argument("--trace-rows", type=pathlib.Path)
    parser.add_argument("--wire-log", type=pathlib.Path)
    parser.add_argument("--json", action="store_true")
    args = parser.parse_args()
    if args.games < 1 or args.nodes < 1 or not 0 < args.random_plies < args.max_plies:
        parser.error("Expected positive games and hard nodes, and opening length below the ply ceiling")
    openings = load_openings(args.openings, args.games, args.random_plies)
    with ExitStack() as receipt:
        trace_rows = receipt.enter_context(args.trace_rows.open("x", encoding="utf-8")) if args.trace_rows else None
        wire_log = receipt.enter_context(args.wire_log.open("x", encoding="utf-8")) if args.wire_log else None
        summary = run_selfplay(args.engine, args.net, openings, args.nodes, args.random_plies, args.max_plies, trace_rows, wire_log)
    if args.json:
        print(json.dumps(summary))
    else:
        for key, value in summary.items():
            print(f"{key}: {value}")


if __name__ == "__main__":
    main()
