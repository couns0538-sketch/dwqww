def variant_pairs(variant):
    n1, n2 = STACK_ARMS[variant]
    return {
        "P1": ("r15d", "r15d_2"),
        "P2": ("K_1", "K_2"),
        "N1": ("K_1", n1),
        "N2": ("K_2", n2),
        "R1": ("r15d", n1),
        "R2": ("r15d_2", n2),
    }


@dataclass
class Known:
    signature: tuple
    hashes: dict
    clang_inputs: dict


@dataclass
class Campaign:
    directory: Path
    manifest: dict
    rows: list
    included_blocks: list
    metrics: dict
    position_means: dict
    summary: dict | None
    complete: bool
    truncated: bool


def require(condition, message):
    if not condition:
        raise ValueError(message)


def load(path):
    return json.loads(path.read_text(encoding="utf-8"))


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def signature(row):
    return tuple(row.get(field) for field in TRACE_FIELDS)


def check(actual, expected, where):
    if isinstance(expected, dict):
        require(isinstance(actual, dict) and expected.keys() <= actual.keys(), f"missing fields at {where}")
        for key, value in expected.items():
            check(actual[key], value, f"{where}.{key}")
    elif isinstance(expected, (tuple, list)):
        require(isinstance(actual, (tuple, list)) and len(actual) == len(expected), f"list shape differs at {where}")
        for index, (one, two) in enumerate(zip(actual, expected)):
            check(one, two, f"{where}[{index}]")
    elif type(expected) is bool:
        require(actual is expected, f"wrong {where}: {actual!r}, expected {expected!r}")
    else:
        equal = math.isclose(actual, expected, rel_tol=1e-9, abs_tol=1e-12) if type(expected) is float else actual == expected
        require(equal, f"wrong {where}: {actual!r}, expected {expected!r}")


def read_rows(path, finished):
    if not path.is_file():
        return [], False
    lines = path.read_text(encoding="utf-8").splitlines(keepends=True)
    rows = []
    for index, line in enumerate(lines, 1):
        try:
            rows.append(json.loads(line))
        except json.JSONDecodeError:
            if not finished and index == len(lines) and not line.endswith("\n"):
                return rows, True
            raise
    return rows, False


def validate_run(directory, stem, row, expected):
    source = f"{directory}/{stem}"
    stdout, stderr = f"{stem}.driver.stdout.txt", f"{stem}.engine.stderr.txt"
    check(row, {"driver_stdout": stdout, "engine_stderr": stderr}, source)
    require((directory / stdout).is_file() and (directory / stderr).is_file(), f"missing driver/engine receipts for {source}")
    actual = signature(row)
    check(row["trace_matches"], actual == expected, f"{source} signature flag")
    start, end = row["wall_start_monotonic_ns"], row["wall_end_monotonic_ns"]
    user, system = row["engine_user_s"], row["engine_system_s"]
    check(row, {"total_wall_ns": end - start, "engine_cpu_s": user + system}, f"{source} durations")
    if row["eligible"] or any(value is not None for value in actual):
        driver = load(directory / stdout)
        check(row, driver, f"{source} driver stdout")
    if row["eligible"]:
        require(actual == expected and row["engine_exit"] == 0 and row["error"] is None
                and user >= 0 and system >= 0 and user + system > 0 and end > start,
                f"{source} marked eligible without correct identity, status and durations")
    else:
        require(bool(row["error"]), f"{source} excluded without a recorded error")


def validate_tape(path, receipt):
    rows, _ = read_rows(path, True)
    last_info_total = final_total = final_lines = zero_last = 0
    hash_value = 14695981039346656037
    current_book_line = 7000
    current_ply = 0
    seen = set()
    for row in rows:
        book_line = row["book_line"]
        require(7001 <= book_line <= 7020 and book_line >= current_book_line, f"unordered book line in {path}")
        current_ply = current_ply + 1 if book_line == current_book_line else 1
        current_book_line = book_line
        seen.add(book_line)
        check(row["ply"], current_ply, f"{path} game {book_line} ply")
        match = re.search(r"\bnodes (\d+)(?=\s|$)", row["last_info"] or "")
        check(row["nodes"], int(match[1]) if match else 0, f"{path} game {book_line} last-info nodes")
        move = row["bestmove_line"].split()
        require(move[:2] == ["bestmove", row["bestmove"]], f"bestmove differs from its UCI line in {path}")
        final = row["final_nodes"]
        if final is None:
            require(row["final_nodes_line"] is None and row["nodes"] == 0, f"missing final soft-node telemetry in {path}")
        else:
            require(final >= row["nodes"], f"final soft nodes below last-info nodes in {path}")
            check(row["final_nodes_line"], f"info string softnodes nodes {final}", f"{path} final soft-node line")
            final_lines += 1
            final_total += final
        last_info_total += row["nodes"]
        zero_last += row["nodes"] == 0
        wire = f"{row['bestmove']},{row['nodes']};"
        check(row["trace_input"], wire, f"{path} trace input")
        for byte in wire.encode("ascii"):
            hash_value = (hash_value ^ byte) * 1099511628211 & ((1 << 64) - 1)
        check(row["trace_fnv1a64"], f"{hash_value:016x}", f"{path} cumulative trace")
    require(seen == set(range(7001, 7021)), f"wrong W book lines in {path}")
    check(receipt, {"searches": len(rows), "total_nodes": last_info_total, "final_nodes": final_total,
                    "trace_fnv1a64": f"{hash_value:016x}", "final_lines": final_lines,
                    "missing_final_lines": len(rows) - final_lines, "zero_last_info_searches": zero_last},
          f"{path} tape summary")


def read_pins(path, labels="AB"):
    lines = path.read_text(encoding="utf-8").splitlines()
    require(len(lines) == len(labels), f"expected one executable receipt per engine in {path}")
    pins = {}
    for arm, line in zip(labels, lines, strict=True):
        sha256, executable = line.split(maxsplit=1)
        require(re.fullmatch(r"[0-9a-f]{64}", sha256), f"invalid executable digest in {path}")
        pins[arm] = {"path": executable, "sha256": sha256}
    return pins


def known_workload(root):
    path = root / "known-answer.txt"
    source = load(path)
    expected = signature(source)
    require(source.keys() == set(TRACE_FIELDS) and expected[0] == 20
            and all(type(value) is int and value > 0 for value in expected[:4])
            and expected[3] >= expected[2]
            and isinstance(expected[4], str) and re.fullmatch(r"[0-9a-f]{16}", expected[4]),
            f"invalid W signature in {path}")
    pins = read_pins(root / "clang-executables.raw.txt")
    for key, arm in zip("AB", ("r15d", "kF")):
        gate = root / "step2"
        data = load(gate / f"{arm}.result.raw.txt")
        validate_run(gate, arm, data, expected)
        check(data["engine_sha256"], pins[key]["sha256"], f"Step 2 {arm} executable")
        require(data["eligible"], f"Step 2 {arm} failed the W gate")
        validate_tape(gate / f"{arm}.searches.raw.txt", data)
    return Known(expected, {"net": NET_SHA256, "book": BOOK_SHA256, "known_answer": digest(path),
                            "driver": digest(root / "soft_book_driver.py.txt")}, pins)


def timing_math(groups):
    blocks = []
    positions = {cohort: {key: [] for key, _ in METRICS} for cohort in ("odd", "even")}
    timed_runs = []
    for group in groups:
        measured = group[1:]
        timed_runs.extend(measured)
        block_index = group[0]["block"]
        cohort = "odd" if block_index % 2 else "even"
        block = {"block": block_index, "order": ORDERS[(block_index - 1) % len(ORDERS)]}
        for key, _ in METRICS:
            totals = {arm: math.fsum(row[key] for row in measured if row["arm"] == arm) for arm in "AB"}
            block[key] = totals["B"] / totals["A"]
            mean = fmean(row[key] for row in measured)
            positions[cohort][key].append([row[key] / mean for row in measured])
        blocks.append(block)
    means = {cohort: {key: [fmean(position) for position in zip(*series)] for key, series in data.items()}
             for cohort, data in positions.items()}
    metrics = {}
    count = len(blocks)
    for key, _ in METRICS:
        if count == 0:
            break
        values = [block[key] for block in blocks]
        ordered = sorted(values)
        block_mean = fmean(values)
        df = count - 1
        below, above = sum(value < 1 for value in values), sum(value > 1 for value in values)
        totals = {arm: math.fsum(row[key] for row in timed_runs if row["arm"] == arm) for arm in "AB"}
        data = {"block_mean": block_mean, "block_median": median(values), "block_min": ordered[0], "block_max": ordered[-1],
                "pooled_b_over_a": totals["B"] / totals["A"], "below_one": below, "above_one": above,
                "equal_one": count - below - above}
        if count == BLOCKS:
            halfwidth = T975 * stdev(values) / math.sqrt(count)
            data.update(block_mean_ci95=[block_mean - halfwidth, block_mean + halfwidth], ci_df=df)
        if count >= 3:
            data.update(block_middle_min=ordered[1], block_middle_max=ordered[-2])
        metrics[key] = data
    return blocks, metrics, means


def read_environment(record):
    wire = b"".join(f"{key}={value}".encode("ascii") + b"\0" for key, value in ENVIRONMENT.items())
    check(record, {"variables": ENVIRONMENT, "bytes": len(wire), "sha256": hashlib.sha256(wire).hexdigest()},
          "fixed engine environment")
    require(record["variables"] == ENVIRONMENT, "the engine environment must have just LC_ALL and TZ")
    return record


def verify_build_evidence(root, label, record, known):
    directory = root / "build-checks" / label
    receipt = directory / "checks.raw.txt"
    if not receipt.is_file():
        require(record["passes"] is False, f"eligible {label} has no checks receipt")
        check(record, {"pins": None, "W": None}, f"unavailable build {label}")
        return
    raw = load(receipt)
    check(record, raw, f"{label} build inventory against checks receipt")
    require(record["sha256"] is not None and Path(record["path"]).is_absolute(), f"invalid {label} executable pin")
    pins = raw["pins"]
    require(set(pins) == set(DEFAULT_COUNTS), f"wrong default count labels in {receipt}")
    default = directory / "default-pins.raw.txt"
    listing = default.read_text(encoding="utf-8")
    require(listing.startswith(f"Executable: {record['path']}\n"), f"wrong executable in {default}")
    labels = tuple(DEFAULT_COUNTS)
    for index, (name, (expected, command, pattern)) in enumerate(DEFAULT_COUNTS.items()):
        _, start, excerpt = listing.partition(f"--- {name} stdin ---\n")
        require(start, f"missing {name} input in {default}")
        if index + 1 < len(labels):
            excerpt = excerpt.partition(f"--- {labels[index + 1]} stdin ---\n")[0]
        stdin, stdout_marker, outputs = excerpt.partition(f"--- {name} stdout ---\n")
        stdout, stderr_marker, trailer = outputs.partition(f"--- {name} stderr ---\n")
        require(stdout_marker and stderr_marker, f"missing raw {name} output in {default}")
        check(stdin, command, f"{default} {name} command")
        chunks = re.findall(pattern, stdout, re.MULTILINE)
        actual = int(chunks[0]) if len(chunks) == 1 else None
        payload = json.dumps(pins[name]) + "\n"
        require(trailer.endswith(payload), f"raw {name} exit/summary differs in {default}")
        stderr = trailer[:-len(payload)]
        passes = pins[name]["exit_status"] == 0 and not stderr and actual == expected
        check(pins[name], {"expected": expected, "actual": actual, "passes": passes}, f"{default} {name} count")
    result = load(directory / "W.result.raw.txt")
    validate_run(directory, "W", result, known.signature)
    check(raw["W"], dict(zip(TRACE_FIELDS, signature(result))), f"{label} raw W")
    check(raw["W_passes"], result["eligible"], f"{label} W gate")
    if result["eligible"]:
        wire = b"LC_ALL=C\0TZ=UTC\0"
        check(result, {"engine_argv0": record["path"], "engine_argv0_length": len(record["path"]),
                       "engine_environment_bytes": len(wire), "engine_environment_sha256": hashlib.sha256(wire).hexdigest()},
              f"{label} W process launch")
    check(raw["passes"], result["eligible"] and all(pin["passes"] for pin in pins.values()), f"{label} build gate")


def verify_manifest(directory, manifest, known, pins, environment, pair):
    check(manifest, {"schema": SCHEMA, "order": ORDERS, "blocks_requested": BLOCKS, "identical_floor": False,
                     "floor": None, "expected_games_searches_nodes_trace": known.signature,
                     "engine_environment": environment}, str(directory))
    read_environment(manifest["engine_environment"])
    affinity = manifest["worker_affinity"]
    check(affinity, [2], f"{directory} Job 16 logical CPU")
    identities = manifest["inputs"]
    for key, sha256 in known.hashes.items():
        check(identities[key]["sha256"], sha256, f"{directory} {key} SHA-256")
    for arm, label in zip("AB", pair):
        check(manifest["arm_labels"][arm], label, f"{directory} arm {arm} label")
        pin = pins[label]
        check(identities[arm], {key: pin[key] for key in ("path", "sha256")}, f"{directory} arm {arm} pin")
        for key in ("length", "inode", "device"):
            if key in identities[arm]:
                check(identities[arm][key], pin[key], f"{directory} arm {arm} {key}")
        argv = [identities["driver"]["path"], "--engine", pin["path"], "--net", identities["net"]["path"],
                "--book", identities["book"]["path"], *WORKLOAD]
        check(manifest["argv_by_arm"][arm], argv, f"{directory} arm {arm} argv")


def engine_metadata(directory, stem, row, pin, environment):
    where = f"{directory}/{stem} launch"
    expected = {"engine_argv0": pin["path"], "engine_argv0_length": pin["length"],
                "engine_environment_bytes": environment["bytes"], "engine_environment_sha256": environment["sha256"]}
    if row["eligible"] or row["engine_pid"] is not None or not expected.keys().isdisjoint(row):
        check(row, expected, where)
    proc = row.get("engine_proc_receipt")
    require(proc in (None, "verified", "unavailable"), f"unexpected procfs receipt at {where}")
    if proc == "verified":
        check(row, {"engine_inode": pin["inode"], "engine_device": pin["device"]}, f"{where} procfs")


def verified_blocks(directory, manifest, rows, known, pins, environment):
    require(len(rows) <= LEGS * BLOCKS, f"too many raw runs in {directory}")
    end = 0
    for index, row in enumerate(rows):
        block, leg = divmod(index, LEGS)
        order = ORDERS[block % len(ORDERS)]
        arm = order[0] if leg == 0 else order[leg - 1]
        stem = f"b{block + 1:02d}-r{leg}-{arm}"
        label = manifest["arm_labels"][arm]
        check(row, {"block": block + 1, "order_index": leg, "arm": arm, "warmup": leg == 0,
                    "engine": label, "engine_sha256": pins[label]["sha256"]}, stem)
        engine_metadata(directory, stem, row, pins[label], environment)
        validate_run(directory, stem, row, known.signature)
        require(row["wall_start_monotonic_ns"] >= end, f"overlapping run at {stem}")
        end = row["wall_end_monotonic_ns"]
    groups = [rows[offset:offset + LEGS] for offset in range(0, len(rows), LEGS)]
    return [group for group in groups if len(group) == LEGS and all(row["eligible"] for row in group)]


def audit(directory, known, pins, environment, pair):
    manifest = load(directory / "inputs.txt")
    verify_manifest(directory, manifest, known, pins, environment, pair)
    summary_path = directory / "summary.txt"
    summary = load(summary_path) if summary_path.is_file() else None
    rows, truncated = read_rows(directory / "runs.jsonl.txt", summary is not None)
    groups = verified_blocks(directory, manifest, rows, known, pins, environment)
    if summary is not None and "verified_inputs_after" not in summary:
        require(str(summary["error"]).startswith("Post-run identity failed:"), f"missing post-run identity in {directory}")
        groups = []
    blocks, metrics, means = timing_math(groups)
    complete = len(blocks) == BLOCKS and len(rows) == BLOCKS * LEGS
    if summary is not None:
        check(summary, {"schema": SCHEMA, "order": ORDERS, "requested_blocks": BLOCKS, "completed_blocks": len(blocks),
                        "comparison_direction": "B / A, less than 1 is faster",
                        "attempted_runs": len(rows), "included_runs": 4 * len(blocks), "included_warmups": len(blocks),
                        "complete": complete, "blocks": blocks, "order_position_means": means,
                        "expected_games_searches_nodes_trace": known.signature,
                        **{key: manifest[key] for key in ("arm_labels", "identical_floor", "floor", "worker_affinity")}},
              str(summary_path))
        require(summary["metrics"].keys() == metrics.keys(), f"unexpected metric coverage in {summary_path}")
        for key, values in metrics.items():
            optional = ("block_middle_min", "block_middle_max")
            actual = summary["metrics"][key]
            check(actual, {item: value for item, value in values.items() if item not in optional or item in actual},
                  f"{summary_path} {key}")
            require(complete or {"block_mean_ci95", "ci_df"}.isdisjoint(actual), f"incomplete {directory} has a 15-df CI")
            require("decision" not in actual, f"unexpected per-campaign decision in {directory}")
        if "engine_environment" in summary:
            check(summary["engine_environment"], environment, f"{summary_path} environment")
        if "verified_inputs_after" in summary:
            check(summary["verified_inputs_after"], manifest["inputs"], f"{summary_path} post-run input pins")
        require(not complete or summary["error"] is None, f"completed campaign records an error in {summary_path}")
    return Campaign(directory, manifest, rows, blocks, metrics, means, summary, complete and summary is not None, truncated)


def interval_direction(interval):
    low, high = interval
    if high < 1:
        return "below"
    if low > 1:
        return "above"
    return "includes"


def completed_metric(campaigns, step, key):
    data = campaigns.get(step)
    return data.metrics[key] if data is not None and data.complete else None

    print("|---|---|---:|---:|---:|---|")
    for key in labels:

            continue

def show_overview(campaigns, pairs, skips, through):
    print("\n| Comparison | A | B | Metric | Blocks | Median | Pooled B/A | Mean | 95% t interval, df 15 | Below 1 |")
    print("|---|---|---|---|---:|---:|---:|---:|---:|---:|")
    for step, (a, b) in pairs.items():
        data = campaigns.get(step)
        for key, label in METRICS:
            if data is None:
                print(f"| {step} | {a} | {b} | {label} | Not run | — | — | — | — | — |")
            elif key not in data.metrics:
                print(f"| {step} | {a} | {b} | {label} | 0/{BLOCKS} | — | — | — | — | — |")
            else:
                values = data.metrics[key]
                count = len(data.included_blocks)
                interval = "—"
                if data.complete:
                    lo, hi = values["block_mean_ci95"]
                    interval = f"[{lo:.4f}, {hi:.4f}]"
                print(f"| {step} | {a} | {b} | {label} | {count}/{BLOCKS} | {values['block_median']:.4f} | "
                      f"{values['pooled_b_over_a']:.4f} | {values['block_mean']:.4f} | {interval} | "
                      f"{values['below_one']}/{count} |")
        if step == through:
            break
    for step, reason in skips.items():
        print(f"\n{step} was skipped. {reason}. [Build receipts](sol2/job21/builds.txt)")


def show_campaign(step, data):
    print(f"\n### {step}\n")
    print(f"Logical CPU {data.manifest['worker_affinity'][0]}. {len(data.rows)} attempted runs, "
          f"{len(data.included_blocks)} included blocks, {len(data.included_blocks) * 4} included timed runs.")
    if data.included_blocks:
        print("\n| Block | Timed order | B/A CPU | B/A wall |")
        print("|---:|---|---:|---:|")
        for row in data.included_blocks:
            print(f"| {row['block']} | {row['order']} | {row['engine_cpu_s']:.4f} | {row['total_wall_ns']:.4f} |")
    for row in data.rows:
        if not row["eligible"]:
            x = row["block"], row["order_index"], row["arm"]
            stem = f"b{x[0]:02d}-r{x[1]}-{x[2]}"
            base = f"sol2/job21/{step}/{stem}"
            print(f"\nExcluded block {x[0]}, leg {x[1]}, arm {x[2]}. Observed `{signature(row)}`. "
                  f"Recorded error `{row['error']}`. [Driver]({base}.driver.stdout.txt) · [Engine]({base}.engine.stderr.txt)")
    if not data.complete:
        print("\nThe campaign is incomplete.")
    if data.summary and data.summary["error"]:
        print(f"\nRecorded summary error `{data.summary['error']}`.")
    if data.truncated:
        print("\nThe last raw run record is partial. The preceding complete receipts were checked.")
    base = f"sol2/job21/{step}"
    links = [f"[Inputs]({base}/inputs.txt)"]
    if (data.directory / "runs.jsonl.txt").is_file():
        links.append(f"[Raw runs]({base}/runs.jsonl.txt)")
    if data.summary is not None:
        links.append(f"[Summary]({base}/summary.txt)")
    command = data.directory.parent / f"{step}-command.raw.txt"
    if command.is_file():
        links.append(f"[Command](sol2/job21/{command.name})")
    print("\n" + " · ".join(links))


def read_builds(root, frozen, known, new_builds):
    path = root / "builds.txt"
    builds = load(path)["builds"]
    require(set(builds) == set(frozen) | set(new_builds), f"wrong build labels in {path}")
    for label in frozen:
        # The frozen executable can be recorded under its original build path or its verified launch copy.
        record = builds[label]
        check(record, {"sha256": frozen[label]["sha256"], "passes": True, "pins": None,
                       "W": dict(zip(TRACE_FIELDS, known.signature))}, f"{path} {label} frozen build")
        require(Path(record["path"]).is_absolute(), f"invalid original path for {label}")
    for label in new_builds:
        record = builds[label]
        require(type(record["passes"]) is bool, f"missing build eligibility for {label}")
        sha = record["sha256"]
        require(sha is not None or not record["passes"], f"no digest for eligible build {label}")
        require(sha is None or re.fullmatch(r"[0-9a-f]{64}", sha), f"invalid digest for {label}")
        require(record.get("twin") is None and record.get("byte_identical_to_twin") is None,
                f"unexpected twin fields for {label}")
        verify_build_evidence(root, label, record, known)
    return builds


def read_arms(root, builds, frozen, prior_environment):
    path = root / "arms.txt"
    document = load(path)
    environment = read_environment(document["engine_environment"])
    check(environment, prior_environment, "engine environment relative to Job 16")
    pins = document["arms"]
    expected = {name for name, record in builds.items() if record["passes"]}
    require(set(pins) == expected, f"{path} arm names disagree with eligible builds")
    for name, arm in pins.items():
        require(Path(arm["path"]).is_absolute() and type(arm["inode"]) is int and arm["inode"] > 0
                and type(arm["device"]) is int and arm["device"] >= 0, f"invalid path/inode/device for {name}")
        check(arm["length"], 72, f"{path} {name} declared path length")
        check(len(arm["path"]), 72, f"{path} {name} actual path length")
        check(arm["sha256"], builds[name]["sha256"], f"{path} {name} build SHA-256")
        if name in frozen:
            check(arm, frozen[name], f"frozen launch identity {name}")
    require(len({pin["path"] for pin in pins.values()}) == len(pins), "build arm paths must be distinct")
    require(len({(pin["device"], pin["inode"]) for pin in pins.values()}) == len(pins),
            "build arm inodes must be distinct")
    return pins, environment


def comparison_plan(root, pins, pairs, *, subset=False):
    skips = {}
    for step, pair in pairs.items():
        unavailable = [label for label in pair if label not in pins]
        if unavailable:
            skips[step] = "Build gate unavailable or failed for " + ", ".join(f"`{label}`" for label in unavailable)
    plan_path = root / "comparison-plan.txt"
    if plan_path.is_file():
        plan = load(plan_path)
        check(plan["comparisons"], pairs, "declared comparison arm assignments")
        recorded = set(plan["skips"])
        check(recorded & pairs.keys() if subset else recorded, set(skips), "gate-based comparison skips in plan")
    skipped_path = root / "skipped.txt"
    if skipped_path.is_file():
        recorded = set(load(skipped_path))
        check(recorded & pairs.keys() if subset else recorded, set(skips), "gate-based comparison skips in receipt")
    return skips


def verify_gate(root, known, frozen, environment, receipt_labels, *, render=False):
    pins = read_pins(root / "clang-executables.raw.txt", tuple(receipt_labels))
    for label, pin in pins.items():
        check(pin, {key: frozen[label][key] for key in ("path", "sha256")}, f"frozen launch pin {label} in {root}")
        check(len(pin["path"]), 72, f"Step 0 {label} path length")
    check(digest(root / "known-answer.txt"), known.hashes["known_answer"], f"{root} known-answer bytes")
    entries = [line.split(maxsplit=1)
               for line in (root / "workload-pins.raw.txt").read_text(encoding="utf-8").splitlines()]
    intake = {Path(path).name: sha256 for sha256, path in entries}
    require(len(intake) == len(entries) == 4, f"{root} must record the four workload input pins")
    for name, key in (("soft_book_driver.py", "driver"), ("known-answer.txt", "known_answer"),
                      ("UHO_Lichess_4852_v1.epd", "book"), ("net768.fnn3", "net")):
        check(intake[name], known.hashes[key], f"{root} workload receipt {name}")
    compiler = root / "step0-command.raw.txt"
    compiler_ok = compiler.is_file() and re.search(r"\bclang version 19\.1\.7\b", compiler.read_text(encoding="utf-8"))
    if not compiler_ok:
        if render:
            print("Step 0 clang 19.1.7 receipt is missing or did not pass. [Compiler and Step 0](sol2/job21/step0-command.raw.txt)")
        return False
    gate = root / "step0"
    eligible = []
    pids = []
    for label, stored in receipt_labels.items():
        path = gate / f"{stored}.result.raw.txt"
        if not path.is_file():
            if render:
                print(f"Step 0 {label} receipt is missing.")
            continue
        row = load(path)
        validate_run(gate, stored, row, known.signature)
        check(row["engine_sha256"], pins[label]["sha256"], f"Step 0 {label} build")
        engine_metadata(gate, stored, row, frozen[label], environment)
        if not row["eligible"]:
            if render:
                print(f"Step 0 {label} failed with `{row['error']}`. [Receipt](sol2/job21/step0/{path.name})")
            continue
        pids.append(row["engine_pid"])
        eligible.append(f"[{label}](sol2/job21/step0/{path.name}#L2-L6)")
    if len(eligible) != len(receipt_labels):
        return False
    require(None not in pids and len(set(pids)) == len(receipt_labels), "Step 0 must use a fresh engine process per arm")
    if render:
        games, searches, nodes, final_nodes, trace = known.signature
        print(f"All three Step 0 builds matched W with {games} games, {searches:,} searches, {nodes:,} last-info nodes, "
              f"{final_nodes:,} final nodes and trace `{trace}`. W uses PC20, Threads 1, Hash 16 MiB, net768 and "
              f"`go softnodes 2000` on book lines 7001–7020. {' · '.join(eligible)} · [W signature](sol2/job21/known-answer.txt#L2-L6)")
    return True


def interval_text(metric):
    low, high = metric["block_mean_ci95"]
    return f"[{low:.4f}, {high:.4f}]"


def reading(metric):
    if metric is None:
        return None
    side = interval_direction(metric["block_mean_ci95"])
    return {"below": "PAYS", "above": "COSTS", "includes": "NOT SEEN"}[side]


def small_stack_reading(n1_cpu, n2_cpu):
    outcomes = (reading(n1_cpu), reading(n2_cpu))
    if all(value in ("PAYS", "NOT SEEN") for value in outcomes):
        return "KEEPS"
    if outcomes == ("COSTS", "COSTS"):
        return "OTHER PIECES PAY"
    return "OPEN"


def metric_cell(metric):
    if metric is None:
        return "—"
    return f"{metric['pooled_b_over_a']:.4f} / {metric['block_mean']:.4f} {interval_text(metric)}"


def percent_cell(metric):
    if metric is None:
        return "—"
    low, high = metric["block_mean_ci95"]
    mean = metric["block_mean"]
    pooled = metric["pooled_b_over_a"]
    return f"{100 * (pooled - 1):+.2f}% / {100 * (mean - 1):+.2f}% [{100 * (low - 1):+.2f}%, {100 * (high - 1):+.2f}%]"


def summary_link(step):
    return f"[{step}](sol2/job21/{step}/summary.txt#L112-L146)"


def show_prior_placebo(anchor, placebos):
    if not placebos:
        return

    history = {
        15: ("T1", "T2", "U40", "U56", "U72", "U88", "V40", "V88"),
        16: ("G1", "G2", "G3", "R1", "F1"),
        17: ("L0", "L1", "L2"),
        18: ("L1", "L2", "L3", "L4"),
        19: ("L1b", "L2b", "L3b", "L4b"),
        20: ("S1", "S2", "S3", "S4", "S1b", "S2b", "S3b", "S4b"),
    }
    gaps = {label: abs(metric["pooled_b_over_a"] - 1) for label, metric in placebos.items()}
    print("\n### Earlier CPU comparisons against the observed placebo gap")
    for label, gap in gaps.items():
        metric = placebos[label]
        require(interval_direction(metric["block_mean_ci95"]) != "includes", f"{label} contains parity")
        print(f"\n{label} differed from parity by {100 * gap:.2f}% in pooled CPU time. "
              f"Its mean ratio was {metric['block_mean']:.4f} with interval {interval_text(metric)}. "
              f"[{label} CPU receipts](sol2/job21/{label}/summary.txt#L113-L129)")

    print("\nEach earlier row uses the archived 16-block engine CPU ratio for the fixed W replay. "
          "The Job 17 ratio sums CPU across three concurrent workers. Path lengths and worker CPUs identify "
          "the launch settings. The gap columns compare absolute pooled changes from parity as descriptive references. "
          "Each interval belongs to its own comparison.")
    gap_headers = "".join(f" | Size vs {label} gap" for label in gaps)
    gap_alignment = " |---:" * len(gaps)
    print(f"\n| Earlier comparison | A → B | Path chars A/B | Worker CPUs | Pooled CPU B/A | Pooled change | "
          f"CPU mean [95% t interval] | Interval side{gap_headers} | Sources |")
    print(f"|---|---|---:|---|---:|---:|---:|---{gap_alignment}|---|")
    for job, steps in history.items():
        for step in steps:
            relative = f"job{job}/{step}"
            directory = anchor / relative
            summary = load(directory / "summary.txt")
            manifest = load(directory / "inputs.txt")
            check(summary, {"complete": True, "completed_blocks": BLOCKS,
                            "expected_games_searches_nodes_trace": W_SIGNATURE,
                            "arm_labels": manifest["arm_labels"], "worker_affinity": manifest["worker_affinity"]}, relative)
            metric = summary["metrics"]["engine_cpu_s"]
            check(metric["ci_df"], BLOCKS - 1, f"{relative} CPU interval degrees of freedom")
            side = {"below": "Below parity", "above": "Above parity", "includes": "Includes parity"}[
                interval_direction(metric["block_mean_ci95"])]
            a, b = (manifest["arm_labels"][arm] for arm in "AB")
            paths = "/".join(str(len(manifest["inputs"][arm]["path"])) for arm in "AB")
            cpus = ", ".join(str(cpu) for cpu in manifest["worker_affinity"])
            pooled = metric["pooled_b_over_a"]
            delta = pooled - 1
            size_cells = [f"{abs(delta) / gap:.2f}×" if gap else "—" for gap in gaps.values()]
            sizes = "".join(f" | {cell}" for cell in size_cells)
            print(f"| Job {job} {step} | {a} → {b} | {paths} | {cpus} | {pooled:.4f} | {100 * delta:+.2f}% | "
                  f"{metric['block_mean']:.4f} {interval_text(metric)} | {side}{sizes} | "
                  f"[CPU receipt](sol2/{relative}/summary.txt#L113-L129) · [Input pins](sol2/{relative}/inputs.txt) |")


def show_readings(campaigns, pairs, variant, prior_anchor):
    print("\n## Declared readings and arithmetic")
    print("\nPool and mean report B/A. Percentages are time differences for B relative to A, shown as "
          "pooled / mean [95% t interval]. Negative values favor B.")
    print("\n| Comparison | A | B | Metric | Pooled / mean [95% t interval] | Time difference | Reading |")
    print("|---|---|---|---|---:|---:|---|")
    cpu = {}
    for step, (a, b) in pairs.items():
        for key, label in METRICS:
            result = completed_metric(campaigns, step, key)
            if key == "engine_cpu_s":
                cpu[step] = result
            state = ("Incomplete" if step in campaigns else "Not run") if result is None else (
                "DESCRIPTIVE" if step.startswith("R") else reading(result))
            link = summary_link(step) if result is not None else step
            print(f"| {link} | {a} | {b} | {label} | {metric_cell(result)} | {percent_cell(result)} | {state} |")
    print("\nA complete interval wholly below 1 gives PAYS. An interval wholly above 1 gives COSTS. "
          "An interval containing 1 gives NOT SEEN.")
    outcome = small_stack_reading(cpu["N1"], cpu["N2"])
    if cpu["N1"] is None or cpu["N2"] is None:
        print(f"\nSmall stack versus kF **{outcome}**. A complete N1 and N2 are needed for the joint CPU reading.")
    elif outcome == "KEEPS":
        print("\nSmall stack versus kF **KEEPS**. Both CPU intervals read PAYS or NOT SEEN under the declared rule. "
              "This leaves the size of any unresolved difference open.")
    elif outcome == "OTHER PIECES PAY":
        print("\nSmall stack versus kF **OTHER PIECES PAY**. Both CPU intervals read COSTS.")
    else:
        print("\nSmall stack versus kF **OPEN**. The two CPU readings lie on different sides of the joint rule.")
    if variant == "fallback":
        print("\nN1 and N2 compare the same fallback N build against the two kF builds. "
              "R1 and R2 also use that same N executable. [Variant receipt](sol2/job21/variant.txt)")
    placebos = {step: cpu[step] for step in ("P1", "P2") if reading(cpu[step]) in ("PAYS", "COSTS")}
    if placebos:
        show_prior_placebo(prior_anchor, placebos)
    elif all(cpu[step] is not None for step in ("P1", "P2")):
        print("\nBoth same-source CPU intervals contain parity. The measurements leave room for build-to-build differences "
              "within the intervals shown above.")


def read_campaigns(root, known, pins, environment, pairs, skips, through=None):
    campaigns = {}
    for step, pair in pairs.items():
        directory = root / step
        if (directory / "inputs.txt").is_file():
            require(step not in skips, f"unexpected timed comparison {step}: {skips.get(step)}")
            campaigns[step] = audit(directory, known, pins, environment, pair)
        else:
            require(not (directory / "runs.jsonl.txt").is_file(), f"raw runs lack a manifest in {directory}")
        if step == through:
            break
    return campaigns


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("job21", type=Path, help="Job 21 raw receipts directory")
    parser.add_argument("job20", type=Path, help="Job 20 receipts directory, beside the earlier jobs")
    parser.add_argument("--job16", type=Path, help="Job 16 frozen launch arms, default beside Job 20")
    parser.add_argument("--job13", type=Path, help="Job 13 W receipts, default beside Job 20")
    parser.add_argument("--through", choices=STEPS, default="R2", help="last Job 21 campaign to read")
    args = parser.parse_args()
    anchor = args.job20.parent
    root = args.job21
    known = known_workload(args.job13 or anchor / "job13")
    check(known.signature, W_SIGNATURE, "declared Job 21 W signature")
    timed = any((root / step / "inputs.txt").is_file() for step in STEPS)
    if not (root / "clang-executables.raw.txt").is_file():
        print("Step 0 frozen executable receipts are unavailable.")
        require(not timed, "timing without Step 0 pins")
        return
    job16 = load((args.job16 or anchor / "job16") / "arms.txt")
    prior_environment = read_environment(job16["engine_environment"])
    frozen = {label: job16["arms"][label] for label in REFERENCE_LABELS}
    for label, arm in zip(("r15d", "K_1"), "AB", strict=True):
        check(frozen[label]["sha256"], known.clang_inputs[arm]["sha256"], f"Job 13 original frozen digest {label}")
    references = load(root / "reference-arms.txt")
    check(set(references["arms"]), set(frozen), "Job 21 Step 0 reference arm inventory")
    check(references["arms"], frozen, "Job 21 Step 0 reference launch receipts")
    check(read_environment(references["engine_environment"]), prior_environment, "Job 21 reference launch environment")
    gate_ok = verify_gate(root, known, frozen, prior_environment,
                          {label: label for label in REFERENCE_LABELS}, render=True)
    if not gate_ok or not (root / "arms.txt").is_file():
        require(not timed, "timing manifests lack valid Step 0 and Step 1 pins")
        return

    variant = load(root / "variant.txt")["variant"]
    require(variant in STACK_ARMS, f"invalid small-stack variant {variant!r}")
    new_builds = ("r15d_2", *dict.fromkeys(STACK_ARMS[variant]))
    builds = read_builds(root, frozen, known, new_builds)
    pins, environment = read_arms(root, builds, frozen, prior_environment)
    pairs = variant_pairs(variant)
    skips = comparison_plan(root, pins, pairs)
    show_arms(pins, environment, (*REFERENCE_LABELS, *new_builds))
    campaigns = read_campaigns(root, known, pins, environment, pairs, skips, args.through)
    show_overview(campaigns, pairs, skips, args.through)
    show_readings(campaigns, pairs, variant, anchor)
    for step, data in campaigns.items():
        show_campaign(step, data)


if __name__ == "__main__":
    main()
