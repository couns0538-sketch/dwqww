#!/usr/bin/env python3
"""Apply the predeclared Job 21 build gates."""
import argparse
import json
from pathlib import Path


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('evidence', type=Path)
    args = parser.parse_args()
    raw = args.evidence
    arms = json.loads((raw / 'arms.txt').read_text())['arms']
    variant = json.loads((raw / 'variant.txt').read_text())['variant']
    n1, n2 = ('N', 'N') if variant == 'fallback' else ('N_1', 'N_2')
    comparisons = {'P1': ('r15d', 'r15d_2'), 'P2': ('K_1', 'K_2'),
                   'N1': ('K_1', n1), 'N2': ('K_2', n2),
                   'R1': ('r15d', n1), 'R2': ('r15d_2', n2)}
    skips = {}
    for step, pair in comparisons.items():
        missing = [label for label in pair if label not in arms]
        if missing:
            skips[step] = f"Build gate unavailable or failed for {', '.join(missing)}."
    print(json.dumps({'comparisons': comparisons, 'skips': skips}, indent=2))


if __name__ == '__main__':
    main()
