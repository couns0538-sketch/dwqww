#define _GNU_SOURCE
#include <linux/perf_event.h>
#include <sys/syscall.h>
#include <sys/ptrace.h>
#include <sys/time.h>
#include <sys/types.h>
#include <sys/wait.h>
#include <stdio.h>
#include <signal.h>
#include <time.h>
#include <unistd.h>
#include <errno.h>
#include <string.h>
int main(void) {
    struct perf_event_attr event = {.type=PERF_TYPE_SOFTWARE, .size=sizeof(event), .config=PERF_COUNT_SW_TASK_CLOCK, .exclude_kernel=1, .exclude_hv=1};
    long fd = syscall(SYS_perf_event_open, &event, 0, -1, -1, 0);
    printf("perf_event_open user task-clock = %ld (%s)\n",fd,fd<0?strerror(errno):"ok");
    timer_t id;
    struct sigevent ev = {.sigev_notify=SIGEV_THREAD_ID, .sigev_signo=SIGUSR2};
    ev._sigev_un._tid = gettid();
    int status = timer_create(CLOCK_THREAD_CPUTIME_ID, &ev, &id);
    printf("per-thread POSIX CPU timer = %d (%s)\n",status,status<0?strerror(errno):"ok");
    status = timer_create(CLOCK_PROCESS_CPUTIME_ID, &ev, &id);
    printf("process POSIX CPU timer = %d (%s)\n",status,status<0?strerror(errno):"ok");
    struct itimerval interval = {};
    status = setitimer(ITIMER_PROF, &interval, NULL);
    printf("setitimer ITIMER_PROF = %d (%s)\n",status,status<0?strerror(errno):"ok");
    pid_t p=fork();
    if (!p) { raise(SIGSTOP); _exit(0); }
    waitpid(p,NULL,WUNTRACED);
    status=ptrace(PTRACE_SEIZE,p,0,0);
    printf("PTRACE_SEIZE task-owned child = %d (%s)\n",status,status<0?strerror(errno):"ok");
    kill(p,SIGKILL); waitpid(p,NULL,0);
}
