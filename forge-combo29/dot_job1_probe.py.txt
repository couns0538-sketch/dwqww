import subprocess, json
EXE='/tmp/forge-dot-base-build/forge'
def run(label,cmd,opts=()):
 p=subprocess.Popen([EXE],stdin=subprocess.PIPE,stdout=subprocess.PIPE,text=True,bufsize=1)
 def send(s): p.stdin.write(s+'\n');p.stdin.flush()
 send('uci')
 while p.stdout.readline().strip()!='uciok': pass
 for s in ('setoption name Threads value 1','setoption name Hash value 16','setoption name OwnBook value false',*opts): send(s)
 send('isready')
 while p.stdout.readline().strip()!='readyok': pass
 send('position startpos');send(cmd)
 rows=[]
 while True:
  s=p.stdout.readline().rstrip(); rows.append(s)
  if s.startswith('bestmove'):break
 send('quit');p.wait(timeout=10)
 print(json.dumps(dict(label=label,command=cmd,options=opts,output=rows)))
run('hard','go nodes 1000')
run('soft','go softnodes 1000')
run('soft_then_hard','go softnodes 1 nodes 1000')
run('hard_then_soft','go nodes 1 softnodes 1000')
run('mixed_depth_clock','go depth 8 wtime 1000000 btime 1000000',('setoption name TmNodeShare value true',))
run('depth_only','go depth 8',('setoption name TmNodeShare value true',))
for val in ['1000','2147483647','2147483648']:
 s='uci\nsetoption name TmSoftDivisor value '+val+'\nuci\nquit\n'
 out=subprocess.run([EXE],input=s,text=True,capture_output=True,timeout=5).stdout
 print(json.dumps(dict(label='spin_'+val,output=[x for x in out.splitlines() if 'TmSoftDivisor' in x])))
