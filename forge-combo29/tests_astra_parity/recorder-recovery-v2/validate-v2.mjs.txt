import fs from 'node:fs';import cp from 'node:child_process';import crypto from 'node:crypto';import assert from 'node:assert/strict';
const here='/workspace/work/parity-equality-preparation-v2',control=here+'/control.mjs',sha=f=>crypto.createHash('sha256').update(fs.readFileSync(f)).digest('hex'),commands=[];
function run(name,argv){const r=cp.spawnSync(argv[0],argv.slice(1),{encoding:'utf8',timeout:100000,maxBuffer:32*1024*1024});fs.writeFileSync(here+'/'+name+'.stdout.txt',r.stdout??'');fs.writeFileSync(here+'/'+name+'.stderr.txt',r.stderr??'');const record={argv,exit:r.status,signal:r.signal,error:r.error?.message??null,stdoutSha256:sha(here+'/'+name+'.stdout.txt'),stderrSha256:sha(here+'/'+name+'.stderr.txt')};fs.writeFileSync(here+'/'+name+'.command.txt',JSON.stringify(record,null,2)+'\n');commands.push(record);return r;}
const migration=run('preview-migration',['node',control,'preview-migration','speed','/workspace/work/parity-run/equality-speed/stage.txt']);assert.equal(migration.status,0,migration.stderr);fs.writeFileSync(here+'/preview-stage.txt',migration.stdout);const m=JSON.parse(migration.stdout);assert.equal(m.migration.nativeExecutions,0);assert.ok(Object.hasOwn(m.toolPins,control));for(const [file,pin]of Object.entries(m.migration.pins))assert.equal(sha(file),pin);
const batch=run('check-real-S1',['node',control,'check-batch','speed','S1-0001']);assert.equal(batch.status,3,batch.stderr);const actual=JSON.parse(batch.stdout),independent=JSON.parse(fs.readFileSync(here+'/S1-0001-independent-audit.stdout.txt'));assert.deepEqual(actual,independent);assert.equal(actual.counts.positions,16719);assert.equal(actual.totalDifferences,0);assert.equal(actual.partialOperationalFaults,1);assert.equal(actual.partialOperationalFindings[0].worker,1);assert.equal(actual.partialOperationalFindings[0].localIndex,4407);
const wrong=run('reject-legacy-E',['node',control,'check-batch','speed','E-0001']);assert.notEqual(wrong.status,0);assert.match(wrong.stderr,/S1-0001/);const code=fs.readFileSync(control,'utf8');assert.ok(!code.includes("'E-'"));assert.ok(!code.includes('/^E-'));assert.ok(!code.includes("dir+'/configs/'"));assert.ok(!code.includes("dir+'/batches/'"));assert.ok(code.includes("prefix=stageName==='speed'?'S1':'S2'"));assert.ok(code.includes("step=stageName==='speed'?'step1':'step2'"));assert.ok(code.includes("base.endsWith('equality-speed-v2')?'S1':'S2'"));
for(const [file,pin]of Object.entries(m.migration.pins))assert.equal(sha(file),pin);assert.ok(!fs.existsSync('/workspace/work/parity-run/equality-speed-v2'));const report={passed:true,nativeExecutions:0,realStageWrites:0,legacyEvidenceUnchanged:true,actualS1PlanAndPinsValidated:true,actualS1AggregateMatchesIndependent:true,legacyEIdRejected:true,S1S2PathsExplicit:true,positions:actual.counts.positions,totalDifferences:actual.totalDifferences,partialOperationalFaults:actual.partialOperationalFaults,speedGate:'BLOCKED',commands,correctedControlSha256:sha(control)};fs.writeFileSync(here+'/VALIDATION.txt',JSON.stringify(report,null,2)+'\n');console.log(JSON.stringify(report));
