// Read-only audit. Final results are frozen before quit; no timeout becomes a completed search.
import fs from 'node:fs';import crypto from 'node:crypto';import assert from 'node:assert/strict';import {finishedDepths} from './audit-core.mjs';
const prep='/workspace/work/parity-release-timeout-0018-preparation',root='/workspace/work/parity-run/release-response-diagnostic-0018',id=process.argv[2],hash=f=>crypto.createHash('sha256').update(fs.readFileSync(f)).digest('hex'),read=f=>JSON.parse(fs.readFileSync(f));
const d=read(prep+'/DECLARATION.txt'),row=d.rows.find(x=>x.id===id);assert.ok(row);const c=d.cases[row.caseIndex],dir=root+'/'+id,meta=read(dir+'/PROCESS.txt'),start=read(dir+'/START.txt'),saved=read(dir+'/DECLARATION.txt');assert.equal(saved.sha256,hash(prep+'/DECLARATION.txt'));assert.deepEqual(saved.row,row);assert.deepEqual(saved.case,c);assert.deepEqual(saved.engine,d.engine);assert.deepEqual(saved.native,d.native);assert.deepEqual(start.sourcePins,d.sourcePins);assert.deepEqual(start.nativeWorkBefore,[]);assert.deepEqual(meta.row,row);assert.deepEqual(meta.argv,start.argv);assert.deepEqual(meta.argv,['timeout','-k','2','75',d.native.binary,d.engine.binary,c.position,c.fen,String(c.enabled),String(row.threads),String(row.depth)]);assert.equal(meta.started,start.started);assert.ok(Date.parse(meta.ended)>=Date.parse(meta.started));assert.equal(hash(meta.stdout),meta.stdoutSha256);assert.equal(hash(meta.stderr),meta.stderrSha256);
const stdout=fs.readFileSync(meta.stdout,'utf8'),stderr=fs.readFileSync(meta.stderr,'utf8'),lines=stdout.split('\n'),parse=k=>lines.filter(x=>x.startsWith(k+' ')).map(x=>JSON.parse(x.slice(k.length+1))),configs=parse('CONFIG'),events=parse('EVENT'),results=parse('RESULT'),diagnostics=[];
for(const [stream,text]of [['stdout',stdout],['stderr',stderr]])for(const [i,line]of text.split('\n').entries())if(/(?:Address|UndefinedBehavior|Leak|Thread|Memory|HWAddress)Sanitizer|runtime error:|Sanitizer CHECK failed|terminate called|Segmentation fault|core dumped/i.test(line))diagnostics.push({stream,line:i+1,text:line});
const errors=[];if(configs.length!==1)errors.push('Expected one CONFIG');else for(const [k,v]of Object.entries({engine:d.engine.binary,position:c.original.context.position,fen:c.original.context.fen,Threads:row.threads,Hash:16,OwnBook:false,UseNNUE:true,MultiPV:1,HindsightDepth:c.enabled,MateAlphaGuard:c.enabled,go:'go depth '+row.depth,target_depth:row.depth,response_allowance_ms:60000,shutdown_allowance_ms:250,fresh_process:true,warm_tt_reproduced:false,soak_credit:0}))assert.equal(configs[0][k],v,k);
const result=results.length===1?results[0]:null;let depths=[];if(!result)errors.push('Missing or multiple RESULT; no completion inferred');else{
 assert.ok(lines.filter(x=>x.trim()).at(-1).startsWith('RESULT '));assert.equal(result.soak_credit,0);assert.equal(result.warm_tt_reproduced,false);assert.equal(result.target_depth,row.depth);assert.equal(result.threads,row.threads);if(result.response_timed_out)assert.equal(result.target_depth_completed_before_quit,0);
 if(result.target_depth_completed_before_quit){assert.equal(result.bestmove_received_before_quit,1);assert.equal(result.depth,row.depth);assert.equal(result.legal_returned_move,1);assert.equal(result.score_in_range,1);assert.equal(result.parser_error,'');}
 if(result.diagnostic_success){assert.equal(meta.exit,0);assert.equal(meta.signal,null);assert.equal(meta.error,null);assert.equal(result.target_depth_completed_before_quit,1);assert.equal(result.normal_exit,1);assert.equal(result.sanitizer_report_observed,0);assert.equal(result.error,'');assert.equal(diagnostics.length,0);}
 const snapshot=events.findIndex(x=>x.event==='pre_quit_snapshot_frozen'),quit=events.findIndex(x=>x.event==='quit_begin');if(quit>=0)assert.ok(snapshot>=0&&snapshot<quit);depths=finishedDepths(stdout);if(result.target_depth_completed_before_quit){assert.equal(depths.at(-1)?.depth,row.depth);assert.equal(depths.at(-1)?.nodes,result.nodes);}
}
console.log(JSON.stringify({id,row,diagnosticOnly:true,soakCredit:0,warmTtReproduced:false,freshPerSearch:true,process:meta,result,finishedDepths:depths,diagnostics,errors,events,evidenceFinalized:!!result&&errors.length===0,searchCompletedBeforeQuit:!!result?.target_depth_completed_before_quit,inputSha256:{[prep+'/DECLARATION.txt']:hash(prep+'/DECLARATION.txt'),[dir+'/PROCESS.txt']:hash(dir+'/PROCESS.txt'),[meta.stdout]:meta.stdoutSha256,[meta.stderr]:meta.stderrSha256}},null,2));process.exitCode=errors.length?1:0;
