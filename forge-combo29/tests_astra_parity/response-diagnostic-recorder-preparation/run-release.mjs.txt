// Root opt-in only. At most four sequential diagnostic searches, never global-state/Git mutation.
import fs from 'node:fs';import cp from 'node:child_process';import crypto from 'node:crypto';import assert from 'node:assert/strict';import {classify} from './classify.mjs';
const run='/workspace/work/parity-run',prep='/workspace/work/parity-release-timeout-0018-preparation',root=run+'/release-response-diagnostic-0018';
const [first,countArg,authorization]=process.argv.slice(2),count=Number(countArg),hash=f=>crypto.createHash('sha256').update(fs.readFileSync(f)).digest('hex'),read=f=>JSON.parse(fs.readFileSync(f));assert.equal(authorization,'--execute-authorized');assert.ok(Number.isInteger(count)&&count>=1&&count<=4);for(const [f,h]of Object.entries(read('/workspace/work/parity-diagnostic-recording-0018-preparation/FROZEN-PINS.txt').files))assert.equal(hash(f),h,f);const d=read(prep+'/DECLARATION.txt'),startIndex=d.rows.findIndex(x=>x.id===first);assert.ok(startIndex>=0&&startIndex+count<=d.rows.length);const rows=d.rows.slice(startIndex,startIndex+count),started=Date.now(),deadline=started+425000;
for(const row of rows){assert.ok(!fs.existsSync(root+'/'+row.id),'Diagnostic already exists');for(const prefix of ['release-diag-','release-diag-audit-'])assert.ok(!fs.existsSync(run+'/logs/'+prefix+row.id.toLowerCase().replace('-','')+'.metadata.txt'),'Logger label already exists');}
function invoke(label,args,timeout){const argv=['node',run+'/command.mjs',label,...args],r=cp.spawnSync(argv[0],argv.slice(1),{cwd:run,encoding:'utf8',timeout,maxBuffer:8*1024*1024}),file=run+'/logs/'+label+'.metadata.txt';assert.ok(fs.existsSync(file),r.stderr||r.stdout);const m=read(file);assert.equal(r.signal,null);assert.equal(m.signal,null);assert.equal(r.status,m.exit);assert.equal(hash(m.stdoutFile),m.stdoutSha256);assert.equal(hash(m.stderrFile),m.stderrSha256);console.log(JSON.stringify({label,exit:m.exit,stdoutFile:m.stdoutFile,stdoutSha256:m.stdoutSha256,stderrFile:m.stderrFile}));return m;}
const outcomes=[];for(const row of rows){if(Date.now()+105000>deadline){console.log(JSON.stringify({status:'bounded runner budget; remaining rows not started',pending:rows.slice(outcomes.length).map(x=>x.id)}));break;}const suffix=row.id.toLowerCase().replace('-','');const control=invoke('release-diag-'+suffix,['node',prep+'/control.mjs',row.id,'--diagnostic-authorized'],90000);assert.ok([0,1].includes(control.exit),'Unexpected diagnostic controller exit; stop');const audit=invoke('release-diag-audit-'+suffix,['node',prep+'/audit.mjs',row.id],12000);assert.equal(audit.exit,0,'Malformed/incomplete audit; stop');assert.equal(fs.readFileSync(audit.stderrFile,'utf8'),'');const result=read(audit.stdoutFile),classification=classify(result);outcomes.push({id:row.id,classification,auditFile:audit.stdoutFile,auditSha256:audit.stdoutSha256});console.log(JSON.stringify(outcomes.at(-1)));if(!classification.continueAllowed){console.log(JSON.stringify({status:'STOP',outcomes,rootStateWrites:0}));process.exitCode=1;break;}}
console.log(JSON.stringify({utc:new Date().toISOString(),elapsedSeconds:(Date.now()-started)/1000,requested:rows.map(x=>x.id),outcomes,soakCredit:0,rootStateWrites:0}));
