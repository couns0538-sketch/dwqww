import assert from 'node:assert/strict';import {classify,summarizeAudits} from './classify.mjs';
const base={soakCredit:0,warmTtReproduced:false,evidenceFinalized:true,errors:[],diagnostics:[],searchCompletedBeforeQuit:true,process:{exit:0,signal:null,error:null},result:{sanitizer_report_observed:0,normal_exit:1,bestmove_received_before_quit:1,legal_returned_move:1,diagnostic_success:1,error:'',parser_error:'',response_timed_out:0,nodes:100,response_ms:2,shutdown:{reaped:1,exit:0,signal:0,timed_out:0,driver_sigterm:0,driver_sigkill:0,error:''}}};let count=0;const test=(n,f)=>{f();count++;console.log('PASS '+n);};const clone=()=>structuredClone(base);

test('timeout with shutdown fault blocks',()=>{const a=timeout();a.result.normal_exit=0;assert.equal(classify(a).continueAllowed,false);});
test('timeout with another parser error blocks',()=>{const a=timeout();a.result.parser_error='softnodes_line_on_usual_search';assert.equal(classify(a).continueAllowed,false);});
test('timeout with observed invalid score blocks',()=>{const a=timeout();a.result.last_exact='info depth 8 score cp 999999 nodes 100 time 2';a.result.score_in_range=0;assert.equal(classify(a).continueAllowed,false);});
