import assert from 'node:assert/strict';
import {merge} from '/workspace/work/parity-soak-time-review/review.mjs';
export const CUTOFF='2026-11-15T04:20:00.000Z',JOB_B_DEADLINE='2026-11-15T05:06:50.313Z';
export function guardClock(now){assert.ok(Number.isFinite(now)&&now>=Date.parse(CUTOFF),'Job A closure is forbidden before real UTC04:20');}
export function verifyTimingSnapshot(snapshot,fresh,statePath,stateSha){assert.deepEqual(fresh.final,snapshot.timing.final,'Timing changed after snapshot');assert.deepEqual(fresh.preliminaryStep1,snapshot.timing.preliminaryStep1);assert.deepEqual(fresh.issues,snapshot.timing.issues);assert.equal(fresh.inputSha256[statePath],stateSha);}
const sum=(rows,key)=>rows.reduce((n,r)=>n+(r[key]??0),0);
export function summarize(snapshot,state,closedUTC){
 guardClock(Date.parse(closedUTC));assert.equal(snapshot.asOf,CUTOFF);assert.equal(state.deadline,CUTOFF);assert.equal(state.jobBDeclaration.deadlineUTC,JOB_B_DEADLINE);
 const intervals=snapshot.timing.final.protocols.flatMap(p=>p.rows),qualified=[],excluded=[];
 for(const p of snapshot.protocols)for(const r of p.rows.filter(r=>r.passed)){const timing=intervals.filter(x=>x.label===r.label);assert.equal(timing.length,1,'Every recorded pass must have one timing record');const t=timing[0];if(t.intervalValid&&t.status==='accepted'&&Date.parse(t.ended)<=Date.parse(CUTOFF))qualified.push({...r,namespace:p.namespace,started:t.started,ended:t.ended});else excluded.push({...r,namespace:p.namespace,reason:'Missing, invalid, unverified or after-cutoff command interval; closure gives zero credit'});}
 const required=snapshot.coverage.conditions.map(x=>x.condition),conditions=required.map(condition=>({condition,covered:qualified.some(r=>r.group===condition),evidence:qualified.filter(r=>r.group===condition).map(r=>({namespace:r.namespace,index:r.index,label:r.label,evidenceIdentity:r.evidenceIdentity}))})),cleanIntervals=merge(qualified.map(r=>({start:Date.parse(r.started),end:Date.parse(r.ended)}))),cleanMilliseconds=cleanIntervals.reduce((n,r)=>n+r.end-r.start,0);
 const step1=state.validation?.step1??{},step2=state.validation?.step2??{},eq1=state.equalityTotals?.step1??{},eq2=step2.equality??{},pins=s=>s?.bench11===312866&&s?.bench13===862329&&s?.perft5===4865609;
 const sourceChecks={step1DefaultPins:pins(step1.pins),step2DefaultPins:pins(step2.pins),step1Options:step1.options?.passed===16&&step1.options?.failed===0,step2Options:step2.options?.passed===16&&step2.options?.failed===0,nineUnitRows:step2.unit?.cases===9&&step2.unit?.failures===0,targetExact:step2.targetPosition?.score===-5&&step2.targetPosition?.nodes===2566746&&step2.targetPosition?.strictCheck?.passed===true,step1Corpus:{completed:eq1.completed??0,rawGatePassed:!!eq1.gatePassed,differences:eq1.differences??null},step2Corpus:{completed:eq2.completed??0,rawGatePassed:!!eq2.gatePassed,effectiveGatePassed:!!state.step2Acceptance?.effectiveGatePassed,rawDifferences:eq2.differences??null,rawAuditUnchanged:!!state.step2Acceptance?.rawStrictAuditUnchanged,acceptanceFile:state.step2Acceptance?.file??null,acceptanceSha256:state.step2Acceptance?.sha256??null}};
 const diagnostics=state.responseDiagnosticResults0018??null,diagnosticsComplete=diagnostics?.completed===42&&diagnostics?.releaseCompleted===40&&diagnostics?.tsanCompleted===2&&diagnostics?.soakCredit===0,shortfalls=[];
 const missing=conditions.filter(x=>!x.covered).map(x=>x.condition);if(missing.length)shortfalls.push('Missing clean latest0018 conditions: '+missing.join(', '));
 if(cleanMilliseconds<7200000)shortfalls.push(`Accepted-clean finalized batch union is ${(cleanMilliseconds/1000).toFixed(3)} seconds, below the requested7200seconds. Elapsed waiting/build/corpus/match/diagnostic time is not sanitizer credit.`);
 if(snapshot.timing.final.unionBatchMilliseconds<7200000)shortfalls.push(`All finalized sanitizer attempts together cover ${(snapshot.timing.final.unionBatchMilliseconds/1000).toFixed(3)} seconds; this is also below7200seconds.`);
 if(excluded.length)shortfalls.push(`${excluded.length} otherwise recorded passing batches excluded from this closure because their deadline-qualified timing is unavailable.`);
 if(snapshot.timing.issues.length)shortfalls.push(`${snapshot.timing.issues.length} interval metadata issues are retained without guessed duration.`);
 if(!diagnosticsComplete)shortfalls.push(`0018 diagnostics incomplete: ${diagnostics?.tsanCompleted??0}/2 TSan and ${diagnostics?.releaseCompleted??0}/40 release searches recorded.`);
 if(Object.values(sourceChecks).some(v=>v===false)||sourceChecks.step1Corpus.completed!==100000||!sourceChecks.step1Corpus.rawGatePassed||sourceChecks.step2Corpus.completed!==100000||!sourceChecks.step2Corpus.effectiveGatePassed)shortfalls.push('One or more source/local/corpus gates are incomplete; see exact sourceChecks fields.');
 if(state.blockFurtherEngineWork)shortfalls.push('An effective global engine-work stop remains set; closure does not clear it.');
 return {schema:1,status:'closed-at-job-a-budget-deadline',deadline:CUTOFF,closedUTC,sourceCommit:state.sameClockCommit,startedUTC:state.start,budgetElapsedSeconds:(Date.parse(CUTOFF)-Date.parse(state.start))/1000,acceptedGames:sum(qualified,'creditedGames'),acceptedSearches:sum(qualified,'creditedSearches'),acceptedBatches:qualified.length,coverage:{covered:conditions.filter(x=>x.covered).length,required:required.length,conditions},protocols:snapshot.protocols.map(p=>({namespace:p.namespace,acceptedGames:sum(qualified.filter(r=>r.namespace===p.namespace),'creditedGames'),acceptedSearches:sum(qualified.filter(r=>r.namespace===p.namespace),'creditedSearches'),acceptedBatches:qualified.filter(r=>r.namespace===p.namespace).length,retainedFailedBatches:p.failedBatches})),retainedFailedBatches:snapshot.retainedFailedBatches,excludedPasses:excluded,timing:{cutoff:CUTOFF,observedFinalizedBatchUnionMilliseconds:snapshot.timing.final.unionBatchMilliseconds,acceptedCleanBatchUnionMilliseconds:cleanMilliseconds,sumBatchDurationsMilliseconds:snapshot.timing.final.sumBatchMilliseconds,elapsedSoakWindowMilliseconds:snapshot.timing.final.elapsedWindowMilliseconds,unaccountedMilliseconds:snapshot.timing.final.unaccountedMilliseconds,issues:snapshot.timing.issues,measurement:'Finalized command intervals wholly within Job A cutoff. Includes startup/protocol/shutdown overhead; not CPU time or continuous testing. Missing/straddling/later intervals are excluded without guessed partial credit.'},sourceChecks,diagnostics:{completed:diagnostics?.completed??0,releaseCompleted:diagnostics?.releaseCompleted??0,tsanCompleted:diagnostics?.tsanCompleted??0,responseTimeouts:diagnostics?.responseTimeouts??null,blocked:diagnostics?.blocked??null,soakCredit:0,report:diagnostics?.report??null,reportSha256:diagnostics?.reportSha256??null,interpretation:diagnostics?.interpretation??'No completed diagnostic interpretation recorded.',originalWarmStateReproduced:false},finding0019:state.smpEndingWorkloadFinding??null,shortfalls,limitations:['The actual owner-PC compiler, ThinLTO/PGO profile, runtime tuning, active network and running PC were not independently tested. All work ran in the cloud.','Warm TT/history reuse does not establish incremental NNUE equals a full recomputation at every node; that per-node comparison was not performed in Job A.','Original failed audits and observations remain unchanged and excluded from clean credit. Resolved acceptance-rule findings are not engine fixes.','Optional split-ASan KRK coverage is not credited by this closure; any separately completed optional evidence must be reported separately.'],jobB:{deadline:JOB_B_DEADLINE,startedUTC:state.jobBDeclaration.startedUTC,declarationSha256:state.jobBDeclaration.sha256,independent:true,clockReset:false,action:'Continue according to its existing declaration and remaining budget; this Job A closure neither closes nor restarts Job B.'},jobCStartedByClosure:false,engineRunsByClosure:0,stateWritesByClosure:0};
}
export function patchProposal(closure,closureSha256,stateSha256,report){return {schema:1,proposedOnly:true,expectedStateSha256AtCapture:stateSha256,requiresJobAClosureAbsent:true,operations:[{op:'add',path:'/jobAClosure',value:{status:closure.status,deadline:closure.deadline,closedUTC:closure.closedUTC,sourceCommit:closure.sourceCommit,report,closureSha256,acceptedGames:closure.acceptedGames,acceptedSearches:closure.acceptedSearches,coverage:{covered:closure.coverage.covered,required:closure.coverage.required},timing:closure.timing,shortfalls:closure.shortfalls,jobBDeadlineUnchanged:JOB_B_DEADLINE,jobBRemainsSeparate:true}}],applyRule:'Root applies only this new field to its current state after checking the capture precondition. If Job B updated state since capture, verify the retained Job A inputs still match, then add this field to the latest state; never replace the full current state with the saved snapshot.'};}
