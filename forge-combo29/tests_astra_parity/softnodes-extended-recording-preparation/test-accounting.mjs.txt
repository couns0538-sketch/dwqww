 const b={...batch,index,label:'synthetic-'+index};

 const soft={...empty(),games:1,searches:xs.length,moves:xs.length,mates:1,soft_x_count:xs.length,soft_x_sum:xs.reduce((a,x)=>a+x,0),soft_mean_x:xs.reduce((a,x)=>a+x,0)/xs.length,soft_hard_cap_reached:xs.filter(x=>x>=10000).length};
 const triggers=[{pair:0,mode:'usual',ply:8,fen:'synthetic',seed:123,threads:1},...xs.map((_,i)=>({pair:0,mode:'soft',ply:8+i,fen:'synthetic',seed:123,threads:1}))];
 const checks=triggers.map(t=>({...t,x:t.mode==='soft'?xs[t.ply-8]:null,item2:1,item3:1,item4:1}));
 const games=[{pair:0,mode:'usual',completed:true},{pair:0,mode:'soft',completed:true}],pairs=[{pair:0,completed:true}];
 const shutdowns=interrupted?[]:[{reaped:1,exit:0,signal:0,timed_out:0,driver_sigterm:0,driver_sigkill:0,quit_sent:1,quit_to_exit_ms:ms}];
 const faults=[];let observedSoft={...soft};
 if(fault||label){const t={pair:1,mode:'soft',ply:8,fen:'failure',seed:123,threads:1};triggers.push(t);checks.push({...t,x:3000,item2:1,item3:label?0:1,item4:1});observedSoft={...soft,searches:soft.searches+1,soft_x_count:soft.soft_x_count+1,soft_x_sum:soft.soft_x_sum+3000,soft_mean_x:(soft.soft_x_sum+3000)/(soft.soft_x_count+1)};if(!interrupted)faults.push({reason:label?'softnodes_item3_label_move':'illegal_move',label_fault:label,context:t});}
 const summary={pairs_completed:1,usual,soft,observed_usual_including_partial_pair:{...usual},observed_soft_including_partial_pair:observedSoft,fault:fault||label?'synthetic_fault':'',label_fault:label,diagnostic_observed:false,unassigned_soft_lines:0};
 const raw=row('CONFIG',{first_pair:0})+triggers.map(t=>row('TRIGGER',t)).join('')+checks.map(c=>row('CHECK',c)).join('')+games.map(g=>row('GAME',g)).join('')+pairs.map(p=>row('PAIR',p)).join('')+shutdowns.map(s=>row('SHUTDOWN',s)).join('')+faults.map(f=>row('FAULT',f)).join('')+(interrupted?'':row('SUMMARY',summary))+(truncated?'CHECK {"pair":2':'');
 const audit={index,label:b.label,exit:interrupted?124:fault||label?1:0,signal:null,complete:!interrupted,passed:!interrupted&&!fault&&!label,coverage:!interrupted,diagnostics:[],faults,summary:interrupted?null:summary,shutdowns,triggerRows:triggers.length,checkRows:checks.length,...(truncated?{auditUnavailable:true}:{})};
 return {b,audit,raw};
}
