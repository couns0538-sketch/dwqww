import assert from 'node:assert/strict';
export const numeric=['games','searches','moves','mates','stalemates','repetitions','fifty','insufficient','ply_caps','opening_terminals','soft_x_count','soft_x_sum','soft_hard_cap_reached'];
export const empty=()=>Object.fromEntries([...numeric.map(k=>[k,0]),['soft_mean_x',null]]);
const add=(to,from)=>{for(const k of numeric){assert.ok(Number.isSafeInteger(from[k])&&from[k]>=0,k);to[k]+=from[k];assert.ok(Number.isSafeInteger(to[k]),'Counter overflow');}to.soft_mean_x=to.soft_x_count?to.soft_x_sum/to.soft_x_count:null;};
export function derive(batch,audit,stdout){
 const lines=stdout.split('\n'),parseIssues=[];
 const rows=prefix=>{const result=[];for(let i=0;i<lines.length;i++){const x=lines[i];if(!x.startsWith(prefix+' '))continue;try{if(audit.auditUnavailable&&i===lines.length-1&&x.length)throw Error('unterminated record');result.push(JSON.parse(x.slice(prefix.length+1)));}catch(error){if(!audit.auditUnavailable)throw error;parseIssues.push({line:i+1,prefix,error:String(error),text:x});}}return result;};
 const checks=rows('CHECK'),games=rows('GAME'),pairs=rows('PAIR'),triggers=rows('TRIGGER'),shutdowns=rows('SHUTDOWN'),faults=rows('FAULT');
 assert.equal(audit.index,batch.index);assert.equal(audit.label,batch.label);assert.equal(audit.checkRows,checks.length);assert.equal(audit.triggerRows,triggers.length);assert.deepEqual(audit.shutdowns,shutdowns);assert.deepEqual(audit.faults,faults);const sums=rows('SUMMARY');assert.deepEqual(audit.summary,sums.length===1?sums[0]:null);
 const credit={pairs:0,usual:empty(),soft:empty()},observed={usual:{games:games.filter(x=>x.mode==='usual').length,searches:checks.filter(x=>x.mode==='usual').length},soft:{games:games.filter(x=>x.mode==='soft').length,searches:checks.filter(x=>x.mode==='soft').length}};
 const configs=rows('CONFIG');if(audit.complete){assert.equal(configs.length,1);const start=configs[0].first_pair,s=audit.summary;assert.ok(Number.isSafeInteger(start)&&start>=0);assert.equal(pairs.length,s.pairs_completed);for(const [i,p]of pairs.entries()){assert.deepEqual(p,{pair:start+i,completed:true});for(const mode of ['usual','soft'])assert.equal(games.filter(g=>g.pair===p.pair&&g.mode===mode&&g.completed).length,1);}const ids=new Set(pairs.map(p=>p.pair));credit.pairs=s.pairs_completed;
  for(const mode of ['usual','soft']){const accepted=s[mode],selected=checks.filter(x=>x.mode===mode&&ids.has(x.pair));assert.equal(accepted.games,pairs.length);assert.equal(accepted.searches,selected.length);assert.equal(accepted.moves,selected.length);add(credit[mode],accepted);observed[mode]={...s['observed_'+mode+'_including_partial_pair']};if(mode==='soft'){const xs=selected.filter(x=>x.x!==null);assert.ok(xs.every(x=>Number.isSafeInteger(x.x)&&x.x>=0));assert.equal(accepted.soft_x_count,xs.length);assert.equal(accepted.soft_x_sum,xs.reduce((n,x)=>n+x.x,0));assert.equal(accepted.soft_hard_cap_reached,xs.filter(x=>x.x>=10000).length);}}
 }
 const forced=s=>!!(s.driver_sigterm||s.driver_sigkill),timingGroups={};for(const s of shutdowns){assert.ok(s.quit_to_exit_ms===null||Number.isFinite(s.quit_to_exit_ms)&&s.quit_to_exit_ms>=0);if(s.quit_to_exit_ms!==null)assert.ok(s.quit_sent&&s.reaped);const key='forced='+forced(s)+':batchComplete='+!!audit.complete,g=timingGroups[key]??={events:0,measured:0,unavailable:0,over100ms:0,maxMs:null};g.events++;if(s.quit_to_exit_ms===null)g.unavailable++;else{g.measured++;g.over100ms+=s.quit_to_exit_ms>100;g.maxMs=g.maxMs===null?s.quit_to_exit_ms:Math.max(g.maxMs,s.quit_to_exit_ms);}}
 const reasons={};for(const f of faults)reasons[f.reason]=(reasons[f.reason]??0)+1;
 const failingLabelChecks=checks.filter(x=>x.mode==='soft'&&(x.item3===0||x.item4===0));
 const labelFault=!!audit.summary?.label_fault||faults.some(x=>x.label_fault)||failingLabelChecks.length>0;
 return {index:batch.index,label:batch.label,seed:batch.seed,group:[batch.mode,batch.HindsightAndGuard?'on':'off','t'+batch.threads,'n'+batch.nodes,'d'+batch.depth].join('/'),mode:batch.mode,threads:batch.threads,enabled:batch.HindsightAndGuard,complete:!!audit.complete,passed:!!audit.passed,coverage:!!audit.coverage,auditUnavailable:!!audit.auditUnavailable,parseIssues,exit:audit.exit,signal:audit.signal,credit:audit.passed?structuredClone(credit):{pairs:0,usual:empty(),soft:empty()},completedPairObservations:credit,observedIncludingPartialPair:observed,rawRows:{checks:checks.length,triggers:triggers.length,games:games.length,pairs:pairs.length},diagnosticLines:audit.diagnostics.length,diagnosticObserved:!!audit.summary?.diagnostic_observed,faultRows:faults.length,faultReasons:reasons,labelFault,failingLabelChecks,labelFaultContexts:triggers.filter(t=>failingLabelChecks.some(x=>x.pair===t.pair&&x.mode===t.mode&&x.ply===t.ply)),unassignedSoftLines:audit.summary?.unassigned_soft_lines??null,summaryFault:audit.summary?.fault??null,shutdowns,timingGroups,exclusion:audit.passed?null:audit.complete?'Finalized batch did not pass: completed-pair observations retained separately; zero accepted-clean coverage.':'Incomplete or externally interrupted batch: zero credited pairs/games/searches; raw observations retained.'};
}
export function totals(rows){const t={recordedBatches:rows.length,completeBatches:0,passedBatches:0,failedBatches:0,incompleteBatches:0,zeroCoverageBatches:0,pairs:0,usual:empty(),soft:empty(),completedPairObservations:{pairs:0,usual:empty(),soft:empty()},diagnosticLines:0,diagnosticBatches:0,faultRows:0,faultBatches:0,labelFaultBatches:0,externalTimeoutBatches:0,signalBatches:0,shutdownTimeouts:0,forcedShutdowns:0,nonzeroEngineExits:0,engineSignals:0,unassignedSoftLines:0,unknownUnassignedSoftBatches:0,faultReasons:{},shutdownTiming:{events:0,measured:0,unavailable:0,over100ms:0,maxMs:null,groups:{}},groups:{},observedIncludingPartialPair:{usual:{games:0,searches:0},soft:{games:0,searches:0}}};const seen=new Set();
 for(const r of rows){assert.ok(!seen.has(r.index),'Duplicate final batch index');seen.add(r.index);t.completeBatches+=r.complete;t.passedBatches+=r.passed;t.failedBatches+=!r.passed;t.incompleteBatches+=!r.complete;t.zeroCoverageBatches+=r.complete&&!r.coverage;t.pairs+=r.credit.pairs;add(t.usual,r.credit.usual);add(t.soft,r.credit.soft);t.completedPairObservations.pairs+=r.completedPairObservations.pairs;for(const mode of ['usual','soft'])add(t.completedPairObservations[mode],r.completedPairObservations[mode]);for(const mode of ['usual','soft'])for(const k of ['games','searches'])t.observedIncludingPartialPair[mode][k]+=r.observedIncludingPartialPair[mode][k];t.diagnosticLines+=r.diagnosticLines;t.diagnosticBatches+=!!(r.diagnosticLines||r.diagnosticObserved);t.faultRows+=r.faultRows;t.faultBatches+=!!(r.faultRows||r.summaryFault||r.diagnosticLines||r.diagnosticObserved||r.labelFault||r.unassignedSoftLines||r.shutdowns.some(x=>x.timed_out||x.exit>0||x.signal>0||!x.reaped));t.labelFaultBatches+=r.labelFault;t.externalTimeoutBatches+=[124,137].includes(r.exit);t.signalBatches+=!!r.signal;t.unassignedSoftLines+=r.unassignedSoftLines??0;t.unknownUnassignedSoftBatches+=r.unassignedSoftLines===null;for(const [k,n]of Object.entries(r.faultReasons))t.faultReasons[k]=(t.faultReasons[k]??0)+n;for(const s of r.shutdowns){t.shutdownTimeouts+=!!s.timed_out;t.forcedShutdowns+=!!(s.driver_sigterm||s.driver_sigkill);t.nonzeroEngineExits+=s.exit>0;t.engineSignals+=s.signal>0;}
  for(const [k,g]of Object.entries(r.timingGroups)){const out=t.shutdownTiming.groups[k]??={events:0,measured:0,unavailable:0,over100ms:0,maxMs:null};for(const dst of [out,t.shutdownTiming]){for(const n of ['events','measured','unavailable','over100ms'])dst[n]+=g[n];if(g.maxMs!==null)dst.maxMs=dst.maxMs===null?g.maxMs:Math.max(dst.maxMs,g.maxMs);}}
  const g=t.groups[r.group]??={batches:0,pairs:0,usual:empty(),soft:empty(),passed:0,failed:0,diagnosticLines:0,faultRows:0};g.batches++;g.pairs+=r.credit.pairs;add(g.usual,r.credit.usual);add(g.soft,r.credit.soft);g.passed+=r.passed;g.failed+=!r.passed;g.diagnosticLines+=r.diagnosticLines;g.faultRows+=r.faultRows;
 }t.games=t.usual.games+t.soft.games;t.searches=t.usual.searches+t.soft.searches;t.completePlan=rows.length===12&&t.completeBatches===12;t.passedPlan=t.completePlan&&t.passedBatches===12;t.softXSum=t.soft.soft_x_sum;t.softXCount=t.soft.soft_x_count;t.softMeanX=t.soft.soft_mean_x;t.softHardCaps=t.soft.soft_hard_cap_reached;t.preliminaryIncluded=false;t.blockFurtherEngineWork=t.labelFaultBatches>0;t.auditUnavailableBatches=rows.filter(r=>r.auditUnavailable).length;t.manualReviewRequired=t.failedBatches>0;t.continuationAllowed=t.failedBatches===0;t.faultCountingUnit='Batches with observed engine, label, diagnostic or shutdown findings; failed/zero-coverage/incomplete batches listed separately.';return t;
}
export function updatedState(state,row){const rows=state.finalPairedSoftnodesRows??[],old=rows.find(r=>r.index===row.index);if(old){assert.equal(old.evidenceIdentity,row.evidenceIdentity,'Conflicting duplicate batch');return {duplicate:true,state};}const result=structuredClone(state);result.finalPairedSoftnodesRows=[...rows,row].sort((a,b)=>a.index-b.index);result.finalPairedSoftnodes=totals(result.finalPairedSoftnodesRows);result.soakTotals??={};result.soakTotals.finalTree??={};const t=result.finalPairedSoftnodes;Object.assign(result.soakTotals.finalTree,{pairedSoftnodes:t,games:t.games,searches:t.searches,faults:t.faultBatches,faultCountingUnit:t.faultCountingUnit,status:t.passedPlan?'All12 paired groups passed':t.blockFurtherEngineWork?'STOP: soft-label item3/4 finding':`${t.passedBatches}/12 paired groups passed; ${t.failedBatches} failed or incomplete; accepted-clean counts only`});result.validation??={};result.validation.finalPairedSoftnodes=t;if(result.active&&(result.active.batch===row.label||result.active.label===row.label))result.active=null;return {duplicate:false,state:result};}
