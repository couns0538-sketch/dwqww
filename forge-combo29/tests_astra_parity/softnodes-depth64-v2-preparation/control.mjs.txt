// Root opt-in only. No Git mutation or global experiment-state writes.
import fs from 'node:fs';
import cp from 'node:child_process';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';
const prep='/workspace/work/parity-softnodes-depth64-v2-preparation',run='/workspace/work/parity-run';
const repo='/workspace/work/dwqww-parity',out=run+'/softnodes-depth64-v2-batches';
const declaration=out+'/declaration.txt',planFile=prep+'/plan.txt',decisionFile=prep+'/FABLE_ANSWER_0013_0014_0015.md',decisionSha='5e2d573b7e553ec49e84918bd1dfd6a28ccfd2ca7236b6834c3b041965945e2f';
const plan=JSON.parse(fs.readFileSync(planFile)),driver=JSON.parse(fs.readFileSync(prep+'/DRIVER-PINS-v3.txt'));
const hash=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');
const put=(p,x)=>fs.writeFileSync(p,JSON.stringify(x,null,2)+'\n',{flag:'wx'});
const [action,arg,authorization]=process.argv.slice(2);
function verify(pins){for(const [p,pin]of Object.entries(pins))assert.equal(hash(p),pin,p);}
if(action==='declare'){
 assert.equal(authorization,'--freeze-authorized');assert.match(arg??'',/^[0-9a-f]{40}$/);
 assert.ok(!fs.existsSync(declaration),'Declaration is immutable');
 verify(driver.inputPins);assert.equal(hash(driver.binary),driver.sha256);assert.equal(hash(decisionFile),decisionSha);
 const git=args=>{const r=cp.spawnSync('git',['-C',repo,...args],{encoding:'utf8',timeout:30000,maxBuffer:32*1024*1024});assert.equal(r.status,0,r.stderr);return r.stdout;};
 const files=git(['ls-tree','-r','--name-only',arg,'--','forge-combo29/src','forge-combo29/include','forge-combo29/CMakeLists.txt']).trim().split('\n');
 const sourcePins={};for(const relative of files){const file=repo+'/'+relative;assert.ok(Buffer.from(git(['show',arg+':'+relative])).equals(fs.readFileSync(file)),relative);sourcePins[file]=hash(file);}
 const engines={};for(const mode of ['asan','tsan']){
  const dir=run+'/builds/final-'+mode,binary=dir+'/forge',flagsFile=dir+'/CMakeFiles/forge.dir/flags.make';
  const flags=fs.readFileSync(flagsFile,'utf8');assert.ok(flags.includes('-DFORGE_MATE_ALPHA_GUARD=1'));
  assert.ok(flags.includes('-fsanitize='+(mode==='asan'?'address,undefined':'thread')));
  const objects=dir+'/CMakeFiles/forge.dir/src';const objectPins=Object.fromEntries(fs.readdirSync(objects).filter(n=>n.endsWith('.cpp.o')).map(n=>[objects+'/'+n,hash(objects+'/'+n)]));
  assert.equal(Object.keys(objectPins).length,8);engines[mode]={binary,sha256:hash(binary),flagsFile,flagsSha256:hash(flagsFile),objectPins};
 }
 fs.mkdirSync(out,{recursive:true});
 const toolPins=Object.fromEntries([decisionFile,planFile,prep+'/control.mjs',prep+'/audit.mjs',prep+'/DRIVER-PINS-v3.txt',driver.binary,...Object.keys(driver.inputPins)].map(p=>[p,hash(p)]));
 put(declaration,{utc:new Date().toISOString(),decision:{commit:'55ddf61a754ceffc4f940b847ad97143148d0434',file:decisionFile,sha256:decisionSha},sourceCommit:arg,sourcePins,engines,toolPins,deadline:plan.deadline,protocol:plan.protocol,engineRuns:0});
 console.log(JSON.stringify({declaration,sha256:hash(declaration),engineRuns:0}));
}else if(action==='run'){
 assert.equal(authorization,'--execute-authorized');const index=Number(arg),b=plan.batches[index];
 assert.ok(Number.isInteger(index)&&b,'Index0..11');assert.ok(Date.now()+475000<Date.parse(plan.deadline),'Need bounded-run and recording margin');
 const d=JSON.parse(fs.readFileSync(declaration));verify(d.sourcePins);verify(d.toolPins);
 const engine=d.engines[b.mode];assert.equal(engine.binary,b.engine);assert.equal(hash(engine.binary),engine.sha256);
 assert.equal(hash(engine.flagsFile),engine.flagsSha256);verify(engine.objectPins);
 const dir=out+'/'+b.label;assert.ok(!fs.existsSync(dir),'Never overwrite or rerun an existing label');
 fs.mkdirSync(dir);put(dir+'/input-pins.txt',{batch:b,declaration,sha256:hash(declaration),engine,driver});
 const argv=['timeout','-k','2','440',...b.argv],started=new Date().toISOString();
 const stdout=dir+'/stdout.txt',stderr=dir+'/stderr.txt',fds=[fs.openSync(stdout,'wx'),fs.openSync(stderr,'wx')];
 put(dir+'/start.txt',{argv,cwd:repo,started,batch:b});console.log(JSON.stringify({started,label:b.label,argv}));
 const r=cp.spawnSync(argv[0],argv.slice(1),{cwd:repo,stdio:['ignore',...fds],timeout:450000});
 fds.forEach(fd=>fs.closeSync(fd));
 const meta={argv,cwd:repo,started,ended:new Date().toISOString(),exit:r.status,signal:r.signal,error:r.error?.message??null,stdout,stderr,stdoutSha256:hash(stdout),stderrSha256:hash(stderr)};
 put(dir+'/command.txt',meta);console.log(JSON.stringify(meta));process.exitCode=r.status??1;
}else throw Error('declare SOURCE_COMMIT --freeze-authorized | run INDEX --execute-authorized');
