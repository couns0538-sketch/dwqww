import fs from 'node:fs';import path from 'node:path';import crypto from 'node:crypto';
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
export function parseCycles(text){const records=[],errors=[];for(const [i,line]of text.split('\n').entries()){if(!line.trim())continue;try{records.push(JSON.parse(line));}catch{errors.push('Invalid/partial JSON at line '+(i+1));}}const configs=records.filter(r=>r.type==='config'),starts=records.filter(r=>r.type==='cycle_started'),cycles=records.filter(r=>r.type==='cycle'),summaries=records.filter(r=>r.type==='summary'),seenStarts=new Set(),seenCycles=new Set();if(configs.length!==1||configs[0].target_cycles!==2000||configs[0].cycles_per_side!==1000)errors.push('Expected one exact2000 config');for(const [i,r]of starts.entries()){if(r.cycle!==i+1||seenStarts.has(r.cycle)||r.cycle>2000)errors.push('Invalid start sequence '+r.cycle);seenStarts.add(r.cycle);if(r.side!==(r.cycle%2?'candidate':'reference'))errors.push('Wrong start side '+r.cycle);}const bySide={candidate:{cycles:0,searched:0,depth9:0,quitSent:0,exitObserved:0,shutdownTimeouts:0,signals:0,nonzeroExits:0,errors:0,budgetInterrupted:0,forced:0,measured:0,unavailable:0,maxQuitToExitMs:null,over100ms:0,tuples:{}},reference:{cycles:0,searched:0,depth9:0,quitSent:0,exitObserved:0,shutdownTimeouts:0,signals:0,nonzeroExits:0,errors:0,budgetInterrupted:0,forced:0,measured:0,unavailable:0,maxQuitToExitMs:null,over100ms:0,tuples:{}}},groups={};for(const [i,r]of cycles.entries()){if(r.cycle!==i+1||seenCycles.has(r.cycle)||!seenStarts.has(r.cycle)||r.cycle>2000)errors.push('Invalid finalized cycle sequence '+r.cycle);seenCycles.add(r.cycle);const expected=r.cycle%2?'candidate':'reference';if(r.side!==expected)errors.push('Wrong alternating side '+r.cycle);const s=bySide[r.side];if(!s)continue;const q=r.shutdown??{};s.cycles++;s.searched+=!!r.searched;s.depth9+=r.searched&&r.depth===9;s.quitSent+=!!q.quit_sent;s.exitObserved+=!!q.reaped;s.shutdownTimeouts+=!!q.timed_out;s.signals+=q.signal>0;s.nonzeroExits+=q.exit_code>0;s.errors+=!!r.error||!!q.error;s.budgetInterrupted+=!!r.budget_interrupted;s.forced+=!!(q.driver_sigterm||q.driver_sigkill);if(q.timing_schema!==1)errors.push('Missing timing schema '+r.cycle);if(q.quit_to_exit_ms!==null&&Number.isFinite(q.quit_to_exit_ms)&&q.quit_to_exit_ms>=0){if(!q.quit_sent||!q.reaped)errors.push('Timing without successful quit/observed exit '+r.cycle);s.measured++;s.maxQuitToExitMs=s.maxQuitToExitMs===null?q.quit_to_exit_ms:Math.max(s.maxQuitToExitMs,q.quit_to_exit_ms);s.over100ms+=q.quit_to_exit_ms>100;}else{s.unavailable++;if(q.quit_sent&&q.reaped)errors.push('Successful quit and observed exit missing timing '+r.cycle);}if(r.searched&&(!['cp','mate'].includes(r.score_kind)||!Number.isInteger(r.score)||!Number.isSafeInteger(r.nodes)||r.nodes<0||!/^([a-h][1-8]){2}[qrbn]?$/.test(r.bestmove)))errors.push('Malformed searched tuple '+r.cycle);const key=r.side+':budget='+!!r.budget_interrupted+':forced='+!!(q.driver_sigterm||q.driver_sigkill),g=groups[key]??={cycles:0,measured:0,maxMs:null,over100ms:0};g.cycles++;if(q.quit_to_exit_ms!==null&&Number.isFinite(q.quit_to_exit_ms)){g.measured++;g.maxMs=g.maxMs===null?q.quit_to_exit_ms:Math.max(g.maxMs,q.quit_to_exit_ms);g.over100ms+=q.quit_to_exit_ms>100;}if(r.searched){const tuple=JSON.stringify([r.depth,r.bestmove,r.score_kind,r.score,r.bound,r.nodes]);s.tuples[tuple]=(s.tuples[tuple]??0)+1;}}
if(summaries.length!==1)errors.push('Missing or multiple native summary');else{const s=summaries[0];for(const [k,n]of Object.entries({finalized_cycles:cycles.length,candidate_cycles:bySide.candidate.cycles,reference_cycles:bySide.reference.cycles,searched:bySide.candidate.searched+bySide.reference.searched,quit_sent:bySide.candidate.quitSent+bySide.reference.quitSent,exit_observed:bySide.candidate.exitObserved+bySide.reference.exitObserved}))if(s[k]!==n)errors.push('Native summary count mismatch '+k);}
return {configs,starts,cycles,summaries,errors,counts:{attempted:starts.length,finalized:cycles.length,unfinalizedStarts:starts.filter(r=>!seenCycles.has(r.cycle)).length},bySide,byBudgetAndForcedStatus:groups,all2000SearchQuitCycles:cycles.length===2000&&starts.length===2000&&Object.values(bySide).every(s=>s.cycles===1000&&s.depth9===1000&&s.quitSent===1000&&s.exitObserved===1000)&&errors.length===0};}
export function auditDirectory(directory){const declaration=JSON.parse(fs.readFileSync(directory+'/DECLARATION.txt')),stdout=directory+'/cycles.jsonl',text=fs.existsSync(stdout)?fs.readFileSync(stdout,'utf8'):'',a=parseCycles(text),tracePins=[],diagnostics=[],expectedFen=fs.readFileSync(directory+'/fen.txt','utf8').trim();for(const r of a.cycles){if(r.fen!==expectedFen)a.errors.push('Wrong FEN '+r.cycle);if(r.binary!==declaration[r.side]?.binary)a.errors.push('Wrong binary '+r.cycle);const expected=path.join(directory,'cycle-'+String(r.cycle).padStart(4,'0')+'-'+r.side+'.transcript.txt');if(r.transcript!==expected){a.errors.push('Unexpected transcript path '+r.cycle);continue;}if(!fs.existsSync(expected)){a.errors.push('Missing transcript '+r.cycle);continue;}const bytes=fs.readFileSync(expected),raw=bytes.toString();tracePins.push({file:expected,bytes:bytes.length,sha256:sha(bytes)});if(/AddressSanitizer|UndefinedBehaviorSanitizer|ThreadSanitizer|MemorySanitizer|LeakSanitizer|runtime error:/.test(raw))diagnostics.push({cycle:r.cycle,side:r.side,transcript:expected});if(r.searched){for(const command of ['setoption name Threads value 1','setoption name Hash value 16','setoption name OwnBook value false','ucinewgame','position fen '+expectedFen,'go depth 9'])if(!raw.includes('> stdin '+command+'\n'))a.errors.push('Missing exact command '+r.cycle+': '+command);if(!r.defaults_ok)a.errors.push('Missing default-false proof '+r.cycle);}if(r.shutdown?.quit_sent&&!raw.includes('> stdin quit\n'))a.errors.push('Missing quit transcript '+r.cycle);}
const processFile=directory+'/PROCESS.txt',process=fs.existsSync(processFile)?JSON.parse(fs.readFileSync(processFile)):null;if(process){if(process.stdoutSha256!==sha(fs.readFileSync(stdout)))a.errors.push('Raw stdout hash differs from PROCESS');const errfile=directory+'/stderr.txt';if(!fs.existsSync(errfile))a.errors.push('Missing raw stderr');else {const err=fs.readFileSync(errfile);if(process.stderrSha256!==sha(err))a.errors.push('Raw stderr hash differs from PROCESS');if(/AddressSanitizer|UndefinedBehaviorSanitizer|ThreadSanitizer|MemorySanitizer|LeakSanitizer|runtime error:/.test(err.toString()))diagnostics.push({stream:'stderr',file:errfile});}}const complete=a.all2000SearchQuitCycles&&a.errors.length===0&&process?.exit===0&&!process?.signal&&!(process?.changedInputs?.length);if(process?.changedInputs?.length)a.errors.push('Frozen inputs changed during diagnostic');return {schema:1,kind:'prescribed-second-candidate-shutdown-diagnostic',declarationSha256:sha(fs.readFileSync(directory+'/DECLARATION.txt')),trigger:declaration.trigger,fen:expectedFen,process,counts:a.counts,bySide:a.bySide,byBudgetAndForcedStatus:a.byBudgetAndForcedStatus,nativeSummary:a.summaries[0]??null,completePrescribed2000:complete,diagnostics,errors:a.errors,tracePins,tupleFields:['depth','bestmove','scoreKind','score','bound','nodes'],note:'2000 TOTAL cycles means1000 per side, strictly alternating fresh processes. Partial runs report actual counts only. Forced/budget exits and unavailable timing are separate. This diagnostic does not prove a physical timing cause.'};}
