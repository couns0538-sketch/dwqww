import fs from 'node:fs';import readline from 'node:readline';import assert from 'node:assert/strict';
import {auditBatch} from './audit-core.mjs';
const equal=(a,b)=>JSON.stringify(a)===JSON.stringify(b);
export function parsePgn(text,candidate){const rows=[],wld={wins:0,losses:0,draws:0},ids=new Set(),errors=[];for(const game of text.split(/(?=^\[Event )/m).filter(x=>x.trim())){const tags=Object.fromEntries([...game.matchAll(/^\[(\w+) "([^"]*)"\]\r?$/gm)].map(m=>[m[1],m[2]])),id=Number(tags.GameId);if(!Number.isInteger(id)||ids.has(id))errors.push('Invalid/duplicate GameId '+tags.GameId);ids.add(id);if(!game.trimEnd().endsWith(tags.Result??'INVALID'))errors.push('Movetext result mismatch '+id);const points=tags.Result==='1/2-1/2'?1:tags.Result==='1-0'?(tags.White===candidate?2:0):tags.Result==='0-1'?(tags.Black===candidate?2:0):null;if(points===null)errors.push('Unfinished PGN '+id);else wld[points===2?'wins':points===1?'draws':'losses']++;rows.push({id,white:tags.White,black:tags.Black,fen:tags.FEN,result:tags.Result,termination:tags.Termination,points});}const bins=[0,0,0,0,0],pairs=[];const byId=new Map(rows.map(r=>[r.id,r]));for(let id=1;id<=Math.max(0,...ids);id+=2){const a=byId.get(id),b=byId.get(id+1);if(a&&b){if(a.fen!==b.fen||a.white!==b.black||a.black!==b.white)errors.push('PGN opening/color mismatch '+id);if(a.points!==null&&b.points!==null){bins[a.points+b.points]++;pairs.push({pair:(id+1)/2,fen:a.fen,games:[id,id+1],points:a.points+b.points});}}}return {games:rows.length,wld,bins,pairs,errors,rows};}
export async function auditExtra(plan,processResult,inputHashes){const summary=await auditBatch(plan,processResult,inputHashes),names=plan.engines.map(e=>e.name),zero=()=>({illegalMoveEvents:0,illegalMoveGames:0,disconnectGames:0,stallGames:0,timeForfeitGames:0,nonzeroProcessExits:0,processSignals:0,crashSignals:0,controllerCleanupSignals:0}),bySide=Object.fromEntries(names.map(n=>[n,zero()])),starts=new Map(),faults=[],unattributed=[],defaults=Object.fromEntries(names.map(n=>[n,new Set()]));
if(fs.existsSync(plan.files.events))for await(const line of readline.createInterface({input:fs.createReadStream(plan.files.events),crlfDelay:Infinity})){if(!line.trim())continue;let e;try{e=JSON.parse(line);}catch{continue;}if(e.event==='game_started')starts.set(e.game,e);if(e.event==='move'&&e.legal!==true){if(bySide[e.engine])bySide[e.engine].illegalMoveEvents++;else unattributed.push(e);}if(e.event==='game_aborted')unattributed.push(e);if(e.event==='game_finished'&&e.termination!=='normal'){const start=starts.get(e.game),loser=e.result==='1-0'?start?.black:e.result==='0-1'?start?.white:null,key={illegal_move:'illegalMoveGames',disconnect:'disconnectGames',stall:'stallGames',time_forfeit:'timeForfeitGames'}[e.termination];if(loser&&key&&bySide[loser])bySide[loser][key]++;else unattributed.push(e);faults.push({...e,side:loser});}}
let lineNo=0;const processEvidence=[],unknownTrace=[];if(fs.existsSync(plan.files.trace))for await(const line of readline.createInterface({input:fs.createReadStream(plan.files.trace),crlfDelay:Infinity})){lineNo++;const m=line.match(/\[Engine\].*?\s(?:<stderr>\s+)?(\S+)\s+---> (.*)$/);if(m&&names.includes(m[1])){const [,,message]=m;if(message.startsWith('option name '))defaults[m[1]].add(message);const exit=message.match(/Process exited normally with status (\d+)/),signal=message.match(/Process terminated by signal (\d+)/);if(exit&&+exit[1]!==0){bySide[m[1]].nonzeroProcessExits++;processEvidence.push({line:lineNo,side:m[1],text:line});}if(signal){const cleanup=(processResult.timedOut||processResult.interrupted)&&[9,15].includes(+signal[1]);bySide[m[1]].processSignals++;bySide[m[1]][cleanup?'controllerCleanupSignals':'crashSignals']++;processEvidence.push({line:lineNo,side:m[1],controllerCleanup:!!cleanup,text:line});}}else if(/Process terminated by signal|Process exited normally with status [1-9]|Unknown status/.test(line))unknownTrace.push({line:lineNo,text:line});}
if(plan.defaultOptions)for(const n of names){if(!equal([...defaults[n]].sort(),plan.defaultOptions))summary.errors.push('Full default option advertisement differs: '+n);}
const pgn=fs.existsSync(plan.files.pgn)?parsePgn(fs.readFileSync(plan.files.pgn,'utf8'),names[0]):{errors:['Missing PGN'],games:0,wld:{wins:0,losses:0,draws:0},bins:[0,0,0,0,0]};for(const e of pgn.errors)summary.errors.push(e);if(!equal(pgn.wld,summary.wldChecks?.events))summary.errors.push('Independent PGN WLD mismatch');if(!equal(pgn.bins,summary.pairBins?.counts))summary.errors.push('Independent PGN pentanomial mismatch');summary.completed=summary.completed&&summary.errors.length===0;summary.passed=summary.passed&&summary.errors.length===0;return {...summary,sideAccounting:{bySide,faultGames:faults,unattributedAbortsOrGames:unattributed,processEvidence,unattributedProcessEvidence:unknownTrace,definition:'Game faults by losing configuration; illegal move attempts retain their event engine. Process exits/signals are separate observations, not extra distinct games. Disconnect does not establish a crash cause. Controller timeout cleanup signals are separated.'},fullAdvertisedDefaults:Object.fromEntries(names.map(n=>[n,[...defaults[n]].sort()])),independentPgn:{games:pgn.games,wld:pgn.wld,bins:pgn.bins,errors:pgn.errors},creditedCompletedGames:summary.completed?summary.counts.finished:0,creditedCompletedPairs:summary.completed?summary.pairBins.completePairs:0,rawIncompleteExcluded:!summary.completed,countsOnly:true};}
