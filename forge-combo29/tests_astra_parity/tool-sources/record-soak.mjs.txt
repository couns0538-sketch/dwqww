import fs from 'node:fs';import cp from 'node:child_process';import crypto from 'node:crypto';import assert from 'node:assert/strict';
const [stage,indexText]=process.argv.slice(2),index=Number(indexText);assert.ok(['step1','final'].includes(stage));assert.ok(Number.isInteger(index)&&index>=0&&index<12);
const run='/workspace/work/parity-run',stateFile=run+'/state.json',s=JSON.parse(fs.readFileSync(stateFile)),key=stage==='step1'?'preliminarySoak':'finalSoak',auditFile=run+'/logs/'+stage+'-soak-audit-'+index+'.stdout.txt';
const a=JSON.parse(fs.readFileSync(auditFile));assert.equal(a.batch.label,a.label);assert.ok(a.label.startsWith(stage==='step1'?'step1-parity-':'parity-'));assert.ok(!(s[key]??[]).some(x=>x.index===index),'Duplicate batch');
const hash=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex'),metaFile=run+'/logs/'+a.label+'.metadata.txt',meta=JSON.parse(fs.readFileSync(metaFile));assert.equal(hash(metaFile),a.metadataSha256);assert.equal(hash(meta.stdoutFile),a.stdoutSha256);assert.equal(hash(meta.stderrFile),a.stderrSha256);
const destination=s.repo+'/forge-combo29/tests_astra_parity/soak-batches/'+stage+'-'+index;
if(!fs.existsSync(destination)){const p=cp.spawnSync('node',[run+'/package-evidence.mjs',run+'/logs/'+a.label,destination],{encoding:'utf8',timeout:60000,maxBuffer:1024*1024});console.log(p.stdout);if(p.stderr)console.error(p.stderr);assert.equal(p.status,0,'Package failed');}
const manifest=JSON.parse(fs.readFileSync(destination+'/MANIFEST.txt'));for(const f of [metaFile,meta.stdoutFile,meta.stderrFile])assert.ok(manifest.files.some(x=>x.source===f&&x.sha256===hash(f)),'Missing published raw proof '+f);
const row={stage,index,label:a.label,counts:a.counts??null,diagnostics:a.diagnostics.length,faults:a.faults,accepted:a.completed&&a.passed,audit:auditFile,auditSha256:hash(auditFile),package:destination};(s[key]??=[]).push(row);
const rows=s[key],accepted=rows.filter(x=>x.accepted),totals={batches:accepted.length,games:0,searches:0,faults:rows.reduce((n,x)=>n+x.diagnostics+x.faults.length+Number(!x.accepted),0),excludedAttempts:rows.filter(x=>!x.accepted).length};for(const r of accepted){totals.games+=r.counts.games;totals.searches+=r.counts.searches;}
s.soakTotals??={};s.soakTotals[stage==='step1'?'preliminaryStep1':'finalTree']=totals;s.active=null;s.phase=stage==='step1'?'Preliminary sanitizer coverage; equality review pending':'Final-tree sanitizer soak';s.progressBatch=a.label+' seed'+a.batch.seed;s.checkpointNote=(stage==='step1'?'Preliminary step1 coverage, separate from required final tree. ':'Final-tree coverage. ')+totals.batches+' accepted batches, '+totals.games+' games, '+totals.searches+' searches, '+totals.faults+' recorded findings. Native commands, diagnostics and exact triggers preserved. '+(stage==='step1'?'Shutdown0012 remains retained; no same-clock source edit yet.':'Original preliminary and equality findings remain in the report.');
fs.writeFileSync(stateFile,JSON.stringify(s,null,2)+'\n');console.log(JSON.stringify({recorded:row.label,accepted:row.accepted,totals}));process.exitCode=row.accepted?0:1;
