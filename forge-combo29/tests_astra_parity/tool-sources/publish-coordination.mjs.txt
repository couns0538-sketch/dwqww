import fs from 'node:fs';import cp from 'node:child_process';import assert from 'node:assert/strict';
const [id,branch,input,leaf]=process.argv.slice(2),repo='/workspace/work/fable-opus-handoff',r='/workspace/work/parity-run',records=[];
assert.match(id??'',/^\d{4}$/);assert.match(branch??'',/^for-fable-and-opus\/astra-[a-z0-9-]+$/);assert.match(leaf??'',/^ASTRA_[A-Z_]+_\d{4}\.md$/);assert.ok(input.startsWith(r+'/inputs/'));const body=fs.readFileSync(input,'utf8');assert.ok(!body.includes('\0'));
const rel='forge-combo29/soak_all/'+leaf,receipt=r+'/inputs/question-'+id+'-receipt.txt';assert.ok(!fs.existsSync(receipt));
function git(args,allowed=[0]){const p=cp.spawnSync('git',args,{cwd:repo,encoding:'utf8',timeout:90000,maxBuffer:4*1024*1024});records.push({utc:new Date().toISOString(),cwd:repo,argv:['git',...args],exit:p.status,signal:p.signal,stdout:p.stdout,stderr:p.stderr});fs.writeFileSync(r+'/logs/question-'+id+'-git.txt',JSON.stringify(records,null,2)+'\n');assert.ok(allowed.includes(p.status),'git failed');return p.stdout.trim();}
const previous=git(['branch','--show-current']);assert.equal(previous,'handoff/fable-opus-combo29-all-20261114');assert.equal(git(['status','--porcelain']),'');assert.equal(git(['ls-remote','--heads','origin','refs/heads/'+branch]),'');git(['show-ref','--verify','--quiet','refs/heads/'+branch],[1]);git(['checkout','-b',branch]);fs.writeFileSync(repo+'/'+rel,body,{flag:'wx'});git(['add','--sparse','--',rel]);assert.equal(git(['diff','--cached','--name-only']),rel);git(['-c','user.name=Codex','-c','user.email=codex@openai.com','commit','-m','Astra coordination '+id+': evidence, classification and next work']);const commit=git(['rev-parse','HEAD']);git(['push','--no-follow-tags','origin','HEAD:refs/heads/'+branch]);git(['checkout',previous]);const result={utc:new Date().toISOString(),branch,commit,pushed:true,path:rel};fs.writeFileSync(receipt,JSON.stringify(result,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify(result));
