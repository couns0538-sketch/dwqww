  14 class TournamentSchedulerBase : public Scheduler {
  15    public:
  16     TournamentSchedulerBase(book::OpeningBook* opening_book, std::size_t players, std::size_t rounds, std::size_t games,
  17                             std::size_t played_games)
  18         : opening_book_(opening_book),
  19           n_players(players),
  20           n_rounds(rounds),
  21           n_games_per_round(games),
  22           current_round(1),
  23           game_counter(played_games),
  24           player1(0),
  25           player2(1),
  26           games_per_pair(0),
  27           pair_counter(played_games / games) {
  28         current_round = (played_games / games) + 1;
  29 
  30         if (n_players < 2 || n_rounds < 1 || n_games_per_round < 1) {
  31             throw fastchess_exception("Invalid number of players, rounds, or games per round");
  32         }
  33     }

  35     // Function to generate the next game pairing
  36     std::optional<Pairing> next() override {
  37         // Check if the current round is beyond the number of rounds
  38         if (current_round > n_rounds) {
  39             // No more rounds left
  40             return std::nullopt;
  41         }
  42 
  43         if (games_per_pair == 0) {
  44             opening = opening_book_->fetchId();
  45         }
  46 
  47         Pairing next_game;
  48         next_game.round_id   = current_round;
  49         next_game.game_id    = ++game_counter;
  50         next_game.player1    = player1;
  51         next_game.player2    = player2;
  52         next_game.pairing_id = pair_counter;
  53         next_game.opening_id = opening;
  54 
  55         // Increment the games between the current player1 and player2
  56         games_per_pair++;
  57 
  58         // If we've exhausted the games between the current player pair, move to the next pair
  59         if (games_per_pair >= n_games_per_round) {
  60             games_per_pair = 0;
  61             player2++;
  62             pair_counter++;
  63 
  64             if (player2 >= n_players) {
  65                 player1++;
  66                 player2 = player1 + 1;
  67             }
  68 
  69             // If we've exhausted all pairs for this round, move to the next round
  70             if (player1 >= getPlayer1Limit()) {
  71                 current_round++;
  72                 player1 = 0;
  73                 player2 = 1;
  74             }
  75         }
  76 
  77         return next_game;
  78     }

  20     template <typename T>
  21     static void shuffle(T& vec) {
  22         for (std::size_t i = 0; i + 2 <= vec.size(); i++) {
  23             auto rand     = random::mersenne_rand();
  24             std::size_t j = i + (rand % (vec.size() - i));
