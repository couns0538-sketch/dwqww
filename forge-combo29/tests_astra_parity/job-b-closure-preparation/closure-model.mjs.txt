import assert from 'node:assert/strict';
export const pairLabels=['LL','LD','DD_or_WL','DW','WW'];
export function closureGate({now,deadline,games,maxGames=6000,activeNative=[],activeState=null,incompleteBatches=0,unreceiptedRaw=[]}){
 assert.ok(Number.isFinite(now)&&Number.isFinite(deadline));assert.ok(Number.isSafeInteger(games)&&games>=0&&games<=maxGames);
 const lastSafeLaunch=deadline-470000,targetMet=games===maxGames,budgetPreventsNextBatch=now>lastSafeLaunch;
 const blockers=[];if(activeNative.length)blockers.push('Native work remains active');if(activeState)blockers.push('Root active slot is occupied');if(incompleteBatches)blockers.push('Incomplete match batch remains recorded');if(unreceiptedRaw.length)blockers.push('Unreceipted native batch artifacts remain');if(!targetMet&&!budgetPreventsNextBatch)blockers.push('Target not reached and another bounded batch still fits');
 return {eligible:!blockers.length,targetMet,budgetPreventsNextBatch,lastSafeLaunchUTC:new Date(lastSafeLaunch).toISOString(),reason:targetMet?'6000-game target reached':budgetPreventsNextBatch?'Remaining declared budget cannot admit another full bounded batch':'Closure not yet eligible',blockers};
}
export function faultSummary(rows){
 const side={},unattributed={abortsOrGames:[],processEvidence:[]},gameFaults=[],batchFailures=[];
 for(const r of rows){for(const [name,counts]of Object.entries(r.sideAccounting.bySide)){const dest=side[name]??={};for(const [key,value]of Object.entries(counts)){assert.ok(Number.isSafeInteger(value)&&value>=0);dest[key]=(dest[key]??0)+value;}}for(const item of r.sideAccounting.unattributedAbortsOrGames)unattributed.abortsOrGames.push({batch:r.id,...item});for(const item of r.sideAccounting.unattributedProcessEvidence)unattributed.processEvidence.push({batch:r.id,...item});for(const item of r.sideAccounting.faultGames)gameFaults.push({batch:r.id,...item});if(!r.passed)batchFailures.push({batch:r.id,completed:r.completed,errors:r.errors,process:r.process,failureCounts:r.failureCounts});}
 return {bySide:side,gameFaults,unattributed,batchFailures,definition:'Crashes/signals, nonzero exits and disconnect games are retained separately; they are not summed as independent crashing games. Counts include raw observations even when a batch is excluded.'};
}

 '',`Stopped after the last recorded native batch at ${clock.lastNativeBatchFinishedUTC??'none'}. Declaration-to-last-native-stop elapsed time was ${clock.declarationToNativeStopSeconds===null?'not available':clock.declarationToNativeStopSeconds.toFixed(3)+' seconds'}. These are actual recorded durations; two continuous hours of play are not claimed.`,
 '',`Closure reason: ${c.gate.reason}. The last safe launch threshold was ${c.gate.lastSafeLaunchUTC}; each launch required at least470 seconds remaining.`,
 '',`Conditions: same binary; HindsightDepth true versus false; MateAlphaGuard false; Threads1, Hash16, OwnBook false; ${conditions.nodes} nodes per move; remaining options at advertised defaults.  '','| Configuration | Illegal games / events | Disconnect games | Stalls | Time forfeits | Nonzero process exits | Crash signals | Cleanup signals |','|---|---|---|---|---|---|---|---|'];

Closure does not clear that stop.`);if(p.asanShutdown0020)lines.push(`Separate ASan shutdown finding0020 paused native launches; its state at closure is ${p.asanShutdown0020.resolved?'resolved':'unresolved'}. Finding UTC: ${p.asanShutdown0020.utc}; fault evidence: ${p.asanShutdown0020.file}, SHA256 ${p.asanShutdown0020.sha256}.  lines.push('',`Aborted games: ${t.rawFailureCounts.abortedGames??0}; raw crash markers: ${t.rawFailureCounts.crashMarkers??0}; unattributed abort/game records: ${f.unattributed.abortsOrGames.length}; unattributed process records: ${f.unattributed.processEvidence.length}.  '',`Target: ${c.target.requestedGames} games; completed ${t.games}; shortfall ${c.target.uncompletedGames}. Optional250000-node leg: not run in this declared experiment. No result from that leg is claimed.`,
 '',`Job A was incomplete and its sanitizer work pending at this match's declaration, as explicitly authorized by0018. This closure changes only JobB status; it does not close sanitizer work or rewrite its findings.`,
 '',`Verified ${c.verification.receipts} receipts and ${c.verification.packages} immutable packages against source artifacts and reconstructed chunks. Package reconstruction bytes: ${c.verification.packagedBytes}. Full raw transcripts, PGNs, events, audits and faults remain linked by the closure index.`,
 '',`Counts only. No Elo, strength or speed conclusion is drawn.`, '');return lines.join('\n');
}
