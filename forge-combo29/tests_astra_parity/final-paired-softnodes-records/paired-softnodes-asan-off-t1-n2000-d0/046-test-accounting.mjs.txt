import assert from 'node:assert/strict';
import {derive,empty,totals,updatedState} from './accounting.mjs';
const batch={index:0,label:'synthetic-0',seed:123,mode:'asan',threads:1,nodes:2000,depth:0,HindsightAndGuard:false};
const row=(p,x)=>p+' '+JSON.stringify(x)+'\n';
function fixture({xs=[2500],fault=false,interrupted=false,truncated=false,label=false,index=0,ms=12}={}){
 const b={...batch,index,label:'synthetic-'+index};
 const usual={...empty(),games:1,searches:1,moves:1,mates:1};
 const soft={...empty(),games:1,searches:xs.length,moves:xs.length,mates:1,soft_x_count:xs.length,soft_x_sum:xs.reduce((a,x)=>a+x,0),soft_mean_x:xs.reduce((a,x)=>a+x,0)/xs.length,soft_hard_cap_reached:xs.filter(x=>x>=10000).length};
 const triggers=[{pair:0,mode:'usual',ply:8,fen:'synthetic',seed:123,threads:1},...xs.map((_,i)=>({pair:0,mode:'soft',ply:8+i,fen:'synthetic',seed:123,threads:1}))];
 const checks=triggers.map(t=>({...t,x:t.mode==='soft'?xs[t.ply-8]:null,item2:1,item3:1,item4:1}));
 const games=[{pair:0,mode:'usual',completed:true},{pair:0,mode:'soft',completed:true}],pairs=[{pair:0,completed:true}];
 const shutdowns=interrupted?[]:[{reaped:1,exit:0,signal:0,timed_out:0,driver_sigterm:0,driver_sigkill:0,quit_sent:1,quit_to_exit_ms:ms}];
 const faults=[];let observedSoft={...soft};
 if(fault||label){const t={pair:1,mode:'soft',ply:8,fen:'failure',seed:123,threads:1};triggers.push(t);checks.push({...t,x:3000,item2:1,item3:label?0:1,item4:1});observedSoft={...soft,searches:soft.searches+1,soft_x_count:soft.soft_x_count+1,soft_x_sum:soft.soft_x_sum+3000,soft_mean_x:(soft.soft_x_sum+3000)/(soft.soft_x_count+1)};if(!interrupted)faults.push({reason:label?'softnodes_item3_label_move':'illegal_move',label_fault:label,context:t});}
 const summary={pairs_completed:1,usual,soft,observed_usual_including_partial_pair:{...usual},observed_soft_including_partial_pair:observedSoft,fault:fault||label?'synthetic_fault':'',label_fault:label,diagnostic_observed:false,unassigned_soft_lines:0};
 const raw=row('CONFIG',{first_pair:0})+triggers.map(t=>row('TRIGGER',t)).join('')+checks.map(c=>row('CHECK',c)).join('')+games.map(g=>row('GAME',g)).join('')+pairs.map(p=>row('PAIR',p)).join('')+shutdowns.map(s=>row('SHUTDOWN',s)).join('')+faults.map(f=>row('FAULT',f)).join('')+(interrupted?'':row('SUMMARY',summary))+(truncated?'CHECK {"pair":2':'');
 const audit={index,label:b.label,exit:interrupted?124:fault||label?1:0,signal:null,complete:!interrupted,passed:!interrupted&&!fault&&!label,coverage:!interrupted,diagnostics:[],faults,summary:interrupted?null:summary,shutdowns,triggerRows:triggers.length,checkRows:checks.length,...(truncated?{auditUnavailable:true}:{})};
 return {b,audit,raw};
}
let n=0;function test(name,fn){fn();n++;console.log('PASS '+name);}
const run=f=>derive(f.b,f.audit,f.raw);
test('clean complete pair accepted once',()=>{const r=run(fixture());assert.equal(r.credit.pairs,1);assert.equal(r.completedPairObservations.pairs,1);assert.equal(totals([r]).games,2);assert.equal(totals([r]).faultBatches,0);});
test('faulted finalized pair retained outside clean coverage',()=>{const r=run(fixture({fault:true})),t=totals([r]);assert.equal(r.credit.pairs,0);assert.equal(r.completedPairObservations.pairs,1);assert.equal(t.games,0);assert.equal(t.completedPairObservations.soft.games,1);assert.equal(t.observedIncludingPartialPair.soft.searches,2);assert.equal(t.faultBatches,1);});
test('interrupted pair rows are observations only',()=>{const r=run(fixture({interrupted:true})),t=totals([r]);assert.equal(r.credit.pairs,0);assert.equal(r.completedPairObservations.pairs,0);assert.equal(t.games,0);assert.equal(t.observedIncludingPartialPair.soft.games,1);assert.equal(t.externalTimeoutBatches,1);assert.equal(t.faultBatches,0);});
test('interrupted item3 failure without summary blocks engine work',()=>{const r=run(fixture({interrupted:true,label:true})),t=totals([r]);assert.equal(r.labelFault,true);assert.equal(r.failingLabelChecks.length,1);assert.equal(r.labelFaultContexts[0].fen,'failure');assert.equal(t.blockFurtherEngineWork,true);assert.equal(t.games,0);});
test('truncated final JSON kept as unavailable audit with zero credit',()=>{const r=run(fixture({interrupted:true,truncated:true}));assert.equal(r.auditUnavailable,true);assert.equal(r.parseIssues.length,1);assert.equal(r.credit.pairs,0);assert.equal(r.rawRows.checks,2);});
test('unequal samples use total X divided by total count',()=>{const a=run(fixture({xs:[2500],index:0})),b=run(fixture({xs:[10000,15000,20000],index:1})),t=totals([a,b]);assert.equal(t.softXSum,47500);assert.equal(t.softXCount,4);assert.equal(t.softMeanX,11875);assert.equal(t.softHardCaps,3);});
test('duplicate index cannot double count',()=>{const r=run(fixture());assert.throws(()=>totals([r,r]),/Duplicate/);const proof={...r,evidenceIdentity:'abc'},s={preliminarySoftnodes:{games:7},soakTotals:{preliminaryStep1:{games:393}},sourceCommits:[{commit:'x'}],active:{batch:'unrelated'},pushes:13};const out=updatedState(s,proof);assert.equal(out.state.soakTotals.finalTree.games,2);assert.deepEqual(out.state.preliminarySoftnodes,s.preliminarySoftnodes);assert.deepEqual(out.state.active,s.active);assert.equal(s.finalPairedSoftnodesRows,undefined);assert.equal(updatedState(out.state,proof).duplicate,true);assert.throws(()=>updatedState(out.state,{...proof,evidenceIdentity:'changed'}),/Conflicting duplicate/);});
test('shutdown over100ms counts once with its actual forced status',()=>{const r=run(fixture({ms:150})),t=totals([r]);assert.equal(t.shutdownTiming.over100ms,1);assert.equal(t.shutdownTiming.maxMs,150);assert.equal(t.shutdownTiming.groups['forced=false:batchComplete=true'].measured,1);});
test('tampered soft X sum rejected',()=>{const f=fixture();f.audit.summary.soft.soft_x_sum++;f.raw=f.raw.replace('"soft_x_sum":2500','"soft_x_sum":2501');assert.throws(()=>run(f));});
test('unknown or duplicate complete pair records rejected',()=>{const f=fixture();f.raw=f.raw.replace('PAIR {"pair":0','PAIR {"pair":4');assert.throws(()=>run(f));});
test('negative soft X rejected',()=>{const f=fixture({xs:[-1]});assert.throws(()=>run(f));});
console.log(JSON.stringify({syntheticCases:n,passed:n,engineRuns:0,stateWrites:0,gitWrites:0}));
