+// Bonus earned by a move that produced a cutoff at `depth` (tunable shape).

+// History gravity: entry += bonus - entry * |bonus| / HistoryMax, with the
+// bonus clamped to +/-HistoryMax, so entries stay inside +/-HistoryMax.
+inline void apply_history(std::int16_t& entry, int bonus) {
+    const int clamped = std::clamp(bonus, -HistoryMax, HistoryMax);
+    entry = static_cast<std::int16_t>(entry + clamped - entry * std::abs(clamped) / HistoryMax);
+}

+#pragma once

+// v2 networks (C4): king-bucketed input layer, loaded from a file at run time.
+// Used only when UCI NNUEKingBuckets > 0 and a matching file is loaded with UCI
+// NNUEFile (forge/search_options.hpp); with neither option set the compiled-in
+// network (forge/nnue.hpp) is used and nothing here runs. A request that cannot
+// be honoured is an error, never a silent fall-back to the compiled-in network
+// (see options::nnue_v2_problem and src/uci.cpp).
+//
+// Feature of piece `pc` on `square` for perspective p (own king on `ksq`):
+//   r      = p == White ? square : square ^ 56       (rank flip, as forge/nnue.hpp)
+//   kr     = p == White ? ksq : ksq ^ 56
+//   flip   = (mirror flag && file_of(kr) >= 4) ? 7 : 0   (own king on files e-h)
+//   bucket = king_bucket_map[kr]                     (64 entries, indexed before mirroring)
+//   index  = bucket * 768 + ((colour(pc) == p ? 0 : 1) * 6 + type(pc) - 1) * 64 + (r ^ flip)
+// With one bucket and no mirroring this is nnue::feature_index(). The output
+// layer has the compiled-in architecture (SCReLU of the side-to-move and the
+// opponent accumulators, one linear output) with the file's w2/b2, and the
+// arithmetic of nnue::evaluate().
+//
+// File format v2 (little-endian), written by tools/nnue/forge_nnue_v2.py:
+//   0   char[8] "FORGENN2"      8  u32 version = 2        12 u32 header_bytes = 160
+//   16  u32 feature_set         20 u32 inputs = 768       24 u32 hidden (= HiddenSize)
+//   28  u32 king_buckets (1-32) 32 u32 output_buckets (1) 36 u32 flags (bit0 residual, bit1 mirror)
+//   40  f32 weight_scale 4096   44 f32 output_scale 4096  48 f32 sigmoid_k 2.25
+//   52  u8 king_bucket_map[64]  116 u8 output_bucket_map[33] (all 0)   149 u8 reserved[11] = 0
+//   160 i16 w1[king_buckets][768][hidden], [i16 wt[ThreatFeatures][hidden] when feature_set bit 0],
+//       [i16 wp[PawnPairFeatures][hidden] when feature_set bit 1],
+//       i16 b1[hidden], i16 w2[output_buckets][2*hidden], i16 b2[output_buckets],
+//       then u64 FNV-1a 64 of every preceding byte.
+// feature_set: 0 = piece-square inputs only (C4); bit 0 = threat inputs (O6, UCI NNUEThreats); bit 1 = pawn-pair
+// inputs (O7, UCI NNUEPawnPairs); definitions below. An engine that does not know a bit rejects the file.
+//
+// Exactness: weights are de-quantised like the compiled-in net (int16 / 4096),
+// so every accumulator value is an integer multiple of 2^-12 below 2^12 in
+// magnitude and every float addition/subtraction on it is exact. Incremental
+// updates, the refresh cache and a full refresh therefore give bit-identical
+// accumulators whatever the order of the terms.
+//
+// Layout discipline: everything here is inline (an out-of-line copy is a COMDAT
+// section) and the search code that uses it lives in .text.forge_v2, so the
+// default build's .text is unchanged (see src/search/evaluation_v2.inc).
+
+#include "forge/nnue.hpp"
+
+#include <algorithm>
+#include <array>
+#include <cstddef>
+#include <cstdint>
+#include <memory>
+#include <new>
+#include <string>
+#include <vector>

+#if defined(__SSE__) || defined(_M_X64) || defined(_M_AMD64)
+#include <xmmintrin.h>
+#define FORGE_NNUE_V2_SSE 1
+#else
+#define FORGE_NNUE_V2_SSE 0
+#endif
+#if defined(__AVX2__)
+#include <immintrin.h>
+#endif
+
+namespace forge::nnue::v2 {
+
+constexpr int MaxKingBuckets = 32;
+constexpr int MaxOutputBuckets = 1;   // reserved field of the format
+constexpr int MaxWorkers = 256;       // refresh caches are per worker id (Threads max)
+constexpr int HeaderBytes = 160;
+
+// Threat inputs (O6; header feature_set bit 0). One feature per pair (attacker A on s, victim V on v) where A
+// attacks the occupied square v in the position: pawn capture squares, knight and king steps, bishop / rook /
+// queen rays up to and including the first piece. Not features (redundant): a king "attacking" the other king
+// (impossible in a legal position) and any piece attacking its own side's king (a king is never captured, so
+// defending it carries no exchange information). Feature of a pair for perspective p with king key k:
+//   rel(piece) = (colour(piece) == p ? 0 : 6) + type(piece) - 1
+//   index      = (rel(A) * 12 + rel(V)) * 64 + ((p == White ? v : v ^ 56) ^ k.flip)
+// The block wt[ThreatFeatures][hidden] is shared by all king buckets. A victim attacked by two pieces of the
+// same kind adds the column twice (the input is the pair count, not a 0/1 flag).
+constexpr int ThreatFeatures = 12 * 12 * 64;
+constexpr int MaxThreatPairs = 512;   // at most 16 attackers per occupied square (8 rays + 8 knight squares) x 32 pieces
+constexpr std::uint32_t FeatureSetThreats = 1U;
+
+// Pawn-pair inputs (O7; header feature_set bit 1). One feature per unordered pair of pawns (any colours) on the
+// same or adjacent files at most two ranks apart, described in the perspective's frame (relative square
+// r = (p == White ? s : s ^ 56) ^ k.flip, colours own / their): the anchor is the pawn on the lower relative rank
+// (on the same rank: the one on the lower file), offset o of the other pawn:
+//   o: 0 (same rank, file +1)  1 (rank +1, file -1)  2 (rank +1, same file)  3 (rank +1, file +1)
+//      4 (rank +2, file -1)    5 (rank +2, same file) 6 (rank +2, file +1)
+//   index = ((colour(anchor) * 2 + colour(other)) * 7 + o) * 64 + r(anchor)      colour: 0 own, 1 their
+// This covers phalanxes, chains (supported pawns), levers, rams, doubled pawns and pawns two squares apart.
+// One block wp[PawnPairFeatures][hidden] shared by all king buckets, after the threat block (if any).
+constexpr int PawnPairOffsets = 7;
+constexpr int PawnPairFeatures = 4 * PawnPairOffsets * 64;
+constexpr int MaxPawnPairs = 16 * PawnPairOffsets;   // a pawn anchors at most 7 pairs
+constexpr std::uint32_t FeatureSetPawnPairs = 2U;
+// The threat pairs of one placement (O6; threat_keys).
+struct ThreatList {
+    int count = 0;
+    std::array<std::uint16_t, MaxThreatPairs> keys;   // pair keys ((A - 1) * 12 + (V - 1)) * 64 + v (absolute)
+};
+
+// The pawn pairs of one placement for one perspective (O7; pawn_pair_features).
+struct PawnPairList {
+    int count = 0;
+    std::array<std::uint16_t, MaxPawnPairs> features;   // feature indices of one perspective (with its flip)
+};
+
+struct KingKey {
+    int bucket = 0;
+    int flip = 0;     // 0, or 7 when the perspective's king stands on files e-h of a mirrored net
+    [[nodiscard]] bool operator==(const KingKey&) const = default;
+};
+
+// 64-byte aligned float array.
+class AlignedFloats {
+public:
+    AlignedFloats() = default;
+    AlignedFloats(const AlignedFloats&) = delete;
+    AlignedFloats& operator=(const AlignedFloats&) = delete;
+    ~AlignedFloats() { release(); }
+    void allocate(std::size_t count) {
+        release();
+        data_ = static_cast<float*>(::operator new(count * sizeof(float), std::align_val_t{64}));
+        size_ = count;
+        std::fill(data_, data_ + count, 0.0F);
+    }
+    [[nodiscard]] float* data() const { return data_; }
+    [[nodiscard]] std::size_t size() const { return size_; }
+
+private:
+    void release() {
+        if (data_ != nullptr) ::operator delete(data_, std::align_val_t{64});
+        data_ = nullptr;
+        size_ = 0;
+    }
+    float* data_ = nullptr;
+    std::size_t size_ = 0;
+};
+
+// Piece bitboards indexed by Piece code (index 0 unused).
+using Pieces = std::array<Bitboard, 13>;
+
+// Per-worker accumulator cache: for each (perspective, bucket, flip) the
+// accumulator of the last piece placement refreshed under that key (piece-square
+// inputs only; threat inputs are added on top, see refresh_full_cached).
+struct RefreshCache {
+    struct Entry {
+        alignas(64) std::array<float, HiddenSize> values;
+        Pieces pieces;
+    };
+    std::vector<Entry> entries;   // index ((perspective * king_buckets + bucket) * 2 + (flip != 0))
+    // O6: pair-count differences of two threat lists (all zero between uses; threat_delta).
+    std::array<std::int8_t, ThreatFeatures> threat_counts{};
+    // O7: the same for two pawn-pair lists of one perspective.
+    std::array<std::int8_t, PawnPairFeatures> pawn_counts{};
+    // O7: the pawn-pair lists (White, Black perspective) of each search-stack entry whose accumulator the extra-input
+    // path computed (index ply + 4); written together with that accumulator, like ply_threats.
+    std::vector<std::array<PawnPairList, 2>> ply_pawns;
+    // O6: the threat list of the placement of each search-stack entry whose accumulator the threat path computed
+    // (index ply + 4); written together with that accumulator, so a valid accumulator always has its list here.
+    std::vector<ThreatList> ply_threats;
+};
+
+struct Net {
+    std::string path;
+    int king_buckets = 1;
+    int output_buckets = 1;
+    bool mirror = false;
+    bool residual = false;
+    bool threats = false;   // O6: feature_set bit 0
+    bool pawn_pairs = false;   // O7: feature_set bit 1
+    std::uint64_t checksum = 0;
+    std::array<std::uint8_t, 64> king_map{};
+    std::array<std::uint8_t, 33> output_map{};
+    AlignedFloats w1;   // [king_buckets][InputCount][HiddenSize]
+    AlignedFloats wt;   // [ThreatFeatures][HiddenSize] (threat nets only)
+    AlignedFloats wp;   // [PawnPairFeatures][HiddenSize] (pawn-pair nets only)
+    AlignedFloats b1;   // [HiddenSize]
+    AlignedFloats w2;   // [output_buckets][2 * HiddenSize]
+    std::vector<float> b2;   // [output_buckets]
+    // Per worker id; slot i is created and used only by the worker with id i.
+    mutable std::array<std::unique_ptr<RefreshCache>, MaxWorkers> caches;
+};
+
+struct Description {

+const Net* select(int king_buckets, std::string* message);
+[[nodiscard]] Description describe(const Net& net);
+// Allocates the refresh cache of one worker (on first use).
+[[nodiscard]] RefreshCache& make_cache(const Net& net, int worker_id);
+
+// The net the search evaluates with (nullptr: the compiled-in network).
+extern const Net* active_net;
+[[nodiscard]] inline const Net* active() { return active_net; }
+
+// ---- evaluation kernels ----
+[[nodiscard]] inline KingKey king_key(const Net& net, Color perspective, int king_square) {
+    const int relative = perspective == White ? king_square : king_square ^ 56;
+    return {net.king_map[static_cast<std::size_t>(relative)], net.mirror && (relative & 7) >= 4 ? 7 : 0};
+}
+
+[[nodiscard]] inline int feature(Color perspective, KingKey key, Piece piece, int square) {
+    const int relative_color = color_of(piece) == perspective ? 0 : 1;
+    const int relative_square = (perspective == White ? square : square ^ 56) ^ key.flip;
+    return key.bucket * InputCount + (relative_color * 6 + piece_type(piece) - 1) * 64 + relative_square;
+}
+
+[[nodiscard]] inline const float* column(const Net& net, Color perspective, KingKey key, Piece piece, int square) {
+    return net.w1.data() + static_cast<std::size_t>(feature(perspective, key, piece, square)) * HiddenSize;
+}
+
+[[nodiscard]] inline Pieces pieces_of(const Board& board) {
+    Pieces pieces{};
+    for (int code = WP; code <= BK; ++code) pieces[static_cast<std::size_t>(code)] = board.pieces(static_cast<Piece>(code));
+    return pieces;
+}
+
+// Takes back the piece changes of one move (`dirty`) on a piece placement.
+inline void undo(Pieces& pieces, const DirtyPieces& dirty) {
+    for (int i = 0; i < dirty.count; ++i) {
+        const PieceChange& change = dirty.changes[static_cast<std::size_t>(i)];
+        Bitboard& bb = pieces[static_cast<std::size_t>(change.piece)];
+        if (change.add) bb &= ~square_bb(change.square);
+        else bb |= square_bb(change.square);
+    }
+}
+
+// O6: applies the piece changes of one move (`dirty`) to a piece placement (the inverse of undo).
+inline void redo(Pieces& pieces, const DirtyPieces& dirty) {
+    for (int i = 0; i < dirty.count; ++i) {
+        const PieceChange& change = dirty.changes[static_cast<std::size_t>(i)];
+        Bitboard& bb = pieces[static_cast<std::size_t>(change.piece)];
+        if (change.add) bb |= square_bb(change.square);
+        else bb &= ~square_bb(change.square);
+    }
+}
+
+// ---- O6 threat inputs (definition at ThreatFeatures; ThreatList above) ----
+
+[[nodiscard]] inline Bitboard attacks_from(int code, int square, Bitboard occupied) {
+    switch (code) {
+    case WP: return attacks::pawn(White, square);
+    case BP: return attacks::pawn(Black, square);
+    case WN: case BN: return attacks::knight(square);
+    case WB: case BB: return attacks::bishop(square, occupied);
+    case WR: case BR: return attacks::rook(square, occupied);
+    case WQ: case BQ: return attacks::queen(square, occupied);
+    default: return attacks::king(square);
+    }
+}
+
+// Every threat pair of a piece placement (attackers in piece-code order, then by square).
+inline void threat_keys(const Pieces& pieces, ThreatList& list) {
+    std::array<std::uint8_t, 64> mailbox{};
+    Bitboard occupied = 0;
+    for (int code = WP; code <= BK; ++code) {
+        Bitboard bb = pieces[static_cast<std::size_t>(code)];
+        occupied |= bb;
+        while (bb != 0) mailbox[static_cast<std::size_t>(pop_lsb(bb))] = static_cast<std::uint8_t>(code);
+    }
+    int n = 0;
+    for (int a = WP; a <= BK; ++a) {
+        const bool a_king = a == WK || a == BK;
+        const bool a_white = a <= WK;
+        Bitboard bb = pieces[static_cast<std::size_t>(a)];
+        while (bb != 0) {
+            const int s = pop_lsb(bb);
+            Bitboard targets = attacks_from(a, s, occupied) & occupied;
+            while (targets != 0) {
+                const int v = pop_lsb(targets);
+                const int b = mailbox[static_cast<std::size_t>(v)];
+                if ((b == WK || b == BK) && (a_king || (b <= WK) == a_white)) continue;   // redundant pairs
+                list.keys[static_cast<std::size_t>(n++)] = static_cast<std::uint16_t>(((a - 1) * 12 + (b - 1)) * 64 + v);
+            }
+        }
+    }
+    list.count = n;
+}
+
+// rel(A) * 12 + rel(V) for the Black perspective, from the absolute (A - 1) * 12 + (V - 1).
+inline constexpr auto BlackThreatKinds = [] {
+    std::array<std::uint8_t, 144> kinds{};
+    for (int a = 0; a < 12; ++a) {
+        for (int b = 0; b < 12; ++b) kinds[static_cast<std::size_t>(a * 12 + b)] = static_cast<std::uint8_t>(((a + 6) % 12) * 12 + (b + 6) % 12);
+    }
+    return kinds;
+}();
+
+[[nodiscard]] inline int threat_feature(Color perspective, KingKey key, int pair_key) {
+    const int square = pair_key & 63;
+    if (perspective == White) return (pair_key & ~63) | (square ^ key.flip);
+    return BlackThreatKinds[static_cast<std::size_t>(pair_key >> 6)] * 64 + ((square ^ 56) ^ key.flip);
+}
+
+[[nodiscard]] inline const float* threat_column(const Net& net, Color perspective, KingKey key, int pair_key) {
+    return net.wt.data() + static_cast<std::size_t>(threat_feature(perspective, key, pair_key)) * HiddenSize;
+}
+
+// Multiset difference after - before of two threat lists (counts: all zero on entry and on return).
+struct ThreatDelta {
+    int removed = 0;
+    int added = 0;
+    std::array<std::uint16_t, MaxThreatPairs> remove_keys;
+    std::array<std::uint16_t, MaxThreatPairs> add_keys;
+};
+inline void threat_delta(const ThreatList& before, const ThreatList& after, std::int8_t* counts, ThreatDelta& delta) {
+    for (int i = 0; i < before.count; ++i) --counts[before.keys[static_cast<std::size_t>(i)]];
+    for (int i = 0; i < after.count; ++i) ++counts[after.keys[static_cast<std::size_t>(i)]];
+    delta.removed = 0;
+    delta.added = 0;
+    for (int i = 0; i < before.count; ++i) {
+        const std::uint16_t k = before.keys[static_cast<std::size_t>(i)];
+        for (; counts[k] < 0; ++counts[k]) delta.remove_keys[static_cast<std::size_t>(delta.removed++)] = k;
+    }
+    for (int i = 0; i < after.count; ++i) {
+        const std::uint16_t k = after.keys[static_cast<std::size_t>(i)];
+        for (; counts[k] > 0; --counts[k]) delta.add_keys[static_cast<std::size_t>(delta.added++)] = k;
+    }
+}
+
+// dst = src - sub[0..subs) + add[0..adds) over the hidden units, in chunks of 64 floats (exact; any order of
+// the terms gives the same bits). src may be dst. Every pointer is 32-byte aligned (accumulators, weight columns).
+// With AVX2 the 64-float chunk stays in eight registers while all columns are applied (plain sub / add, no FMA).
+inline void apply_lists(const float* src, float* dst, const float* const* sub, int subs, const float* const* add, int adds) {
+    constexpr int Chunk = 64;
+    static_assert(HiddenSize % Chunk == 0, "HiddenSize must be a multiple of the chunk");
+#if defined(__AVX2__)
+    for (int h0 = 0; h0 < HiddenSize; h0 += Chunk) {
+        __m256 v0 = _mm256_load_ps(src + h0);
+        __m256 v1 = _mm256_load_ps(src + h0 + 8);
+        __m256 v2 = _mm256_load_ps(src + h0 + 16);
+        __m256 v3 = _mm256_load_ps(src + h0 + 24);
+        __m256 v4 = _mm256_load_ps(src + h0 + 32);
+        __m256 v5 = _mm256_load_ps(src + h0 + 40);
+        __m256 v6 = _mm256_load_ps(src + h0 + 48);
+        __m256 v7 = _mm256_load_ps(src + h0 + 56);
+        for (int i = 0; i < subs; ++i) {
+            const float* col = sub[i] + h0;
+            v0 = _mm256_sub_ps(v0, _mm256_load_ps(col));
+            v1 = _mm256_sub_ps(v1, _mm256_load_ps(col + 8));
+            v2 = _mm256_sub_ps(v2, _mm256_load_ps(col + 16));
+            v3 = _mm256_sub_ps(v3, _mm256_load_ps(col + 24));
+            v4 = _mm256_sub_ps(v4, _mm256_load_ps(col + 32));
+            v5 = _mm256_sub_ps(v5, _mm256_load_ps(col + 40));
+            v6 = _mm256_sub_ps(v6, _mm256_load_ps(col + 48));
+            v7 = _mm256_sub_ps(v7, _mm256_load_ps(col + 56));
+        }
+        for (int i = 0; i < adds; ++i) {
+            const float* col = add[i] + h0;
+            v0 = _mm256_add_ps(v0, _mm256_load_ps(col));
+            v1 = _mm256_add_ps(v1, _mm256_load_ps(col + 8));
+            v2 = _mm256_add_ps(v2, _mm256_load_ps(col + 16));
+            v3 = _mm256_add_ps(v3, _mm256_load_ps(col + 24));
+            v4 = _mm256_add_ps(v4, _mm256_load_ps(col + 32));
+            v5 = _mm256_add_ps(v5, _mm256_load_ps(col + 40));
+            v6 = _mm256_add_ps(v6, _mm256_load_ps(col + 48));
+            v7 = _mm256_add_ps(v7, _mm256_load_ps(col + 56));
+        }
+        _mm256_store_ps(dst + h0, v0);
+        _mm256_store_ps(dst + h0 + 8, v1);
+        _mm256_store_ps(dst + h0 + 16, v2);
+        _mm256_store_ps(dst + h0 + 24, v3);
+        _mm256_store_ps(dst + h0 + 32, v4);
+        _mm256_store_ps(dst + h0 + 40, v5);
+        _mm256_store_ps(dst + h0 + 48, v6);
+        _mm256_store_ps(dst + h0 + 56, v7);
+    }
+#else
+    for (int h0 = 0; h0 < HiddenSize; h0 += Chunk) {
+        float chunk[Chunk];
+        for (int k = 0; k < Chunk; ++k) chunk[k] = src[h0 + k];
+        for (int i = 0; i < subs; ++i) {
+            const float* col = sub[i] + h0;
+            for (int k = 0; k < Chunk; ++k) chunk[k] -= col[k];
+        }
+        for (int i = 0; i < adds; ++i) {
+            const float* col = add[i] + h0;
+            for (int k = 0; k < Chunk; ++k) chunk[k] += col[k];
+        }
+        for (int k = 0; k < Chunk; ++k) dst[h0 + k] = chunk[k];
+    }
+#endif
+}
+
+// values += every threat column of `list` for this perspective and king key.
+inline void add_threat_columns(const Net& net, Color perspective, KingKey key, const ThreatList& list, float* values) {
+    std::array<const float*, MaxThreatPairs> add;
+    for (int i = 0; i < list.count; ++i) add[static_cast<std::size_t>(i)] = threat_column(net, perspective, key, list.keys[static_cast<std::size_t>(i)]);
+    apply_lists(values, values, nullptr, 0, add.data(), list.count);
+}

+// ---- O7 pawn-pair inputs (definition at PawnPairFeatures) ----
+
+[[nodiscard]] inline Bitboard mirror_files(Bitboard bb) {   // square s -> s ^ 7
+    constexpr Bitboard k1 = 0x5555555555555555ULL;
+    constexpr Bitboard k2 = 0x3333333333333333ULL;
+    constexpr Bitboard k4 = 0x0F0F0F0F0F0F0F0FULL;
+    bb = ((bb >> 1U) & k1) | ((bb & k1) << 1U);
+    bb = ((bb >> 2U) & k2) | ((bb & k2) << 2U);
+    return ((bb >> 4U) & k4) | ((bb & k4) << 4U);
+}
+
+// A bitboard in the perspective's frame: square s -> (p == White ? s : s ^ 56) ^ flip.
+[[nodiscard]] inline Bitboard relative_bb(Bitboard bb, Color perspective, int flip) {
+    if (perspective == Black) bb = __builtin_bswap64(bb);
+    return flip != 0 ? mirror_files(bb) : bb;
+}
+
+// Every pawn pair of a placement for one perspective (its king key's flip).
+inline void pawn_pair_features(const Pieces& pieces, Color perspective, int flip, PawnPairList& list) {
+    struct Offset { int shift; Bitboard anchors; };   // other pawn = anchor + shift; anchors whose other square is on the board
+    constexpr std::array<Offset, PawnPairOffsets> offsets{{{1, ~FileHBB}, {7, ~FileABB}, {8, ~0ULL}, {9, ~FileHBB},
+                                                           {15, ~FileABB}, {16, ~0ULL}, {17, ~FileHBB}}};
+    const std::array<Bitboard, 2> side{relative_bb(pieces[static_cast<std::size_t>(perspective == White ? WP : BP)], perspective, flip),
+                                       relative_bb(pieces[static_cast<std::size_t>(perspective == White ? BP : WP)], perspective, flip)};
+    int n = 0;
+    for (int o = 0; o < PawnPairOffsets; ++o) {
+        const Offset& offset = offsets[static_cast<std::size_t>(o)];
+        for (int c1 = 0; c1 < 2; ++c1) {
+            for (int c2 = 0; c2 < 2; ++c2) {
+                Bitboard anchors = side[static_cast<std::size_t>(c1)] & offset.anchors &
+                                   (side[static_cast<std::size_t>(c2)] >> static_cast<unsigned>(offset.shift));
+                while (anchors != 0) {
+                    list.features[static_cast<std::size_t>(n++)] = static_cast<std::uint16_t>(((c1 * 2 + c2) * PawnPairOffsets + o) * 64 + pop_lsb(anchors));
+                }
+            }
+        }
+    }
+    list.count = n;
+}
+
+[[nodiscard]] inline const float* pawn_pair_column(const Net& net, int feature) {
+    return net.wp.data() + static_cast<std::size_t>(feature) * HiddenSize;
+}
+
+// Difference after - before of two feature lists of one perspective (counts: all zero on entry and on return).
+struct PawnPairDelta {
+    int removed = 0;
+    int added = 0;
+    std::array<std::uint16_t, MaxPawnPairs> remove_features;
+    std::array<std::uint16_t, MaxPawnPairs> add_features;
+};
+inline void pawn_pair_delta(const PawnPairList& before, const PawnPairList& after, std::int8_t* counts, PawnPairDelta& delta) {
+    for (int i = 0; i < before.count; ++i) --counts[before.features[static_cast<std::size_t>(i)]];
+    for (int i = 0; i < after.count; ++i) ++counts[after.features[static_cast<std::size_t>(i)]];
+    delta.removed = 0;
+    delta.added = 0;
+    for (int i = 0; i < before.count; ++i) {
+        const std::uint16_t f = before.features[static_cast<std::size_t>(i)];
+        for (; counts[f] < 0; ++counts[f]) delta.remove_features[static_cast<std::size_t>(delta.removed++)] = f;
+    }
+    for (int i = 0; i < after.count; ++i) {
+        const std::uint16_t f = after.features[static_cast<std::size_t>(i)];
+        for (; counts[f] > 0; --counts[f]) delta.add_features[static_cast<std::size_t>(delta.added++)] = f;
+    }
+}
+
+inline void add_pawn_pair_columns(const Net& net, const PawnPairList& list, float* values) {
+    std::array<const float*, MaxPawnPairs> add;
+    for (int i = 0; i < list.count; ++i) add[static_cast<std::size_t>(i)] = pawn_pair_column(net, list.features[static_cast<std::size_t>(i)]);
+    apply_lists(values, values, nullptr, 0, add.data(), list.count);
+}
+
+[[nodiscard]] inline bool touches_pawns(const DirtyPieces& dirty) {
+    for (int i = 0; i < dirty.count; ++i) {
+        const Piece piece = dirty.changes[static_cast<std::size_t>(i)].piece;
+        if (piece == WP || piece == BP) return true;
+    }
+    return false;
+}

+[[nodiscard]] inline RefreshCache& cache(const Net& net, int worker_id) {
+    const std::unique_ptr<RefreshCache>& slot = net.caches[static_cast<std::size_t>(std::clamp(worker_id, 0, MaxWorkers - 1))];
+    if (slot == nullptr) [[unlikely]] return make_cache(net, worker_id);
+    return *slot;
+}
+
+// dst = src - sub[0] - ... + add[0] + ... (exact; any order gives the same bits).
+template <int Subs, int Adds>
+inline void apply_columns(const float* src, float* dst, const float* const* sub, const float* const* add) {
+    for (int h = 0; h < HiddenSize; ++h) {
+        float value = src[h];
+        for (int i = 0; i < Subs; ++i) value -= sub[i][h];
+        for (int i = 0; i < Adds; ++i) value += add[i][h];
+        dst[h] = value;
+    }
+}

+// Accumulator of one perspective from scratch (reference implementation; with
+// the threat inputs of a threat net).
+inline void refresh(const Net& net, const Board& board, Color perspective, float* values) {
+    const KingKey key = king_key(net, perspective, board.king_square(perspective));
+    std::copy(net.b1.data(), net.b1.data() + HiddenSize, values);
+    Bitboard occupied = board.occupied();
+    while (occupied != 0) {
+        const int square = pop_lsb(occupied);
+        const float* col = column(net, perspective, key, board.piece_at(square), square);
+        for (int h = 0; h < HiddenSize; ++h) values[h] += col[h];
+    }
+    if (net.threats) {
+        ThreatList list;
+        threat_keys(pieces_of(board), list);
+        for (int i = 0; i < list.count; ++i) {
+            const float* col = threat_column(net, perspective, key, list.keys[static_cast<std::size_t>(i)]);
+            for (int h = 0; h < HiddenSize; ++h) values[h] += col[h];
+        }
+    }
+    if (net.pawn_pairs) {   // O7
+        PawnPairList list;
+        pawn_pair_features(pieces_of(board), perspective, key.flip, list);
+        for (int i = 0; i < list.count; ++i) {
+            const float* col = pawn_pair_column(net, list.features[static_cast<std::size_t>(i)]);
+            for (int h = 0; h < HiddenSize; ++h) values[h] += col[h];
+        }
+    }
+}
+
+// Accumulator of one perspective for the placement `pieces` (own king on
+// `king_square`) through the cache: only the difference to the cached
+// placement under the same key is applied; the entry is updated.
+inline void refresh_cached(const Net& net, RefreshCache& cache, const Pieces& pieces, int king_square,
+                           Color perspective, float* values) {
+    const KingKey key = king_key(net, perspective, king_square);
+    RefreshCache::Entry& entry = cache.entries[(static_cast<std::size_t>(perspective) * static_cast<std::size_t>(net.king_buckets) +
+                                                static_cast<std::size_t>(key.bucket)) * 2U + (key.flip != 0 ? 1U : 0U)];
+    // At most one removal and one addition per square.
+    std::array<const float*, 64> sub;
+    std::array<const float*, 64> add;
+    int subs = 0;
+    int adds = 0;
+    for (int code = WP; code <= BK; ++code) {
+        const Piece piece = static_cast<Piece>(code);
+        const Bitboard now = pieces[static_cast<std::size_t>(code)];
+        Bitboard removed = entry.pieces[static_cast<std::size_t>(code)] & ~now;
+        Bitboard added = now & ~entry.pieces[static_cast<std::size_t>(code)];
+        while (removed != 0) sub[static_cast<std::size_t>(subs++)] = column(net, perspective, key, piece, pop_lsb(removed));
+        while (added != 0) add[static_cast<std::size_t>(adds++)] = column(net, perspective, key, piece, pop_lsb(added));
+        entry.pieces[static_cast<std::size_t>(code)] = now;
+    }
+    constexpr int Chunk = 64;
+    static_assert(HiddenSize % Chunk == 0, "HiddenSize must be a multiple of the refresh chunk");
+    for (int h0 = 0; h0 < HiddenSize; h0 += Chunk) {
+        float chunk[Chunk];
+        for (int k = 0; k < Chunk; ++k) chunk[k] = entry.values[static_cast<std::size_t>(h0 + k)];
+        for (int i = 0; i < subs; ++i) {
+            const float* col = sub[static_cast<std::size_t>(i)] + h0;
+            for (int k = 0; k < Chunk; ++k) chunk[k] -= col[k];
+        }
+        for (int i = 0; i < adds; ++i) {
+            const float* col = add[static_cast<std::size_t>(i)] + h0;
+            for (int k = 0; k < Chunk; ++k) chunk[k] += col[k];
+        }
+        for (int k = 0; k < Chunk; ++k) {
+            entry.values[static_cast<std::size_t>(h0 + k)] = chunk[k];
+            values[h0 + k] = chunk[k];
+        }
+    }
+}
+
+// O6: refresh_cached plus, for a threat net, the threat columns of `threats` (the
+// threat list of the same placement); O7: plus, for a pawn-pair net, the pawn-pair
+// columns of this perspective (the list is also stored in `pawn_list` if given).
+// The cache entries hold piece-square inputs only.
+inline void refresh_full_cached(const Net& net, RefreshCache& cache, const Pieces& pieces, const ThreatList* threats,
+                                int king_square, Color perspective, float* values, PawnPairList* pawn_list = nullptr) {
+    refresh_cached(net, cache, pieces, king_square, perspective, values);
+    const KingKey key = king_key(net, perspective, king_square);
+    if (threats != nullptr) add_threat_columns(net, perspective, key, *threats, values);
+    if (net.pawn_pairs) {
+        PawnPairList local;
+        PawnPairList& list = pawn_list != nullptr ? *pawn_list : local;
+        pawn_pair_features(pieces, perspective, key.flip, list);
+        add_pawn_pair_columns(net, list, values);
+    }
+}
+
+// Incremental update of one perspective whose king key did not change.
+inline void update(const Net& net, Color perspective, KingKey key, const float* parent, float* child,
+                   const DirtyPieces& dirty) {
+    std::array<const float*, 4> sub{};
+    std::array<const float*, 4> add{};
+    int subs = 0;
+    int adds = 0;
+    for (int i = 0; i < dirty.count; ++i) {
+        const PieceChange& change = dirty.changes[static_cast<std::size_t>(i)];
+        const float* col = column(net, perspective, key, change.piece, change.square);
+        if (change.add) add[static_cast<std::size_t>(adds++)] = col;
+        else sub[static_cast<std::size_t>(subs++)] = col;
+    }
+    if (subs == 1 && adds == 1) apply_columns<1, 1>(parent, child, sub.data(), add.data());
+    else if (subs == 2 && adds == 1) apply_columns<2, 1>(parent, child, sub.data(), add.data());
+    else if (subs == 2 && adds == 2) apply_columns<2, 2>(parent, child, sub.data(), add.data());
+    else {
+        apply_columns<0, 0>(parent, child, sub.data(), add.data());
+        for (int i = 0; i < subs; ++i) apply_columns<1, 0>(child, child, &sub[static_cast<std::size_t>(i)], add.data());
+        for (int i = 0; i < adds; ++i) apply_columns<0, 1>(child, child, sub.data(), &add[static_cast<std::size_t>(i)]);
+    }
+}

+// Network output before the conversion to centipawns; nnue::evaluate()'s arithmetic.
+[[nodiscard]] inline float output(const Net& net, const Board& board, const Accumulator& accumulator) {
+    const Color us = board.side_to_move();
+    const auto& ours = accumulator.values[static_cast<std::size_t>(us)];
+    const auto& theirs = accumulator.values[static_cast<std::size_t>(opposite(us))];
+    const float* w2 = net.w2.data();
+    constexpr int Lanes = 16;
+    static_assert(HiddenSize % Lanes == 0, "HiddenSize must be a multiple of the lane count");
+    alignas(64) float lanes[Lanes];
+#if defined(__AVX2__)

+        }
+    }
+#endif
+    float out = net.b2[0];
+    for (int j = 0; j < Lanes; ++j) out += lanes[j];

+    nodes_.store(local_nodes_, std::memory_order_relaxed);
+    if (shared_.stop.load(std::memory_order_relaxed)) return true;
+    if (!is_main()) return false;
+    if (node_limit_specified_ && total_nodes() >= node_limit_) return true;
+    if (hard_limit_ms_ > 0) {
+        if (pondering_ && shared_.ponder_hit_ns.load(std::memory_order_relaxed) == 0) return false;
+        if (elapsed_ms() >= hard_limit_ms_) return true;
+    }
+    return false;
+}

+    (void)ply;

+        return !board.in_check() || board.has_legal_move();

+// Hash of the piece counts of both sides, for material-indexed statistics.
+[[nodiscard]] std::uint64_t material_key(const Board& board) {
+    std::uint64_t key = 0;
+    for (const Color color : {White, Black}) {
+        for (int type = Pawn; type <= Queen; ++type) {
+            key = key * 1099511628211ULL + static_cast<std::uint64_t>(popcount(board.pieces(color, static_cast<PieceType>(type))) + 1);
+        }
+    }
+    return key * 0x9E3779B97F4A7C15ULL;

+int Worker::static_evaluate(const Board& board, SearchStack* ss) {
+    if (options::nnue_v2) [[unlikely]] {
+        // Layout pad: with it the dispatch grows this function by exactly 64
+        // bytes in the PC build (MinGW g++ 15.2), so the search functions
+        // emitted after it keep their AL1 s198 addresses mod 64.
+        FORGE_LAYOUT_PAD(32);
+        return static_evaluate_v2(board, ss);
+    }
+    if (!nnue::enabled || !nnue::available()) return evaluator_.evaluate(board);
+    const bool residual = nnue::residual();
+    if (!ss->accumulator.valid) {
+        // Walk back to the nearest valid accumulator, then replay the moves.
+        int back = 0;
+        while (back < ss->ply && !(ss - back - 1)->accumulator.valid) ++back;
+        if (back >= ss->ply && !(ss - back - 1)->accumulator.valid) {
+            nnue::refresh(board, ss->accumulator);
+            if (residual) ss->runtime_psqt = Evaluator::refresh_runtime_psqt(board);
+        } else {
+            for (int i = back; i >= 0; --i) {
+                nnue::update((ss - i - 1)->accumulator, (ss - i)->accumulator, (ss - i)->dirty);
+                if (residual) Evaluator::update_runtime_psqt((ss - i - 1)->runtime_psqt, (ss - i)->runtime_psqt, (ss - i)->dirty);
+            }
+        }
+    }
+    const int network = nnue::evaluate(board, ss->accumulator);
+    if (!residual) return network;
+    if (!ss->runtime_psqt.valid) ss->runtime_psqt = Evaluator::refresh_runtime_psqt(board);
+    // A residual network is a learned correction on top of the hand-crafted evaluation.
+    return std::clamp(evaluator_.evaluate_runtime(board, ss->runtime_psqt) + network, -MateInMaxPly + 1, MateInMaxPly - 1);
+}

+int Worker::corrected_eval(const Board& board, int raw_eval) const {
+    const std::size_t side = static_cast<std::size_t>(board.side_to_move());

+void Worker::update_correction(const Board& board, int depth, int best_score, int static_eval) {
+    const std::size_t side = static_cast<std::size_t>(board.side_to_move());

+    auto apply = [&](std::int32_t& entry) {
+        entry = std::clamp(entry + diff * weight / 256, -CorrectionLimit, CorrectionLimit);
+    };

+    if (tune::MaterialCorrectionWeight > 0) apply((*material_correction_)[side][material_key(board) & (CorrectionSize - 1U)]);
+}

+// NNUEThreats) and / or pawn-pair inputs (O7; UCI NNUEPawnPairs), both OFF by default; the feature definitions are
+// in forge/nnue_v2.hpp at ThreatFeatures and PawnPairFeatures. Compiled as part of src/search.cpp (included exactly
+// once, before evaluation_v2.inc, whose static_evaluate_v2 calls it: GCC emits a caller after its callees). The
+// function lives in section .text.forge_v2 like the rest of the v2 code, so the default build's .text is unchanged.
+//
+// Reached only through static_evaluate_v2 when the active net has an extra block. Accumulators follow C4's lazy
+// scheme: an invalid ply walks back to the nearest valid ancestor and replays the recorded piece changes; every
+// replayed ply becomes valid. Per replayed ply, the child's piece placement is the parent's with that ply's piece
+// changes applied (starting from the board with every replayed ply taken back). Threat inputs: the child's threat
+// list is generated and the multiset difference to the parent's list is applied (each computed ply's list is kept
+// in the worker's cache, RefreshCache::ply_threats, so the valid ancestor's list is read). Pawn-pair inputs: each
+// perspective's pawn-pair list is kept per ply in the worker's cache (RefreshCache::ply_pawns) next to the accumulator;
+// it is regenerated only on a ply that moves,
+// captures or promotes a pawn (other plies cannot change pawn pairs), and the difference is applied. All changes of
+// a ply go through one chunked pass over each perspective's accumulator. On a king-key change the perspective is
+// refreshed: piece-square inputs through the refresh cache, then every threat / pawn-pair column of the child.
+// All weights are int16 / 4096, so every accumulator value is an exact multiple of 2^-12 and incremental, cached
+// and from-scratch accumulators are bit-identical (tests/nnue_threat_tests.cpp, tests/nnue_pawn_pair_tests.cpp).

+__attribute__((section(".text.forge_v2"), noinline)) int Worker::static_evaluate_extra(const Board& board, SearchStack* ss) {
+    const nnue::v2::Net& net = *nnue::v2::active();
+    const bool residual = net.residual;
+    const bool threats = net.threats;
+    const bool pawns = net.pawn_pairs;
+    if (!ss->accumulator.valid) {
+        nnue::v2::RefreshCache& cache = nnue::v2::cache(net, id_);

+        auto list_of = [&cache](const SearchStack* entry) -> nnue::v2::ThreatList& {
+            return cache.ply_threats[static_cast<std::size_t>(entry->ply + 4)];
+        };
+        if (pawns && cache.ply_pawns.empty()) [[unlikely]] cache.ply_pawns.resize(static_cast<std::size_t>(MaxPly + 8));
+        auto pawns_of = [&cache](const SearchStack* entry) -> std::array<nnue::v2::PawnPairList, 2>& {
+            return cache.ply_pawns[static_cast<std::size_t>(entry->ply + 4)];
+        };

+        while (back < ss->ply && !(ss - back - 1)->accumulator.valid) ++back;
+        nnue::v2::Pieces pieces = nnue::v2::pieces_of(board);
+        if (back >= ss->ply && !(ss - back - 1)->accumulator.valid) {
+            if (threats) nnue::v2::threat_keys(pieces, list_of(ss));
+            for (const Color color : {White, Black}) {
+                const std::size_t c = static_cast<std::size_t>(color);
+                nnue::v2::refresh_full_cached(net, cache, pieces, threats ? &list_of(ss) : nullptr, board.king_square(color), color,
+                                              ss->accumulator.values[c].data(), pawns ? &pawns_of(ss)[c] : nullptr);
+            }
+            ss->accumulator.valid = true;
+            if (residual) ss->runtime_psqt = Evaluator::refresh_runtime_psqt(board);
+        } else {

+            auto king_of = [&pieces](Color color) { return lsb(pieces[static_cast<std::size_t>(make_piece(color, King))]); };
+            std::array<nnue::v2::KingKey, 2> key{nnue::v2::king_key(net, White, king_of(White)),

+                if (pawns && !pawn_move) pawns_of(child) = pawns_of(parent);

+                    int adds = 0;

+                        }
+                        for (int k = 0; k < delta.added; ++k) {
+                            add[static_cast<std::size_t>(adds++)] = nnue::v2::threat_column(net, color, key[c], delta.add_keys[static_cast<std::size_t>(k)]);
+                        }
+                    }
+                    if (pawn_move) {
+                        nnue::v2::PawnPairList& pawn_child = pawns_of(child)[c];
+                        nnue::v2::pawn_pair_features(pieces, color, key[c].flip, pawn_child);
+                        nnue::v2::pawn_pair_delta(pawns_of(parent)[c], pawn_child, cache.pawn_counts.data(), pawn_delta);
+                        for (int k = 0; k < pawn_delta.removed; ++k) {

+                    }

+            }

+// Search module: static evaluation with a v2 network (C4: king-bucketed input
+// layer; UCI NNUEFile + NNUEKingBuckets, OFF by default; forge/nnue_v2.hpp).
+// Compiled as part of src/search.cpp (included exactly once, before
+// evaluation.inc so that GCC keeps emitting static_evaluate at its AL1 s198
+// position: a caller is emitted after its callees).
+//
+// Reached only through the options::nnue_v2 branch at the top of
+// Worker::static_evaluate. This code lives in section .text.forge_v2 (or is
+// cold), so the default build's search functions keep their addresses.
+//
+// Accumulators reuse SearchStack::accumulator (a v2 net has the engine's
+// HiddenSize) with the lazy scheme of the compiled-in network: an invalid ply
+// walks back to the nearest valid ancestor and replays the recorded piece
+// changes, and every replayed ply becomes valid. New for king buckets: on a ply
+// where a king changes its own perspective's (bucket, flip) key, that
+// perspective is refreshed through the worker's refresh cache. The piece
+// placement of that ply is the board with the later plies' piece changes taken
+// back, so no board copy is kept per ply.
+
+#define FORGE_SEARCH_V2 __attribute__((section(".text.forge_v2"), noinline))

+namespace forge {

+FORGE_SEARCH_V2 int Worker::static_evaluate_v2(const Board& board, SearchStack* ss) {
+    // nnue_v2 is only set while a v2 net is active; the UCI layer re-selects
+    // after every change, so this fallback is defensive.
+    const nnue::v2::Net* active = nnue::v2::active();
+    if (active == nullptr || !nnue::enabled) return evaluator_.evaluate(board);
+    const nnue::v2::Net& net = *active;
+    // O6 / O7: a net with threat or pawn-pair inputs has its own update path (evaluation_extra.inc);
+    // apply_nnue_v2 activates one only with the matching NNUEThreats / NNUEPawnPairs.
+    if (net.threats || net.pawn_pairs) [[unlikely]] return static_evaluate_extra(board, ss);
+    const bool residual = net.residual;
+    if (!ss->accumulator.valid) {
+        int back = 0;
+        while (back < ss->ply && !(ss - back - 1)->accumulator.valid) ++back;
+        if (back >= ss->ply && !(ss - back - 1)->accumulator.valid) {
+            nnue::v2::RefreshCache& cache = nnue::v2::cache(net, id_);
+            const nnue::v2::Pieces pieces = nnue::v2::pieces_of(board);
+            nnue::v2::refresh_cached(net, cache, pieces, board.king_square(White), White, ss->accumulator.values[White].data());
+            nnue::v2::refresh_cached(net, cache, pieces, board.king_square(Black), Black, ss->accumulator.values[Black].data());
+            ss->accumulator.valid = true;
+            if (residual) ss->runtime_psqt = Evaluator::refresh_runtime_psqt(board);
+        } else {
+            // King squares of the ancestor position: take back the replayed king
+            // moves (changes[0] is always the moving piece, removed from `from`).
+            std::array<int, 2> king{board.king_square(White), board.king_square(Black)};
+            for (int i = 0; i <= back; ++i) {
+                const nnue::DirtyPieces& dirty = (ss - i)->dirty;
+                if (dirty.count > 0 && piece_type(dirty.changes[0].piece) == King) {
+                    king[static_cast<std::size_t>(color_of(dirty.changes[0].piece))] = dirty.changes[0].square;
+                }
+            }
+            std::array<nnue::v2::KingKey, 2> key{nnue::v2::king_key(net, White, king[0]),
+                                                 nnue::v2::king_key(net, Black, king[1])};
+            for (int i = back; i >= 0; --i) {
+                const SearchStack* parent = ss - i - 1;
+                SearchStack* child = ss - i;
+                const nnue::DirtyPieces& dirty = child->dirty;
+                for (const Color color : {White, Black}) {
+                    const std::size_t c = static_cast<std::size_t>(color);
+                    if (dirty.count > 0 && dirty.changes[0].piece == make_piece(color, King)) {
+                        // The king's destination: the added change of the same piece.
+                        int to = dirty.changes[0].square;
+                        for (int k = 1; k < dirty.count; ++k) {
+                            const nnue::PieceChange& change = dirty.changes[static_cast<std::size_t>(k)];
+                            if (change.add && change.piece == dirty.changes[0].piece) to = change.square;
+                        }
+                        const nnue::v2::KingKey moved = nnue::v2::king_key(net, color, to);
+                        if (!(moved == key[c])) [[unlikely]] {
+                            key[c] = moved;
+                            nnue::v2::Pieces pieces = nnue::v2::pieces_of(board);
+                            for (int k = 0; k < i; ++k) nnue::v2::undo(pieces, (ss - k)->dirty);
+                            nnue::v2::refresh_cached(net, nnue::v2::cache(net, id_), pieces, to, color,
+                                                     child->accumulator.values[c].data());

+                        }
+                    }

+                                     child->accumulator.values[c].data(), dirty);
+                }

+                if (residual) Evaluator::update_runtime_psqt(parent->runtime_psqt, child->runtime_psqt, dirty);
+            }
+        }

+    return std::clamp(evaluator_.evaluate_runtime(board, ss->runtime_psqt) + network, -MateInMaxPly + 1, MateInMaxPly - 1);

+[[gnu::cold]] bool apply_nnue_v2(std::string* message) {
+    // UseNNUE false disables every network (hand-crafted evaluation only).
+    // O6 / O7: NNUEThreats 1 or NNUEPawnPairs 1 with NNUEKingBuckets 0 asks for a net without king buckets.
+    const bool threats = nnue_threats != 0;
+    const bool pawns = nnue_pawn_pairs != 0;
+    const int wanted = nnue_king_buckets > 0 ? nnue_king_buckets : (threats || pawns ? 1 : 0);
+    const nnue::v2::Net* net = nnue::v2::select(nnue::enabled ? wanted : 0, message);
+    if (net != nullptr && (net->threats != threats || net->pawn_pairs != pawns)) {
+        // O6 / O7: the header's extra-input bits must agree with NNUEThreats / NNUEPawnPairs (nnue_v2_problem
+        // names the mismatch).
+        (void)nnue::v2::select(0, nullptr);
+        net = nullptr;
+        if (message != nullptr) *message = "the loaded net's threat / pawn-pair inputs differ from NNUEThreats / NNUEPawnPairs (not selected)";
+    }
+    if (!nnue::enabled && message != nullptr) *message = "UseNNUE is false: hand-crafted evaluation only";
+    nnue_v2 = net != nullptr;
+    return nnue_v2;
+}

+[[gnu::cold]] std::string nnue_v2_problem(const std::string& file, const std::string& load_error) {
+    const bool threats = nnue_threats != 0;
+    const bool pawns = nnue_pawn_pairs != 0;
+    const bool extras = threats || pawns;
+    if (!nnue::enabled) return {};   // UseNNUE false: every network off on purpose
+    if (file.empty() && nnue_king_buckets <= 0 && !extras) return {};   // nothing requested
+    if (!file.empty() && !load_error.empty()) return "cannot load NNUEFile " + file + ": " + load_error;
+    const nnue::v2::Net* net = nnue::v2::loaded();
+    if (net == nullptr) {
+        if (!file.empty()) return "NNUEFile " + file + " is not loaded";
+        if (nnue_king_buckets <= 0) {   // O6 / O7: NNUEThreats or NNUEPawnPairs alone
+            return threats ? "NNUEThreats is 1 but no v2 net is loaded (NNUEFile is empty); NNUEThreats needs a net with threat "
+                             "inputs (trainer --threats)"
+                           : "NNUEPawnPairs is 1 but no v2 net is loaded (NNUEFile is empty); NNUEPawnPairs needs a net with "
+                             "pawn-pair inputs (trainer --pawn-pairs)";
+        }
+        return "NNUEKingBuckets " + std::to_string(nnue_king_buckets) + " is set but no v2 net is loaded (NNUEFile is empty)";
+    }
+    if (nnue_v2 && nnue::v2::active() == net) return {};
+    const std::string buckets = std::to_string(net->king_buckets) + (net->mirror ? " mirrored" : "") + " king buckets";
+    // O6 / O7: the extra-input bits of the file and NNUEThreats / NNUEPawnPairs must agree.
+    const std::string threat_net = "NNUEFile " + net->path + " has threat inputs (trainer --threats); set NNUEThreats 1 to use it";

+void Worker::update_continuation(SearchStack* ss, Piece piece, int to, int bonus) {
+    for (const int back : {1, 2, 4}) {
+        const SearchStack* previous = ss - back;
+        if (previous->continuation == nullptr || previous->current_move.is_none()) continue;
+        apply_history((*previous->continuation)[static_cast<std::size_t>(piece)][static_cast<std::size_t>(to)],
+                      back == 4 ? bonus / 2 : bonus);
+    }

+        (*histories_.counter_moves)[static_cast<std::size_t>(previous->moved_piece)][previous->current_move.to] = move;
+    }
+    const Piece piece = board.moved_piece(move);

+int Worker::qsearch(Board& board, SearchStack* ss, int alpha, int beta) {

+

+

+    const bool tt_pv = pv_node || (tt_hit && tte.pv());

+        (tt_bound == BoundExact || (tt_bound == BoundLower && tt_score >= beta) ||
+         (tt_bound == BoundUpper && tt_score <= alpha))) {
+        return tt_score;
+    }

+    int futility_base;
+    if (in_check) {
+        ss->static_eval = ValueNone;

+             tt_bound == BoundExact)) {
+            eval = tt_score;
+        }
+        best_score = eval;
+        if (best_score >= beta) {
+            if (!tt_hit) shared_.tt.store(tt_probe, board.key(), score_to_tt(best_score, ply), ss->raw_eval, Move::none(), 0, BoundLower, false, board.halfmove_clock());
+            return best_score;

+                best_score = std::max(best_score, futility_base + capture_value(board, move));
+                continue;
+            }
+            if (!board.see_ge(move, -tune::QsSeeMargin)) continue;
+        }
+        ss->current_move = move;
+        ss->moved_piece = board.moved_piece(move);

+            best_score = score;

+                best_move = move;

+                    ss->pv[0] = move;

+                    ss->pv_length = (ss + 1)->pv_length + 1;
+                }

+            }
+        }
+    }
+    if (in_check && move_count == 0) return mated_in(ply);
+
+    const Bound bound = best_score >= beta ? BoundLower : (pv_node && !best_move.is_none() ? BoundExact : BoundUpper);
+    shared_.tt.store(tt_probe, board.key(), score_to_tt(best_score, ply), in_check ? ValueNone : ss->raw_eval, best_move, 0, bound, tt_pv, board.halfmove_clock());

+// Soft/hard budgets (ms) and whether the search is clock-managed (only those
+// stop early on the soft budget): movetime -> soft = hard = movetime - overhead;
+// clock -> from the remaining time, increment and movestogo; infinite, depth,
+// nodes and mate searches -> no budget.
+inline void allocate(const SearchLimits& limits, std::int64_t& soft_ms, std::int64_t& hard_ms, bool& managed) {
+    const std::int64_t overhead = std::max<std::int64_t>(0, limits.move_overhead_ms);
+    soft_ms = hard_ms = 0;
+    managed = false;
+    if (limits.infinite) {
+        soft_ms = hard_ms = 0;
+    } else if (limits.movetime_ms >= 0) {
+        hard_ms = std::max<std::int64_t>(1, limits.movetime_ms - overhead);
+        soft_ms = hard_ms;
+    } else if (limits.time_ms >= 0) {
+        const std::int64_t time = limits.time_ms;
+        const std::int64_t increment = std::max<std::int64_t>(0, limits.increment_ms);
+        const std::int64_t available = std::max<std::int64_t>(1, time - overhead);
+        std::int64_t soft;
+        std::int64_t hard;
+        if (limits.moves_to_go > 0) {
+            const std::int64_t mtg = std::min<std::int64_t>(limits.moves_to_go, 50);
+            soft = available / (mtg + 1) + increment * 3 / 4;
+            hard = std::min(available * 4 / 5, soft * 4);
+        } else {
+            soft = available / tune::TmSoftDivisor + increment * tune::TmIncrementPercent / 100;
+            hard = std::min(available * tune::TmHardPercent / 100, soft * tune::TmHardSoftMultiple);
+        }
+        soft_ms = std::max<std::int64_t>(1, std::min(soft, available));
+        hard_ms = std::max<std::int64_t>(1, std::min(hard, available));
+        managed = true;
+    }
+}
+
+// Stop early when the best move has been stable; spend longer when it keeps changing.
+[[nodiscard]] inline double stability_factor(int stable_iterations, int depth) {
+    double factor = 1.0;
+    if (stable_iterations >= 6) factor = tune::TmStableFactor / 100.0;
+    else if (stable_iterations >= 3) factor = tune::TmSemiStableFactor / 100.0;
+    else if (stable_iterations == 0 && depth >= 6) factor = tune::TmUnstableFactor / 100.0;
+    return factor;
+}
+
+// A best move that absorbed most of the effort (share of the iteration's
+// nodes spent below it) is clearly best; spend longer when the alternatives
+// still draw attention.
+[[nodiscard]] inline double effort_scale(double effort) {
+    return std::clamp(tune::TmEffortBase / 100.0 - effort, 0.6, 1.4);
+}

+[[nodiscard]] inline bool eval_fell(int previous_score, int score) {
+    return previous_score != ValueNone && !is_mate_score(score) && score < previous_score;
+}

+    const int drop = std::min(previous_score - score, 150);
+    return 1.0 + drop * tune::TmFallingEvalPercent / 10000.0;
+}

+thread_local const TranspositionTable::ExclusiveLease* TranspositionTable::active_lease_ = nullptr;

+TranspositionTable::ExclusiveLease::ExclusiveLease(TranspositionTable& table)
+    : table_(table), previous_(active_lease_), gate_(table.exclusive_gate_, std::defer_lock) {
+    if (!table_.has_exclusive_access()) {
+        gate_.lock();
+        table_.exclusive_requested_.store(true, std::memory_order_release);
+        try {
+            // A foreign accessor either completes before its stripe drains,
+            // or observes the request after acquiring that drained stripe.
+            for (std::size_t i = 0; i < StripeCount; ++i) {
+                const std::lock_guard<std::mutex> drain(table_.stripes_[i].mutex);
+            }
+        } catch (...) {
+            table_.exclusive_requested_.store(false, std::memory_order_release);
+            throw;
+        }
+    }
+    active_lease_ = this;
+}
+
+TranspositionTable::ExclusiveLease::~ExclusiveLease() {
+    // Publish completed writes before member destruction releases the gate.
+    active_lease_ = previous_;
+    if (gate_.owns_lock()) table_.exclusive_requested_.store(false, std::memory_order_release);
+}
+
+bool TranspositionTable::has_exclusive_access() const {
+    for (const ExclusiveLease* lease = active_lease_; lease != nullptr; lease = lease->previous_) {
+        if (&lease->table_ == this) return true;
+    }
+    return false;
+}
+
+std::unique_lock<std::mutex> TranspositionTable::lock_stripe(std::size_t index) const {
+    std::mutex& stripe = stripes_[index % StripeCount].mutex;
+    if (has_exclusive_access()) return std::unique_lock<std::mutex>(stripe, std::defer_lock);
+    for (;;) {
+        std::unique_lock<std::mutex> lock(stripe);
+        if (!exclusive_requested_.load(std::memory_order_acquire)) return lock;
+        // Never wait for the gate while holding a stripe: the owner drains it.
+        lock.unlock();
+        const std::lock_guard<std::mutex> wait(exclusive_gate_);
+    }
+}

+TTProbe TranspositionTable::probe(std::uint64_t key) const {
+    TTCluster& cluster = cluster_for(key);
+    const std::size_t index = static_cast<std::size_t>(&cluster - clusters_.get());
+    const auto lock = lock_stripe(index);
+    auto snapshot = [&](const TTEntry& entry, bool hit) {
+        TTProbe result;
+        result.entry = entry;
+        result.hit = hit;
+        result.owner_ = this;
+        result.cluster_ = index;
+        result.slot_ = static_cast<std::uint8_t>(&entry - cluster.entries.data());
+        result.epoch_ = epoch_;
+        result.key_ = key;
+        return result;
+    };
+    const std::uint32_t key32 = verification_key(key);
+    for (TTEntry& entry : cluster.entries) {
+        if (entry.key32 == key32 && entry.occupied()) {
+            // Reused entries remain recent even when search returns a TT cutoff.
+            if (entry.generation() != generation_) {
+                entry.gen_bound = static_cast<std::uint8_t>(
+                    (static_cast<unsigned>(generation_) << 3U) | (entry.gen_bound & 7U));
+            }
+            return snapshot(entry, true);
+        }
+    }
+    const TTEntry* replace = &cluster.entries[0];
+    auto relative_age = [this](const TTEntry& entry) {
+        return static_cast<int>((32 + generation_ - entry.generation()) & 31U);
+    };
+    for (const TTEntry& entry : cluster.entries) {
+        if (!entry.occupied()) return snapshot(entry, false);
+        if (entry.depth - 4 * relative_age(entry) < replace->depth - 4 * relative_age(*replace)) replace = &entry;
+    }
+    return snapshot(*replace, false);
+}

+void TranspositionTable::store(const TTProbe& probe, std::uint64_t key, int score, int eval, Move move, int depth,
+                               Bound bound, bool pv, std::uint16_t halfmove_clock) {
+    // Reject default, foreign or lifecycle-stale handles without touching storage.
+    // Lifecycle operations themselves must not run concurrently with search.
+    if (probe.owner_ != this || probe.epoch_ != epoch_ || probe.key_ != key ||
+        probe.cluster_ >= cluster_count_ || probe.slot_ >= 4) return;
+    const auto lock = lock_stripe(probe.cluster_);
+    TTEntry* entry = &clusters_[probe.cluster_].entries[probe.slot_];
+    const std::uint32_t key32 = verification_key(key);
+    const bool same_position = entry->key32 == key32 && entry->occupied();
+    if (!move.is_none() || !same_position) entry->move = move;
+    if (bound == BoundExact || !same_position || depth + 4 > entry->depth || entry->generation() != generation_) {
+        entry->key32 = key32;
+        entry->score = static_cast<std::int16_t>(std::clamp(score, -32767, 32767));
+        entry->rule50_clock = static_cast<std::uint8_t>(std::min<int>(halfmove_clock, 100));
+        entry->eval = static_cast<std::int16_t>(std::clamp(eval, -32767, 32767));
+        entry->depth = static_cast<std::uint8_t>(std::clamp(depth, 0, 255));
+        entry->gen_bound = static_cast<std::uint8_t>((static_cast<unsigned>(generation_) << 3U) | (pv ? 4U : 0U) | static_cast<unsigned>(bound));
+    }
+}

+struct ReductionTable {
+    std::array<std::array<std::uint8_t, 64>, 64> quiet{};
+    std::array<std::array<std::uint8_t, 64>, 64> noisy{};
+    ReductionTable() { refresh(); }
+    void refresh() {
+        for (int depth = 1; depth < 64; ++depth) {
+            for (int moves = 1; moves < 64; ++moves) {
+                const double product = std::log(static_cast<double>(depth)) * std::log(static_cast<double>(moves));
+                quiet[static_cast<std::size_t>(depth)][static_cast<std::size_t>(moves)] = static_cast<std::uint8_t>(
+                    std::max(0.0, tune::LmrQuietBase / 100.0 + product / (tune::LmrQuietDivisor / 100.0)));
+                noisy[static_cast<std::size_t>(depth)][static_cast<std::size_t>(moves)] = static_cast<std::uint8_t>(
+                    std::max(0.0, tune::LmrNoisyBase / 100.0 + product / (tune::LmrNoisyDivisor / 100.0)));
+            }
+        }
+    }
+};
+ReductionTable Reductions;
+
+[[nodiscard]] int reduction(bool quiet, int depth, int moves) {
+    const std::size_t d = static_cast<std::size_t>(std::min(depth, 63));
+    const std::size_t m = static_cast<std::size_t>(std::min(moves, 63));
+    return quiet ? Reductions.quiet[d][m] : Reductions.noisy[d][m];
+}
+
+[[nodiscard]] int lmp_threshold(int depth, bool improving) {
+    return improving ? tune::LmpBase + depth * depth : (tune::LmpBase + depth * depth) / 2;
+}

+
+void tune::refresh() { Reductions.refresh(); }
